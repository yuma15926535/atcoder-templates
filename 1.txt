//答えを直接求めるのは難しいが、ある整数が性質を満たすかを求めるのは簡単(貪欲法)
//性質を満たす整数は非負整数全体の集合を二分割する->答えで二分探索法
//末尾にLを追加しK+1以上であるかを判定することで実装を楽にする
int N,L,K;
vector<int> A;
bool solv(int target){
    ll cl=0,c=0,b=0;
    rep(i,0,N+1){
        cl=A[i]-b;
        if(cl>=target){
            c++;
            cl=0;
            b=A[i];
        }
    }
    if(c>=K+1)return true;
    else return false;
}

int main()
{
    cin>>N>>L>>K;
    A=vector<int>(N);
    rep(i,0,N)cin>>A[i];
    A.push_back(L);
    int left=0,right=1000000000;
    while(right-left>1){
        int mid=(right+left)/2;
        if(solv(mid))left=mid;
        else right=mid;
    }
    cout<<left<<"\n";
    return 0;
}
