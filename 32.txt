#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    int N;
    cin>>N;
    vector<vector<int>> A(N,vector<int>(N));
    rep(i,0,N)rep(j,0,N)cin>>A[i][j];
    vector<set<int>> ng(N);
    int M;
    cin>>M;
    rep(i,0,M){
        int x,y;
        cin>>x>>y;
        x--;y--;
        ng[x].insert(y);
        ng[y].insert(x);
    }
    vector<int> pos(N);
    int ans=100000;
    rep(i,0,N)pos[i]=i;
    //Nが十分小さいので順列全探索
    do{
        int time=0;
        bool b=true;
        rep(i,0,N-1){
            time+=A[pos[i]][i];
            if(ng[pos[i]].count(pos[i+1]))b=false;
        }
        time+=A[pos[N-1]][N-1];
        if(b)ans=min(ans,time);
    }while(next_p(pos));
    if(ans==100000)cout<<-1<<"\n";
    else cout<<ans<<"\n";
    return 0;
}
