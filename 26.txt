#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor

vector<vector<int>> G;
vector<int> col;
//DFSを用いて彩色を実装
void dfs(int pos,int c){
    col[pos]=c;
    for(int a:G[pos]){
        if(col[a]==0)dfs(a,c*(-1));
    }
}

int main()
{
    int N;
    cin>>N;
    G=vector<vector<int>>(N);
    col=vector<int>(N,0);
    rep(i,0,N-1){
        int a,b;
        cin>>a>>b;
        a--;b--;
        G[a].push_back(b);
        G[b].push_back(a);
    }
    dfs(0,1);
    int c1=0;
    for(int c:col)if(c==1)c1++;
    if(c1>=N/2)c1=1;
    else c1=-1;
    int count=0;
    rep(i,0,N)if(col[i]==c1){
        count++;
        cout<<i+1<<" ";
        if(count==N/2)break;
    }
    cout<<"\n";
    return 0;
}
