ll base8(string snum){
    ll num=0;
    rep(i,0,snum.size())num=num*8+(int)(snum[i]-'0');
    return num;
}

string base9(ll num){
    string snum="";
    while(num>=9){
        snum=(char)('0'+num%9)+snum;
        num/=9;
    }
    snum=(char)('0'+num)+snum;
    return snum;
}
