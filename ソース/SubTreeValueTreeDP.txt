struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

vector<vector<int>> make_random_tree(
    int N,
    mt19937 &rng,
    bool shuffle_vertex = true
){
    vector<int> perm(N);
    iota(perm.begin(), perm.end(), 0);

    // 頂点番号をシャッフル
    if(shuffle_vertex){
        shuffle(perm.begin(), perm.end(), rng);
    }

    vector<vector<int>> G(N);

    for(int i = 1; i < N; i++){

        // [0, i-1] のどれかに接続
        int p = uniform_int_distribution<int>(0, i - 1)(rng);

        int u = perm[i];
        int v = perm[p];

        G[u].push_back(v);
        G[v].push_back(u);
    }

    return G;
}

struct TreeDP {

    int N;
    vector<vector<int>> g;

    // ★ 頂点の値
    vector<ll> val;

    // ★ 部分木和
    vector<ll> sub_sum;

    // コンストラクタ
    TreeDP(
        const vector<vector<int>>& graph,
        const vector<ll>& values
    )
        : g(graph), val(values)
    {
        N = g.size();

        // ★ 部分木和初期化
        sub_sum.assign(N, 0);
    }

    // ===== 木DP DFS =====
    void dfs(int v, int p){

        // ★ 自身の値
        sub_sum[v] = val[v];

        for(int to : g[v]){

            if(to == p) continue;

            dfs(to, v);

            // ★ 子部分木和を加算
            sub_sum[v] += sub_sum[to];
        }
    }

    // 実行
    void build(int root=0){
        dfs(root, -1);
    }
};

void random_test(){

    mt19937 rng(
        chrono::steady_clock::now().time_since_epoch().count()
    );

    for(int tc = 0; tc < 10000; tc++){

        // ===== ランダム木生成 =====
        int N = uniform_int_distribution<int>(1, 100)(rng);

        auto G = make_random_tree(N, rng);

        // ===== 頂点値生成 =====
        vector<ll> val(N);

        for(int i = 0; i < N; i++){

            val[i] =
                uniform_int_distribution<ll>(-100, 100)(rng);
        }

        // ===== root =====
        int root =
            uniform_int_distribution<int>(0, N - 1)(rng);

        // ===== 木DP実行 =====
        TreeDP tree_dp(G, val);

        tree_dp.build(root);

        // ===== ナイーブに部分木和計算 =====
        vector<ll> naive(N);

        auto dfs = [&](auto self, int v, int p)->ll{

            ll sum = val[v];

            for(int to : G[v]){

                if(to == p) continue;

                sum += self(self, to, v);
            }

            naive[v] = sum;

            return sum;
        };

        dfs(dfs, root, -1);

        // ===== 比較 =====
        for(int v = 0; v < N; v++){

            if(tree_dp.sub_sum[v] != naive[v]){

                cout << "NG\n";

                cout << "N = " << N << "\n";

                cout << "root = " << root << "\n";

                cout << "vertex = " << v << "\n";

                cout << "TreeDP = "
                     << tree_dp.sub_sum[v] << "\n";

                cout << "Naive = "
                     << naive[v] << "\n";

                cout << "\nValues\n";

                for(int i = 0; i < N; i++){

                    cout << i
                         << " : "
                         << val[i]
                         << "\n";
                }

                cout << "\nEdges\n";

                for(int i = 0; i < N; i++){
                    for(int to : G[i]){
                        if(i < to){
                            cout << i << " " << to << "\n";
                        }
                    }
                }

                return;
            }
        }
    }

    cout << "Random Test OK\n";
}

void test(){

    /*
            0(5)
          /  |  \
       1(2) 2(4) 3(1)
              |
            4(3)

        部分木和:
        0 -> 15
        1 -> 2
        2 -> 7
        3 -> 1
        4 -> 3
    */

    int N = 5;

    vector<vector<int>> G(N);

    auto add_edge = [&](int u, int v){

        G[u].push_back(v);
        G[v].push_back(u);
    };

    add_edge(0, 1);
    add_edge(0, 2);
    add_edge(0, 3);
    add_edge(2, 4);

    // ===== 頂点値 =====
    vector<ll> val = {
        5,
        2,
        4,
        1,
        3
    };

    // ===== 木DP実行 =====
    TreeDP tree_dp(G, val);

    tree_dp.build(0);

    // ===== 出力 =====
    cout << "部分木和一覧\n";

    for(int v = 0; v < N; v++){

        cout << "頂点 "
             << v
             << " : "
             << tree_dp.sub_sum[v]
             << "\n";
    }
}
