//メモ化再帰
//単純な再帰だと呼び出し数が多く、与えられる引数の種類が限られている場合に有効
unordered_map<ll,ll> m;
ll rec(ll n){
    if(n==0)return 1;
    if(m.count(n))return m[n];
    return m[n]=rec(n/2)+rec(n/3);
}

int main()
{
    ll N;
    cin>>N;
    cout<<rec(N)<<"\n";
    return 0;
}
