    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }

    ){

                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }

    }

struct S {

    ll sum;   // 区間和

    ll pref;  // 最大 prefix 和

    ll suff;  // 最大 suffix 和

    ll best;  // 最大部分列和

    int len;  // 区間長
};

/*
=====================================================
★ 更新情報
=====================================================
*/

using F = ll;

/*
=====================================================
★ 演算定義
=====================================================
*/

// ノード結合
S combine(S l, S r) {

    // dummy node 対策
    if(l.len == 0)
        return r;

    if(r.len == 0)
        return l;

    return {

        // sum
        l.sum + r.sum,

        // pref
        max(l.pref, l.sum + r.pref),

        // suff
        max(r.suff, r.sum + l.suff),

        // best
        max({
            l.best,
            r.best,
            l.suff + r.pref
        }),

        // len
        l.len + r.len
    };
}

// 更新適用
S apply(F f, S x) {

    if(f == -LINF)
        return x;

    ll total = f * x.len;

    ll val;

    // 空部分列不許可
    if(f > 0)
        val = total;
    else
        val = f;

    return {
        total,
        val,
        val,
        val,
        x.len
    };
}

// 更新の合成
F compose(F f, F g) {

    return (f != -LINF ? f : g);
}

S idS() {

    return {
        0,
        -LINF,
        -LINF,
        -LINF,
        0
    };
}

F idF() { return -LINF; }

/*
=====================================================
Lazy SegTree 本体
=====================================================
*/

struct LazyKadaneSegTree {

    int n;

    vector<S> seg;

    vector<F> lazy;

    /*
    =====================================================
    ★ 値 → ノード
    =====================================================
    */

    S make_node(ll x) {

        return {
            x,
            x,
            x,
            x,
            1
        };
    }

    /*
    =====================================================
    ★ construct
    =====================================================
    */

    LazyKadaneSegTree(const vector<ll>& v) {

        int sz = v.size();

        n = 1;

        while(n < sz)
            n <<= 1;

        seg.assign(2 * n, idS());

        lazy.assign(2 * n, idF());

        for(int i = 0; i < sz; i++)
            seg[n + i] = make_node(v[i]);

        /*
        =============================================
        ★ dummy leaf
        =============================================
        */

        for(int i = sz; i < n; i++) {

            seg[n + i] = {
                0,
                -LINF,
                -LINF,
                -LINF,
                1
            };
        }

        for(int i = n - 1; i > 0; i--)
            seg[i] = combine(seg[i << 1], seg[i << 1 | 1]);
    }

    /*
    =====================================================
    ★ apply node
    =====================================================
    */

    void apply_node(int k, F f) {

        seg[k] = apply(f, seg[k]);

        if(k < n)
            lazy[k] = compose(f, lazy[k]);
    }

    /*
    =====================================================
    ★ push
    =====================================================
    */

    void push(int k) {

        if(lazy[k] == idF())
            return;

        apply_node(k << 1, lazy[k]);

        apply_node(k << 1 | 1, lazy[k]);

        lazy[k] = idF();
    }

    /*
    =====================================================
    ★ range apply
    =====================================================
    */

    void range_apply(
        int l,
        int r,
        F f,
        int k,
        int a,
        int b
    ) {

        if(r <= a || b <= l)
            return;

        if(l <= a && b <= r) {

            apply_node(k, f);

            return;
        }

        push(k);

        int mid = (a + b) / 2;

        range_apply(l, r, f, k << 1, a, mid);

        range_apply(l, r, f, k << 1 | 1, mid, b);

        seg[k] = combine(
            seg[k << 1],
            seg[k << 1 | 1]
        );
    }

    void range_apply(int l, int r, F f) {

        range_apply(l, r, f, 1, 0, n);
    }

    /*
    =====================================================
    ★ range query
    =====================================================
    */

    S range(
        int l,
        int r,
        int k,
        int a,
        int b
    ) {

        if(r <= a || b <= l)
            return idS();

        if(l <= a && b <= r)
            return seg[k];

        push(k);

        int mid = (a + b) / 2;

        return combine(

            range(l, r, k << 1, a, mid),

            range(l, r, k << 1 | 1, mid, b)
        );
    }

    /*
    =====================================================
    ★ 最大部分列和
    =====================================================
    */

    ll range_max_subarray(int l, int r) {

        return range(l, r, 1, 0, n).best;
    }

    /*
    =====================================================
    ★ 全体最大部分列和
    =====================================================
    */

    ll all_max_subarray() {

        return seg[1].best;
    }
};

void random_test_lazy_kadane() {

    random_device rd;

    mt19937 rng(rd());

    rep(_, 0, 10000) {

        /*
        =====================================================
        ★ 配列生成
        =====================================================
        */

        int N = uniform_int_distribution<int>(1, 20)(rng);

        vector<ll> A(N);

        rep(i, 0, N) {

            A[i] = uniform_int_distribution<ll>(-20, 20)(rng);
        }

        LazyKadaneSegTree seg(A);

        /*
        =====================================================
        ★ ランダムクエリ
        =====================================================
        */

        rep(q, 0, 200) {

            int type = uniform_int_distribution<int>(0, 1)(rng);

            /*
            =================================================
            ★ 区間代入更新
            =================================================
            */

            if(type == 0) {

                int l =
                    uniform_int_distribution<int>(0, N - 1)(rng);

                int r =
                    uniform_int_distribution<int>(l + 1, N)(rng);

                ll x =
                    uniform_int_distribution<ll>(-20, 20)(rng);

                /*
                =============================================
                ★ brute
                =============================================
                */

                rep(i, l, r)
                    A[i] = x;

                /*
                =============================================
                ★ segtree
                =============================================
                */

                seg.range_apply(l, r, x);
            }

            /*
            =================================================
            ★ 区間最大部分列和
            =================================================
            */

            else {

                int l =
                    uniform_int_distribution<int>(0, N - 1)(rng);

                int r =
                    uniform_int_distribution<int>(l + 1, N)(rng);

                /*
                =============================================
                ★ brute
                =============================================
                */

                ll brute = -LINF;

                rep(i, l, r) {

                    ll sum = 0;

                    rep(j, i, r) {

                        sum += A[j];

                        brute = max(brute, sum);
                    }
                }

                /*
                =============================================
                ★ segtree
                =============================================
                */

                ll seg_ans =
                    seg.range_max_subarray(l, r);

                /*
                =============================================
                ★ verify
                =============================================
                */

                if(brute != seg_ans) {

                    cerr << "WA detected\n";

                    cerr << "N = " << N << "\n";

                    cerr << "A : ";

                    for(auto x : A)
                        cerr << x << " ";

                    cerr << "\n";

                    cerr << "l r = "
                         << l << " "
                         << r << "\n";

                    cerr << "brute = "
                         << brute << "\n";

                    cerr << "seg   = "
                         << seg_ans << "\n";

                    assert(false);
                }
            }
        }
    }

    cerr << "Lazy Kadane SegTree Random Test Passed\n";
}

void solv(){
    random_test_lazy_kadane();
    return;
}
