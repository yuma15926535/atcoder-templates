/* ====================================
   Dijkstra（経路復元付き）
   ====================================

【概要】
- 非負重みグラフの単一始点最短路
- priority_queue を使用

【制約目安】
- 頂点数 N ≤ 2×10^5
- 辺数   M ≤ 2×10^5
- 辺の重みは 0 以上

【戻り値】
- dist[i] : 始点から i への最短距離
- prev[i] : 最短経路における直前の頂点（経路復元用）
*/
pair<vector<ll>, vector<int>> dijkstra_with_prev(
    int N,
    vector<vector<pair<int,ll>>> &G, // G[u] = {v, cost}
    int s
){
    const ll INF = (1LL<<60);
    vector<ll> dist(N, INF);
    vector<int> prev(N, -1);

    priority_queue<pair<ll,int>, vector<pair<ll,int>>, greater<pair<ll,int>>> pq;
    dist[s] = 0;
    pq.push({0, s});

    while(!pq.empty()){
        auto [d, u] = pq.top(); pq.pop();
        if(d > dist[u]) continue;

        for(auto [v, cost] : G[u]){
            if(dist[v] > dist[u] + cost){
                dist[v] = dist[u] + cost;
                prev[v] = u;
                pq.push({dist[v], v});
            }
        }
    }

    return {dist, prev};
}


/* ====================================
   01-BFS（経路復元付き）
   ====================================

【概要】
- 辺の重みが 0 または 1 の場合の最短路
- deque を用いて O(N+M)

【制約目安】
- 頂点数 N ≤ 2×10^5
- 辺数   M ≤ 2×10^5
- 辺の重みは 0 or 1 のみ

【戻り値】
- dist[i] : 始点から i への最短距離
- prev[i] : 経路復元用
*/
pair<vector<ll>, vector<int>> bfs01_with_prev(
    int N,
    vector<vector<pair<int,int>>> &G, // cost は 0 or 1
    int s
){
    const ll INF = (1LL<<60);
    vector<ll> dist(N, INF);
    vector<int> prev(N, -1);
    deque<int> deq;

    dist[s] = 0;
    deq.push_front(s);

    while(!deq.empty()){
        int u = deq.front();
        deq.pop_front();

        for(auto [v, cost] : G[u]){
            if(dist[v] > dist[u] + cost){
                dist[v] = dist[u] + cost;
                prev[v] = u;
                if(cost == 0) deq.push_front(v);
                else          deq.push_back(v);
            }
        }
    }

    return {dist, prev};
}


/* ====================================
   Bellman-Ford（負閉路検出付き）
   ====================================

【概要】
- 負の辺を含む単一始点最短路
- 負閉路の存在を検出可能

【制約目安】
- 頂点数 N ≤ 2000
- 辺数   M ≤ 2×10^5
- 計算量 O(NM) のため重い

【戻り値】
- dist[i] : 最短距離
- has_negative_cycle : 負閉路が存在するか
*/
struct Edge {
    int from, to;
    ll cost;
};

pair<vector<ll>, bool> bellman_ford(
    int N,
    vector<Edge> &edges,
    int s
){
    const ll INF = (1LL<<60);
    vector<ll> dist(N, INF);
    dist[s] = 0;

    // N-1 回緩和
    for(int i = 0; i < N-1; i++){
        bool updated = false;
        for(auto &e : edges){
            if(dist[e.from] == INF) continue;
            if(dist[e.to] > dist[e.from] + e.cost){
                dist[e.to] = dist[e.from] + e.cost;
                updated = true;
            }
        }
        if(!updated) break;
    }

    // 負閉路検出
    bool has_negative_cycle = false;
    for(auto &e : edges){
        if(dist[e.from] == INF) continue;
        if(dist[e.to] > dist[e.from] + e.cost){
            has_negative_cycle = true;
            break;
        }
    }

    return {dist, has_negative_cycle};
}


/* ====================================
   Warshall-Floyd（全点対最短路）
   ====================================

【概要】
- 全頂点対の最短距離を求める
- 負の辺 OK
- dist[i][i] < 0 なら負閉路あり

【制約目安】
- 頂点数 N ≤ 400
- 計算量 O(N^3)

【引数】
- dist : 初期距離行列（未接続は INF）
*/
vector<vector<ll>> warshall_floyd(
    int N,
    vector<vector<ll>> dist
){
    const ll INF = (1LL<<60);

    for(int k = 0; k < N; k++){
        for(int i = 0; i < N; i++){
            if(dist[i][k] == INF) continue;
            for(int j = 0; j < N; j++){
                if(dist[k][j] == INF) continue;
                dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j]);
            }
        }
    }
    return dist;
}


/* ====================================
   経路復元ユーティリティ
   ====================================

【概要】
- prev 配列から終点 t までの経路を復元
- Dijkstra / 01-BFS 用

【制約】
- prev は -1 で始点を表す
*/
vector<int> restore_path(int t, vector<int> &prev){
    vector<int> path;
    for(int v = t; v != -1; v = prev[v]){
        path.push_back(v);
    }
    reverse(path.begin(), path.end());
    return path;
}
