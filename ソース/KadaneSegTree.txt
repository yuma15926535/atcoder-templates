struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
=====================================================
★ ノード情報（Kadane SegTree）
=====================================================
*/

struct S {
    ll sum;   // 区間和
    ll pref;  // 最大prefix和
    ll suff;  // 最大suffix和
    ll best;  // 最大部分列和
};

/*
=====================================================
★ 演算定義
=====================================================
*/

// 単位元
S e() {
    const ll INF = (1LL << 60);
    return {
        0,
        -INF,
        -INF,
        -INF
    };
}

// ノード結合
S op(S l, S r) {
    return {
        l.sum + r.sum,

        max(l.pref, l.sum + r.pref),

        max(r.suff, r.sum + l.suff),

        max({
            l.best,
            r.best,
            l.suff + r.pref
        })
    };
}

/*
=====================================================
★ 値 → ノード変換
=====================================================
*/

S make_node(ll x) {
    return {
        x,
        x,
        x,
        x
    };
}

/*
=====================================================
★ Kadane SegTree
=====================================================
*/

struct KadaneSegTree {
    int n;
    segtree<S, op, e> seg;

    // サイズ指定
    KadaneSegTree(int sz)
        : n(sz), seg(vector<S>(sz, make_node(0))) {}

    // 配列構築
    KadaneSegTree(const vector<ll>& v)
        : n(v.size()) {

        vector<S> init(n);

        rep(i, 0, n) {
            init[i] = make_node(v[i]);
        }

        seg = segtree<S, op, e>(init);
    }

    /*
    =====================================================
    ★ 1点更新
    =====================================================
    */

    void set_val(int p, ll x) {
        seg.set(p, make_node(x));
    }

    /*
    =====================================================
    ★ 区間取得
    =====================================================
    */

    S prod(int l, int r) {
        return seg.prod(l, r);
    }

    /*
    =====================================================
    ★ 区間最大部分列和
    =====================================================
    */

    ll max_subarray(int l, int r) {
        return prod(l, r).best;
    }

    /*
    =====================================================
    ★ 全体最大部分列和
    =====================================================
    */

    ll all_max_subarray() {
        return seg.all_prod().best;
    }
};

void random_test_kadane() {

    random_device rd;
    mt19937 rng(rd());

    rep(_, 0, 10000) {

        /*
        =====================================================
        ★ ランダム配列生成
        =====================================================
        */

        int N = uniform_int_distribution<int>(1, 20)(rng);

        vector<ll> A(N);

        rep(i, 0, N) {
            A[i] = uniform_int_distribution<ll>(-20, 20)(rng);
        }

        KadaneSegTree seg(A);

        /*
        =====================================================
        ★ ランダムクエリ
        =====================================================
        */

        rep(q, 0, 200) {

            int type = uniform_int_distribution<int>(0, 1)(rng);

            /*
            =================================================
            ★ 1点更新
            =================================================
            */

            if(type == 0) {

                int p = uniform_int_distribution<int>(0, N - 1)(rng);

                ll x = uniform_int_distribution<ll>(-20, 20)(rng);

                A[p] = x;

                seg.set_val(p, x);
            }

            /*
            =================================================
            ★ 区間最大部分列和
            =================================================
            */

            else {

                int l = uniform_int_distribution<int>(0, N - 1)(rng);

                int r = uniform_int_distribution<int>(l + 1, N)(rng);

                /*
                =============================================
                ★ 愚直
                =============================================
                */

                ll brute = -(1LL << 60);

                rep(i, l, r) {

                    ll sum = 0;

                    rep(j, i, r) {

                        sum += A[j];

                        brute = max(brute, sum);
                    }
                }

                /*
                =============================================
                ★ segtree
                =============================================
                */

                ll seg_ans = seg.max_subarray(l, r);

                /*
                =============================================
                ★ verify
                =============================================
                */

                if(brute != seg_ans) {

                    cerr << "WA detected\n";

                    cerr << "N = " << N << "\n";

                    cerr << "A : ";

                    for(auto x : A) {
                        cerr << x << " ";
                    }

                    cerr << "\n";

                    cerr << "l r = " << l << " " << r << "\n";

                    cerr << "brute = " << brute << "\n";

                    cerr << "seg   = " << seg_ans << "\n";

                    assert(false);
                }
            }
        }
    }

    cerr << "Kadane SegTree Random Test Passed\n";
}

void solv(){
    random_test_kadane();

}
