//値の追加と最小値の取得を高速で行うpriority_queue
//全体加算する値を別で持ち取得時に処理
//追加時に加算する値を引くことで以後の加算地のみ反映される
//加算では全体の大小関係は変化しない
struct QueryingMultiset{
    priority_queue<ll, vector<ll>, greater<ll>>pq;
    ll offset;
    QueryingMultiset(){
        offset=0;
    }
    void query1(ll x){
        pq.push(x-offset);
    }
    void query2(ll x){
        offset+=x;
    }
    ll query3(){
        ll val=pq.top();
        pq.pop();
        return val+offset;
    }
};

int main(){
    int Q;

        }else if(q==2){

            cin>>x;
            qm.query2(x);
        }else cout<<qm.query3()<<"\n";
    }
    return 0;
}
