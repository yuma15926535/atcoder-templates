struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/**
 * @brief Range Add Point Query BIT (区間加算・一点取得 Fenwick Tree)
 * 
 * @details 区間に対する一括加算と、任意のインデックスの一点取得を高速に行うデータ構造。
 * 内部的には、Difference Array (差分配列) を Fenwick Tree 上で管理することで実現している。
 * 
 * 注意点:
 * - 取得できるのは一点の値のみであり、区間の総和を求めることはできない。
 * - 範囲外アクセスに関しては、アサーションによって動作を保証する。
 * 
 * 用途:
 * - 区間加算クエリと一点取得クエリの混在する状況のシミュレーション
 * 
 * 計算量:
 * - 構築: O(N)
 * - 区間加算 (add, all_add): O(log N)
 * - 一点取得 (get, operator[]): O(log N)
 * - 全要素取得 (values): O(N log N)
 * - クリア (clear): O(N)
 * 
 * API一覧:
 * - RangeAddPointQueryBIT(n), RangeAddPointQueryBIT(a)
 * - build(a), assign(a), clear()
 * - add(l, r, x), all_add(x)
 * - get(i), operator[](i), values()
 * - size(), empty()
 */
template <class T>
class RangeAddPointQueryBIT {
private:
    int n;
    std::vector<T> tree;

    // 内部用の Fenwick Tree sum [0, r)
    T sum(int r) const {
        T s = 0;
        for (int idx = r; idx > 0; idx -= idx & -idx) {
            s += tree[idx];
        }
        return s;
    }

    // 内部用の Fenwick Tree add
    void add_point(int i, T x) {
        for (int idx = i + 1; idx <= n; idx += idx & -idx) {
            tree[idx] += x;
        }
    }

public:
    /**
     * @brief コンストラクタ (サイズ指定)
     * @param n 要素数
     * @note 計算量: O(N)
     */
    explicit RangeAddPointQueryBIT(int n = 0) : n(n), tree(n + 1, 0) {}

    /**
     * @brief コンストラクタ (初期配列指定)
     * @param a 初期配列
     * @note 計算量: O(N)
     */
    explicit RangeAddPointQueryBIT(const std::vector<T>& a) {
        build(a);
    }

    /**
     * @brief 初期配列からO(N)で構築
     * @param a 初期配列
     * @note 計算量: O(N)
     */
    void build(const std::vector<T>& a) {
        n = a.size();
        tree.assign(n + 1, 0);
        std::vector<T> diff(n, 0);
        for (int i = 0; i < n; ++i) {
            diff[i] = a[i] - (i > 0 ? a[i - 1] : 0);
        }
        for (int i = 0; i < n; ++i) {
            tree[i + 1] = diff[i];
        }
        for (int i = 1; i <= n; ++i) {
            int ni = i + (i & -i);
            if (ni <= n) {
                tree[ni] += tree[i];
            }
        }
    }

    /**
     * @brief 新しい配列で再初期化
     * @param a 新しい配列
     * @note 計算量: O(N)
     */
    void assign(const std::vector<T>& a) {
        build(a);
    }

    /**
     * @brief 要素数を取得
     * @return 要素数
     * @note 計算量: O(1)
     */
    int size() const {
        return n;
    }

    /**
     * @brief 空であるか判定
     * @return 空ならtrue、そうでなければfalse
     * @note 計算量: O(1)
     */
    bool empty() const {
        return n == 0;
    }

    /**
     * @brief すべての要素を0にする
     * @note 計算量: O(N)
     */
    void clear() {
        std::fill(tree.begin(), tree.end(), 0);
    }

    /**
     * @brief 半開区間 [l, r) に値 x を加算する
     * @param l 左端のインデックス (0-index)
     * @param r 右端のインデックス (0-index)
     * @param x 加算する値
     * @note 計算量: O(log N)
     */
    void add(int l, int r, T x) {
        assert(0 <= l && l <= r && r <= n);
        if (l == r) return;
        add_point(l, x);
        if (r < n) {
            add_point(r, -x);
        }
    }

    /**
     * @brief すべての要素に値 x を加算する
     * @param x 加算する値
     * @note 計算量: O(log N)
     */
    void all_add(T x) {
        add(0, n, x);
    }

    /**
     * @brief 位置 i の現在の値を取得する (a[i])
     * @param i インデックス (0-index)
     * @return 位置 i の値
     * @note 計算量: O(log N)
     */
    T get(int i) const {
        assert(0 <= i && i < n);
        return sum(i + 1);
    }

    /**
     * @brief 位置 i の現在の値を取得する (getと同じ)
     * @param i インデックス (0-index)
     * @return 位置 i の値
     * @note 計算量: O(log N)
     */
    T operator[](int i) const {
        return get(i);
    }

    /**
     * @brief 現在の全要素を格納した vector を取得する
     * @return 全要素の配列
     * @note 計算量: O(N log N)
     */
    std::vector<T> values() const {
        std::vector<T> res(n);
        for (int i = 0; i < n; ++i) {
            res[i] = get(i);
        }
        return res;
    }
};

// 動作確認用テスト
void test_RangeAddPointQueryBIT() {
    std::cout << "--- test_RangeAddPointQueryBIT 開始 ---" << std::endl;

    std::vector<int> a = {1, 2, 3, 4, 5};
    RangeAddPointQueryBIT<int> bit(a);

    std::cout << "初期配列: ";
    for (int i = 0; i < bit.size(); ++i) std::cout << bit[i] << " ";
    std::cout << std::endl;

    bit.add(1, 4, 10);
    std::cout << "add(1, 4, 10) 実行後 (半開区間 [1, 4) に 10 加算):" << std::endl;
    std::cout << "values(): ";
    for (int x : bit.values()) std::cout << x << " ";
    std::cout << " (期待値: 1 12 13 14 5)" << std::endl;

    bit.all_add(5);
    std::cout << "all_add(5) 実行後 (全区間に 5 加算):" << std::endl;
    std::cout << "values(): ";
    for (int x : bit.values()) std::cout << x << " ";
    std::cout << " (期待値: 6 17 18 19 10)" << std::endl;

    std::cout << "get(2): " << bit.get(2) << " (期待値: 18)" << std::endl;

    bit.clear();
    std::cout << "clear() 実行後:" << std::endl;
    std::cout << "get(2): " << bit.get(2) << " (期待値: 0)" << std::endl;

    std::cout << "--- test_RangeAddPointQueryBIT 終了 ---" << std::endl << std::endl;
}

// ランダムテスト
void random_test_RangeAddPointQueryBIT() {
    std::cout << "--- random_test_RangeAddPointQueryBIT 開始 ---" << std::endl;

    std::mt19937 mt(42);
    int n = 500;
    int queries = 5000;

    std::vector<long long> naive(n);
    std::uniform_int_distribution<long long> val_dist(-1000, 1000);
    for (int i = 0; i < n; ++i) {
        naive[i] = val_dist(mt);
    }

    RangeAddPointQueryBIT<long long> bit(naive);

    std::uniform_int_distribution<int> op_dist(0, 3);
    std::uniform_int_distribution<int> idx_dist(0, n - 1);

    for (int q = 0; q < queries; ++q) {
        int op = op_dist(mt);
        if (op == 0) {
            // add
            int l = idx_dist(mt);
            int r = idx_dist(mt);
            if (l > r) std::swap(l, r);
            long long x = val_dist(mt);
            for (int i = l; i < r; ++i) {
                naive[i] += x;
            }
            bit.add(l, r, x);
        } else if (op == 1) {
            // all_add
            long long x = val_dist(mt);
            for (int i = 0; i < n; ++i) {
                naive[i] += x;
            }
            bit.all_add(x);
        } else if (op == 2) {
            // get / operator[]
            int i = idx_dist(mt);
            assert(bit.get(i) == naive[i]);
            assert(bit[i] == naive[i]);
        } else {
            // values
            std::vector<long long> b_vals = bit.values();
            assert(b_vals == naive);
        }
    }

    std::cout << "OK" << std::endl;
    std::cout << "--- random_test_RangeAddPointQueryBIT 終了 ---" << std::endl << std::endl;
}

    test_RangeAddPointQueryBIT();
    random_test_RangeAddPointQueryBIT();
    return;
}
