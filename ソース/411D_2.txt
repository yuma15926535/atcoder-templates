struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

struct LazyBidirectionalString {

    // reverse で保持
    string left;

    // 通常順
    string right;

    // lazy reverse
    bool rev = false;

    // ===== lazy materialize =====

    mutable string cache;
    mutable bool dirty = true;

    //--------------------------------------------------------
    // constructor
    //--------------------------------------------------------

    //空文字で初期化
    LazyBidirectionalString(){}

    //初期化
    LazyBidirectionalString(const string& s){
        right = s;
        dirty = true;
    }

    //--------------------------------------------------------
    // size
    //--------------------------------------------------------
    int size() const {
        return left.size() + right.size();
    }

    //--------------------------------------------------------
    // empty
    //--------------------------------------------------------
    bool empty() const {
        return size() == 0;
    }

    //--------------------------------------------------------
    // reserve
    //--------------------------------------------------------
    void reserve(size_t n){
        left.reserve(n);
        right.reserve(n);
    }

    //--------------------------------------------------------
    // prepend(char)
    //--------------------------------------------------------
    void prepend(char c){

        dirty = true;

        if(!rev) left.push_back(c);
        else right.push_back(c);
    }

    //--------------------------------------------------------
    // prepend(string)
    //--------------------------------------------------------
    void prepend(const string& s){

        dirty = true;

        if(!rev){
            for(auto it = s.rbegin(); it != s.rend(); ++it)
                left.push_back(*it);
        }
        else{
            for(auto it = s.rbegin(); it != s.rend(); ++it)
                right.push_back(*it);
        }
    }

    //--------------------------------------------------------
    // append(char)
    //--------------------------------------------------------
    void append(char c){

        dirty = true;

        if(!rev) right.push_back(c);
        else left.push_back(c);
    }

    //--------------------------------------------------------
    // append(string)
    //--------------------------------------------------------
    void append(const string& s){

        dirty = true;

        if(!rev){
            right += s;
        }
        else{
            left += s;
        }
    }

    //--------------------------------------------------------
    // reverse
    //--------------------------------------------------------
    void reverse(){
        rev ^= 1;
        dirty = true;
    }

    //--------------------------------------------------------
    // cache rebuild
    //--------------------------------------------------------
    void build_cache() const {

        if(!dirty) return;

        cache.clear();
        cache.reserve(size());

        if(!rev){

            for(auto it = left.rbegin(); it != left.rend(); ++it)
                cache.push_back(*it);

            cache += right;

        }
        else{

            for(auto it = right.rbegin(); it != right.rend(); ++it)
                cache.push_back(*it);

            cache += left;
        }

        dirty = false;
    }

    //--------------------------------------------------------
    // index access
    //--------------------------------------------------------
    char operator[](size_t i) const {

        size_t L = left.size();
        size_t R = right.size();

        if(!rev){

            if(i < L) return left[L - 1 - i];
            else return right[i - L];

        }else{

            if(i < R) return right[R - 1 - i];
            else return left[i - R];
        }
    }

    //--------------------------------------------------------
    // front
    //--------------------------------------------------------
    char front() const {
        return (*this)[0];
    }

    //--------------------------------------------------------
    // back
    //--------------------------------------------------------
    char back() const {
        return (*this)[size() - 1];
    }

    //--------------------------------------------------------
    // substr
    //--------------------------------------------------------
    string substr(size_t pos, size_t len) const {

        size_t L = left.size();
        size_t R = right.size();

        string res;
        res.reserve(len);

        if(!rev){

            // real = reverse(left) + right

            if(pos < L){

                size_t take = min(len, L - pos);

                for(size_t i = 0; i < take; i++)
                    res.push_back(left[L - 1 - (pos + i)]);

                pos += take;
                len -= take;
            }

            if(len){
                res.append(right, pos - L, len);
            }

        }else{

            // real = reverse(right) + left

            if(pos < R){

                size_t take = min(len, R - pos);

                for(size_t i = 0; i < take; i++)
                    res.push_back(right[R - 1 - (pos + i)]);

                pos += take;
                len -= take;
            }

            if(len){
                res.append(left, pos - R, len);
            }
        }

        return res;
    }

    //--------------------------------------------------------
    // str
    //--------------------------------------------------------
    const string& str() const {

        build_cache();
        return cache;
    }

    //--------------------------------------------------------
    // clear
    //--------------------------------------------------------
    void clear(){
        left.clear();
        right.clear();
        rev = false;

        cache.clear();
        dirty = true;
    }
};

struct Node {
    int parent;
    string add;
};

void solv(){
    int N,Q;
    cin>>N>>Q;
    //参照と追加文字列
    vector<Node> nodes;
    nodes.push_back({-1, ""});
    //各pcの参照node
    vector<int> pc(N + 1, 0);
    //サーバの参照node
    int server = 0;
    while(Q--){
        int q,p;
        cin>>q>>p;
        if(q==1){
            pc[p]=server;
        }else if(q==2){
            string s;
            cin>>s;
            nodes.push_back({pc[p],s});
            pc[p]=sizei(nodes)-1;
        }else{
            server=pc[p];
        }
    }
    LazyBidirectionalString ans;
    int cur = server;
    while (cur != 0) {
        ans.prepend(nodes[cur].add);
        cur = nodes[cur].parent;
    }
    cout << ans.str() << "\n";
    return;
}

    //cin>>T;
    while(T--)solv();
