                adj[v].push_back(u);

struct S {
    ll min_val;
    int cnt;
};

/*
=====================================================
★ 更新情報
=====================================================
*/

using F = ll;

/*
=====================================================
★ 演算定義
=====================================================
*/

// ノード結合
S combine(S a, S b) {

    if(a.min_val < b.min_val)
        return a;

    if(a.min_val > b.min_val)
        return b;

    return {
        a.min_val,
        a.cnt + b.cnt
    };
}

// 更新適用
S apply(F f, S x) {

    if (x.min_val != LINF)
        x.min_val += f;

    return x;
}

// 更新の合成
F compose(F f, F g) {
    return f + g;
}

S idS() {
    return {LINF, 0};
}

F idF() {
    return 0;
}

/*
=====================================================
Lazy SegTree 本体
=====================================================
*/

struct LazySegTree {

    int n;

    vector<S> seg;

    vector<F> lazy;

    LazySegTree(const vector<ll>& v) {

        int sz = v.size();

        n = 1;

        while (n < sz)
            n <<= 1;

        seg.assign(2 * n, idS());

        lazy.assign(2 * n, idF());

        for (int i = 0; i < sz; i++)
            seg[n + i] = {v[i], 1};

        // n+i >= sz の要素は idS() のまま

        for (int i = n - 1; i > 0; i--)
            seg[i] = combine(seg[i << 1], seg[i << 1 | 1]);
    }

    void push(int k) {

        if (lazy[k] == idF())
            return;

        apply_node(k << 1, lazy[k]);

        apply_node(k << 1 | 1, lazy[k]);

        lazy[k] = idF();
    }

    void apply_node(int k, F f) {

        seg[k] = apply(f, seg[k]);

        if (k < n)
            lazy[k] = compose(f, lazy[k]);
    }

    void range_apply(int l, int r, F f, int k, int a, int b) {

        if (r <= a || b <= l)
            return;

        if (l <= a && b <= r) {

            apply_node(k, f);

            return;
        }

        push(k);

        int mid = (a + b) / 2;

        range_apply(l, r, f, k << 1, a, mid);

        range_apply(l, r, f, k << 1 | 1, mid, b);

        seg[k] = combine(seg[k << 1], seg[k << 1 | 1]);
    }

    void range_apply(int l, int r, F f) {
        range_apply(l, r, f, 1, 0, n);
    }

    S range(int l, int r, int k, int a, int b) {

        if (r <= a || b <= l)
            return idS();

        if (l <= a && b <= r)
            return seg[k];

        push(k);

        int mid = (a + b) / 2;

        return combine(
            range(l, r, k << 1, a, mid),
            range(l, r, k << 1 | 1, mid, b)
        );
    }

    S range(int l, int r) {
        return range(l, r, 1, 0, n);
    }

    ll range_min(int l, int r) {
        return range(l, r).min_val;
    }

    int range_count(int l, int r) {
        return range(l, r).cnt;
    }
};

void random_test() {

    mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());

    rep(_,0,10000){

        // ========================
        // サイズ
        // ========================

        int N = rng()%20 + 1;

        vector<ll> A(N);

        rep(i,0,N)
            A[i] = rng()%21 - 10;

        // ========================
        // segtree 構築
        // ========================

        LazySegTree seg(A);

        // ========================
        // クエリ
        // ========================

        rep(q,0,1000){

            int type = rng()%2;

            // ====================================
            // 区間加算
            // ====================================

            if(type == 0){

                int l = rng()%N;
                int r = rng()%N;

                if(l > r)
                    swap(l,r);

                r++;

                ll x = rng()%21 - 10;

                // 愚直
                rep(i,l,r)
                    A[i] += x;

                // segtree
                seg.range_apply(l,r,x);
            }

            // ====================================
            // 区間取得
            // ====================================

            else{

                int l = rng()%N;
                int r = rng()%N;

                if(l > r)
                    swap(l,r);

                r++;

                // ------------------------
                // 愚直
                // ------------------------

                ll mn = LINF;
                int cnt = 0;

                rep(i,l,r){

                    if(A[i] < mn){

                        mn = A[i];
                        cnt = 1;
                    }

                    else if(A[i] == mn){

                        cnt++;
                    }
                }

                // ------------------------
                // segtree
                // ------------------------

                auto res = seg.range(l,r);

                // ------------------------
                // check
                // ------------------------

                if(res.min_val != mn || res.cnt != cnt){

                    cout << "ERROR\n";

                    cout << "N = " << N << endl;

                    cout << "A : ";

                    for(auto x : A)
                        cout << x << " ";

                    cout << endl;

                    cout << "l r = " << l << " " << r << endl;

                    cout << "expected : ";
                    cout << mn << " " << cnt << endl;

                    cout << "actual   : ";
                    cout << res.min_val << " " << res.cnt << endl;

                    exit(0);
                }
            }
        }
    }

    cout << "OK\n";
}
