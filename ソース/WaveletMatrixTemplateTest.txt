    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }

struct BitVector{
    int N;

    // 64bitごとのビット列
    vector<unsigned long long> bit;

    // sum[i] = block[0..i-1] の1の個数
    vector<int> sum;

    BitVector() : N(0) {}

    BitVector(const vector<int>& B){
        build(B);
    }

    void build(const vector<int>& B){
        N = (int)B.size();

        int M = (N + 63) >> 6;

        bit.assign(M, 0);
        sum.assign(M + 1, 0);

        for(int i=0;i<N;i++){
            if(B[i]){
                bit[i >> 6] |= (1ULL << (i & 63));
            }
        }

        for(int i=0;i<M;i++){
            sum[i + 1] = sum[i] + __builtin_popcountll(bit[i]);
        }
    }

    // B[i]
    int access(int i) const{
        return (bit[i >> 6] >> (i & 63)) & 1ULL;
    }

    // [0,r) の1の個数
    int rank1(int r) const{
        if(r <= 0) return 0;

        int block = r >> 6;
        int offset = r & 63;

        int res = sum[block];

        if(offset){
            unsigned long long mask = (1ULL << offset) - 1;
            res += __builtin_popcountll(bit[block] & mask);
        }

        return res;
    }

    // [0,r) の0の個数
    int rank0(int r) const{
        return r - rank1(r);
    }
};

struct WaveletMatrix{
    using ull = unsigned long long;

    int N;
    static constexpr int LOG = 64;

    vector<ll> A;      // 元の配列
    vector<ull> U;     // Wavelet Matrix内部表現
    vector<BitVector> bv;
    vector<int> mid;
    // ランダムテスト用
    vector<vector<ull>> level_data;

    // long long → unsigned long long
    static ull encode(ll x){
        return (ull)x ^ (1ULL << 63);
    }

    // unsigned long long → long long
    static ll decode(ull x){
        return (ll)(x ^ (1ULL << 63));
    }

    WaveletMatrix() : N(0) {}

    WaveletMatrix(const vector<ll>& v){
        build(v);
    }

    void build(const vector<ll>& v){
        A = v;
        N = (int)v.size();

        U.resize(N);

        for(int i=0;i<N;i++){
            U[i] = encode(v[i]);
        }

        bv.resize(LOG);
        mid.resize(LOG);

        // ランダムテスト用
        level_data.clear();
        level_data.resize(LOG + 1);

        vector<ull> cur = U;

        level_data[0] = cur;

        for(int level=0; level<LOG; level++){

            int bitpos = LOG - 1 - level;

            vector<int> bits(N);

            vector<ull> zero;
            vector<ull> one;

            zero.reserve(N);
            one.reserve(N);

            for(int i=0;i<N;i++){

                int b = (cur[i] >> bitpos) & 1ULL;

                bits[i] = b;

                if(b==0){
                    zero.push_back(cur[i]);
                }
                else{
                    one.push_back(cur[i]);
                }
            }

            bv[level].build(bits);

            mid[level] = (int)zero.size();

            cur.clear();

            for(auto x : zero){
                cur.push_back(x);
            }

            for(auto x : one){
                cur.push_back(x);
            }

            level_data[level+1] = cur;
        }
    }

    ll access(int i) const{

        ull x = 0;

        for(int level=0; level<LOG; level++){

            int bitpos = LOG - 1 - level;

            int b = bv[level].access(i);

            if(b){
                x |= (1ULL << bitpos);

                i = mid[level]
                    + bv[level].rank1(i);
            }
            else{
                i = bv[level].rank0(i);
            }
        }

        return decode(x);
    }

    int range_count_eq(int l,int r,ll x) const{

        ull u = encode(x);

        for(int level=0; level<LOG; level++){

            int bitpos = LOG - 1 - level;

            int b = (u >> bitpos) & 1ULL;

            if(b==0){
                l = bv[level].rank0(l);
                r = bv[level].rank0(r);
            }
            else{
                l = mid[level] + bv[level].rank1(l);
                r = mid[level] + bv[level].rank1(r);
            }
        }

        return r - l;
    }

    int range_count_lt(int l,int r,ll x) const{

        ull u = encode(x);

        int ans = 0;

        for(int level=0; level<LOG; level++){

            int bitpos = LOG - 1 - level;

            int b = (u >> bitpos) & 1ULL;

            if(b==0){

                l = bv[level].rank0(l);
                r = bv[level].rank0(r);
            }
            else{

                ans +=
                    bv[level].rank0(r)
                    - bv[level].rank0(l);

                l = mid[level]
                    + bv[level].rank1(l);

                r = mid[level]
                    + bv[level].rank1(r);
            }
        }

        return ans;
    }

    // x未満の個数
    int range_rank(int l,int r,ll x) const{
        return range_count_lt(l,r,x);
    }

    // x以下の個数
    int range_count_le(int l,int r,ll x) const{
        return range_count_lt(l,r,x)
             + range_count_eq(l,r,x);
    }

    // xより大きい個数
    int range_count_gt(int l,int r,ll x) const{
        return (r-l) - range_count_le(l,r,x);
    }

    // x以上の個数
    int range_count_ge(int l,int r,ll x) const{
        return (r-l) - range_count_lt(l,r,x);
    }

    ll range_kth(int l,int r,int k) const{

        assert(0 <= k && k < r-l);

        ull ans = 0;

        for(int level=0; level<LOG; level++){

            int bitpos = LOG - 1 - level;

            int zero =
                bv[level].rank0(r)
                - bv[level].rank0(l);

            if(k < zero){

                l = bv[level].rank0(l);
                r = bv[level].rank0(r);
            }
            else{

                ans |= (1ULL << bitpos);

                k -= zero;

                l = mid[level]
                    + bv[level].rank1(l);

                r = mid[level]
                    + bv[level].rank1(r);
            }
        }

        return decode(ans);
    }

    ll range_min(int l,int r) const{

        assert(l < r);

        return range_kth(l,r,0);
    }

    ll range_max(int l,int r) const{

        assert(l < r);

        return range_kth(l,r,r-l-1);
    }

    ll range_prev_eq(int l,int r,ll x) const{

        int cnt = range_count_le(l,r,x);

        if(cnt == 0){
            return LLONG_MIN; // or INF扱い（設計次第）
        }

        return range_kth(l,r,cnt - 1);
    }

    ll range_prev(int l,int r,ll x) const{

        if(x == LLONG_MIN){
            return LLONG_MIN;
        }

        return range_prev_eq(l,r,x - 1);
    }

    ll range_next_eq(int l,int r,ll x) const{

        int cnt = range_count_lt(l,r,x);
        int len = r - l;

        if(cnt == len){
            return LLONG_MAX; // 存在しない
        }

        return range_kth(l,r,cnt);
    }

    ll range_next(int l,int r,ll x) const{

        int le = range_count_le(l,r,x);
        int len = r - l;

        if(le == len){
            return LLONG_MAX;
        }

        return range_kth(l,r,le);
    }
};

void random_test_step0(){
    mt19937_64 rng(
        chrono::steady_clock::now().time_since_epoch().count()
    );

    for(int t=0;t<100000;t++){
        ll x = (ll)rng();

        auto u = WaveletMatrix::encode(x);
        auto y = WaveletMatrix::decode(u);

        assert(x == y);
    }

    for(int t=0;t<100000;t++){
        ll x = (ll)rng();
        ll y = (ll)rng();

        bool a = (x < y);
        bool b = (
            WaveletMatrix::encode(x)
            < WaveletMatrix::encode(y)
        );

        assert(a == b);
    }

    cout << "Step0 Random Test Passed" << endl;
}

void random_test_bitvector(){
    mt19937 rng(
        chrono::steady_clock::now().time_since_epoch().count()
    );

    for(int tc=0;tc<100000;tc++){
        int N = rng() % 100 + 1;

        vector<int> B(N);

        for(int i=0;i<N;i++){
            B[i] = rng() & 1;
        }

        BitVector bv(B);

        // access
        for(int i=0;i<N;i++){
            assert(bv.access(i) == B[i]);
        }

        // rank
        for(int r=0;r<=N;r++){
            int one = 0;

            for(int i=0;i<r;i++){
                one += B[i];
            }

            int zero = r - one;

            assert(bv.rank1(r) == one);
            assert(bv.rank0(r) == zero);
        }
    }

    cout << "Step1 Random Test Passed\n";
}

void random_test_step2(){

    mt19937_64 rng(
        chrono::steady_clock::now().time_since_epoch().count()
    );

    for(int tc=0; tc<10000; tc++){

        int N = rng()%50 + 1;

        vector<ll> A(N);

        for(int i=0;i<N;i++){
            A[i]=(ll)rng();
        }

        WaveletMatrix wm(A);

        vector<WaveletMatrix::ull> cur(N);

        for(int i=0;i<N;i++){
            cur[i]=WaveletMatrix::encode(A[i]);
        }

        for(int level=0; level<WaveletMatrix::LOG; level++){

            int bitpos=
                WaveletMatrix::LOG-1-level;

            vector<WaveletMatrix::ull> zero;
            vector<WaveletMatrix::ull> one;

            int zero_cnt=0;

            for(int i=0;i<N;i++){

                int b=(cur[i]>>bitpos)&1ULL;

                // BitVector確認
                assert(
                    wm.bv[level].access(i)
                    == b
                );

                if(b==0){
                    zero.push_back(cur[i]);
                    zero_cnt++;
                }
                else{
                    one.push_back(cur[i]);
                }
            }

            // mid確認
            assert(
                wm.mid[level]
                == zero_cnt
            );

            vector<WaveletMatrix::ull> nxt;

            for(auto x:zero){
                nxt.push_back(x);
            }

            for(auto x:one){
                nxt.push_back(x);
            }

            // 並び確認
            assert(
                nxt
                == wm.level_data[level+1]
            );

            cur=nxt;
        }
    }

    cout << "Step2 Random Test Passed\n";
}

void random_test_step4(){

    mt19937_64 rng(
        chrono::steady_clock::now().time_since_epoch().count()
    );

    for(int tc=0; tc<10000; tc++){

        int N = rng()%50 + 1;

        vector<ll> A(N);

        for(int i=0;i<N;i++){
            A[i] = (ll)rng();
        }

        WaveletMatrix wm(A);

        for(int q=0;q<100;q++){

            int l = rng()%(N+1);
            int r = rng()%(N+1);

            if(l>r) swap(l,r);

            ll x;

            // 半分は存在する値を使う
            if((rng()&1) && N>0){
                x = A[rng()%N];
            }
            else{
                x = (ll)rng();
            }

            int ans = 0;

            for(int i=l;i<r;i++){
                if(A[i]==x){
                    ans++;
                }
            }

            assert(
                wm.range_count_eq(l,r,x)
                == ans
            );
        }
    }

    cout << "Step4 Random Test Passed\n";
}
