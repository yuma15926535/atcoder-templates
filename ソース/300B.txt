int H,W;
vector<vector<char>> A,B;
int h,w;
vector<int> hindex,windex;
int geth(int i){
    return (i+h)%H;
}
int getw(int i){
    return (i+w)%W;
}

int main()
{
    cin>>H>>W;
    A=vector<vector<char>>(H,vector<char>(W));
    B=A;
    rep(i,0,H)rep(j,0,W)cin>>A[i][j];
    rep(i,0,H)rep(j,0,W)cin>>B[i][j];
    hindex.resize(H);
    windex.resize(W);
    rep(i,0,H)hindex[i]=i;
    rep(i,0,W)windex[i]=i;
    //gotowp用いたデバッグ
    // goto debug;
    for(h=0;h<H;h++)for(w=0;w<W;w++){
        bool b=true;
        rep(i,0,H)rep(j,0,W)if(B[i][j]!=A[hindex[geth((int)i)]][windex[getw((int)j)]]){
            b=false;
            i=H;
            j=W;
        }
        if(b){
            YES;
            return 0;
        }
    }
    NO;
    // debug:
