struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
    RectangleDiagonalQuery

    矩形
        L <= x <= R
        D <= y <= U

    と

        y = x + c
        y = -x + c

    の交点となる格子点情報を O(1) で取得する。

    返却値

        exist : 交点が存在するか
        first : x が最小の格子点
        last  : x が最大の格子点
        count : 格子点数
*/
struct RectangleDiagonalQuery {

    struct Result {

        bool exist = false;

        pair<ll,ll> first;
        pair<ll,ll> last;

        ll count = 0;
    };

    ll L,R,D,U;

    RectangleDiagonalQuery(ll L,ll R,ll D,ll U)
        : L(L),R(R),D(D),U(U){}

    /*
        y = x + c
    */
    Result plus(ll c) const{

        ll l=max(L,D-c);
        ll r=min(R,U-c);

        if(l>r) return {};

        Result res;

        res.exist=true;
        res.count=r-l+1;

        res.first={l,l+c};
        res.last ={r,r+c};

        return res;
    }

    /*
        y = -x + c
    */
    Result minus(ll c) const{

        ll l=max(L,c-U);
        ll r=min(R,c-D);

        if(l>r) return {};

        Result res;

        res.exist=true;
        res.count=r-l+1;

        res.first={l,c-l};
        res.last ={r,c-r};

        return res;
    }
};

void random_test_RectangleDiagonalQuery(){

    mt19937 rng(0);

    uniform_int_distribution<int> dist(-5,5);

    for(int tc=0;tc<100000;tc++){

        ll x1=dist(rng);
        ll x2=dist(rng);
        ll y1=dist(rng);
        ll y2=dist(rng);

        ll L=min(x1,x2);
        ll R=max(x1,x2);
        ll D=min(y1,y2);
        ll U=max(y1,y2);

        RectangleDiagonalQuery solver(L,R,D,U);

        for(ll c=-10;c<=10;c++){

            //=========================
            // y=x+c
            //=========================

            vector<pair<ll,ll>> brute;

            for(ll x=L;x<=R;x++){
                for(ll y=D;y<=U;y++){
                    if(y==x+c) brute.push_back({x,y});
                }
            }

            auto res=solver.plus(c);

            assert(res.exist==(!brute.empty()));
            assert(res.count==(ll)brute.size());

            if(!brute.empty()){
                assert(res.first==brute.front());
                assert(res.last==brute.back());
            }

            //=========================
            // y=-x+c
            //=========================

            brute.clear();

            for(ll x=L;x<=R;x++){
                for(ll y=D;y<=U;y++){
                    if(y==-x+c) brute.push_back({x,y});
                }
            }

            res=solver.minus(c);

            assert(res.exist==(!brute.empty()));
            assert(res.count==(ll)brute.size());

            if(!brute.empty()){
                assert(res.first==brute.front());
                assert(res.last==brute.back());
            }
        }
    }

    cout<<"RectangleDiagonalQuery Random Test Passed"<<endl;
}

void test_RectangleDiagonalQuery(){

    cout << "===== RectangleDiagonalQuery 動作確認 =====\n\n";

    RectangleDiagonalQuery rect(-2, 4, -1, 3);

    cout << "矩形\n";
    cout << "  -2 <= x <= 4\n";
    cout << "  -1 <= y <= 3\n\n";

    auto print = [](const string& name,
                    ll c,
                    const RectangleDiagonalQuery::Result& res){

        cout << name << " (切片 = " << c << ")\n";

        if(!res.exist){
            cout << "  共通部分なし\n\n";
            return;
        }

        cout << "  格子点数 : " << res.count << "\n";

        cout << "  始点     : ("
             << res.first.first << ", "
             << res.first.second << ")\n";

        cout << "  終点     : ("
             << res.last.first << ", "
             << res.last.second << ")\n\n";
    };

    print("y = x + c", 1, rect.plus(1));
    print("y = x + c", 5, rect.plus(5));

    print("y = -x + c", 2, rect.minus(2));
    print("y = -x + c", 10, rect.minus(10));

    cout << "===== 動作確認終了 =====\n";
}

    return;
