struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

struct RollbackUF {
    int n, cc;//cc:連結成分数
    vector<int> parent, sz;
    stack<pair<int,int>> history;

    RollbackUF(int n): n(n), cc(n), parent(n), sz(n,1) {
        iota(parent.begin(), parent.end(), 0);
    }

    int find(int x) {
        while(x != parent[x]) x = parent[x];
        return x;
    }

    bool unite(int a, int b) {
        a = find(a);
        b = find(b);

        if(a == b) {
            history.emplace(-1, -1);
            history.emplace(-1, -1);
            return false;
        }

        if(sz[a] < sz[b]) swap(a,b);

        // 変更前保存
        history.emplace(a, sz[a]);
        history.emplace(b, sz[b]);

        parent[b] = a;
        sz[a] += sz[b];
        cc--;

        return true;
    }

    void undo() {
        auto [b, sz_b] = history.top(); history.pop();
        auto [a, sz_a] = history.top(); history.pop();

        if(a == -1) return;

        parent[b] = b;
        sz[b] = sz_b;
        sz[a] = sz_a;
        cc++;
    }

    int snapshot() {
        return (int)history.size();
    }

    void rollback(int snap) {
        while((int)history.size() > snap) {
            undo();
        }
    }

    bool same(int a, int b) {
        return find(a) == find(b);
    }
};

struct OfflineDynamicConnectivity {
    int n, Q;
    vector<vector<pair<int,int>>> seg;
    RollbackUF uf;
    vector<bool> ans;

    OfflineDynamicConnectivity(int n, int Q)
        : n(n), Q(Q), seg(4*Q), uf(n), ans(Q) {}

    void add_edge(int l, int r, pair<int,int> e, int node, int nl, int nr) {
        if(r <= nl || nr <= l) return;
        if(l <= nl && nr <= r) {
            seg[node].push_back(e);
            return;
        }
        int mid = (nl + nr) / 2;
        add_edge(l,r,e,node*2,nl,mid);
        add_edge(l,r,e,node*2+1,mid,nr);
    }

    void dfs(int node, int l, int r,
             const vector<tuple<int,int,int>>& queries) {

        int snap = uf.snapshot();

        for(auto [u,v]: seg[node]) uf.unite(u,v);

        if(r - l == 1) {
            auto [type,u,v] = queries[l];
            if(type == 2) {
                ans[l] = uf.same(u,v);
            }
        } else {
            int mid = (l + r) / 2;
            dfs(node*2,l,mid,queries);
            dfs(node*2+1,mid,r,queries);
        }

        uf.rollback(snap);
    }
};

vector<bool> solve_dynamic_connectivity(
    int n,
    vector<tuple<int,int,int>> queries
) {
    int Q = queries.size();

    OfflineDynamicConnectivity solver(n,Q);

    map<pair<int,int>, vector<int>> start;

    for(int i=0;i<Q;i++) {
        auto [type,u,v] = queries[i];
        if(u > v) swap(u,v);

        if(type == 0) {
            start[{u,v}].push_back(i);
        }
        else if(type == 1) {
            if(!start[{u,v}].empty()) {
                int s = start[{u,v}].back();
                start[{u,v}].pop_back();
                solver.add_edge(s, i, {u,v}, 1, 0, Q);
            }
        }
    }

    // 残ってる辺（最後まで生存）
    for(auto& [e, vec] : start) {
        for(int s : vec) {
            solver.add_edge(s, Q, e, 1, 0, Q);
        }
    }

    solver.dfs(1,0,Q,queries);

    return solver.ans;
}

void demo_dynamic_connectivity() {
    cout << "=== 動的連結性 デモ開始 ===\n";

    int n = 5;

    /*
    クエリ形式:
    0 u v : 辺追加
    1 u v : 辺削除
    2 u v : 連結判定
    */

    vector<tuple<int,int,int>> queries = {
        {2, 0, 1}, // 初期状態
        {0, 0, 1}, // 0-1追加
        {2, 0, 1},
        {0, 1, 2}, // 1-2追加
        {2, 0, 2},
        {1, 0, 1}, // 0-1削除
        {2, 0, 2},
        {0, 2, 3}, // 2-3追加
        {2, 0, 3},
        {0, 3, 4}, // 3-4追加
        {2, 0, 4},
        {1, 1, 2}, // 1-2削除
        {2, 0, 4},
    };

    auto ans = solve_dynamic_connectivity(n, queries);

    cout << "クエリ実行結果:\n";

    for(int i=0;i<(int)queries.size();i++) {
        auto [t,u,v] = queries[i];

        if(t == 0) {
            cout << "辺追加: (" << u << ", " << v << ")\n";
        }
        else if(t == 1) {
            cout << "辺削除: (" << u << ", " << v << ")\n";
        }
        else {
            cout << "連結判定: (" << u << ", " << v << ") → ";
            cout << (ans[i] ? "連結" : "非連結") << "\n";
        }
    }

    cout << "=== デモ終了 ===\n";
}
