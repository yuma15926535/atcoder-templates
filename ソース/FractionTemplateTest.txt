struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
============================================================
 Fraction（有理数構造体）
------------------------------------------------------------
 long long 分子・分母で分数を管理（常に既約・分母>0）

 対応機能
  ・四則演算 (+ - * /)
  ・比較演算 (< <= > >= == !=)
  ・floor / ceil / sign
  ・abs / reciprocal
  ・小数出力
  ・複合演算子 (+= -= *= /=)

 overflow対策
  ・cross reduction
  ・比較は __int128

 計算量
  ・四則演算 : O(log N)   (gcd)
  ・比較     : O(1)
============================================================
*/
struct Fraction {
    long long num;
    long long den;

    // 正規化
    void normalize() {
        if (den < 0) {
            num = -num;
            den = -den;
        }
        long long g = gcd(llabs(num), den);
        num /= g;
        den /= g;
    }

    // 整数コンストラクタ
    Fraction(long long n = 0) {
        num = n;
        den = 1;
    }

    // 分数コンストラクタ
    Fraction(long long n, long long d) {
        assert(d != 0);
        num = n;
        den = d;
        normalize();
    }

    // + 
    Fraction operator+(const Fraction& r) const {
        long long g = gcd(den, r.den);
        long long l = den / g;
        long long num2 = num * (r.den / g) + r.num * l;
        long long den2 = l * r.den;
        return Fraction(num2, den2);
    }

    // -
    Fraction operator-(const Fraction& r) const {
        long long g = gcd(den, r.den);
        long long l = den / g;
        long long num2 = num * (r.den / g) - r.num * l;
        long long den2 = l * r.den;
        return Fraction(num2, den2);
    }

    // *
    Fraction operator*(const Fraction& r) const {
        long long g1 = gcd(llabs(num), r.den);
        long long g2 = gcd(llabs(r.num), den);

        long long a = num / g1;
        long long b = den / g2;
        long long c = r.num / g2;
        long long d = r.den / g1;

        return Fraction(a * c, b * d);
    }

    // /
    Fraction operator/(const Fraction& r) const {
        assert(r.num != 0);

        long long g1 = gcd(llabs(num), llabs(r.num));
        long long g2 = gcd(den, r.den);

        long long a = num / g1;
        long long b = den / g2;
        long long c = r.num / g1;
        long long d = r.den / g2;

        return Fraction(a * d, b * c);
    }

    // 比較
    bool operator<(const Fraction& r) const {
        return (__int128)num * r.den < (__int128)r.num * den;
    }

    bool operator>(const Fraction& r) const { return r < *this; }
    bool operator<=(const Fraction& r) const { return !(r < *this); }
    bool operator>=(const Fraction& r) const { return !(*this < r); }
    bool operator==(const Fraction& r) const { return num == r.num && den == r.den; }
    bool operator!=(const Fraction& r) const { return !(*this == r); }

    // sign
    int sign() const {
        if (num > 0) return 1;
        if (num < 0) return -1;
        return 0;
    }

    // floor
    long long floor() const {
        if (num >= 0) return num / den;
        return -((-num + den - 1) / den);
    }

    // ceil
    long long ceil() const {
        if (num >= 0) return (num + den - 1) / den;
        return -((-num) / den);
    }

    // 小数出力
    long double to_ld() const {
        return (long double)num / den;
    }

    string to_decimal(int digits) const {
        stringstream ss;
        ss << fixed << setprecision(digits) << to_ld();
        return ss.str();
    }

    // 絶対値
    Fraction abs() const {
        return Fraction(llabs(num), den);
    }

    // 逆数
    Fraction reciprocal() const {
        assert(num != 0);
        return Fraction(den, num);
    }

    // 複合演算子
    Fraction& operator+=(const Fraction& r) {
        *this = *this + r;
        return *this;
    }

    Fraction& operator-=(const Fraction& r) {
        *this = *this - r;
        return *this;
    }

    Fraction& operator*=(const Fraction& r) {
        *this = *this * r;
        return *this;
    }

    Fraction& operator/=(const Fraction& r) {
        *this = *this / r;
        return *this;
    }

    // デバッグ出力
    friend ostream& operator<<(ostream& os, const Fraction& f) {
        os << f.num << "/" << f.den;
        return os;
    }
};

void solv(){
    cout << "===== Fraction テンプレ 動作確認 =====\n\n";

    cout << "---- コンストラクタと正規化 ----\n";

    Fraction a(3,4);
    Fraction b(6,8);
    Fraction c(3,-4);
    Fraction d(-6,-8);
    Fraction e(5);

    cout << "3/4 -> " << a << "\n";
    cout << "6/8 (約分) -> " << b << "\n";
    cout << "3/-4 (符号調整) -> " << c << "\n";
    cout << "-6/-8 (正に統一) -> " << d << "\n";
    cout << "整数 5 -> " << e << "\n\n";


    cout << "---- 四則演算 ----\n";

    Fraction f(1,3);
    Fraction g(1,6);

    cout << "1/3 + 1/6 = " << (f + g) << " (期待 1/2)\n";
    cout << "1/3 - 1/6 = " << (f - g) << " (期待 1/6)\n";
    cout << "1/3 * 1/6 = " << (f * g) << " (期待 1/18)\n";
    cout << "1/3 / 1/6 = " << (f / g) << " (期待 2/1)\n\n";


    cout << "---- 比較 ----\n";

    Fraction h(2,3);
    Fraction i(3,4);

    cout << "2/3 < 3/4 : " << (h < i) << " (期待 1)\n";
    cout << "2/3 > 3/4 : " << (h > i) << " (期待 0)\n";
    cout << "2/3 == 4/6 : " << (h == Fraction(4,6)) << " (期待 1)\n\n";


    cout << "---- sign ----\n";

    cout << "sign(3/5) = " << Fraction(3,5).sign() << " (期待 1)\n";
    cout << "sign(-3/5) = " << Fraction(-3,5).sign() << " (期待 -1)\n";
    cout << "sign(0) = " << Fraction(0).sign() << " (期待 0)\n\n";


    cout << "---- floor / ceil ----\n";

    Fraction j(7,3);
    Fraction k(-7,3);

    cout << "floor(7/3) = " << j.floor() << " (期待 2)\n";
    cout << "ceil(7/3) = " << j.ceil() << " (期待 3)\n";

    cout << "floor(-7/3) = " << k.floor() << " (期待 -3)\n";
    cout << "ceil(-7/3) = " << k.ceil() << " (期待 -2)\n\n";


    cout << "---- abs ----\n";

    Fraction l(-5,7);

    cout << "abs(-5/7) = " << l.abs() << " (期待 5/7)\n\n";


    cout << "---- reciprocal ----\n";

    Fraction m(3,5);

    cout << "reciprocal(3/5) = " << m.reciprocal() << " (期待 5/3)\n\n";


    cout << "---- 複合演算子 ----\n";

    Fraction n(1,4);

    n += Fraction(1,4);
    cout << "1/4 += 1/4 -> " << n << " (期待 1/2)\n";

    n *= Fraction(2);
    cout << "(1/2) *= 2 -> " << n << " (期待 1/1)\n";

    n -= Fraction(1,3);
    cout << "1 - 1/3 -> " << n << " (期待 2/3)\n";

    n /= Fraction(2);
    cout << "(2/3) / 2 -> " << n << " (期待 1/3)\n\n";


    cout << "---- 小数出力 ----\n";

    Fraction p(1,3);

    cout << "1/3 (小数10桁) = " << p.to_decimal(10) << "\n\n";


    cout << "---- 大きめの数（overflow回避確認） ----\n";

    Fraction q(1000000000,3);
    Fraction r(2000000000,7);

    cout << "1000000000/3 + 2000000000/7 = " << (q + r) << "\n";
    cout << "1000000000/3 * 2000000000/7 = " << (q * r) << "\n\n";


    cout << "===== 動作確認終了 =====\n";
    return;
}
