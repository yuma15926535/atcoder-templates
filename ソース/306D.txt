#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    int N;
    cin>>N;
    //状態が高々2通りしかないのでDP
    //解毒は毒有り無しにかかわらず食べられるので更新に注意する
    vector<vector<ll>> dp(2,vector<ll>(N,0));
    bool b=true;
    rep(i,0,N){
        ll x,y;
        cin>>x>>y;
        if(i==0){
            if(x){
                if(y>0){
                    dp[x][i]=y;
                    b=false;
                }
            }else dp[x][i]=max(y,0LL);
        }else{
            dp[0][i]=dp[0][i-1];
            dp[1][i]=dp[1][i-1];
            if(x){
                if(b){
                    if(y>0){
                        dp[x][i]=dp[0][i-1]+y;
                        b=false;
                    }
                }else dp[x][i]=max(dp[x][i-1],dp[0][i-1]+y);
            }else{
                dp[x][i]=max(dp[x][i-1],dp[x][i-1]+y);
                if(!b)dp[x][i]=max(dp[x][i],dp[1][i-1]+y);
            }
        }
    }
    ll ans=dp[0][N-1];
    if(!b)ans=max(ans,dp[1][N-1]);
    cout<<ans<<"\n";
    return 0;
}
