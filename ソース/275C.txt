// ================================
// 90度回転（任意中心・左右統合）
// ================================
// left = true : 左90度（反時計）
// left = false: 右90度（時計）
pair<ll,ll> rot90(pair<ll,ll> p, pair<ll,ll> c, bool left){
    ll x = p.first  - c.first;
    ll y = p.second - c.second;

    ll nx, ny;
    if(left){
        nx = -y;
        ny =  x;
    }else{
        nx =  y;
        ny = -x;
    }
    return {nx + c.first, ny + c.second};
}

// ================================
// グリッド範囲判定（動的サイズ対応）
// ================================
struct GridRange {
    int H, W;

    GridRange(int h, int w) : H(h), W(w) {}

    GridRange(const vector<string>& g){
        H = g.size();
        W = (H ? g[0].size() : 0);
    }

    template<class T>
    GridRange(const vector<vector<T>>& g){
        H = g.size();
        W = (H ? g[0].size() : 0);
    }

    bool in(int x, int y) const {
        return (0 <= x && x < H && 0 <= y && y < W);
    }

    bool in(pair<int,int> p) const {
        return in(p.first, p.second);
    }
};

// ================================
// トーラスグリッド（上下左右が接続）
// ================================
struct TorusGrid {
    int H, W;

    TorusGrid(int h, int w) : H(h), W(w) {}

    TorusGrid(const vector<string>& g){
        H = g.size();
        W = (H ? g[0].size() : 0);
    }

    template<class T>
    TorusGrid(const vector<vector<T>>& g){
        H = g.size();
        W = (H ? g[0].size() : 0);
    }

    pair<int,int> norm(int x, int y) const {
        x = (x % H + H) % H;
        y = (y % W + W) % W;
        return {x, y};
    }
};

// ================================
// マンハッタン距離制限
// ================================
struct Manhattan {
    int cx, cy, D;

    Manhattan(int x, int y, int d) : cx(x), cy(y), D(d) {}

    bool in(int x, int y) const {
        return abs(x - cx) + abs(y - cy) <= D;
    }

    bool in(pair<int,int> p) const {
        return in(p.first, p.second);
    }
};

// ================================
// チェビシェフ距離制限
// ================================
struct Chebyshev {
    int cx, cy, D;

    Chebyshev(int x, int y, int d) : cx(x), cy(y), D(d) {}

    bool in(int x, int y) const {
        return max(abs(x - cx), abs(y - cy)) <= D;
    }

    bool in(pair<int,int> p) const {
        return in(p.first, p.second);
    }
};

// ================================
// 円（ユークリッド距離）制限
// ================================
struct Circle {
    ll cx, cy, R2;

    Circle(ll x, ll y, ll r) : cx(x), cy(y), R2(r*r) {}

    bool in(ll x, ll y) const {
        ll dx = x - cx;
        ll dy = y - cy;
        return dx*dx + dy*dy <= R2;
    }

    bool in(pair<ll,ll> p) const {
        return in(p.first, p.second);
    }
};

void solv(){
    vector<string>S(9);
