//エラストテネスの篩
//isPrime:素数のみtrue
//戻り値は素数を列挙したvector
//引数num:列挙の最大値
//num=400000000で2sを少し超える
//num=1000000000がギリギリ
vector<bool> isPrime;
vector<ll> hurui(ll num){
    isPrime.resize(num+1,true);
    isPrime[0]=false;
    isPrime[1]=false;
    if(num>=2)for(ll i=2;i*i<=num;i++)if(isPrime[i])for(ll j=2;i*j<=num;j++)isPrime[i*j]=false;
    vector<ll> ret;
    for(ll i=2;i<=num;i++)if(isPrime[i])ret.push_back(i);
    return ret;
}

int main()
{
    ll N;
    cin>>N;
    ll ans=0;
    auto v=hurui(round(sqrt(N)));
    for(ll a:v){
        if(a*a*a*a*a*a*a*a>N)break;
        ans++;
    }
    for(ll a:v){
        if(a*a*a*a>N)break;
        for(ll b:v){
            if(b<=a)continue;
            if(a*a*b*b<=N)ans++;
            else break;
        }
    }
    cout<<ans<<"\n";
    return 0;
}
