struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
============================================================
 LazyBidirectionalString
------------------------------------------------------------
 前後から高速に文字列を構築できる構造体
 + lazy reverse
 + lazy materialize（遅延文字列構築）

 内部構造
   left  : reverse で保持
   right : 通常順で保持

 rev == false
   実際の文字列 = reverse(left) + right

 rev == true
   実際の文字列 = reverse(right) + left

------------------------------------------------------------
 Lazy Materialize
------------------------------------------------------------
 実体文字列を cache に保持する。

 更新操作
   dirty = true

 str() / substr() / operator[] など
   dirty のときだけ rebuild

------------------------------------------------------------
 計算量
------------------------------------------------------------
 prepend/append(char)     O(1)
 prepend/append(string)   O(|s|)
 reverse()                O(1)

 operator[]               O(1)
 front/back               O(1)

 substr                   O(len)

 str()
   初回 O(N)
   以降 O(1)

------------------------------------------------------------
 用途
------------------------------------------------------------
 ・前後から文字列構築
 ・reverse が多い問題
 ・文字列クエリ問題
 ・deque + reverse の代替
============================================================
*/
struct LazyBidirectionalString {

    // reverse で保持
    string left;

    // 通常順
    string right;

    // lazy reverse
    bool rev = false;

    // ===== lazy materialize =====

    mutable string cache;
    mutable bool dirty = true;

    //--------------------------------------------------------
    // constructor
    //--------------------------------------------------------

    //空文字で初期化
    LazyBidirectionalString(){}

    //初期化
    LazyBidirectionalString(const string& s){
        right = s;
        dirty = true;
    }

    //--------------------------------------------------------
    // size
    //--------------------------------------------------------
    int size() const {
        return left.size() + right.size();
    }

    //--------------------------------------------------------
    // empty
    //--------------------------------------------------------
    bool empty() const {
        return size() == 0;
    }

    //--------------------------------------------------------
    // reserve
    //--------------------------------------------------------
    void reserve(size_t n){
        left.reserve(n);
        right.reserve(n);
    }

    //--------------------------------------------------------
    // prepend(char)
    //--------------------------------------------------------
    void prepend(char c){

        dirty = true;

        if(!rev) left.push_back(c);
        else right.push_back(c);
    }

    //--------------------------------------------------------
    // prepend(string)
    //--------------------------------------------------------
    void prepend(const string& s){

        dirty = true;

        if(!rev){
            for(auto it = s.rbegin(); it != s.rend(); ++it)
                left.push_back(*it);
        }
        else{
            for(auto it = s.rbegin(); it != s.rend(); ++it)
                right.push_back(*it);
        }
    }

    //--------------------------------------------------------
    // append(char)
    //--------------------------------------------------------
    void append(char c){

        dirty = true;

        if(!rev) right.push_back(c);
        else left.push_back(c);
    }

    //--------------------------------------------------------
    // append(string)
    //--------------------------------------------------------
    void append(const string& s){

        dirty = true;

        if(!rev){
            right += s;
        }
        else{
            left += s;
        }
    }

    //--------------------------------------------------------
    // reverse
    //--------------------------------------------------------
    void reverse(){
        rev ^= 1;
        dirty = true;
    }

    //--------------------------------------------------------
    // cache rebuild
    //--------------------------------------------------------
    void build_cache() const {

        if(!dirty) return;

        cache.clear();
        cache.reserve(size());

        if(!rev){

            for(auto it = left.rbegin(); it != left.rend(); ++it)
                cache.push_back(*it);

            cache += right;

        }
        else{

            for(auto it = right.rbegin(); it != right.rend(); ++it)
                cache.push_back(*it);

            cache += left;
        }

        dirty = false;
    }

    //--------------------------------------------------------
    // index access
    //--------------------------------------------------------
    char operator[](size_t i) const {

        size_t L = left.size();
        size_t R = right.size();

        if(!rev){

            if(i < L) return left[L - 1 - i];
            else return right[i - L];

        }else{

            if(i < R) return right[R - 1 - i];
            else return left[i - R];
        }
    }

    //--------------------------------------------------------
    // front
    //--------------------------------------------------------
    char front() const {
        return (*this)[0];
    }

    //--------------------------------------------------------
    // back
    //--------------------------------------------------------
    char back() const {
        return (*this)[size() - 1];
    }

    //--------------------------------------------------------
    // substr
    //--------------------------------------------------------
    string substr(size_t pos, size_t len) const {

        size_t L = left.size();
        size_t R = right.size();

        string res;
        res.reserve(len);

        if(!rev){

            // real = reverse(left) + right

            if(pos < L){

                size_t take = min(len, L - pos);

                for(size_t i = 0; i < take; i++)
                    res.push_back(left[L - 1 - (pos + i)]);

                pos += take;
                len -= take;
            }

            if(len){
                res.append(right, pos - L, len);
            }

        }else{

            // real = reverse(right) + left

            if(pos < R){

                size_t take = min(len, R - pos);

                for(size_t i = 0; i < take; i++)
                    res.push_back(right[R - 1 - (pos + i)]);

                pos += take;
                len -= take;
            }

            if(len){
                res.append(left, pos - R, len);
            }
        }

        return res;
    }

    //--------------------------------------------------------
    // str
    //--------------------------------------------------------
    const string& str() const {

        build_cache();
        return cache;
    }

    //--------------------------------------------------------
    // clear
    //--------------------------------------------------------
    void clear(){
        left.clear();
        right.clear();
        rev = false;

        cache.clear();
        dirty = true;
    }
};

// LazyBidirectionalString ランダムテスト

void random_test(){

    cout << "\n==============================\n";
    cout << " ランダムテスト開始\n";
    cout << "==============================\n";

    LazyBidirectionalString s;
    string t;

    mt19937 rng(0);

    const int TEST = 100000;

    for(int step = 1; step <= TEST; step++){

        int op = rng() % 6;

        //----------------------------------------------------
        // prepend char
        //----------------------------------------------------
        if(op == 0){

            char c = 'a' + rng()%26;

            s.prepend(c);
            t = c + t;

        }

        //----------------------------------------------------
        // append char
        //----------------------------------------------------
        if(op == 1){

            char c = 'a' + rng()%26;

            s.append(c);
            t += c;

        }

        //----------------------------------------------------
        // prepend string
        //----------------------------------------------------
        if(op == 2){

            int len = rng()%5;

            string x;
            for(int i=0;i<len;i++)
                x += char('a'+rng()%26);

            s.prepend(x);
            t = x + t;

        }

        //----------------------------------------------------
        // append string
        //----------------------------------------------------
        if(op == 3){

            int len = rng()%5;

            string x;
            for(int i=0;i<len;i++)
                x += char('a'+rng()%26);

            s.append(x);
            t += x;

        }

        //----------------------------------------------------
        // reverse
        //----------------------------------------------------
        if(op == 4){

            s.reverse();
            reverse(t.begin(), t.end());

        }

        //----------------------------------------------------
        // substr check
        //----------------------------------------------------
        if(op == 5 && !t.empty()){

            int pos = rng()%t.size();
            int len = rng()%(t.size()-pos);

            string a = s.substr(pos,len);
            string b = t.substr(pos,len);

            if(a != b){

                cout << "\nsubstr 不一致発生\n";
                cout << "pos=" << pos << " len=" << len << "\n";
                cout << "LazyBidirectionalString: " << a << "\n";
                cout << "std::string: " << b << "\n";

                exit(0);
            }
        }

        //----------------------------------------------------
        // 全体一致確認
        //----------------------------------------------------

        if(s.str() != t){

            cout << "\n文字列不一致発生\n";
            cout << "step=" << step << "\n";

            cout << "LazyBidirectionalString: " << s.str() << "\n";
            cout << "std::string: " << t << "\n";

            exit(0);
        }

        //----------------------------------------------------
        // index check
        //----------------------------------------------------

        for(int i=0;i<t.size();i++){

            if(s[i] != t[i]){

                cout << "\nindex 不一致発生\n";
                cout << "step=" << step << " index=" << i << "\n";

                cout << "LazyBidirectionalString: " << s[i] << "\n";
                cout << "std::string: " << t[i] << "\n";

                exit(0);
            }
        }
    }

    cout << "ランダムテスト " << TEST << " 回成功\n";
}
