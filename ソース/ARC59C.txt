ll power(ll base,ll exp){
    ll result=1;
    while(exp>0){
        if(exp&1)result*=base;
        base*=base;
        exp>>=1;
    }
    return result;
}

int main(){
    int N;
    cin>>N;
    vector<ll>A(N);
    rep(i,0,N)cin>>A[i];
    ll sum=0;
    rep(i,0,N)sum+=A[i];
    //繰り下がり、繰り上がり
    ll ave1=sum/N,ave2=(sum+N-1)/N;
    ll ans1=0,ans2=0;
    rep(i,0,N){
        ans1+=power(A[i]-ave1,2);
        ans2+=power(A[i]-ave2,2);
    }
    cout<<min(ans1,ans2)<<'\n';
    return 0;
}
