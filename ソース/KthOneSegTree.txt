struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
=====================================================
★ モノイド定義（k-th One SegTree）
=====================================================

0/1配列を管理

S : 区間内の1の個数
combine : 和
id : 0

=====================================================
*/

using S = int;

// ★区間結合
S combine(S a, S b) {
    return a + b;
}

// ★単位元
S id() {
    return 0;
}

/*
=====================================================
SegTree 本体
=====================================================
*/

struct SegTree {
    int n;
    vector<S> seg;

    SegTree(int sz) {
        n = 1;
        while (n < sz) n <<= 1;
        seg.assign(2*n, id());
    }

    // 配列初期化
    SegTree(const vector<S>& v) {
        int sz = v.size();

        n = 1;
        while (n < sz) n <<= 1;

        seg.assign(2*n, id());

        for (int i = 0; i < sz; i++)
            seg[n+i] = v[i];

        for (int i = n-1; i > 0; i--)
            seg[i] = combine(seg[i<<1], seg[i<<1|1]);
    }

    // ========================
    // 点更新
    // ========================
    void set(int i, S x) {
        i += n;

        seg[i] = x;

        while (i >>= 1)
            seg[i] = combine(seg[i<<1], seg[i<<1|1]);
    }

    // ========================
    // 区間取得 [l,r)
    // ========================
    S range(int l, int r) {
        S L = id(), R = id();

        for (l += n, r += n; l < r; l >>= 1, r >>= 1) {

            if (l & 1)
                L = combine(L, seg[l++]);

            if (r & 1)
                R = combine(seg[--r], R);
        }

        return combine(L, R);
    }

    S get(int i) {
        return seg[n+i];
    }

    S all() {
        return seg[1];
    }

    // ========================
    // k-th One
    // ========================
    // 1-indexed
    // k番目の1の位置を返す
    // 存在しないなら -1
    // ========================
    int kth_one(int k) {

        if (seg[1] < k)
            return -1;

        int i = 1;

        while (i < n) {

            if (seg[i<<1] >= k) {
                i <<= 1;
            }
            else {
                k -= seg[i<<1];
                i = i<<1 | 1;
            }
        }

        return i - n;
    }
};

void random_test() {

    mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());

    rep(_,0,10000){

        int N = rng()%20 + 1;

        vector<int> a(N);

        rep(i,0,N)
            a[i] = rng()%2;

        SegTree seg(a);

        rep(q,0,1000){

            int type = rng()%3;

            // ========================
            // 点更新
            // ========================
            if(type==0){

                int idx = rng()%N;
                int x = rng()%2;

                a[idx] = x;
                seg.set(idx,x);
            }

            // ========================
            // 区間和
            // ========================
            else if(type==1){

                int l = rng()%N;
                int r = rng()%N;

                if(l>r)swap(l,r);

                r++;

                int ans1 = 0;

                rep(i,l,r)
                    ans1 += a[i];

                int ans2 = seg.range(l,r);

                if(ans1!=ans2){

                    cerr << "range error\n";

                    cerr << "N = " << N << "\n";

                    cerr << "array : ";
                    for(auto x:a)cerr << x << " ";
                    cerr << "\n";

                    cerr << "l r = " << l << " " << r << "\n";

                    cerr << "naive = " << ans1 << "\n";
                    cerr << "seg   = " << ans2 << "\n";

                    exit(0);
                }
            }

            // ========================
            // kth_one
            // ========================
            else{

                int total = 0;

                for(auto x:a)
                    total += x;

                int k = rng()%(total+2); // 存在しないケース込み

                if(k==0)k=1;

                int ans1 = -1;

                if(k<=total){

                    int cnt = 0;

                    rep(i,0,N){

                        if(a[i])cnt++;

                        if(cnt==k){
                            ans1 = i;
                            break;
                        }
                    }
                }

                int ans2 = seg.kth_one(k);

                if(ans1!=ans2){

                    cerr << "kth error\n";

                    cerr << "N = " << N << "\n";

                    cerr << "array : ";
                    for(auto x:a)cerr << x << " ";
                    cerr << "\n";

                    cerr << "k = " << k << "\n";

                    cerr << "naive = " << ans1 << "\n";
                    cerr << "seg   = " << ans2 << "\n";

                    exit(0);
                }
            }
        }
    }

    cerr << "OK\n";
}
