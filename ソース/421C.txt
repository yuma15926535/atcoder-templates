#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor

template<class T> using PQ = priority_queue<T, vector<T>>;//大きい順
template<class T> using PQ_g = priority_queue<T, vector<T>, greater<T>>;//小さい順
#define chmax(x,y) x = max(x,y)
#define chmin(x,y) x = min(x,y)

int main(){
    int N;
    string S;
    cin>>N>>S;
    ll sz=S.size();
    vector<int>Apos;
    rep(i,0,sz)if(S[i]=='A')Apos.push_back(i);
    int aidx=0;
    ll ans=1e15;
    //状態が2つなのですべて試す
    rep(i,0,2){
        char c;
        if(i)c='A';
        else c='B';
        string s=S;
        ll cnt=0;
        aidx=-1;
        //A同士を入れ替える必要がないのでAの順序は不変
        //文字が2つしかないのでAのみを合わせればBも合う
        rep(j,0,sz){
            if(c=='A')c='B';
            else c='A';
            if(c!='A')continue;
            aidx++;
            if(Apos[aidx]==j)continue;
            cnt+=abs(Apos[aidx]-j);
            s[Apos[aidx]]='B';
        }
        chmin(ans,cnt);
    }
    cout<<ans<<'\n';
    return 0;
}
