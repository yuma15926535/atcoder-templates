/**
 * @brief Rolling Hash（2^61-1）
 *
 * 文字列の部分文字列ハッシュを O(1) で取得する。
 * 部分文字列比較・LCP・辞書順比較・ハッシュ連結に対応。
 *
 * Time Complexity:
 *   構築           : O(N)
 *   get系          : O(1)
 *   concat         : O(1)
 *   lcp            : O(log N)
 *   less_than      : O(log N)
 */
struct RollingHash {
    using ull = unsigned long long;
    static constexpr ull mod = (1ULL << 61) - 1;

    /**
     * @brief ハッシュ情報
     *
     * value : ハッシュ値
     * left  : 元文字列での開始位置（0-index）
     * right : 元文字列での終了位置（0-index, 半開区間）
     * len   : 部分文字列長
     *
     * ※ concat() で生成した Hash は len のみ有効で、
     *    left==-1 の場合は元文字列に対応しない Hash を表す。
     */
    struct Hash {
        ull value;
        int left;
        int right;
        int len;

        operator ull() const { return value; }

        bool operator==(const Hash& rhs) const {
            return value == rhs.value;
        }

        bool operator!=(const Hash& rhs) const {
            return value != rhs.value;
        }

        bool operator<(const Hash& rhs) const {
            return value < rhs.value;
        }
    };

    static ull base;
    static bool initialized;
    static vector<ull> power;

    vector<ull> h;

    /**
     * @brief base をランダムに初期化
     *
     * 初回のみ呼ばれ、base と power を初期化する。
     * コンストラクタで呼ばれるため個別での実行不要
     *
     * Time Complexity:
     *   O(1)
     */
    static void init() {
        if (initialized) return;
        initialized = true;

        mt19937_64 rng(
            chrono::steady_clock::now().time_since_epoch().count()
        );
        base = uniform_int_distribution<ull>(
            100000000ULL,
            1000000000ULL
        )(rng);

        power = {1};
    }

    /**
     * @brief 文字列から Rolling Hash を構築
     *
     * @param s 元文字列
     *
     * Time Complexity:
     *   O(N)
     */
    RollingHash(const string& s) {
        init();

        int n = (int)s.size();
        ensure_power(n);

        h.assign(n + 1, 0);

        for (int i = 0; i < n; i++) {
            h[i + 1] = add(
                mul(h[i], base),
                (ull)(unsigned char)s[i] + 1
            );
        }
    }

    /**
     * @brief 文字列長を取得
     *
     * @return 文字列長
     *
     * Time Complexity:
     *   O(1)
     */
    int size() const {
        return (int)h.size() - 1;
    }

    /**
     * @brief 部分文字列 [l,r) のハッシュを取得
     *
     * @param l 開始位置（0-index）
     * @param r 終了位置（0-index, 半開区間）
     * @return Hash
     *
     * Time Complexity:
     *   O(1)
     */
    Hash get(int l, int r) const {
        return {
            add(h[r], mod - mul(h[l], power[r - l])),
            l,
            r,
            r - l
        };
    }

    /**
     * @brief 長さ指定で部分文字列のハッシュを取得
     *
     * s[l,l+len) のハッシュを返す。
     *
     * @param l 開始位置
     * @param len 長さ
     * @return Hash
     *
     * Time Complexity:
     *   O(1)
     */
    Hash get_len(int l, int len) const {
        return get(l, l + len);
    }

    /**
     * @brief 接尾辞のハッシュを取得
     *
     * s[l,n) のハッシュを返す。
     *
     * @param l 開始位置
     * @return Hash
     *
     * Time Complexity:
     *   O(1)
     */
    Hash get_from(int l = 0) const {
        return get(l, size());
    }

    /**
     * @brief 空文字列のハッシュを取得
     *
     * concat() の単位元として利用できる。
     *
     * @return Hash
     *
     * Time Complexity:
     *   O(1)
     */
    static Hash empty_hash() {
        return {
            0,
            -1,
            -1,
            0
        };
    }

    /**
     * @brief 2つのハッシュを連結
     *
     * a + b のハッシュを返す。
     * 異なる RollingHash 同士でも利用可能。
     *
     * ※返却される Hash は len のみ有効で、
     *   left==-1 のため元文字列には対応しない。
     *
     * @param a 前半
     * @param b 後半
     * @return Hash
     *
     * Time Complexity:
     *   O(1)
     */
    static Hash concat(const Hash& a, const Hash& b) {
        ensure_power(b.len);

        return {
            add(
                mul(a.value, power[b.len]),
                b.value
            ),
            -1,
            -1,
            a.len + b.len
        };
    }

    /**
     * @brief 最長共通接頭辞長（LCP）を取得
     *
     * a[i,n) と b[j,n) の LCP を返す。
     *
     * @param a 文字列A
     * @param i 開始位置
     * @param b 文字列B
     * @param j 開始位置
     * @return LCP長
     *
     * Time Complexity:
     *   O(log N)
     */
    static int lcp(
        const RollingHash& a, int i,
        const RollingHash& b, int j
    ) {
        int low = 0;
        int high = min(a.size() - i, b.size() - j) + 1;

        while (high - low > 1) {
            int mid = (low + high) / 2;

            if (a.get(i, i + mid) == b.get(j, j + mid))
                low = mid;
            else
                high = mid;
        }

        return low;
    }

    /**
     * @brief 接尾辞同士を辞書順比較
     *
     * a[i,n) < b[j,n) なら true を返す。
     *
     * @param a 文字列A
     * @param i 開始位置
     * @param b 文字列B
     * @param j 開始位置
     * @return 比較結果
     *
     * Time Complexity:
     *   O(log N)
     */
    static bool less_than(
        const RollingHash& a, int i,
        const RollingHash& b, int j
    ) {
        int l = lcp(a, i, b, j);

        if (i + l == a.size() || j + l == b.size()) {
            return (a.size() - i) < (b.size() - j);
        }

        return a.get(i + l, i + l + 1).value
             < b.get(j + l, j + l + 1).value;
    }

    #ifdef LOCAL
    /**
     * @brief Hash の情報を表示（デバッグ用）
     *
     * ハッシュ値・区間・長さ・対応する部分文字列を表示する。
     * 提出時（-DLOCAL なし）は何もしない。
     *
     * @param s 元文字列
     * @param hash 表示する Hash
     *
     * Time Complexity:
     *   O(len)
     */
    void print_hash(const string& s, const Hash& hash) const {
        cout << DBG_COLOR;
        cout << "hash  = " << hash.value << '\n';
        cout << "left  = " << hash.left << '\n';
        cout << "right = " << hash.right << '\n';
        cout << "len   = " << hash.len << '\n';

        if (hash.left == -1) {
            cout << "str   = (concat 等により元文字列に対応しません)\n";
        } else {
            cout << "str   = "
                 << s.substr(hash.left, hash.len) << '\n';
        }

        cout << DBG_RESET;
    }
    #else
    /**
     * @brief Hash の情報を表示（デバッグ用）
     *
     * ハッシュ値・区間・長さ・対応する部分文字列を表示する。
     * 提出時（-DLOCAL なし）は何もしない。
     *
     * @param s 元文字列
     * @param hash 表示する Hash
     *
     * Time Complexity:
     *   O(len)
     */
    void print_hash(const string&, const Hash&) const {}
    #endif

    static ull mul(ull a, ull b) {
        __uint128_t t = (__uint128_t)a * b;
        t = (t >> 61) + (t & mod);
        if (t >= mod) t -= mod;
        return (ull)t;
    }

    static ull add(ull a, ull b) {
        a += b;
        if (a >= mod) a -= mod;
        return a;
    }

    static void ensure_power(int n) {
        while ((int)power.size() <= n) {
            power.push_back(
                mul(power.back(), base)
            );
        }
    }
};

// ランダムテスト
void test_RollingHashRandom() {
    cout << "===== RollingHash ランダムテスト =====" << endl;

    mt19937 rng(123456789);

    const string CH = "abc";

    rep(_, 5000) {

        int n1 = rng() % 25;
        int n2 = rng() % 25;

        string s, t;

        rep(i, n1) s += CH[rng() % CH.size()];
        rep(i, n2) t += CH[rng() % CH.size()];

        RollingHash rhs(s);
        RollingHash rht(t);

        // get(), get_len(), get_from()
        rep(l, n1 + 1) {
            rep(r, l, n1 + 1) {

                auto h = rhs.get(l, r);

                RollingHash naive(s.substr(l, r - l));

                assert(h == naive.get_from());

                auto h2 = rhs.get_len(l, r - l);

                assert(h == h2);

                auto h3 = rhs.get_from(l);

                RollingHash naive2(s.substr(l));

                assert(h3 == naive2.get_from());
            }
        }

        // concat()
        rep(l1, n1 + 1) {
            rep(r1, l1, n1 + 1) {
                auto a = rhs.get(l1, r1);

                rep(l2, n2 + 1) {
                    rep(r2, l2, n2 + 1) {

                        auto b = rht.get(l2, r2);

                        auto c = RollingHash::concat(a, b);

                        RollingHash naive(
                            s.substr(l1, r1 - l1) +
                            t.substr(l2, r2 - l2)
                        );

                        assert(c == naive.get_from());
                        assert(c.len == a.len + b.len);
                    }
                }
            }
        }

        // empty_hash()
        {
            auto e = RollingHash::empty_hash();

            auto all = rhs.get_from();

            assert(RollingHash::concat(e, all) == all);
            assert(RollingHash::concat(all, e) == all);
        }

        // lcp()
        rep(i, n1 + 1) {
            rep(j, n2 + 1) {

                int naive = 0;

                while (i + naive < n1 &&
                       j + naive < n2 &&
                       s[i + naive] == t[j + naive])
                    naive++;

                assert(
                    RollingHash::lcp(rhs, i, rht, j)
                    == naive
                );
            }
        }

        // less_than()
        rep(i, n1 + 1) {
            rep(j, n2 + 1) {

                bool naive =
                    s.substr(i) < t.substr(j);

                assert(
                    RollingHash::less_than(rhs, i, rht, j)
                    == naive
                );
            }
        }

        // Hash演算子
        rep(l, n1 + 1) {
            rep(r, l, n1 + 1) {

                auto h = rhs.get(l, r);

                assert(h == h);
                assert(!(h != h));

                ull x = h;

                assert(x == h.value);
            }
        }
    }

    cout << "RollingHash ランダムテスト OK" << endl;
}

void test_RollingHashConcatAssociative() {
    cout << "===== RollingHash concat 結合則ランダムテスト =====" << endl;

    mt19937 rng(123456789);

    const string CH = "abc";

    rep(_, 100000) {

        auto make_string = [&](int n) {
            string s;
            rep(i, n) s += CH[rng() % CH.size()];
            return s;
        };

        string s1 = make_string(rng() % 20);
        string s2 = make_string(rng() % 20);
        string s3 = make_string(rng() % 20);

        RollingHash rh1(s1);
        RollingHash rh2(s2);
        RollingHash rh3(s3);

        auto random_hash = [&](const RollingHash& rh) {
            int n = rh.size();
            int l = rng() % (n + 1);
            int r = l + rng() % (n - l + 1);
            return rh.get(l, r);
        };

        auto A = random_hash(rh1);
        auto B = random_hash(rh2);
        auto C = random_hash(rh3);

        auto left = RollingHash::concat(
            RollingHash::concat(A, B),
            C
        );

        auto right = RollingHash::concat(
            A,
            RollingHash::concat(B, C)
        );

        assert(left == right);
        assert(left.len == A.len + B.len + C.len);

        // 念のため愚直とも比較
        string naive =
            s1.substr(A.left, A.len) +
            s2.substr(B.left, B.len) +
            s3.substr(C.left, C.len);

        RollingHash rhs(naive);

        assert(left == rhs.get_from());
    }

    cout << "concat 結合則ランダムテスト OK" << endl;
}

void test_RollingHash() {
    cout << "===== RollingHash 動作確認 =====" << endl;

    string s = "abracadabra";
    string t = "cadabra";

    RollingHash rhs(s);
    RollingHash rht(t);

    cout << "元文字列1 : " << s << endl;
    cout << "元文字列2 : " << t << endl;
    cout << endl;

    auto h1 = rhs.get(0, 4);      // "abra"
    auto h2 = rhs.get_len(7, 4);  // "abra"

    cout << "get(0,4) == get_len(7,4) : "
         << (h1 == h2 ? "はい" : "いいえ") << endl;

    cout << "Hash の情報" << endl;
    cout << "  left : " << h1.left << endl;
    cout << "  right: " << h1.right << endl;
    cout << "  len  : " << h1.len << endl;
    rhs.print_hash(s, h1);
    cout << endl;

    auto suffix = rhs.get_from(4);

    cout << "get_from(4)" << endl;
    cout << "  len : " << suffix.len << endl;
    rhs.print_hash(s, suffix);
    cout << endl;

    auto c = RollingHash::concat(
        rhs.get(0, 4),   // "abra"
        rht.get(0, 3)    // "cad"
    );

    cout << "concat(\"abra\", \"cad\")" << endl;
    cout << "  len : " << c.len << endl;
    rhs.print_hash(s, c);
    cout << endl;

    cout << "LCP(\"cadabra\", \"cadabra\") : "
         << RollingHash::lcp(rhs, 4, rht, 0)
         << endl;

    cout << "辞書順比較" << endl;
    cout << "  \"abracadabra\" < \"cadabra\" : "
         << (RollingHash::less_than(rhs, 0, rht, 0) ? "はい" : "いいえ")
         << endl;
    cout << endl;

    auto e = RollingHash::empty_hash();

    cout << "empty_hash()" << endl;
    cout << "  len : " << e.len << endl;
    cout << "  empty + \"abra\" == \"abra\" : "
         << (RollingHash::concat(e, h1) == h1 ? "はい" : "いいえ")
         << endl;
    cout << "  \"abra\" + empty == \"abra\" : "
         << (RollingHash::concat(h1, e) == h1 ? "はい" : "いいえ")
         << endl;
}
