template <typename T>

    using value_type = T;
    static T identity() { return 0; }
    static T op(const T& a, const T& b) { return a + b; }
};

    using value_type=T;
    static T identity(){ return numeric_limits<T>::max(); }
    static T op(const T&a,const T&b){ return min(a,b); }
};

    using value_type=T;
    static T identity(){ return numeric_limits<T>::lowest(); }
    static T op(const T&a,const T&b){ return max(a,b); }
};

};

/*
    B+Tree Sequence

    【概要】
    ・配列をB+Treeで管理する平衡木
    ・insert / erase / get / set / prod を高速に処理できる
    ・Monoidを変更することで区間和・区間最小値・区間最大値などへ変更可能

    【計算量】
        size()      : O(1)
        empty()     : O(1)
        get()       : O(log_B N)
        set()       : O(log_B N)
        insert()    : O(B log_B N)
        erase()     : O(B log_B N)
        prod()      : O(B log_B N)
        clear()     : O(N)

    【添字】
        0-index

    【区間】
        prod(l,r) は半開区間 [l,r)

    【Monoid】
        Monoid は

            using value_type
            static identity()
            static op(a,b)

        を実装した構造体を渡す。

    【用途】
        ・動的配列
        ・Rope
        ・Text Editor
        ・Insert/Delete付き区間クエリ
*/
template <typename Monoid, int B = 64>
class BPlusTree {
public:
    using T = typename Monoid::value_type;

private:
    struct Node {
        bool is_leaf;
        int len;
        int subtree_size;
        T prod;
        Node* children[B + 1]; 
        T values[B + 1];       

        Node(bool leaf) : is_leaf(leaf), len(0), subtree_size(0), prod(Monoid::identity()) {
            for (int i = 0; i <= B; ++i) {
                children[i] = nullptr;
                values[i] = Monoid::identity();
            }
        }

        void update() {
            subtree_size = 0;
            prod = Monoid::identity();
            if (is_leaf) {
                subtree_size = len;
                for (int i = 0; i < len; ++i) {
                    prod = Monoid::op(prod, values[i]);
                }
            } else {
                for (int i = 0; i < len; ++i) {
                    subtree_size += children[i]->subtree_size;
                    prod = Monoid::op(prod, children[i]->prod);
                }
                for (int i = len; i <= B; ++i) {
                    children[i] = nullptr;
                }
            }
        }
    };

    Node* root;
    const int MIN_DEG = (B + 1) / 2;

    void destroy(Node* node) {
        if (!node) return;
        if (!node->is_leaf) {
            for (int i = 0; i < node->len; ++i) {
                destroy(node->children[i]);
            }
        }
        delete node;
    }

    Node* find_leaf(int& pos, std::vector<Node*>& path) {
        Node* curr = root;
        while (!curr->is_leaf) {
            path.push_back(curr);
            int idx = 0;
            while (idx < curr->len - 1) {
                int sz = curr->children[idx]->subtree_size;
                if (pos < sz) break;
                pos -= sz;
                idx++;
            }
            curr = curr->children[idx];
        }
        return curr;
    }

    T prod_rec(Node* node, int l, int r) {
        if (l >= node->subtree_size || r <= 0) return Monoid::identity(); 
        if (l <= 0 && r >= node->subtree_size) return node->prod;         

        T res = Monoid::identity();
        if (node->is_leaf) {
            int max_l = std::max(0, l);
            int min_r = std::min(node->len, r);
            for (int i = max_l; i < min_r; ++i) {
                res = Monoid::op(res, node->values[i]);
            }
        } else {
            int offset = 0;
            for (int i = 0; i < node->len; ++i) {
                int sz = node->children[i]->subtree_size;
                if (offset < r && offset + sz > l) {
                    res = Monoid::op(res, prod_rec(node->children[i], l - offset, r - offset));
                }
                offset += sz;
                if (offset >= r) break;
            }
        }
        return res;
    }

    int check_node(Node* node, int depth, int& leaf_depth) const {
        if (!node) return 0;
        
        if (node != root) {
            assert(node->len >= MIN_DEG);
            assert(node->len <= B);
        } else {
            assert(node->len <= B);
        }

        int actual_subtree_size = 0;
        T actual_prod = Monoid::identity();

        if (node->is_leaf) {
            if (leaf_depth == -1) leaf_depth = depth;
            assert(leaf_depth == depth); 
            for (int i = 0; i < node->len; ++i) {
                actual_subtree_size++;
                actual_prod = Monoid::op(actual_prod, node->values[i]);
            }
        } else {
            for (int i = 0; i < node->len; ++i) {
                assert(node->children[i] != nullptr);
                actual_subtree_size += check_node(node->children[i], depth + 1, leaf_depth);
                actual_prod = Monoid::op(actual_prod, node->children[i]->prod);
            }
            for (int i = node->len; i <= B; ++i) {
                assert(node->children[i] == nullptr);
            }
        }
        assert(node->subtree_size == actual_subtree_size);
        assert(node->prod == actual_prod); 
        return actual_subtree_size;
    }

public:
    BPlusTree() {
        root = new Node(true);
    }

    ~BPlusTree() {
        clear();
    }

    // 全要素削除
    void clear() {
        if (root) {
            destroy(root);
            root = nullptr;
        }
    }

    // 要素数
    int size() const {
        return root ? root->subtree_size : 0;
    }

    // 空判定
    bool empty() const {
        return size() == 0;
    }

    // pos に val を挿入
    // O(B log_B N)
    void insert(int pos, T val) {
        if (!root) root = new Node(true);
        if (pos < 0 || pos > size()) return;
        
        int local_pos = pos;
        std::vector<Node*> path;
        Node* leaf = find_leaf(local_pos, path);

        for (int i = leaf->len; i > local_pos; --i) {
            leaf->values[i] = leaf->values[i - 1];
        }
        leaf->values[local_pos] = val;
        leaf->len++;
        leaf->update();
        
        Node* curr = leaf;
        while (curr->len > B) {
            Node* right = new Node(curr->is_leaf);
            int mid = curr->len / 2;
            int right_len = curr->len - mid;

            if (curr->is_leaf) {
                for (int i = 0; i < right_len; ++i) {
                    right->values[i] = curr->values[mid + i];
                    curr->values[mid + i] = Monoid::identity();
                }
            } else {
                for (int i = 0; i < right_len; ++i) {
                    right->children[i] = curr->children[mid + i];
                    curr->children[mid + i] = nullptr;
                }
            }
            right->len = right_len;
            curr->len = mid;
            curr->update();
            right->update();

            if (path.empty()) {
                Node* new_root = new Node(false);
                new_root->children[0] = curr;
                new_root->children[1] = right;
                new_root->len = 2;
                new_root->update();
                root = new_root;
                return; // 新しい根を作った場合はここで完了
            }

            Node* p = path.back();
            path.pop_back();

            int idx = 0;
            while (idx < p->len && p->children[idx] != curr) idx++;

            for (int i = p->len; i > idx + 1; --i) {
                p->children[i] = p->children[i - 1];
            }
            p->children[idx + 1] = right;
            p->len++;
            
            curr = p; // 上の階層へ
        }

        curr->update();
        while (!path.empty()) {
            Node* anc = path.back();
            path.pop_back();
            anc->update();
        }
    }

    // pos を削除
    // O(B log_B N)
    void erase(int pos) {
        if (!root || pos < 0 || pos >= size()) return;
        
        int local_pos = pos;
        std::vector<Node*> path;
        Node* leaf = find_leaf(local_pos, path);

        for (int i = local_pos; i < leaf->len - 1; ++i) {
            leaf->values[i] = leaf->values[i + 1];
        }
        leaf->values[leaf->len - 1] = Monoid::identity();
        leaf->len--;
        leaf->update();
        
        Node* curr = leaf;
        while (curr != root && curr->len < MIN_DEG) {
            if (path.empty()) break;
            
            Node* p = path.back();
            path.pop_back();

            int idx = 0;
            while (idx < p->len && p->children[idx] != curr) idx++;

            Node* left_sib = (idx > 0) ? p->children[idx - 1] : nullptr;
            Node* right_sib = (idx < p->len - 1) ? p->children[idx + 1] : nullptr;

            if (left_sib && left_sib->len > MIN_DEG) {
                if (curr->is_leaf) {
                    for (int i = curr->len; i > 0; --i) curr->values[i] = curr->values[i - 1];
                    curr->values[0] = left_sib->values[left_sib->len - 1];
                    left_sib->values[left_sib->len - 1] = Monoid::identity();
                } else {
                    for (int i = curr->len; i > 0; --i) curr->children[i] = curr->children[i - 1];
                    curr->children[0] = left_sib->children[left_sib->len - 1];
                    left_sib->children[left_sib->len - 1] = nullptr;
                }
                curr->len++;
                left_sib->len--;
                left_sib->update();
                curr->update();
                curr = p;
                break; // 借用完了
            } else if (right_sib && right_sib->len > MIN_DEG) {
                if (curr->is_leaf) {
                    curr->values[curr->len] = right_sib->values[0];
                    for (int i = 0; i < right_sib->len - 1; ++i) right_sib->values[i] = right_sib->values[i + 1];
                    right_sib->values[right_sib->len - 1] = Monoid::identity();
                } else {
                    curr->children[curr->len] = right_sib->children[0];
                    for (int i = 0; i < right_sib->len - 1; ++i) right_sib->children[i] = right_sib->children[i + 1];
                    right_sib->children[right_sib->len - 1] = nullptr;
                }
                curr->len++;
                right_sib->len--;
                right_sib->update();
                curr->update();
                curr = p;
                break; // 借用完了
            } else {
                if (left_sib) {
                    if (curr->is_leaf) {
                        for (int i = 0; i < curr->len; ++i) left_sib->values[left_sib->len + i] = curr->values[i];
                    } else {
                        for (int i = 0; i < curr->len; ++i) {
                            left_sib->children[left_sib->len + i] = curr->children[i];
                            curr->children[i] = nullptr; 
                        }
                    }
                    left_sib->len += curr->len;
                    left_sib->update();

                    for (int i = idx; i < p->len - 1; ++i) p->children[i] = p->children[i + 1];
                    p->children[p->len - 1] = nullptr;
                    p->len--;
                    delete curr;
                } else if (right_sib) {
                    if (curr->is_leaf) {
                        for (int i = 0; i < right_sib->len; ++i) curr->values[curr->len + i] = right_sib->values[i];
                    } else {
                        for (int i = 0; i < right_sib->len; ++i) {
                            curr->children[curr->len + i] = right_sib->children[i];
                            right_sib->children[i] = nullptr;
                        }
                    }
                    curr->len += right_sib->len;
                    curr->update();

                    for (int i = idx + 1; i < p->len - 1; ++i) p->children[i] = p->children[i + 1];
                    p->children[p->len - 1] = nullptr;
                    p->len--;
                    delete right_sib;
                }
                curr = p; // マージしたので親を継続確認
            }
        }

        if (curr == root && !root->is_leaf && root->len == 1) {
            Node* old_root = root;
            root = root->children[0];
            old_root->children[0] = nullptr;
            delete old_root;

            root->update(); // 念のため最新状態にする
            return;
        }

        curr->update();
        while (!path.empty()) {
            Node* anc = path.back();
            path.pop_back();
            anc->update();
        }
    }

    // pos の値取得
    // O(log_B N)
    T get(int pos) {
        assert(pos >= 0 && pos < size());
        int local_pos = pos;
        std::vector<Node*> path;
        Node* leaf = find_leaf(local_pos, path);
        return leaf->values[local_pos];
    }

    // pos を val に変更
    // O(log_B N)
    void set(int pos, T val) {
        assert(pos >= 0 && pos < size());
        int local_pos = pos;
        std::vector<Node*> path;
        Node* leaf = find_leaf(local_pos, path);
        leaf->values[local_pos] = val;
        leaf->update();
        while (!path.empty()) {
            Node* anc = path.back();
            path.pop_back();
            anc->update();
        }
    }

    // 半開区間 [l,r) のモノイド積
    // O(B log_B N)
    T prod(int l, int r) {
        if (!root || l >= r || l >= size() || r <= 0) return Monoid::identity();
        return prod_rec(root, l, r);
    }

    // 内部構造の整合性チェック（デバッグ用）
    void debug_check() const {
        if (!root) return;
        int leaf_depth = -1;
        check_node(root, 0, leaf_depth);
    }

    // 全要素をvectorへ出力（デバッグ用）
    void export_to_vector(std::vector<T>& out) const {
        out.clear();
        auto dfs = [&](auto& self, Node* n) -> void {
            if (!n) return;
            if (n->is_leaf) {
                for (int i = 0; i < n->len; ++i) out.push_back(n->values[i]);
            } else {
                for (int i = 0; i < n->len; ++i) self(self, n->children[i]);
            }
        };
        dfs(dfs, root);
    }
};

int get_insert_pos(int size, std::mt19937& rng) {
    if (size == 0) return 0;
    int r = rng() % 3;
    if (r == 0) return 0;           
    if (r == 1) return size;        
    return rng() % (size + 1);      
}

int get_erase_pos(int size, std::mt19937& rng) {
    if (size <= 0) return 0; 
    int r = rng() % 3;
    if (r == 0) return 0;
    if (r == 1) return size - 1; 
    return rng() % size;
}

// 徹底的な全検証
void verify_against_naive(BPlusTree<SumMonoid<long long>, 4>& tree, const std::vector<long long>& naive) {
    assert(tree.size() == (int)naive.size());
    
    // 1. 順序の完全一致
    std::vector<long long> exported;
    tree.export_to_vector(exported);
    assert(exported == naive);
    
    // 2. 個別アクセスの完全一致
    for (int i = 0; i < (int)naive.size(); ++i) {
        assert(tree.get(i) == naive[i]);
    }
    
    // 3. 区間積の全体一致
    long long naive_sum = 0;
    for (long long x : naive) naive_sum += x;
    assert(tree.prod(0, tree.size()) == naive_sum);
    
    // 4. B+Treeとしての構造条件一致
    tree.debug_check();
}

void verify_small_B() {
    std::cout << "--- verify_small_B (B=4, Q=10000) ---" << std::endl;
    BPlusTree<SumMonoid<long long>, 4> tree;
    std::vector<long long> naive;
    std::mt19937 rng(1337);

    int Q = 10000;
    for (int i = 0; i < Q; ++i) {
        int op = rng() % 100;
        if (op < 35 || tree.empty()) { 
            // 35%: Insert
            int pos = get_insert_pos(tree.size(), rng);
            long long val = rng() % 1000;
            tree.insert(pos, val);
            naive.insert(naive.begin() + pos, val);
        } else if (op < 60) {
            // 25%: Erase
            int pos = get_erase_pos(tree.size(), rng);
            tree.erase(pos);
            naive.erase(naive.begin() + pos);
        } else if (op < 80) {
            // 20%: Set
            int pos = rng() % tree.size();
            long long val = rng() % 1000;
            tree.set(pos, val);
            naive[pos] = val;
        } else {
            // 20%: Prod
            int l = rng() % tree.size();
            int r = rng() % tree.size();
            if (l > r) std::swap(l, r);
            r++;
            long long tree_ans = tree.prod(l, r);
            long long naive_ans = 0;
            for (int k = l; k < r; ++k) naive_ans += naive[k];
            assert(tree_ans == naive_ans);
        }
        
        // 毎回状態を徹底チェック
        verify_against_naive(tree, naive);
    }
    std::cout << "verify_small_B PASSED!" << std::endl;
}

void reuse_test() {
    std::cout << "--- reuse_test ---" << std::endl;

    BPlusTree<SumMonoid<long long>, 64> tree;

    // 空
    assert(tree.empty());
    assert(tree.size() == 0);

    // 挿入
    tree.insert(0, 5);
    assert(!tree.empty());
    assert(tree.size() == 1);
    assert(tree.get(0) == 5);
    assert(tree.prod(0, 1) == 5);
    tree.debug_check();

    // 空に戻す
    tree.erase(0);
    assert(tree.empty());
    assert(tree.size() == 0);
    tree.debug_check();

    // clear()後も問題なく使えるか
    tree.clear();
    assert(tree.empty());
    assert(tree.size() == 0);

    // 再利用
    tree.insert(0, 10);
    tree.insert(1, 20);
    tree.insert(1, 15);

    assert(tree.size() == 3);
    assert(tree.get(0) == 10);
    assert(tree.get(1) == 15);
    assert(tree.get(2) == 20);
    assert(tree.prod(0, 3) == 45);

    std::vector<long long> expected = {10, 15, 20};
    std::vector<long long> exported;
    tree.export_to_vector(exported);
    assert(exported == expected);

    tree.debug_check();

    std::cout << "reuse_test PASSED!" << std::endl;
}

void height_transition_test() {
    std::cout << "--- height_transition_test ---" << std::endl;

    constexpr int B = 4;
    constexpr int N = 1000;
    constexpr int REP = 100;

    for (int rep = 0; rep < REP; rep++) {
        BPlusTree<SumMonoid<long long>, B> tree;
        std::vector<long long> naive;

        // 大量挿入（高さを上げる）
        for (int i = 0; i < N; i++) {
            tree.insert(tree.size(), i);
            naive.push_back(i);
        }
        verify_against_naive(tree, naive);

        // 全削除（高さを下げて空にする）
        while (!tree.empty()) {
            tree.erase(tree.size() - 1);
            naive.pop_back();
        }
        verify_against_naive(tree, naive);

        // 再利用（もう一度高さを上げる）
        for (int i = 0; i < N; i++) {
            tree.insert(0, i);
            naive.insert(naive.begin(), i);
        }
        verify_against_naive(tree, naive);

        // もう一度全削除（今度は先頭から）
        while (!tree.empty()) {
            tree.erase(0);
            naive.erase(naive.begin());
        }
        verify_against_naive(tree, naive);
    }

    std::cout << "height_transition_test PASSED!" << std::endl;
}
