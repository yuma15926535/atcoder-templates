#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())
template <typename T> constexpr auto ssize(const T& container) -> std::ptrdiff_t {return static_cast<std::ptrdiff_t>(container.size());}

struct xy{
    int x1;
    int y1;
    int x2;
    int y2;
    pair<int,int> get1(){
        return {x1,y1};
    }
    pair<int,int> get2(){
        return {x2,y2};
    }
};

int main(){
    int N;
    cin>>N;
    vector<string>S(N);
    rep(i,0,N)cin>>S[i];
    //Nが十分に小さいのでN^4でBFSを行う
    vector visited(N,vector(N,vector(N,vector(N,true))));
    vector dis(N,vector(N,vector(N,vector(N,1e9))));
    pair<int,int> s1={-1,-1},s2;
    rep(i,0,N)rep(j,0,N)if(S[i][j]=='P'){
        if(s1.first==-1)s1={i,j},S[i][j]='.';
        else s2={i,j},S[i][j]='.';
    }
    queue<xy>q;
    q.push({s1.first,s1.second,s2.first,s2.second});
    dis[s1.first][s1.second][s2.first][s2.second]=0;
    visited[s1.first][s1.second][s2.first][s2.second]=false;
    auto move=[&](int x,int y,int p)->pair<int,int>{
        vector<int>dx={-1,1,0,0},dy={0,0,-1,1};
        if(0<=x+dx[p]&&x+dx[p]<N&&0<=y+dy[p]&&y+dy[p]<N){
            if(S[x+dx[p]][y+dy[p]]=='.')return {x+dx[p],y+dy[p]};
            else return {x,y};
        }
        return {x,y};
    };
    while(!q.empty()){
        auto t=q.front();
        q.pop();
        auto[a,b]=t.get1();
        auto[c,d]=t.get2();
        if(a==c&&b==d){
            cout<<dis[a][b][c][d]<<"\n";
            return 0;
        }
        rep(i,0,4){
            auto[x1,y1]=move(a,b,i);
            auto[x2,y2]=move(c,d,i);
            if(visited[x1][y1][x2][y2]){
                visited[x1][y1][x2][y2]=false;
                dis[x1][y1][x2][y2]=dis[a][b][c][d]+1;
                q.push({x1,y1,x2,y2});
            }
        }
    }
    cout<<-1<<"\n";
    return 0;
}
