//xor(排他的論理和)の性質
//x^x=0
//x^0=x
//N=4で考えて
//   x2^x3^x4=a1
//x1^   x3^x4=a2
//x1^x2^   x4=a3
//x1^x2^x3   =a4
//全てでxorを取ると各xiで奇数回xorになるので
//x1^x2^x3^x4=a1^a2^a3^a4=B
//さらにi番目の等式とxorを取るとxi以外が偶数回になって消える
//xi=B^ai
int main(){
    int N;
    cin>>N;
    vector<ll>A(N);
    ll B=0;
    rep(i,0,N){
        cin>>A[i];
        B^=A[i];
    }
    rep(i,0,N)cout<<(B^A[i])<<" ";
    cout<<"\n";
    return 0;
}
