#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    const string target="ABC";
    cin.tie(nullptr);
    ios::sync_with_stdio(false);
    int N,Q,ans=0;
    string S;
    cin>>N>>Q>>S;
    set<int> s;
    erep(i,0,N-3)if(S.substr(i,3)==target){
        ans++;
        s.insert(i);
    }
    //最初の位置がどこに来るかを記録
    map<char,int> m;
    m['A']=0;
    m['B']=1;
    m['C']=2;
    rep(i,0,Q){
        int x;char c;
        cin>>x>>c;
        x--;
        if(c!=S[x]){
            if(m.count(S[x])){
                int y=m[S[x]];
                if(0<=x-y&&x-y<N&&s.count(x-y)){
                    //s.erase(s.find(x-y));でもok
                    s.erase(x-y);
                    ans--;
                }
            }
            S[x]=c;
            if(m.count(S[x])){
                int y=m[S[x]];
                if(0<=x-y&&x-y<N&&S.substr(x-y,3)==target){
                    s.insert(x-y);
                    ans++;
                }
            }
        }
        cout<<ans<<"\n";
    }
    return 0;
}
