//エラトステネスの篩で引数以下の素数列を生成
vector<int> makepl(int n){
    vector<bool> rec(n+1,true);
    vector<int> ret;
    erep(i,2,n)if(rec[i]){
        ret.push_back(i);
        for(ll j=2;i*j<=n;j++)rec[i*j]=false;
    }
    return ret;
}

int main()
{
    cin.tie(nullptr);
    ios::sync_with_stdio(false);
    int T;
    cin>>T;
    vector<int> pl=makepl(2080083);
    rep(i,0,T){
        ll N,p,q;
        cin>>N;
        for(ll r:pl)if(N%r==0){
            N/=r;
            if(N%r){
                q=r;
                p=sqrt(N);
                while(p*p!=N){
                    if(p*p<N)p++;
                    else p--;
                }
            }else{
                p=r;
                q=N/r;
            }
            break;
        }
        cout<<p<<" "<<q<<"\n";
    }
    return 0;
}
