struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// ============================================================
// 数列・連結系テンプレ（非素数 mod 対応）
// ============================================================
//
// 含まれる関数
//
// sum_arithmetic  : 等差数列の和
// geom_sum_core   : r^n と (1+r+...+r^(n-1))
// sum_geometric   : 等比数列の和
// concat          : 2つの数の連結
// repeat_concat   : 同じ数をk回連結
//
// すべて非素数 mod でも使用可能
// ============================================================

// ============================================================
// 等差数列の和
// a + (a+d) + ... + (a+(n-1)d)
// 計算量: O(1)
// ============================================================
ll sum_arithmetic(ll a, ll d, ll n, ll mod){
    if(n == 0) return 0;
    __int128 term = (__int128)n * (2*a + (n-1)*d);
    term /= 2; // 必ず割り切れる
    return (ll)(term % mod);
}

// ============================================================
// 等比数列コア
// r^n と (1 + r + r^2 + ... + r^(n-1)) を同時計算
//
// 戻り値
// first  = r^n
// second = 1 + r + ... + r^(n-1)
//
// 計算量: O(log n)
// ============================================================
pair<ll,ll> geom_sum_core(ll r, ll n, ll mod){
    if(n == 0) return {1,0};
    auto [p, s] = geom_sum_core(r, n/2, mod);
    ll p2 = (__int128)p * p % mod;
    ll s2 = (__int128)s * (1 + p) % mod;
    if(n % 2 == 0){
        return {p2, s2};
    }else{
        ll p3 = (__int128)p2 * r % mod;
        ll s3 = (s2 + p2) % mod;
        return {p3, s3};
    }
}

// ============================================================
// 等比数列の和
// a + ar + ar^2 + ... + ar^(n-1)
// 計算量: O(log n)
// ============================================================
ll sum_geometric(ll a, ll r, ll n, ll mod){
    auto [p, s] = geom_sum_core(r % mod, n, mod);
    return (__int128)(a % mod) * s % mod;
}

// ============================================================
// 同じ数字 digit を n 回並べた数 mod
//
// 例
// digit=3 n=5 → 33333
// digit=7 n=4 → 7777
//
// 計算量: O(log n)
// 非素数 mod OK
// ============================================================
long long repunit_mod(long long digit, long long n, long long mod){
    digit %= mod;
    return digit * sum_geometric(1, 10, n, mod) % mod;
}

// ============================================================
// b の桁数を取得
// ============================================================
ll digit_count(ll x){
    if(x == 0) return 1;
    ll d = 0;
    while(x){
        d++;
        x/=10;
    }
    return d;
}

ll mod_pow(ll a,ll n,ll mod){
    ll r=1;
    while(n){
        if(n&1) r=(__int128)r*a%mod;
        a=(__int128)a*a%mod;
        n>>=1;
    }
    return r;
}

// ============================================================
// 数の連結
// concat(a,b)
//
// 例
// a=123 b=45
// → 12345
//
// 計算量: O(digit(b))
// ============================================================
ll concat(ll a, ll b, ll mod){
    ll d = digit_count(b);
    ll p = mod_pow(10,d, mod);
    return ((__int128)a * p + b) % mod;
}

// ============================================================
// 同じ数を k 回連結
//
// x=123 k=3
// → 123123123
//
// 計算量: O(log k)
// ============================================================
ll repeat_concat(ll x, ll k, ll mod){
    if(k == 0) return 0;
    ll d = digit_count(x);
    ll r = mod_pow(10,d, mod);
    return sum_geometric(x, r, k, mod);
}

void solv(){
    ll K,M;
    cin>>K>>M;

    ll mod = M*10007;

    ll Nmod = 0;

    for(int i=0;i<K;i++){

        ll c,l;
        cin>>c>>l;

        ll p=mod_pow(10,l,mod);

        Nmod = (__int128)Nmod*p%mod;

        Nmod = (Nmod + repunit_mod(c,l,mod))%mod;
    }

    cout << Nmod/M << "\n";
    return;
}
