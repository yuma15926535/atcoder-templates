#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor

#define YN {cout<<"Yes\n";}else{cout<<"No\n";}

template<class T> using PQ = priority_queue<T, vector<T>>;//大きい順
template<class T> using PQ_g = priority_queue<T, vector<T>, greater<T>>;//小さい順
#define chmax(x,y) x = max(x,y)
#define chmin(x,y) x = min(x,y)

int main(){
    int T;
    cin>>T;
    rep(i,0,T){
        int N;
        string S;
        cin>>N>>S;
        S='0'+S;
        unordered_map<ll,ll>m;
        auto pow2=[&](auto pow2,ll n)->ll{
            if(m.count(n))return m[n];
            if(n==0)return m[n]=1;
            return m[n]=pow2(pow2,n-1)*2;
        };
        //薬が追加される方向にしか遷移しないので小さい順にdpで結果を記録していけば再帰する必要がない
        vector<bool>dp(pow2(pow2,N),false);
        dp[0]=true;
        //ビット演算で集合を管理
        rep(i,0,pow2(pow2,N)){
            if(!dp[i])continue;
            rep(j,0,N){
                if(i&(1<<j))continue;
                int nxt=(i|pow2(pow2,j));
                if(S[nxt]=='0')dp[nxt]=true;
            }
        }
        if(dp[pow2(pow2,N)-1])YN;
    }
    return 0;
}
