//累積和(1D)
struct PrefixSum1D {
    int N;
    vector<long long> ps; // ps[i]: [0, i) の和

    // 0-index の配列を渡す
    PrefixSum1D(const vector<long long>& a) {
        N = a.size();
        ps.assign(N + 1, 0);
        for (int i = 0; i < N; i++) {
            ps[i + 1] = ps[i] + a[i];
        }
    }

    // ===== 0-index =====

    // 半開区間 [l, r)
    long long sum0_open(int l, int r) {
        return ps[r] - ps[l];
    }

    // 閉区間 [l, r]
    long long sum0_close(int l, int r) {
        return ps[r + 1] - ps[l];
    }

    // ===== 1-index =====

    // 半開区間 [l, r)（1-index）
    long long sum1_open(int l, int r) {
        // [l, r) -> [l-1, r-1)
        return ps[r - 1] - ps[l - 1];
    }

    // 閉区間 [l, r]（1-index）
    long long sum1_close(int l, int r) {
        // [l, r] -> [l-1, r)
        return ps[r] - ps[l - 1];
    }
};

//累積和(2D)
struct PrefixSum2D {
    int H, W;
    vector<vector<long long>> ps;

    // a は 0-index の H×W 行列
    PrefixSum2D(const vector<vector<long long>>& a) {
        H = a.size();
        W = a[0].size();
        ps.assign(H + 1, vector<long long>(W + 1, 0));

        for (int i = 0; i < H; i++) {
            for (int j = 0; j < W; j++) {
                ps[i + 1][j + 1] =
                    ps[i][j + 1] + ps[i + 1][j] - ps[i][j] + a[i][j];
            }
        }
    }

    // ===== 0-index =====

    // 半開区間 [x1, y1) × [x2, y2)
    long long sum0_open(int x1, int y1, int x2, int y2) {
        return ps[y1][y2]
             - ps[x1][y2]
             - ps[y1][x2]
             + ps[x1][x2];
    }

    // 閉区間 [x1, y1] × [x2, y2]
    long long sum0_close(int x1, int y1, int x2, int y2) {
        return sum0_open(x1, y1 + 1, x2, y2 + 1);
    }

    // ===== 1-index =====

    // 半開区間 [x1, y1) × [x2, y2)
    long long sum1_open(int x1, int y1, int x2, int y2) {
        return sum0_open(x1 - 1, y1 - 1, x2 - 1, y2 - 1);
    }

    // 閉区間 [x1, y1] × [x2, y2]
    long long sum1_close(int x1, int y1, int x2, int y2) {
        return sum0_open(x1 - 1, y1, x2 - 1, y2);
    }
};
