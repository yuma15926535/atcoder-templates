struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

vector<vector<int>> make_random_tree(
    int N,
    mt19937 &rng,
    bool shuffle_vertex = true
){
    vector<int> perm(N);
    iota(perm.begin(), perm.end(), 0);

    // 頂点番号をシャッフル
    if(shuffle_vertex){
        shuffle(perm.begin(), perm.end(), rng);
    }

    vector<vector<int>> G(N);

    for(int i = 1; i < N; i++){

        // [0, i-1] のどれかに接続
        int p = uniform_int_distribution<int>(0, i - 1)(rng);

        int u = perm[i];
        int v = perm[p];

        G[u].push_back(v);
        G[v].push_back(u);
    }

    return G;
}

struct TreeDP {

    int N;
    vector<vector<int>> g;

    // ★ dp0[v]
    // v を親とマッチさせない場合の最大 matching
    vector<int> dp0;

    // ★ dp1[v]
    // v を子の誰か1つとマッチさせる場合の最大 matching
    vector<int> dp1;

    // ★ 答え
    int ans;

    // コンストラクタ
    TreeDP(const vector<vector<int>>& graph)
        : g(graph)
    {
        N = g.size();

        // ★ 初期化
        dp0.assign(N, 0);
        dp1.assign(N, 0);

        ans = 0;
    }

    // ===== 木DP DFS =====
    void dfs(int v, int p){

        // ★ 子を自由に処理した総和
        int base = 0;

        for(int to : g[v]){

            if(to == p) continue;

            dfs(to, v);

            base += max(dp0[to], dp1[to]);
        }

        // ★ v を子とマッチさせない
        dp0[v] = base;

        // ★ v を子1つとマッチさせる
        dp1[v] = 0;

        for(int to : g[v]){

            if(to == p) continue;

            int cand = base;

            // ★ to を自由状態から外す
            cand -= max(dp0[to], dp1[to]);

            // ★ to は親(v)とマッチ済みなので
            // ★ 子側は dp0
            cand += dp0[to];

            // ★ 辺 (v,to) を使う
            cand += 1;

            dp1[v] = max(dp1[v], cand);
        }
    }

    // 実行
    void build(int root=0){

        dfs(root, -1);

        ans = max(dp0[root], dp1[root]);
    }
};

// ===== ランダムテスト =====
void random_test(){

    mt19937 rng(
        chrono::steady_clock::now().time_since_epoch().count()
    );

    for(int tc = 0; tc < 10000; tc++){

        // ===== ランダム木生成 =====
        int N = uniform_int_distribution<int>(1, 20)(rng);

        auto G = make_random_tree(N, rng);

        // ===== root =====
        int root =
            uniform_int_distribution<int>(0, N - 1)(rng);

        // ===== 木DP実行 =====
        TreeDP tree_dp(G);

        tree_dp.build(root);

        // ===== 辺一覧 =====
        vector<pair<int,int>> edges;

        for(int v = 0; v < N; v++){

            for(int to : G[v]){

                if(v < to){

                    edges.push_back({v, to});
                }
            }
        }

        int M = edges.size();

        // ===== 全探索 =====
        int naive = 0;

        for(int bit = 0; bit < (1 << M); bit++){

            vector<int> used(N, 0);

            bool ok = true;

            int cnt = 0;

            for(int i = 0; i < M; i++){

                if(((bit >> i) & 1) == 0) continue;

                auto [u, v] = edges[i];

                // ★ matching 条件
                if(used[u] || used[v]){

                    ok = false;
                    break;
                }

                used[u] = 1;
                used[v] = 1;

                cnt++;
            }

            if(ok){

                naive = max(naive, cnt);
            }
        }

        // ===== 比較 =====
        if(tree_dp.ans != naive){

            cout << "NG\n";

            cout << "N = " << N << "\n";

            cout << "root = " << root << "\n";

            cout << "TreeDP = "
                 << tree_dp.ans
                 << "\n";

            cout << "Naive = "
                 << naive
                 << "\n";

            cout << "\nEdges\n";

            for(auto [u, v] : edges){

                cout << u
                     << " "
                     << v
                     << "\n";
            }

            return;
        }
    }

    cout << "Random Test OK\n";
}

void test(){

    /*
            0
           / \
          1   2
             / \
            3   4

        最大 matching の一例:
        (0,1)
        (2,3)

        matching 数 = 2
    */

    int N = 5;

    vector<vector<int>> G(N);

    auto add_edge = [&](int u, int v){

        G[u].push_back(v);
        G[v].push_back(u);
    };

    add_edge(0, 1);
    add_edge(0, 2);
    add_edge(2, 3);
    add_edge(2, 4);

    // ===== 木DP実行 =====
    TreeDP tree_dp(G);

    tree_dp.build(0);

    // ===== 出力 =====
    cout << "最大 matching 数 : "
         << tree_dp.ans
         << "\n";

    cout << "\n";

    cout << "各頂点のDP\n";

    for(int v = 0; v < N; v++){

        cout << "頂点 "
             << v
             << " : ";

        cout << "マッチしない = "
             << tree_dp.dp0[v]
             << " , ";

        cout << "子とマッチする = "
             << tree_dp.dp1[v]
             << "\n";
    }
}
