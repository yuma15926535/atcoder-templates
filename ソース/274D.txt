#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor

#define YN {cout<<"Yes\n";}else{cout<<"No\n";}

template<class T> using PQ = priority_queue<T, vector<T>>;//大きい順

int main(){
    int N,x,y;
    cin>>N>>x>>y;
    vector<ll>A(N);
    rep(i,0,N)cin>>A[i];
    //計算量
    vector<bool>dp1(2e4+5,false);
    vector<bool>dp2(2e4+5,false);
    vector<bool>dp(2e4+5,false);
    int base=1e4+2;
    dp1[base+A[0]]=true;
    dp2[base]=true;
    rep(i,1,N){
        dp.assign(2e4+5,false);
        if(i%2){
            rep(j,0,2e4+5)if(dp2[j]){
                dp[j-A[i]]=true;
                dp[j+A[i]]=true;
            }
            dp2=dp;
        }else{
            rep(j,0,2e4+5)if(dp1[j]){
                dp[j-A[i]]=true;
                dp[j+A[i]]=true;
            }
            dp1=dp;
        }
    }
    if(dp1[x+base]&&dp2[y+base])YN;
    return 0;
}
