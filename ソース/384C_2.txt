#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

struct cp{
    int p;
    char f;
};
//比較関数を用意
//trueの方が先に並ぶ
bool pc(pair<int,string> p,pair<int,string> q){
    if(p.first!=q.first)return p.first>q.first;
    return p.second.compare(q.second)<0;
}

int main()
{
    vector<cp> v(5);
    char f='A';
    rep(i,0,5){
        int p;
        cin>>p;
        cp &q=v[i];
        q.f=f;q.p=p;
        f+=1;
    }
    vector<pair<int,string>> vp;
    rep(i,1,1<<5){
        string s="";
        int point=0;
        rep(j,0,5)if(((i>>j)&1)==1){
            s+=v[j].f;
            point+=v[j].p;
        }
        vp.push_back(make_pair(point,s));
    }
    //sortで比較関数を指定することで昇順、降順が混ざっていてもok
    sort(nall(vp),pc);
    rep(i,0,31)cout<<vp[i].second<<"\n";
    return 0;
}
