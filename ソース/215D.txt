#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる

//エラトステネスの篩で引数以下の素数列を生成
vector<int> makepl(int n){
    vector<bool> rec(n+1,true);
    vector<int> ret;
    erep(i,2,n)if(rec[i]){
        ret.push_back(i);
        for(ll j=2;i*j<=n;j++)rec[i*j]=false;
    }
    return ret;
}
vector<int> pl;
//引数の素因数列を生成
vector<int> pfac(int a){
    vector<int> ret;
    for(int p:pl){
        if(p*p>a)break;
        if(a%p==0){
            ret.push_back(p);
            while(a%p==0)a/=p;
        }
    }
    if(a!=1)ret.push_back(a);
    return ret;
}

int main()
{
    int N,M;
    cin>>N>>M;
    pl=makepl(317);
    vector<bool> v(M+1,true);
    int sz=M;
    //複数のgcdをを素因数に分けて考えるのは典型
    //各素因数で割れなければgcdは1
    rep(i,0,N){
        int a;
        cin>>a;
        vector<int> fac=pfac(a);
        for(int p:fac){
            if(!v[p])continue;
            for(int n=1;p*n<=M;n++){
                if(v[p*n]){
                    sz--;
                    v[p*n]=false;
                }
            }
        }
    }
    cout<<sz<<"\n";
    erep(i,1,M)if(v[i])cout<<i<<"\n";
    return 0;
}
