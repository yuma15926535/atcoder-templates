template<class T> using PQ = priority_queue<T, vector<T>>;//大きい順

#define chmax(x,y) x = max(x,y)
#define chmin(x,y) x = min(x,y)
#define popcount __builtin_popcountll

int main(){
    //木の辺彩色
    int N;
    cin >> N;
    vector<vector<pair<int,int>>> G(N);
    rep(i,0,N-1){
        int a,b;
        cin >> a >> b;
        --a; --b;
        G[a].push_back({b,i});
        G[b].push_back({a,i});
    }

    vector<int> color(N-1, -1);
    int K = 0;
    rep(i,0,N) chmax(K, (int)G[i].size());

    queue<int> q;
    q.push(0);
    //BFSで頂点に来た時使われている色は親の1色だけ
    //1色の記録だけでいいのでvector
    vector<int> parent_color(N, -1);

    while(!q.empty()){
        int v = q.front(); q.pop();
        int c = 1;
        for(auto [to,id]: G[v]){
            if(color[id] != -1) continue;
            //親の色を避ける
            if(c == parent_color[v]) c++;
            color[id] = c;
            //qの初期値が1点なので根付き木のように考えられて
            //toのparent_colorがqに入っている頂点の彩色によって変化することはない
            //変化すれば0->u->toと0->v->toが別経路で存在し閉路となるので矛盾
            parent_color[to] = c;
            q.push(to);
            c++;
        }
    }

    cout << K << '\n';
    rep(i,0,N-1) cout << color[i] << '\n';

    return 0;
}
