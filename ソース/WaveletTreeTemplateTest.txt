/*
==================================================
 Wavelet Tree（静的区間クエリ）
==================================================

【概要】
値で分割するセグ木のような構造。
各ノードで「左に行く個数の累積(mid)」を持つことで、
区間クエリを O(log σ) で処理できる。

【できること】
- kth(l, r, k)              : k番目（0-index）
- count(l, r, x)            : xの個数
- count_leq(l, r, x)        : x以下の個数
- count_less(l, r, x)       : x未満の個数
- count_range(l, r, a, b)   : [a,b] の個数
- prev_value(l, r, x)       : 最大の ≤ x
- next_value(l, r, x)       : 最小の ≥ x
- range_kth(l, r, a, b, k)  : [a,b] 内でk番目

【計算量】
- build : O(n log σ)
- query : O(log σ)

【前提・仕様】
- 配列は静的（更新なし）
- 区間は [l, r)
- kth は 0-index
- 座圧は内部で自動処理

【重要ポイント】
- mid[i] = [0,i) のうち「左に行く個数」
- 右側に行くときは index を変換する：
    left_l  = mid[l]
    right_l = l - mid[l]
- nulloptの例外処理：if(res)でelseに行く。結果は*resで使う。

【用途】
- 区間kth
- 区間freq
- 値域制限クエリ

【注意】
- 動的更新は不可
- distinct数には向かない（Moの方が強い）
==================================================
*/
template<typename T>
struct WaveletTree {
    struct Node {
        int lo, hi;
        vector<int> mid; // 左に行く個数の累積
        Node *l = nullptr, *r = nullptr;

        Node(int lo, int hi): lo(lo), hi(hi) {}
    };

    Node* root;
    vector<T> xs; // 復元用

    // build：値域 [lo,hi] で再帰構築
    // mid[i] = [0,i) のうち左に行く個数
    Node* build(vector<int>& a, int lo, int hi) {
        Node* node = new Node(lo, hi);
        int n = a.size();

        node->mid.resize(n + 1);
        node->mid[0] = 0;

        if (lo == hi) {
            for (int i = 0; i < n; i++) {
                node->mid[i+1] = node->mid[i] + 1;
            }
            return node;
        }

        int midv = (lo + hi) >> 1;

        vector<int> left, right;
        left.reserve(n);
        right.reserve(n);

        for (int i = 0; i < n; i++) {
            if (a[i] <= midv) {
                left.push_back(a[i]);
                node->mid[i+1] = node->mid[i] + 1;
            } else {
                right.push_back(a[i]);
                node->mid[i+1] = node->mid[i];
            }
        }

        if (!left.empty()) node->l = build(left, lo, midv);
        if (!right.empty()) node->r = build(right, midv+1, hi);

        return node;
    }

    // ---- constructor (vector<T>) ----
    WaveletTree(const vector<T>& a) {
        xs = a;
        sort(xs.begin(), xs.end());
        xs.erase(unique(xs.begin(), xs.end()), xs.end());

        vector<int> comp(a.size());
        for (int i = 0; i < (int)a.size(); i++) {
            comp[i] = lower_bound(xs.begin(), xs.end(), a[i]) - xs.begin();
        }

        root = build(comp, 0, (int)xs.size()-1);
    }

    // ---- constructor (string) ----
    WaveletTree(const string& s) {
        vector<char> a(s.begin(), s.end());
        *this = WaveletTree<char>(a);
    }

    int count(Node* node, int l, int r, int x) {
        if (!node || l >= r) return 0;

        if (node->lo == node->hi) {
            return r - l;
        }

        int midv = (node->lo + node->hi) >> 1;

        int left_l = node->mid[l];
        int left_r = node->mid[r];

        if (x <= midv) {
            return count(node->l, left_l, left_r, x);
        } else {
            int right_l = l - left_l;
            int right_r = r - left_r;
            return count(node->r, right_l, right_r, x);
        }
    }

    int kth(Node* node, int l, int r, int k) {
        if (node->lo == node->hi) {
            return node->lo;
        }

        int left_l = node->mid[l];
        int left_r = node->mid[r];
        int left_cnt = left_r - left_l;

        if (k < left_cnt) {
            return kth(node->l, left_l, left_r, k);
        } else {
            int right_l = l - left_l;
            int right_r = r - left_r;
            return kth(node->r, right_l, right_r, k - left_cnt);
        }
    }

    int count_leq(Node* node, int l, int r, int x) {
        if (!node || l >= r) return 0;

        // 範囲外
        if (x < node->lo) return 0;

        // 全部OK
        if (node->hi <= x) return r - l;

        int left_l = node->mid[l];
        int left_r = node->mid[r];

        int res = 0;

        if (node->l) {
            res += count_leq(node->l, left_l, left_r, x);
        }

        if (node->r) {
            int right_l = l - left_l;
            int right_r = r - left_r;
            res += count_leq(node->r, right_l, right_r, x);
        }

        return res;
    }

    // count(l,r,x)：xの個数（存在しない場合は0）
    int count(int l, int r, T x) {
        int id = lower_bound(xs.begin(), xs.end(), x) - xs.begin();

        if (id == (int)xs.size() || xs[id] != x) return 0;

        return count(root, l, r, id);
    }

    // kth(l,r,k)：k番目（0-index）
    // 範囲外は例外
    T kth(int l, int r, int k) {
        int len = r - l;
        if (k < 0 || k >= len) {
            throw out_of_range("kth: k out of range");
        }

        int id = kth(root, l, r, k);
        return xs[id];
    }

    // range_kth：区間 [a,b] に制限したkth
    // base + k で通常kthに変換
    // 0-index
    optional<T> range_kth(int l, int r, T a, T b, int k) {
        int cnt = count_range(l, r, a, b);
        if (k < 0 || k >= cnt) return nullopt;

        int base = 0;
        auto it = lower_bound(xs.begin(), xs.end(), a);
        if (it != xs.begin()) {
            T prev = *(it - 1);
            base = count_leq(l, r, prev);
        }

        return kth(l, r, base + k);
    }

    // count_leq(l,r,x)：x以下の個数（upper_bound使用）
    int count_leq(int l, int r, T x) {
        if (xs.empty()) return 0;

        int id = upper_bound(xs.begin(), xs.end(), x) - xs.begin() - 1;

        if (id < 0) return 0;

        return count_leq(root, l, r, id);
    }

    // count_less(l,r,x)：x未満の個数（lower_bound→1つ前）
    int count_less(int l, int r, T x) {
        auto it = lower_bound(xs.begin(), xs.end(), x);
        if (it == xs.begin()) return 0;
        T prev = *(it - 1);
        return count_leq(l, r, prev);
    }

    // count_range(l,r,a,b)：[a,b] の個数
    // = count_leq(b) - count_less(a)
    int count_range(int l, int r, T a, T b) {
        if (l >= r) return 0;
        if (a > b) return 0;

        int right = count_leq(l, r, b);

        int left = 0;
        auto it = lower_bound(xs.begin(), xs.end(), a);
        if (it != xs.begin()) {
            T prev = *(it - 1);
            left = count_leq(l, r, prev);
        }

        return right - left;
    }

    // prev_value：最大の ≤ x（なければnullopt）
    optional<T> prev_value(int l, int r, T x) {
        int cnt = count_leq(l, r, x);
        if (cnt == 0) return nullopt;
        return kth(l, r, cnt - 1);
    }

    // next_value：最小の ≥ x（なければnullopt）
    optional<T> next_value(int l, int r, T x) {
        int less = 0;

        auto it = lower_bound(xs.begin(), xs.end(), x);
        if (it != xs.begin()) {
            T prev = *(it - 1);
            less = count_leq(l, r, prev);
        }

        int len = r - l;
        if (less == len) return nullopt;

        return kth(l, r, less);
    }
};

template<typename T>
bool check_node(typename WaveletTree<T>::Node* node) {
    if (!node) return true;

    int n = node->mid.size() - 1;

    // mid[0] == 0
    if (node->mid[0] != 0) return false;

    // 単調増加
    for (int i = 0; i < n; i++) {
        if (node->mid[i] > node->mid[i+1]) return false;
    }

    // 範囲チェック
    if (node->lo == node->hi) {
        // 全部左に行くはず
        if (node->mid[n] != n) return false;
    } else {
        int left_size = node->mid[n];
        int right_size = n - left_size;

        if (node->l) {
            if ((int)node->l->mid.size() - 1 != left_size) return false;
        }
        if (node->r) {
            if ((int)node->r->mid.size() - 1 != right_size) return false;
        }
    }

    return check_node<T>(node->l) && check_node<T>(node->r);
}

template<typename T>
int naive_count(const vector<T>& a, int l, int r, T x) {
    int cnt = 0;
    for (int i = l; i < r; i++) {
        if (a[i] == x) cnt++;
    }
    return cnt;
}

template<typename T>
T naive_kth(vector<T> a, int l, int r, int k) {
    vector<T> b;
    for (int i = l; i < r; i++) b.push_back(a[i]);
    sort(b.begin(), b.end());
    return b[k];
}

template<typename T>
int naive_count_leq(const vector<T>& a, int l, int r, T x) {
    int cnt = 0;
    for (int i = l; i < r; i++) {
        if (a[i] <= x) cnt++;
    }
    return cnt;
}

template<typename T>
int naive_count_range(const vector<T>& a, int l, int r, T x, T y) {
    int cnt = 0;
    for (int i = l; i < r; i++) {
        if (x <= a[i] && a[i] <= y) cnt++;
    }
    return cnt;
}

template<typename T>
optional<T> naive_prev(const vector<T>& a, int l, int r, T x) {
    optional<T> res;
    for (int i = l; i < r; i++) {
        if (a[i] <= x) {
            if (!res || a[i] > *res) res = a[i];
        }
    }
    return res;
}

template<typename T>
optional<T> naive_next(const vector<T>& a, int l, int r, T x) {
    optional<T> res;
    for (int i = l; i < r; i++) {
        if (a[i] >= x) {
            if (!res || a[i] < *res) res = a[i];
        }
    }
    return res;
}

template<typename T>
int naive_count_less(const vector<T>& a, int l, int r, T x) {
    int cnt = 0;
    for (int i = l; i < r; i++) {
        if (a[i] < x) cnt++;
    }
    return cnt;
}

template<typename T>
optional<T> naive_range_kth(vector<T> a, int l, int r, T x, T y, int k) {
    vector<T> b;
    for (int i = l; i < r; i++) {
        if (x <= a[i] && a[i] <= y) b.push_back(a[i]);
    }
    if (k < 0 || k >= (int)b.size()) return nullopt;
    sort(b.begin(), b.end());
    return b[k];
}

void random_test_count_less_range_kth() {
    for (int t = 0; t < 1000; t++) {
        int n = rand() % 50 + 1;

        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            a[i] = rand() % 20;
        }

        WaveletTree<int> wt(a);

        for (int q = 0; q < 200; q++) {
            int l = rand() % n;
            int r = rand() % n;
            if (l > r) swap(l, r);
            r++;

            int x = rand() % 25 - 5;
            int y = rand() % 25 - 5;
            if (x > y) swap(x, y);

            // --- count_less ---
            int e1 = naive_count_less(a, l, r, x);
            int a1 = wt.count_less(l, r, x);

            if (e1 != a1) {
                cout << "ERROR count_less\n";
                return;
            }

            // --- range_kth ---
            int cnt = naive_count_range(a, l, r, x, y);
            int k = (cnt == 0 ? 0 : rand() % (cnt + 2) - 1); // 敢えて範囲外も入れる

            auto e2 = naive_range_kth(a, l, r, x, y, k);
            auto a2 = wt.range_kth(l, r, x, y, k);

            if (e2 != a2) {
                cout << "ERROR range_kth\n";
                cout << "array: ";
                for (auto v : a) cout << v << " ";
                cout << "\n";

                cout << "l r x y k = "
                     << l << " " << r << " "
                     << x << " " << y << " "
                     << k << "\n";

                if (e2) cout << "expected = " << *e2 << "\n";
                else cout << "expected = null\n";

                if (a2) cout << "actual = " << *a2 << "\n";
                else cout << "actual = null\n";

                return;
            }

            // --- 整合性チェック（重要） ---
            int less = wt.count_less(l, r, x);
            int leq  = wt.count_leq(l, r, x);

            if (!(less <= leq)) {
                cout << "ERROR consistency less<=leq\n";
                return;
            }

            int range_cnt = wt.count_range(l, r, x, y);
            int check = wt.count_leq(l, r, y) - wt.count_less(l, r, x);

            if (range_cnt != check) {
                cout << "ERROR consistency range\n";
                return;
            }
        }
    }
    cout << "OK count_less & range_kth\n";
}

void demo_wavelet_tree() {
    vector<int> a = {5, 1, 4, 2, 3, 2, 5};
    WaveletTree<int> wt(a);

    int l = 1, r = 6; // [1,6) = {1,4,2,3,2}

    cout << "array: ";
    for (auto x : a) cout << x << " ";
    cout << "\n";

    cout << "range = [" << l << "," << r << ")\n";

    // --- kth ---
    cout << "kth:\n";
    for (int k = 0; k < r-l; k++) {
        cout << k << " -> " << wt.kth(l,r,k) << "\n";
    }

    // --- count ---
    cout << "count(2) = " << wt.count(l,r,2) << "\n";

    // --- leq / less ---
    cout << "count_leq(3) = " << wt.count_leq(l,r,3) << "\n";
    cout << "count_less(3) = " << wt.count_less(l,r,3) << "\n";

    // --- range ---
    cout << "count_range(2,4) = " << wt.count_range(l,r,2,4) << "\n";

    // --- prev / next ---
    auto p = wt.prev_value(l,r,3);
    auto n = wt.next_value(l,r,3);

    cout << "prev(3) = " << (p ? to_string(*p) : "null") << "\n";
    cout << "next(3) = " << (n ? to_string(*n) : "null") << "\n";

    // --- range_kth ---
    cout << "range_kth [2,4]:\n";
    for (int k = 0; k < 10; k++) {
        auto v = wt.range_kth(l,r,2,4,k);
        if (!v) break;
        cout << k << " -> " << *v << "\n";
    }
}

void demo_wavelet_tree_string() {
    string s = "bacbcab";
    WaveletTree<char> wt(s);

    int l = 1, r = 6; // [1,6) = "acbca"

    cout << "string: " << s << "\n";
    cout << "range = [" << l << "," << r << ") -> ";
    for (int i = l; i < r; i++) cout << s[i];
    cout << "\n";

    // --- kth ---
    cout << "kth:\n";
    for (int k = 0; k < r - l; k++) {
        cout << k << " -> " << wt.kth(l, r, k) << "\n";
    }

    // --- count ---
    cout << "count('c') = " << wt.count(l, r, 'c') << "\n";

    // --- leq / less ---
    cout << "count_leq('c') = " << wt.count_leq(l, r, 'c') << "\n";
    cout << "count_less('c') = " << wt.count_less(l, r, 'c') << "\n";

    // --- range ---
    cout << "count_range('b','c') = " << wt.count_range(l, r, 'b', 'c') << "\n";

    // --- prev / next ---
    auto p = wt.prev_value(l, r, 'c');
    auto n = wt.next_value(l, r, 'c');

    cout << "prev('c') = " << (p ? string(1, *p) : "null") << "\n";
    cout << "next('c') = " << (n ? string(1, *n) : "null") << "\n";

    // --- range_kth ---
    cout << "range_kth ['b','c']:\n";
    for (int k = 0; k < 10; k++) {
        auto v = wt.range_kth(l, r, 'b', 'c', k);
        if (!v) break;
        cout << k << " -> " << *v << "\n";
    }
}
