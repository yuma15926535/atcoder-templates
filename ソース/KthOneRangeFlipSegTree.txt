/*
=====================================================
★ ノード情報（区間反転 + k-th One）
=====================================================

0/1配列を管理

sum : 区間内の1の個数
len : 区間長

=====================================================
*/

struct S {
    int sum;
    int len;
};

/*
=====================================================
★ 更新情報
=====================================================

false : no-op
true  : flip (0↔1)

=====================================================
*/

using F = bool;

/*
=====================================================
★ 演算定義
=====================================================
*/

// ノード結合
S combine(S a, S b) {
    return {
        a.sum + b.sum,
        a.len + b.len
    };
}

// 更新適用
S apply(F f, S x) {

    if (!f)
        return x;

    return {
        x.len - x.sum,
        x.len
    };
}

// 更新の合成
F compose(F f, F g) {

    return f ^ g;
}

S idS() {
    return {0,0};
}

F idF() {
    return false;
}

/*
=====================================================
Lazy SegTree 本体
=====================================================
*/

struct LazySegTree {

    int n;

    vector<S> seg;

    vector<F> lazy;

    LazySegTree(const vector<int>& v) {

        int sz = v.size();

        n = 1;

        while (n < sz)
            n <<= 1;

        seg.assign(2 * n, idS());

        lazy.assign(2 * n, idF());

        for (int i = 0; i < sz; i++)
            seg[n + i] = {v[i],1};

        for (int i = sz; i < n; i++)
            seg[n + i] = {0,1};

        for (int i = n - 1; i > 0; i--)
            seg[i] = combine(seg[i << 1], seg[i << 1 | 1]);
    }

    void apply_node(int k, F f) {

        seg[k] = apply(f, seg[k]);

        if (k < n)
            lazy[k] = compose(f, lazy[k]);
    }

    void push(int k) {

        if (lazy[k] == idF())
            return;

        apply_node(k << 1, lazy[k]);

        apply_node(k << 1 | 1, lazy[k]);

        lazy[k] = idF();
    }

    void range_apply(int l, int r, F f, int k, int a, int b) {

        if (r <= a || b <= l)
            return;

        if (l <= a && b <= r) {
            apply_node(k, f);
            return;
        }

        push(k);

        int mid = (a + b) / 2;

        range_apply(l, r, f, k << 1, a, mid);

        range_apply(l, r, f, k << 1 | 1, mid, b);

        seg[k] = combine(seg[k << 1], seg[k << 1 | 1]);
    }

    // ========================
    // 区間反転 [l,r)
    // ========================
    void range_flip(int l, int r) {
        range_apply(l, r, true, 1, 0, n);
    }

    S range(int l, int r, int k, int a, int b) {

        if (r <= a || b <= l)
            return idS();

        if (l <= a && b <= r)
            return seg[k];

        push(k);

        int mid = (a + b) / 2;

        return combine(
            range(l, r, k << 1, a, mid),
            range(l, r, k << 1 | 1, mid, b)
        );
    }

    // ========================
    // 区間和 [l,r)
    // ========================
    int range_sum(int l, int r) {
        return range(l, r, 1, 0, n).sum;
    }

    // ========================
    // k-th One
    // ========================
    // 1-indexed
    // k番目の1の位置
    // 存在しないなら -1
    // ========================
    int kth_one(int k) {

        if (seg[1].sum < k)
            return -1;

        int i = 1;

        while (i < n) {

            push(i);

            if (seg[i << 1].sum >= k) {
                i <<= 1;
            }
            else {
                k -= seg[i << 1].sum;
                i = i << 1 | 1;
            }
        }

        return i - n;
    }
};

void random_test() {

    mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());

    rep(_,0,10000){

        int N = rng()%20 + 1;

        vector<int> a(N);

        rep(i,0,N)
            a[i] = rng()%2;

        LazySegTree seg(a);

        rep(q,0,1000){

            int type = rng()%3;

            // ========================
            // 区間反転
            // ========================
            if(type==0){

                int l = rng()%N;
                int r = rng()%N;

                if(l>r)
                    swap(l,r);

                r++;

                rep(i,l,r)
                    a[i] ^= 1;

                seg.range_flip(l,r);
            }

            // ========================
            // 区間和
            // ========================
            else if(type==1){

                int l = rng()%N;
                int r = rng()%N;

                if(l>r)
                    swap(l,r);

                r++;

                int ans1 = 0;

                rep(i,l,r)
                    ans1 += a[i];

                int ans2 = seg.range_sum(l,r);

                if(ans1!=ans2){

                    cerr << "range_sum error\n";

                    cerr << "N = " << N << "\n";

                    cerr << "array : ";

                    for(auto x:a)
                        cerr << x << " ";

                    cerr << "\n";

                    cerr << "l r = " << l << " " << r << "\n";

                    cerr << "naive = " << ans1 << "\n";

                    cerr << "seg   = " << ans2 << "\n";

                    exit(0);
                }
            }

            // ========================
            // kth_one
            // ========================
            else{

                int total = 0;

                for(auto x:a)
                    total += x;

                int k = rng()%(total+2);

                if(k==0)
                    k=1;

                int ans1 = -1;

                if(k<=total){

                    int cnt = 0;

                    rep(i,0,N){

                        if(a[i])
                            cnt++;

                        if(cnt==k){
                            ans1 = i;
                            break;
                        }
                    }
                }

                int ans2 = seg.kth_one(k);

                if(ans1!=ans2){

                    cerr << "kth_one error\n";

                    cerr << "N = " << N << "\n";

                    cerr << "array : ";

                    for(auto x:a)
                        cerr << x << " ";

                    cerr << "\n";

                    cerr << "k = " << k << "\n";

                    cerr << "naive = " << ans1 << "\n";

                    cerr << "seg   = " << ans2 << "\n";

                    exit(0);
                }
            }
        }
    }

    cerr << "OK\n";
}
