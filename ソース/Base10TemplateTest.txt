struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
============================================================
 n進数テンプレ（最小構成）
------------------------------------------------------------
 1. to_base     : 10進 → base文字列
 2. from_base   : base文字列 → 10進
 3. leq_base    : s(base) ≤ limit 判定（オーバーフロー対策）
 4. digits      : 桁配列取得
 5. digit_sum   : 桁和
============================================================
*/

//10進 → base
string to_base(long long x, int base) {
    if (x == 0) return "0";
    string s;
    while (x > 0) {
        s += char('0' + x % base);
        x /= base;
    }
    reverse(s.begin(), s.end());
    return s;
}

//base → 10進
long long from_base(const string& s, int base) {
    long long x = 0;
    for (char c : s) {
        x = x * base + (c - '0');
    }
    return x;
}

//base表現 ≤ limit 判定
bool leq_base(const string& s, long long limit, long long base) {
    long long x = 0;
    for (char c : s) {
        int d = c - '0';
        if (x > limit / base) return false;
        x = x * base + d;
        if (x > limit) return false;
    }
    return true;
}

//桁取得
//11進法以上には注意する
vector<int> digits(long long x, int base) {
    vector<int> d;
    while (x) {
        d.push_back(x % base);
        x /= base;
    }
    reverse(d.begin(), d.end());
    return d;
}

//桁和
long long digit_sum(long long x, int base) {
    long long s = 0;
    while (x) {
        s += x % base;
        x /= base;
    }
    return s;
}

void solv(){
    cout << "==== n進数テンプレ 動作確認 ====\n\n";

    //--------------------------------------------------------
    // 1. to_base
    //--------------------------------------------------------
    cout << "[to_base]\n";
    cout << "13 を 2進数 → " << to_base(13, 2) << "\n";
    cout << "255 を 16進数 → " << to_base(255, 16) << "\n\n";

    //--------------------------------------------------------
    // 2. from_base
    //--------------------------------------------------------
    cout << "[from_base]\n";
    cout << "1101 (base2) → " << from_base("1101", 2) << "\n";
    cout << "123 (base4) → " << from_base("123", 4) << "\n\n";

    //--------------------------------------------------------
    // 3. digits
    //--------------------------------------------------------
    cout << "[digits]\n";
    auto d = digits(29, 3);
    cout << "29 を 3進数の桁配列 → ";
    for (int x : d) cout << x << " ";
    cout << "\n\n";

    //--------------------------------------------------------
    // 4. digit_sum
    //--------------------------------------------------------
    cout << "[digit_sum]\n";
    cout << "29 の 3進数桁和 → " << digit_sum(29, 3) << "\n\n";

    //--------------------------------------------------------
    // 5. leq_base
    //--------------------------------------------------------
    cout << "[leq_base]\n";
    string s = "123";
    long long limit = 100;

    cout << "123(base4) <= 100 ? → ";
    cout << (leq_base(s, limit, 4) ? "Yes" : "No") << "\n";

    cout << "123(base5) <= 100 ? → ";
    cout << (leq_base(s, limit, 5) ? "Yes" : "No") << "\n";

    cout << "999999999999999999(base10) <= 1000000000000000000 ? → ";
    cout << (leq_base("999999999999999999", 1e18, 10) ? "Yes" : "No") << "\n";

    cout << "999999999999999999(base11) <= 1000000000000000000 ? → ";
    cout << (leq_base("999999999999999999", 1e18, 11) ? "Yes" : "No") << "\n";

    cout << "\n==== 動作確認終了 ====\n";
    return;
}
