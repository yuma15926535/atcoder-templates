#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    int N,Q;
    cin>>N>>Q;
    vector<ll> A(N),K(Q),ans(Q);
    vector<int> id(Q);
    rep(i,0,N)cin>>A[i];
    rep(i,0,Q)cin>>K[i];
    //階段状に初期化
    iota(nall(id),0);
    //sort関数を外に置かずソート
    sort(nall(id),[&](int a,int b){return K[a]<K[b];});
    int t=0;
    ll sum=0,pre=0;
    //クエリ先読みで計算量を減らす
    //ソートした順番に解くことで計算量を減らすことができる
    rep(i,0,N){
        while(t<Q&&K[id[t]]<=sum+A[i]-pre-1){
            ans[id[t]]=K[id[t]]-sum+pre;
            t++;
        }
        sum+=A[i]-pre-1;
        pre=A[i];
    }
    rep(i,t,Q)ans[id[i]]=K[id[i]]-sum+pre;
    rep(i,0,Q)cout<<ans[i]<<"\n";
    return 0;
}
