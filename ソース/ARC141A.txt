    if constexpr(I<sizeof...(Ts)){

    }
}

template<class T>
std::enable_if_t<is_iter_v<T>,ostream&>
operator<<(ostream&os,const T&v){
    using V = std::decay_t<decltype(*begin(v))>;
    if constexpr(is_iter_v<V>){
        bool first_row=true;
        os<<"\n";
        for(auto&&row:v){
            if(!first_row) os<<"\n";
            first_row=false;
            bool f=0;
            for(auto&&x:row){
                if(f) os<<" ";
                os<<x;
                f=1;
            }
        }
    }
    else{
        os<<"[";
        bool f=0;
        for(auto&&x:v){
            if(f) os<<", ";
            os<<x;
            f=1;
        }
        os<<"]";
    }
    return os;
}
void _dbg(const char*){
    cerr<<DBG_COLOR<<"========================================\n"<<DBG_RESET;
}
template<class T,class...U>
void _dbg(const char*names,T&&x,U&&...y){
    const char*comma=strchr(names,',');
    if(!comma){
        cerr<<DBG_COLOR<<names<<" = "<<x<<"\n"<<DBG_RESET;
    }else{
        cerr<<DBG_COLOR;
        cerr.write(names,comma-names)<<" = "<<x<<" | ";
        _dbg(comma+1,y...);
    }
}

//部分文字列の数値化
inline long long parse(const string& s, int pos, int len) {
    long long x = 0;
    for (int i = pos, end = pos + len; i < end; i++)
        x = x * 10 + (s[i] - '0');
    return x;
}
//数値の結合
inline long long concat_number(long long A, long long B, int digitsB = -1) {
    long long p = 1;
    if (digitsB < 0) {
        do { p *= 10; } while (B >= p);
    } else {
        while (digitsB--) p *= 10;
    }
    return A * p + B;
}

}
