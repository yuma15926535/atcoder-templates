struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
==================================================
 永続セグメント木（区間k番目・個数）
==================================================

【概要】
値を座圧し、「出現回数」を管理する永続セグメント木。
prefixごとの状態を保持し、差分を取ることで区間クエリを実現する。

【できること】
・区間 [l, r) の k番目の値（kth）
・区間 [l, r) の「x以下の個数」

【内部の考え方】
roots[i] は「prefix [0, i) の頻度」
→ 区間 [l, r) = roots[r] - roots[l]

【計算量】
構築：O(N log N)
クエリ：O(log N)
（二分探索版kthは O(log^2 N)）

【注意】
・kは1-index（1 ≤ k ≤ r-l）
・区間は [l, r)
・座圧後のindexで処理し、最後に値へ復元する

==================================================
*/
struct PersistentKth {
    struct Node {
        int val;
        Node *l, *r;
        Node(int v=0, Node* l=nullptr, Node* r=nullptr)
            : val(v), l(l), r(r) {}
    };

    int n;                  // 座圧後サイズ
    vector<int> xs;         // index → 元の値
    vector<Node*> roots;    // roots[i] = prefix [0, i)

    // --- セグ木の初期構築（全て0） ---
    Node* build(int l, int r) {
        if (l + 1 == r) return new Node(0);
        int mid = (l + r) >> 1;
        Node* left = build(l, mid);
        Node* right = build(mid, r);
        return new Node(0, left, right);
    }

    // --- posにdelta加算（非破壊） ---
    Node* add(Node* node, int l, int r, int pos, int delta) {
        if (l + 1 == r) {
            return new Node(node->val + delta);
        }
        int mid = (l + r) >> 1;
        if (pos < mid) {
            Node* nl = add(node->l, l, mid, pos, delta);
            return new Node(nl->val + node->r->val, nl, node->r);
        } else {
            Node* nr = add(node->r, mid, r, pos, delta);
            return new Node(node->l->val + nr->val, node->l, nr);
        }
    }

    // --- kth（差分）---
    // a=roots[l], b=roots[r]
    // 区間[l, r)のk番目（indexを返す）
    int kth(Node* a, Node* b, int l, int r, int k) {
        if (l + 1 == r) return l;
        int mid = (l + r) >> 1;
        int cnt = b->l->val - a->l->val;
        if (k <= cnt) return kth(a->l, b->l, l, mid, k);
        else return kth(a->r, b->r, mid, r, k - cnt);
    }

    int count_leq(Node* node, int l, int r, int qr) {
        // [l, r) のうち [0, qr) の個数
        if (qr <= l) return 0;
        if (r <= qr) return node->val;
        int mid = (l + r) >> 1;
        return count_leq(node->l, l, mid, qr)
             + count_leq(node->r, mid, r, qr);
    }

    // --- x以下の個数（差分） ---
    int count_leq(Node* a, Node* b, int l, int r, int qr) {
        if (qr <= l) return 0;
        if (r <= qr) return b->val - a->val;
        int mid = (l + r) >> 1;
        return count_leq(a->l, b->l, l, mid, qr)
             + count_leq(a->r, b->r, mid, r, qr);
    }

    int kth_by_bs(int l, int r, int k) {
        int low = 0, high = n - 1;
        int ans = n - 1;

        while (low <= high) {
            int mid = (low + high) >> 1;

            int cnt = count_leq(roots[l], roots[r], 0, n, mid + 1);

            if (cnt >= k) {
                ans = mid;
                high = mid - 1;
            } else {
                low = mid + 1;
            }
        }

        return xs[ans];
    }

    // --- 配列から構築 ---
    // 座圧 + prefix構築
    void build_from_array(const vector<int>& a) {
        // 座圧
        xs = a;
        sort(xs.begin(), xs.end());
        xs.erase(unique(xs.begin(), xs.end()), xs.end());
        n = xs.size();

        auto get_id = [&](int x) {
            return lower_bound(xs.begin(), xs.end(), x) - xs.begin();
        };

        // roots構築
        roots.clear();
        roots.push_back(build(0, n));

        for (int i = 0; i < (int)a.size(); i++) {
            int id = get_id(a[i]);
            roots.push_back(add(roots.back(), 0, n, id, 1));
        }
    }

    // --- クエリ：k番目 ---
    // 区間 [l, r) の k番目（1-index）
    int query_kth(int l, int r, int k) {
        int idx = kth(roots[l], roots[r], 0, n, k);
        return xs[idx];
    }

    // --- クエリ：x以下の個数 ---
    int query_leq(int l, int r, int x) {
        int id = upper_bound(xs.begin(), xs.end(), x) - xs.begin();
        return count_leq(roots[l], roots[r], 0, n, id);
    }

    // --- クエリ：二分探索kth ---
    // count_leqを使ってkthを再現（理解用）
    int query_kth_bs(int l, int r, int k) {
        int low = 0, high = n - 1;
        int ans = n - 1;

        while (low <= high) {
            int mid = (low + high) >> 1;
            int cnt = count_leq(roots[l], roots[r], 0, n, mid + 1);

            if (cnt >= k) {
                ans = mid;
                high = mid - 1;
            } else {
                low = mid + 1;
            }
        }

        return xs[ans];
    }
};

mt19937 rng(0);

void random_test_step4_final() {
    const int N = 50;
    const int Q = 1000;
    const int MAXV = 100;

    vector<int> a(N);
    for (int i = 0; i < N; i++) {
        a[i] = rng() % MAXV;
    }

    PersistentKth pst;
    pst.build_from_array(a);

    for (int q = 0; q < Q; q++) {
        int l = rng() % N;
        int r = rng() % N;
        if (l > r) swap(l, r);
        r++;

        int len = r - l;
        int k = rng() % len + 1;

        int res = pst.query_kth(l, r, k);

        // naive
        vector<int> v;
        for (int i = l; i < r; i++) v.push_back(a[i]);
        sort(v.begin(), v.end());
        int naive = v[k-1];

        if (res != naive) {
            cout << "Mismatch!\n";
            cout << "l r k = " << l << " " << r << " " << k << "\n";
            cout << "seg = " << res << " naive = " << naive << "\n";
            exit(0);
        }
    }

    cout << "Final OK" << endl;
}

void random_test_count_leq() {
    const int N = 50;
    const int Q = 1000;
    const int MAXV = 100;

    mt19937 rng(0);

    vector<int> a(N);
    for (int i = 0; i < N; i++) {
        a[i] = rng() % MAXV;
    }

    PersistentKth pst;
    pst.build_from_array(a);

    for (int q = 0; q < Q; q++) {
        int l = rng() % N;
        int r = rng() % N;
        if (l > r) swap(l, r);
        r++;

        int x = rng() % MAXV;

        int res = pst.query_leq(l, r, x);

        // naive
        int naive = 0;
        for (int i = l; i < r; i++) {
            if (a[i] <= x) naive++;
        }

        if (res != naive) {
            cout << "Mismatch!\n";
            cout << "l r x = " << l << " " << r << " " << x << "\n";
            cout << "seg = " << res << " naive = " << naive << "\n";
            exit(0);
        }
    }

    cout << "count_leq OK" << endl;
}

void random_test_kth_bs() {
    const int N = 50;
    const int Q = 1000;
    const int MAXV = 100;

    mt19937 rng(0);

    vector<int> a(N);
    for (int i = 0; i < N; i++) {
        a[i] = rng() % MAXV;
    }

    PersistentKth pst;
    pst.build_from_array(a);

    for (int q = 0; q < Q; q++) {
        int l = rng() % N;
        int r = rng() % N;
        if (l > r) swap(l, r);
        r++;

        int len = r - l;
        int k = rng() % len + 1;

        int res1 = pst.query_kth(l, r, k);
        int res2 = pst.query_kth_bs(l, r, k);

        if (res1 != res2) {
            cout << "Mismatch!\n";
            cout << "l r k = " << l << " " << r << " " << k << "\n";
            cout << "direct = " << res1 << " bs = " << res2 << "\n";
            exit(0);
        }
    }

    cout << "kth_bs OK" << endl;
}

void demo_persistent_kth() {
    cout << "==== 永続セグ木（kth）の動作確認 ====\n";

    vector<int> a = {5, 1, 4, 2, 3};
    cout << "元の配列: ";
    for (int x : a) cout << x << " ";
    cout << "\n";

    PersistentKth pst;
    pst.build_from_array(a);

    int l = 1, r = 5; // [1,5)
    cout << "\n区間 [" << l << ", " << r << ") を考える\n";

    cout << "部分配列: ";
    for (int i = l; i < r; i++) cout << a[i] << " ";
    cout << "\n";

    // kth
    for (int k = 1; k <= r - l; k++) {
        int val = pst.query_kth(l, r, k);
        cout << k << "番目の値: " << val << "\n";
    }

    // count_leq
    cout << "\n--- x以下の個数 ---\n";
    for (int x = 0; x <= 5; x++) {
        int cnt = pst.query_leq(l, r, x);
        cout << x << "以下の個数: " << cnt << "\n";
    }

    // kth（二分探索版）
    cout << "\n--- kth（二分探索版） ---\n";
    for (int k = 1; k <= r - l; k++) {
        int val = pst.query_kth_bs(l, r, k);
        cout << k << "番目（二分探索）: " << val << "\n";
    }

    cout << "==== 確認終了 ====\n";
}
