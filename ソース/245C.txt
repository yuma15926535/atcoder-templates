#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int N,K;
vector<int> A,B;

int main()
{
    cin>>N>>K;
    A.resize(N);
    B.resize(N);
    rep(i,0,N)cin>>A[i];
    rep(i,0,N)cin>>B[i];
    const int a=-1,b=1,c=0,d=2,e=-2;
    vector<int> dp(N,e);
    dp[0]=c;
    rep(i,1,N){
        if(dp[i-1]==c){
            if(abs(A[i-1]-A[i])<=K||abs(B[i-1]-A[i])<=K)dp[i]=a;
            if(abs(A[i-1]-B[i])<=K||abs(B[i-1]-B[i])<=K){
                if(dp[i]==a)dp[i]=c;
                else dp[i]=b;
            }
        }else if(dp[i-1]==a){
            if(abs(A[i-1]-A[i])<=K)dp[i]=a;
            if(abs(A[i-1]-B[i])<=K){
                if(dp[i]==a)dp[i]=c;
                else dp[i]=b;
            }
        }else if(dp[i-1]==b){
            if(abs(B[i-1]-A[i])<=K)dp[i]=a;
            if(abs(B[i-1]-B[i])<=K){
                if(dp[i]==a)dp[i]=c;
                else dp[i]=b;
            }
        }
        if(dp[i]==e)dp[i]=d;
    }
    if(dp[N-1]!=d)YN;
    return 0;
}
