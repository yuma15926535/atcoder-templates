struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// odd[i]  : iを中心とする回文半径(中心を含む)
//           回文長 = odd[i]*2-1
//
// even[i] : i-1,i間を中心とする回文半径
//           回文長 = even[i]*2
//
// 計算量 O(N)
pair<vector<int>,vector<int>> manacher_raw(const string& S){
    int N=sizei(S);

    vector<int> odd(N),even(N);

    // odd
    for(int l=0,r=-1,i=0;i<N;i++){
        int k=(i>r?1:min(odd[l+r-i],r-i+1));
        while(0<=i-k&&i+k<N&&S[i-k]==S[i+k])k++;
        odd[i]=k--;
        if(i+k>r){
            l=i-k;
            r=i+k;
        }
    }

    // even
    for(int l=0,r=-1,i=0;i<N;i++){
        int k=(i>r?0:min(even[l+r-i+1],r-i+1));
        while(0<=i-k-1&&i+k<N&&S[i-k-1]==S[i+k])k++;
        even[i]=k--;
        if(i+k>r){
            l=i-k-1;
            r=i+k;
        }
    }

    return {odd,even};
}

// Manacher Algorithm
//
// 構築 O(N)
//
// odd_radius[i]  : iを中心とする奇数長回文の半径
// even_radius[i] : i-1,i間を中心とする偶数長回文の半径
// odd_length[i]  : 奇数長回文の最大長
// even_length[i] : 偶数長回文の最大長
//
// is_palindrome(l,r) : S[l,r]が回文か判定 O(1)
struct Manacher{
    vector<int> odd_radius;
    vector<int> even_radius;

    vector<int> odd_length;
    vector<int> even_length;

    Manacher(const string& S){
        tie(odd_radius,even_radius)=manacher_raw(S);

        int N=sizei(S);
        odd_length.resize(N);
        even_length.resize(N);

        rep(i,N){
            odd_length[i]=odd_radius[i]*2-1;
            even_length[i]=even_radius[i]*2;
        }
    }

    // S[l,r]が回文か
    bool is_palindrome(int l,int r)const{
        int len=r-l+1;
        if(len&1){
            int c=(l+r)/2;
            return odd_radius[c]>=len/2+1;
        }else{
            int c=(l+r+1)/2;
            return even_radius[c]>=len/2;
        }
    }
};

bool is_palindrome_naive(const string& S,int l,int r){
    while(l<r){
        if(S[l]!=S[r])return false;
        l++;
        r--;
    }
    return true;
}

void random_test_manacher(){
    mt19937 rng(random_device{}());

    for(int tc=0;tc<100000;tc++){
        int N=uniform_int_distribution<int>(0,20)(rng);

        string S;
        for(int i=0;i<N;i++){
            S.push_back(char('a'+uniform_int_distribution<int>(0,2)(rng)));
        }

        Manacher man(S);

        // is_palindrome
        for(int l=0;l<N;l++){
            for(int r=l;r<N;r++){
                bool ans1=man.is_palindrome(l,r);
                bool ans2=is_palindrome_naive(S,l,r);

                if(ans1!=ans2){
                    cerr<<"is_palindrome failed\n";
                    cerr<<"S = "<<S<<"\n";
                    cerr<<"l = "<<l<<" r = "<<r<<"\n";
                    assert(false);
                }
            }
        }

        // odd
        for(int c=0;c<N;c++){
            int rad=0;
            while(c-rad>=0&&c+rad<N&&S[c-rad]==S[c+rad])rad++;

            if(man.odd_radius[c]!=rad){
                cerr<<"odd_radius failed\n";
                cerr<<"S = "<<S<<"\n";
                cerr<<"center = "<<c<<"\n";
                cerr<<"expected = "<<rad<<"\n";
                cerr<<"actual = "<<man.odd_radius[c]<<"\n";
                assert(false);
            }

            if(man.odd_length[c]!=rad*2-1){
                cerr<<"odd_length failed\n";
                cerr<<"S = "<<S<<"\n";
                cerr<<"center = "<<c<<"\n";
                assert(false);
            }
        }

        // even
        for(int c=0;c<N;c++){
            int rad=0;
            while(c-rad-1>=0&&c+rad<N&&S[c-rad-1]==S[c+rad])rad++;

            if(man.even_radius[c]!=rad){
                cerr<<"even_radius failed\n";
                cerr<<"S = "<<S<<"\n";
                cerr<<"center = "<<c<<"\n";
                cerr<<"expected = "<<rad<<"\n";
                cerr<<"actual = "<<man.even_radius[c]<<"\n";
                assert(false);
            }

            if(man.even_length[c]!=rad*2){
                cerr<<"even_length failed\n";
                cerr<<"S = "<<S<<"\n";
                cerr<<"center = "<<c<<"\n";
                assert(false);
            }
        }
    }

    cerr<<"ALL TEST PASSED\n";
}

// odd_radius[i] >= k
//     → S[i-k+1 ～ i+k-1] が回文
//
// even_radius[i] >= k
//     → S[i-k ～ i+k-1] が回文
//
// is_palindrome(l,r)
//     → S[l,r] が回文か O(1)
void debug_manacher(const string& S){
    Manacher man(S);

    cout<<"文字列: "<<S<<"\n\n";

    cout<<"奇数長回文\n";
    cout<<"中心  半径  長さ\n";
    rep(i,sizei(S)){
        cout<<setw(4)<<i
            <<setw(6)<<man.odd_radius[i]
            <<setw(6)<<man.odd_length[i]
            <<"\n";
    }

    cout<<"\n偶数長回文\n";
    cout<<"中心(左|右)  半径  長さ\n";
    rep(i,sizei(S)){
        cout<<setw(5)<<i-1<<"|"<<i
            <<setw(6)<<man.even_radius[i]
            <<setw(6)<<man.even_length[i]
            <<"\n";
    }

    cout<<"\n回文となる区間一覧\n";
    rep(l,sizei(S)){
        rep(r,l,sizei(S)){
            if(man.is_palindrome(l,r)){
                cout<<"["<<l<<","<<r<<"] : "
                    <<S.substr(l,r-l+1)
                    <<"\n";
            }
        }
    }
}
