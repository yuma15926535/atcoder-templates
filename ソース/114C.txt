struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// ========================
// 状態遷移DFSテンプレート（753数）
// ========================
// 【変えるところ】
// - Stateの中身
// - 遷移（forの中身）
// - 初期状態
// - 処理（条件判定）
// - 枝刈り条件

struct State {
    long long val; // 作っている数
    bool f3, f5, f7; // 3,5,7を使ったか

    // ※問題によっては
    // - 桁数
    // - mod
    // - bitmask
    // などを追加
};

long long N;
long long ans = 0;

void dfs(State cur) {
    // 【任意】枝刈り
    if (cur.val > N) return;

    // 【変える】処理（条件を満たすか）
    if (cur.f3 && cur.f5 && cur.f7) {
        ans++;
    }

    // --- 遷移 ---
    // 【変える】追加する候補
    // 例：3,5,7 を追加
    {
        State nxt = cur;
        nxt.val = cur.val * 10 + 3;
        nxt.f3 = true;
        dfs(nxt);
    }
    {
        State nxt = cur;
        nxt.val = cur.val * 10 + 5;
        nxt.f5 = true;
        dfs(nxt);
    }
    {
        State nxt = cur;
        nxt.val = cur.val * 10 + 7;
        nxt.f7 = true;
        dfs(nxt);
    }
}

void solv(){
    cin >> N;

    // 【変える】初期状態
    State init;
    init.val = 0;
    init.f3 = init.f5 = init.f7 = false;

    dfs(init);

    cout << ans << endl;
    return;
}

int main(){
    BUFF;
    int T=1;
    //cin>>T;
    while(T--)solv();
    return 0;
}
