struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

template<class T>
struct BaseCounter {

    unordered_map<T,long long> cnt;

    long long total_size = 0;
    long long distinct_size = 0;

protected:
    /*
        Hook関数
        派生Layerが更新検知に使う
    */
    virtual void on_update(
        const T&, long long, long long){}

public:

    // =============================
    // insert
    // =============================
    void insert(const T& x, long long k=1){
        if(k<=0) return;

        long long old = cnt[x];
        long long now = old + k;

        cnt[x] = now;

        if(old==0) distinct_size++;
        total_size += k;

        on_update(x, old, now);
    }

    // =============================
    // erase
    // =============================
    void erase(const T& x, long long k=1){

        auto it = cnt.find(x);
        if(it==cnt.end()) return;

        long long old = it->second;
        long long dec = min(k, old);
        long long now = old - dec;

        total_size -= dec;

        if(now==0){
            cnt.erase(it);
            distinct_size--;
        }else{
            it->second = now;
        }

        on_update(x, old, now);
    }

    long long count(const T& x) const{
        auto it=cnt.find(x);
        return it==cnt.end()?0:it->second;
    }

    bool exist(const T& x) const{
        return count(x)>0;
    }

    long long size() const{
        return total_size;
    }

    long long distinct() const{
        return distinct_size;
    }

    bool empty() const{
        return total_size==0;
    }
};

void solv(){
    int H,W,N,h,w;
    cin>>H>>W>>N>>h>>w;
    vector<vector<int>>A(H,vector<int>(W));
    rep(i,0,H)cin>>A[i];
    BaseCounter<int> C;
    vector<vector<int>>ans(H-h+1,vector<int>(W-w+1));
    {
        BaseCounter<int> tmp;
        erep(i,0,H-h)erep(j,0,W-w){
            if(j==0){
                if(i==0){
                    rep(k,0,H)rep(l,0,W)if(k>=h||l>=w)C.insert(A[k][l]);
                    tmp=C;
                }else{
                    C=tmp;
                    rep(k,0,w){
                        C.insert(A[i-1][j+k]);
                        C.erase(A[i+h-1][j+k]);
                    }
                    tmp=C;
                }
            }else{
                debug();
                rep(k,0,h){
                    C.insert(A[i+k][j-1]);
                    C.erase(A[i+k][j+w-1]);
                }
            }
            ans[i][j]=C.distinct_size;
        }
    }
    erep(i,0,H-h){
        erep(j,0,W-w)cout<<ans[i][j]<<' ';
        cout<<'\n';
    }
    return;
}
