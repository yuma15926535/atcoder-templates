struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

constexpr int MOD = 998244353;
using mint = static_modint<MOD>;

// Linear Recurrence
// a[i] = c[0] * a[i-1] + ... + c[d-1] * a[i-d]
// 第k項をBostan-Moriで求める
//
// constructor : O(d)
// kth         : O(M(d) log k)
//   M(d): d次多項式の乗算
//
// c : 漸化式の係数
// a : 初期値 a[0], ..., a[d-1]
struct LinearRecurrence {
    vector<mint> c, a;

    LinearRecurrence(const vector<int>& C, const vector<int>& A)
        : c(C.begin(), C.end()), a(A.begin(), A.end()) {}

    int kth(ll k) {
        int d = c.size();

        // 初期値
        if (k < d) return a[k].val();

        // Q(x) = 1 - c[0]x - c[1]x^2 - ... - c[d-1]x^d
        vector<mint> Q(d + 1);
        Q[0] = 1;
        for (int i = 0; i < d; i++) {
            Q[i + 1] = -c[i];
        }

        // P(x) = A(x)Q(x) mod x^d
        auto tmp = convolution(a, Q);
        vector<mint> P(tmp.begin(), tmp.begin() + d);

        // [x^k] P(x)/Q(x) を求める
        while (k > 0) {
            // Q(-x)
            vector<mint> Qneg = Q;
            for (int i = 1; i < (int)Qneg.size(); i += 2) {
                Qneg[i] = -Qneg[i];
            }

            // P(x)Q(-x)
            auto PQ = convolution(P, Qneg);

            // Q(x)Q(-x)
            auto QQ = convolution(Q, Qneg);

            // kの偶奇に応じてPを抽出
            vector<mint> nP;
            int parity = k & 1;
            for (int i = parity; i < (int)PQ.size(); i += 2) {
                nP.push_back(PQ[i]);
            }

            // Qは偶数次数だけ抽出
            vector<mint> nQ;
            for (int i = 0; i < (int)QQ.size(); i += 2) {
                nQ.push_back(QQ[i]);
            }

            P = move(nP);
            Q = move(nQ);

            k >>= 1;
        }

        return (P[0] / Q[0]).val();
    }
};

void test_LinearRecurrence_Random() {
    mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());

    constexpr int TEST = 10000;

    for (int tc = 1; tc <= TEST; tc++) {
        // 小さい次数をランダム生成
        int d = rng() % 8 + 1;

        // 係数・初期値をランダム生成
        vector<int> c(d), a(d);

        for (int i = 0; i < d; i++) {
            c[i] = rng() % 11 - 5;  // [-5, 5]
            a[i] = rng() % 11 - 5;  // [-5, 5]
        }

        LinearRecurrence lr(c, a);

        // 小さいkは愚直計算
        int N = 100;

        vector<mint> naive(N + 1);
        for (int i = 0; i < d; i++) {
            naive[i] = a[i];
        }

        for (int i = d; i <= N; i++) {
            for (int j = 0; j < d; j++) {
                naive[i] += mint(c[j]) * naive[i - j - 1];
            }
        }

        // 0～Nをすべて比較
        for (int k = 0; k <= N; k++) {
            int expected = naive[k].val();
            int actual = lr.kth(k);

            if (expected != actual) {
                cout << "ランダムテスト失敗\n";
                cout << "テストケース: " << tc << '\n';
                cout << "次数: " << d << '\n';

                cout << "係数: ";
                for (int x : c) cout << x << ' ';
                cout << '\n';

                cout << "初期値: ";
                for (int x : a) cout << x << ' ';
                cout << '\n';

                cout << "k: " << k << '\n';
                cout << "期待値: " << expected << '\n';
                cout << "実際値: " << actual << '\n';

                return;
            }
        }
    }

    cout << "ランダムテスト: ALL TEST PASSED\n";
}

void test_LinearRecurrence() {
    cout << "=== LinearRecurrence 動作確認 ===\n";

    // Fibonacci
    // a[n] = a[n-1] + a[n-2]
    // a[0] = 0, a[1] = 1
    {
        LinearRecurrence lr({1, 1}, {0, 1});

        cout << "Fibonacci: ";
        for (int k = 0; k <= 10; k++) {
            cout << lr.kth(k);
            if (k != 10) cout << ' ';
        }
        cout << '\n';

        cout << "Fibonacci 第10項 = " << lr.kth(10) << '\n';
    }

    // a[n] = 2*a[n-1] + 3*a[n-2]
    // a[0] = 1, a[1] = 2
    {
        LinearRecurrence lr({2, 3}, {1, 2});

        cout << "a[n] = 2a[n-1] + 3a[n-2]: ";
        for (int k = 0; k <= 8; k++) {
            cout << lr.kth(k);
            if (k != 8) cout << ' ';
        }
        cout << '\n';
    }

    // 大きなkでも求められることを確認
    {
        LinearRecurrence lr({1, 1}, {0, 1});

        cout << "Fibonacci 第10^18項 = "
             << lr.kth(1000000000000000000LL) << '\n';
    }

    cout << "=== 動作確認終了 ===\n";
}
