    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/**
 * @brief 2D BIT (2次元 Fenwick Tree)
 * 
 * @details 2次元グリッド上の点加算、および矩形領域の総和 (矩形和) を高速に処理するデータ構造。
 * 外部APIはすべて 0-index かつ半開区間 [x1, x2) × [y1, y2) で統一している。
 * 
 * 2D Fenwick Tree の仕組み:
 * - 1次元の Fenwick Tree を入れ子にした構造。tree[i][j] は縦方向と横方向の双方について
 *   Fenwick Tree の定義に基づく区間の和を保持している。
 * 
 * 矩形和が包除原理で求められる理由:
 * - 領域 [x1, x2) × [y1, y2) の総和は、4つの左上隅からの累積和を用いて以下のように算出できる。
 *   Sum([x1, x2) × [y1, y2)) = Sum([0, x2) × [0, y2))
 *                           - Sum([0, x1) × [0, y2))
 *                           - Sum([0, x2) × [0, y1))
 *                           + Sum([0, x1) × [0, y1))
 * - これはベン図における面積の重なり合い（包除原理）と同様の理由に基づく。
 * 
 * 用途:
 * - 平面上の動的な点更新、および範囲クエリの処理
 * 
 * 計算量:
 * - 構築: O(H * W)
 * - 点加算 (add): O(log H * log W)
 * - 一点更新 (set): O(log H * log W)
 * - 一点取得 (get, operator()): O(log H * log W)
 * - 矩形和取得 (sum): O(log H * log W)
 * - クリア (clear): O(H * W)
 * 
 * API一覧:
 * - BIT2D(H, W), BIT2D(a)
 * - build(a), assign(a), clear()
 * - add(x, y, v), set(x, y, v)
 * - get(x, y), operator()(x, y), values()
 * - sum(x, y), sum(x1, y1, x2, y2), all_sum()
 * - height(), width(), empty()
 */
template <class T>
class BIT2D {
private:
    int h;
    int w;
    std::vector<std::vector<T>> tree;

public:
    /**
     * @brief コンストラクタ (サイズ指定)
     * @param H 行数 (縦サイズ)
     * @param W 列数 (横サイズ)
     * @note 計算量: O(H * W)
     */
    BIT2D(int H = 0, int W = 0) : h(H), w(W), tree(H + 1, std::vector<T>(W + 1, 0)) {}

    /**
     * @brief コンストラクタ (初期2次元配列指定)
     * @param a 初期2次元配列
     * @note 計算量: O(H * W)
     */
    explicit BIT2D(const std::vector<std::vector<T>>& a) {
        build(a);
    }

    /**
     * @brief 初期2次元配列からO(H * W)で構築
     * @param a 初期2次元配列
     * @note 計算量: O(H * W)。まず各行に対して1次元のO(W)構築を行い、その後各列に対して1次元のO(H)構築を行うことで、
     * 単純な逐次挿入 (O(HW log H log W)) より高速な線形時間構築を実現している。
     */
    void build(const std::vector<std::vector<T>>& a) {
        h = a.size();
        w = h > 0 ? a[0].size() : 0;
        tree.assign(h + 1, std::vector<T>(w + 1, 0));
        
        for (int i = 0; i < h; ++i) {
            for (int j = 0; j < w; ++j) {
                tree[i + 1][j + 1] = a[i][j];
            }
        }
        
        // 行方向の1次元構築 (各行に対して右へ伝播)
        for (int i = 1; i <= h; ++i) {
            for (int j = 1; j <= w; ++j) {
                int nj = j + (j & -j);
                if (nj <= w) {
                    tree[i][nj] += tree[i][j];
                }
            }
        }
        
        // 列方向の1次元構築 (各列に対して下へ伝播)
        for (int i = 1; i <= h; ++i) {
            int ni = i + (i & -i);
            if (ni <= h) {
                for (int j = 1; j <= w; ++j) {
                    tree[ni][j] += tree[i][j];
                }
            }
        }
    }

    /**
     * @brief 新しい配列で再初期化
     * @param a 新しい2次元配列
     * @note 計算量: O(H * W)
     */
    void assign(const std::vector<std::vector<T>>& a) {
        build(a);
    }

    /**
     * @brief 行数 (縦サイズ) を取得
     * @return 行数
     * @note 計算量: O(1)
     */
    int height() const {
        return h;
    }

    /**
     * @brief 列数 (横サイズ) を取得
     * @return 列数
     * @note 計算量: O(1)
     */
    int width() const {
        return w;
    }

    /**
     * @brief グリッドが空であるか判定
     * @return 行数または列数が0ならtrue、そうでなければfalse
     * @note 計算量: O(1)
     */
    bool empty() const {
        return h == 0 || w == 0;
    }

    /**
     * @brief すべての要素を0にする
     * @note 計算量: O(H * W)
     */
    void clear() {
        for (auto& row : tree) {
            std::fill(row.begin(), row.end(), 0);
        }
    }

    /**
     * @brief 位置 (x, y) に値 v を加算する (a[x][y] += v)
     * @param x 行インデックス (0-index)
     * @param y 列インデックス (0-index)
     * @param v 加算する値
     * @note 計算量: O(log H * log W)
     */
    void add(int x, int y, T v) {
        assert(0 <= x && x < h && 0 <= y && y < w);
        for (int i = x + 1; i <= h; i += i & -i) {
            for (int j = y + 1; j <= w; j += j & -j) {
                tree[i][j] += v;
            }
        }
    }

    /**
     * @brief 位置 (x, y) の値を v に更新する (a[x][y] = v)
     * @param x 行インデックス (0-index)
     * @param y 列インデックス (0-index)
     * @param v 更新する値
     * @note 計算量: O(log H * log W)
     */
    void set(int x, int y, T v) {
        assert(0 <= x && x < h && 0 <= y && y < w);
        T cur = get(x, y);
        add(x, y, v - cur);
    }

    /**
     * @brief 位置 (x, y) の値を取得する (a[x][y])
     * @param x 行インデックス (0-index)
     * @param y 列インデックス (0-index)
     * @return 位置 (x, y) の値
     * @note 計算量: O(log H * log W)
     */
    T get(int x, int y) const {
        assert(0 <= x && x < h && 0 <= y && y < w);
        return sum(x, y, x + 1, y + 1);
    }

    /**
     * @brief 位置 (x, y) の値を取得する (getと同じ)
     * @param x 行インデックス (0-index)
     * @param y 列インデックス (0-index)
     * @return 位置 (x, y) の値
     * @note 計算量: O(log H * log W)
     */
    T operator()(int x, int y) const {
        return get(x, y);
    }

    /**
     * @brief 領域 [0, x) × [0, y) の累積和を取得する
     * @param x 右端の行境界
     * @param y 右端の列境界
     * @return 累積和
     * @note 計算量: O(log H * log W)
     */
    T sum(int x, int y) const {
        assert(0 <= x && x <= h && 0 <= y && y <= w);
        T s = 0;
        for (int i = x; i > 0; i -= i & -i) {
            for (int j = y; j > 0; j -= j & -j) {
                s += tree[i][j];
            }
        }
        return s;
    }

    /**
     * @brief 矩形領域 [x1, x2) × [y1, y2) の総和を取得する
     * @param x1 左上行インデックス
     * @param y1 左上列インデックス
     * @param x2 右下行境界
     * @param y2 右下列境界
     * @return 矩形和
     * @note 計算量: O(log H * log W)
     */
    T sum(int x1, int y1, int x2, int y2) const {
        assert(0 <= x1 && x1 <= x2 && x2 <= h);
        assert(0 <= y1 && y1 <= y2 && y2 <= w);
        return sum(x2, y2) - sum(x1, y2) - sum(x2, y1) + sum(x1, y1);
    }

    /**
     * @brief グリッド全体の総和を取得する
     * @return 全体の総和
     * @note 計算量: O(log H * log W)
     */
    T all_sum() const {
        return sum(h, w);
    }

    /**
     * @brief 現在の全要素を表す2次元配列を取得する
     * @return 2次元配列
     * @note 計算量: O(H * W * log H * log W)
     */
    std::vector<std::vector<T>> values() const {
        std::vector<std::vector<T>> res(h, std::vector<T>(w));
        for (int i = 0; i < h; ++i) {
            for (int j = 0; j < w; ++j) {
                res[i][j] = get(i, j);
            }
        }
        return res;
    }
};

// 動作確認用テスト
void test_BIT2D() {
    std::cout << "--- test_BIT2D 開始 ---" << std::endl;

    std::vector<std::vector<int>> a = {
        {1, 2, 3},
        {4, 5, 6},
        {7, 8, 9}
    };
    BIT2D<int> bit(a);

    std::cout << "height(): " << bit.height() << " (期待値: 3)" << std::endl;
    std::cout << "width(): " << bit.width() << " (期待値: 3)" << std::endl;

    std::cout << "初期値:" << std::endl;
    for (int i = 0; i < 3; ++i) {
        for (int j = 0; j < 3; ++j) {
            std::cout << bit(i, j) << " ";
        }
        std::cout << std::endl;
    }

    std::cout << "sum(2, 2): " << bit.sum(2, 2) << " (期待値: 12) (1+2+4+5)" << std::endl;
    std::cout << "sum(1, 1, 3, 3): " << bit.sum(1, 1, 3, 3) << " (期待値: 28) (5+6+8+9)" << std::endl;
    std::cout << "all_sum(): " << bit.all_sum() << " (期待値: 45)" << std::endl;

    bit.add(1, 1, 10);
    std::cout << "add(1, 1, 10) 実行後 (位置 (1, 1) の 5 に 10 加算):" << std::endl;
    std::cout << "get(1, 1): " << bit.get(1, 1) << " (期待値: 15)" << std::endl;
    std::cout << "sum(1, 1, 3, 3): " << bit.sum(1, 1, 3, 3) << " (期待値: 38)" << std::endl;

    bit.set(1, 1, 5);
    std::cout << "set(1, 1, 5) 実行後 (位置 (1, 1) を 5 に更新):" << std::endl;
    std::cout << "get(1, 1): " << bit.get(1, 1) << " (期待値: 5)" << std::endl;

    bit.clear();
    std::cout << "clear() 実行後 all_sum(): " << bit.all_sum() << " (期待値: 0)" << std::endl;

    std::cout << "--- test_BIT2D 終了 ---" << std::endl << std::endl;
}

// ランダムテスト
void random_test_BIT2D() {
    std::cout << "--- random_test_BIT2D 開始 ---" << std::endl;

    std::mt19937 mt(42);
    int H = 30;
    int W = 40;
    int queries = 5000;

    std::vector<std::vector<long long>> naive(H, std::vector<long long>(W));
    std::uniform_int_distribution<long long> val_dist(-100, 100);
    for (int i = 0; i < H; ++i) {
        for (int j = 0; j < W; ++j) {
            naive[i][j] = val_dist(mt);
        }
    }

    BIT2D<long long> bit(naive);

    std::uniform_int_distribution<int> op_dist(0, 4);
    std::uniform_int_distribution<int> x_dist(0, H - 1);
    std::uniform_int_distribution<int> y_dist(0, W - 1);

    for (int q = 0; q < queries; ++q) {
        int op = op_dist(mt);
        if (op == 0) {
            // add
            int x = x_dist(mt);
            int y = y_dist(mt);
            long long v = val_dist(mt);
            naive[x][y] += v;
            bit.add(x, y, v);
        } else if (op == 1) {
            // set
            int x = x_dist(mt);
            int y = y_dist(mt);
            long long v = val_dist(mt);
            naive[x][y] = v;
            bit.set(x, y, v);
        } else if (op == 2) {
            // sum(x1, y1, x2, y2)
            int x1 = x_dist(mt);
            int x2 = x_dist(mt);
            int y1 = y_dist(mt);
            int y2 = y_dist(mt);
            if (x1 > x2) std::swap(x1, x2);
            if (y1 > y2) std::swap(y1, y2);
            x2++; // 半開区間化
            y2++;
            
            long long naive_sum = 0;
            for (int i = x1; i < x2; ++i) {
                for (int j = y1; j < y2; ++j) {
                    naive_sum += naive[i][j];
                }
            }
            assert(bit.sum(x1, y1, x2, y2) == naive_sum);
        } else if (op == 3) {
            // get / operator()
            int x = x_dist(mt);
            int y = y_dist(mt);
            assert(bit.get(x, y) == naive[x][y]);
            assert(bit(x, y) == naive[x][y]);
        } else {
            // values, all_sum
            std::vector<std::vector<long long>> b_vals = bit.values();
            assert(b_vals == naive);

            long long naive_all_sum = 0;
            for (int i = 0; i < H; ++i) {
                for (int j = 0; j < W; ++j) {
                    naive_all_sum += naive[i][j];
                }
            }
            assert(bit.all_sum() == naive_all_sum);
        }
    }

    std::cout << "OK" << std::endl;
    std::cout << "--- random_test_BIT2D 終了 ---" << std::endl << std::endl;
}

    test_BIT2D();

}
