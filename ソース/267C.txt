#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    cin.tie(nullptr);
    ios::sync_with_stdio(false);
    //階段状の和を高速で計算
    //答えが負になることもあるので0で初期化できない
    int N,M;
    cin>>N>>M;
    vector<ll> A(N),S(N-M+1,0);
    rep(i,0,N)cin>>A[i];
    rep(i,0,N-M+1){
        if(i==0)rep(j,0,M)S[0]+=A[j];
        else S[i]=S[i-1]-A[i-1]+A[i+M-1];
    }
    ll ans,rec=0;
    rep(i,0,N-M+1){
        if(i==0){
            rep(j,0,M)rec+=(j+1)*A[j];
            ans=rec;
        }else{
            rec=rec-S[i-1]+M*A[i+M-1];
            ans=max(ans,rec);
        }
    }
    cout<<ans<<"\n";
    return 0;
}
