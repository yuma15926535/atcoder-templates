template<class T> using PQ = priority_queue<T, vector<T>>;//大きい順
template<class T> using PQ_g = priority_queue<T, vector<T>, greater<T>>;//小さい順
#define chmax(x,y) x = max(x,y)
#define chmin(x,y) x = min(x,y)
#define popcount __builtin_popcountll

struct maxsum{
    PQ<ll> pq;
    ll ret=0;
    const ll yet=-1;
    int sz;
    maxsum(int K):sz(K){}
    ll solv(ll b){
        if(pq.size()<sz){
            pq.push(b);
            ret+=b;
        }else if(pq.top()>b){
            ret-=pq.top();
            pq.pop();
            pq.push(b);
            ret+=b;
        }
        if(pq.size()!=sz)return yet;
        return ret;
    }
};

int main(){
    BUFF;
    int T;
    cin>>T;
    //maxとsumはsumの方が扱いやすいのでmaxを固定(1元のみ固定なので固定しやすい、もある)
    rep(_,0,T){
        int N,K;
        cin>>N>>K;
        vector<ll>A(N),B(N);
        rep(i,0,N)cin>>A[i];
        rep(i,0,N)cin>>B[i];
        vector<int>idx(N);
        iota(nall(idx),0);
        sort(nall(idx),[&](int a,int b){return A[a]<A[b];});
        ll ans=1e18;
        maxsum ms(K);
        rep(i,0,N){
            ll num=ms.solv(B[idx[i]]);
            if(num!=-1)chmin(ans,A[idx[i]]*num);
        }
        cout<<ans<<'\n';
    }
    return 0;
}
