struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// ============================================================
// IndexSet（set版）
// 値 → その値を持つ index の集合を管理する
// 想定用途：
//   ・値の出現位置管理
//   ・区間内存在判定
//   ・最近距離取得
//   ・動的更新
// 計算量：基本すべて O(log N)
// ============================================================
template<class T = int>
struct IndexSet {

    // 値 → その値を持つ index の昇順集合
    map<T, set<int>> mp;

    // --------------------------------------------------------
    // 値 val を index idx に追加
    // --------------------------------------------------------
    void add(T val, int idx){
        mp[val].insert(idx);
    }

    // --------------------------------------------------------
    // 値 val の index idx を削除
    // （空になったら map からも消す）
    // --------------------------------------------------------
    void remove(T val, int idx){
        if(!mp.count(val)) return;
        mp[val].erase(idx);
        if(mp[val].empty()) mp.erase(val);
    }

    // --------------------------------------------------------
    // 値 val が存在するか
    // --------------------------------------------------------
    bool contains_value(T val) const {
        return mp.count(val);
    }

    // --------------------------------------------------------
    // 値 val の出現回数
    // --------------------------------------------------------
    int size(T val) const {
        if(!mp.count(val)) return 0;
        return mp.at(val).size();
    }

    // --------------------------------------------------------
    // 値 val の最小 index
    // 存在しない場合 -1
    // --------------------------------------------------------
    int min_index(T val) const {
        if(!mp.count(val)) return -1;
        return *mp.at(val).begin();
    }

    // --------------------------------------------------------
    // 値 val の最大 index
    // 存在しない場合 -1
    // --------------------------------------------------------
    int max_index(T val) const {
        if(!mp.count(val)) return -1;
        return *mp.at(val).rbegin();
    }

    // --------------------------------------------------------
    // 値 val が区間 [l, r) に存在するか
    // 超頻出機能
    // --------------------------------------------------------
    bool exist_in_range(T val, int l, int r) const {
        if(!mp.count(val)) return false;
        const auto& st = mp.at(val);
        auto it = st.lower_bound(l);
        return (it != st.end() && *it < r);
    }

    // --------------------------------------------------------
    // 値 val の index x 以上で最小の index
    // 存在しない場合 -1
    // --------------------------------------------------------
    int next_index(T val, int x) const {
        if(!mp.count(val)) return -1;
        const auto& st = mp.at(val);
        auto it = st.lower_bound(x);
        if(it == st.end()) return -1;
        return *it;
    }

    // --------------------------------------------------------
    // 値 val の index x より小さい最大の index
    // 存在しない場合 -1
    // --------------------------------------------------------
    int prev_index(T val, int x) const {
        if(!mp.count(val)) return -1;
        const auto& st = mp.at(val);
        auto it = st.lower_bound(x);
        if(it == st.begin()) return -1;
        --it;
        return *it;
    }

    // --------------------------------------------------------
    // 値 val の index x に最も近い index を返す
    // 存在しない場合 -1
    // --------------------------------------------------------
    int nearest_index(T val, int x) const {
        if(!mp.count(val)) return -1;
        const auto& st = mp.at(val);

        auto it = st.lower_bound(x);

        int ans = -1;

        // 右側候補
        if(it != st.end()){
            ans = *it;
        }

        // 左側候補
        if(it != st.begin()){
            int cand = *prev(it);
            if(ans == -1 || abs(cand - x) <= abs(ans - x)){
                ans = cand;
            }
        }

        return ans;
    }

    // --------------------------------------------------------
    // 値 val の区間 [l, r) 内の個数
    // ※ 注意：O(N) の可能性あり（distance使用）
    // --------------------------------------------------------
    int count_in_range(T val, int l, int r) const {
        if(!mp.count(val)) return 0;
        const auto& st = mp.at(val);
        return distance(st.lower_bound(l), st.lower_bound(r));
    }

};

template<class T = int>
struct NaiveIndexSet {
    map<T, set<int>> mp;

    void add(T val, int idx) {
        mp[val].insert(idx);
    }

    void remove(T val, int idx) {
        if (!mp.count(val)) return;
        mp[val].erase(idx);
        if (mp[val].empty()) mp.erase(val);
    }

    bool contains_value(T val) const {
        return mp.count(val) > 0;
    }

    int size(T val) const {
        if (!mp.count(val)) return 0;
        return (int)mp.at(val).size();
    }

    int min_index(T val) const {
        if (!mp.count(val)) return -1;
        return *mp.at(val).begin();
    }

    int max_index(T val) const {
        if (!mp.count(val)) return -1;
        return *mp.at(val).rbegin();
    }

    bool exist_in_range(T val, int l, int r) const {
        if (!mp.count(val)) return false;
        const auto& st = mp.at(val);
        auto it = st.lower_bound(l);
        return (it != st.end() && *it < r);
    }

    int next_index(T val, int x) const {
        if (!mp.count(val)) return -1;
        const auto& st = mp.at(val);
        auto it = st.lower_bound(x);
        if (it == st.end()) return -1;
        return *it;
    }

    int prev_index(T val, int x) const {
        if (!mp.count(val)) return -1;
        const auto& st = mp.at(val);
        auto it = st.lower_bound(x);
        if (it == st.begin()) return -1;
        --it;
        return *it;
    }

    int nearest_index(T val, int x) const {
        if (!mp.count(val)) return -1;
        const auto& st = mp.at(val);
        auto it = st.lower_bound(x);
        int ans = -1;
        if (it != st.end()) ans = *it;
        if (it != st.begin()) {
            int cand = *prev(it);
            if (ans == -1 || abs(cand - x) <= abs(ans - x)) {
                ans = cand;
            }
        }
        return ans;
    }

    int count_in_range(T val, int l, int r) const {
        if (!mp.count(val)) return 0;
        const auto& st = mp.at(val);
        int cnt = 0;
        for (int v : st) {
            if (l <= v && v < r) cnt++;
        }
        return cnt;
    }
};

void run_random_test(int num_ops, int max_val, int max_idx, uint32_t seed) {
    cout << "シード値 " << seed << " でランダムテストを開始します (試行回数: " << num_ops << ")..." << endl;

    IndexSet<int> iset;
    NaiveIndexSet<int> naive;
    mt19937 mt(seed);

    for (int i = 0; i < num_ops; ++i) {
        int op = mt() % 11;
        int val = mt() % max_val;
        int idx = mt() % max_idx;

        try {
            if (op == 0) { // add
                iset.add(val, idx);
                naive.add(val, idx);
            } else if (op == 1) { // remove
                iset.remove(val, idx);
                naive.remove(val, idx);
            } else if (op == 2) { // contains_value
                assert(iset.contains_value(val) == naive.contains_value(val));
            } else if (op == 3) { // size
                assert(iset.size(val) == naive.size(val));
            } else if (op == 4) { // min_index
                assert(iset.min_index(val) == naive.min_index(val));
            } else if (op == 5) { // max_index
                assert(iset.max_index(val) == naive.max_index(val));
            } else if (op == 6) { // exist_in_range
                int l = mt() % max_idx;
                int r = mt() % max_idx;
                if (l > r) swap(l, r);
                assert(iset.exist_in_range(val, l, r) == naive.exist_in_range(val, l, r));
            } else if (op == 7) { // next_index
                assert(iset.next_index(val, idx) == naive.next_index(val, idx));
            } else if (op == 8) { // prev_index
                assert(iset.prev_index(val, idx) == naive.prev_index(val, idx));
            } else if (op == 9) { // nearest_index
                assert(iset.nearest_index(val, idx) == naive.nearest_index(val, idx));
            } else if (op == 10) { // count_in_range
                int l = mt() % max_idx;
                int r = mt() % max_idx;
                if (l > r) swap(l, r);
                assert(iset.count_in_range(val, l, r) == naive.count_in_range(val, l, r));
            }
        } catch (...) {
            cout << "エラー発生: ループ " << i << ", 操作 " << op << ", 値 " << val << ", インデックス " << idx << endl;
            throw;
        }

        if ((i + 1) % 1000 == 0) {
            cout << (i + 1) << " 回の操作を完了..." << endl;
        }
    }

    cout << "テストが正常に終了しました！" << endl;
}
