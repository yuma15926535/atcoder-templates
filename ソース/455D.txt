struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
==================================================
 配列で作る双方向連結リスト
==================================================

【用途】
・順序を持つ要素の「つなぎ替え」
・区間を切り出して別の場所へ移動
・前後関係のみが重要（ランダムアクセス不要）

【計算量】
・link / cut / insert：すべて O(1)

【注意】
・番兵（ダミー）を使うとバグりにくい
==================================================
*/
struct LinkedList {
    int n;
    vector<int> nxt, prv;

    LinkedList(int n) : n(n), nxt(n+1, -1), prv(n+1, -1) {}

    // x の後ろに y をつなぐ（x -> y）
    void link(int x, int y) {
        nxt[x] = y;
        if (y != -1) prv[y] = x;
    }

    // x をリストから外す
    void cut(int x) {
        int p = prv[x];
        int q = nxt[x];
        if (p != -1) nxt[p] = q;
        if (q != -1) prv[q] = p;
        prv[x] = nxt[x] = -1;
    }

    // x の後ろに y を挿入（y は単体 or 別リストでもOK）
    void insert_after(int x, int y) {
        int z = nxt[x];
        link(x, y);
        link(y, z);
    }

    // 区間 [l, r] を切り出す（l と r は同一リスト内）
    // 戻り値：(前のノード, 次のノード)
    pair<int,int> cut_segment(int l, int r) {
        int p = prv[l];
        int q = nxt[r];
        if (p != -1) nxt[p] = q;
        if (q != -1) prv[q] = p;
        prv[l] = nxt[r] = -1;
        return {p, q};
    }

    // 区間 [l, r] を x の後ろに挿入
    void insert_segment(int x, int l, int r) {
        int z = nxt[x];
        link(x, l);
        link(r, z);
    }
};

void solv(){
    int N,Q;
    cin>>N>>Q;
    LinkedList llist(2*N);
    rep(i,0,N)llist.nxt[N+i]=i,llist.prv[i]=N+i;
    rep(i,0,Q){
        int C,P;
        cin>>C>>P;
        C--;P--; 
        int p=llist.prv[C];
        llist.prv[C]=P;
        llist.nxt[P]=C;
        llist.nxt[p]=-1;
    }
    rep(i,0,N){
        int x=N+i,ans=0;
        while(llist.nxt[x]!=-1){
            x=llist.nxt[x];
            ans++;
        }
        cout<<ans<<' ';
    }
    cout<<'\n';
    return;
}

    while(T--)solv();
