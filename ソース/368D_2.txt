int main(){
    int N,K;
    cin>>N>>K;
    vector<vector<int>>G(N);
    rep(i,0,N-1){
        int a,b;
        cin>>a>>b;
        a--;b--;
        G[a].push_back(b);
        G[b].push_back(a);
    }
    vector<bool>V(N,false);
    int v;
    rep(i,0,K){
        cin>>v;
        v--;
        V[v]=true;
    }
    vector<bool>ans(N,false);
    //木DP:dp配列にindexを根とする部分木の値を持たせる
    auto dfs=[&](int pos,int pre,auto dfs)->bool{
        bool ret=V[pos];
        for(int nxt:G[pos]){
            if(nxt==pre)continue;
            if(dfs(nxt,pos,dfs))ret=true;
        }
        ans[pos]=ret;
        return ret;
    };
    dfs(v,-1,dfs);
    cout<<count(nall(ans),true)<<'\n';
    return 0;
}
