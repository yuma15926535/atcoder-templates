struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

using S = ll;

// ★区間結合
S combine(S a, S b) {
    return max(a, b); // 区間max
}

// ★単位元
S id() {
    return -(1LL << 60);
}

/*
=====================================================
SegTree 本体
=====================================================
*/

struct SegTree {
    int n;
    vector<S> seg;

    SegTree(int sz) {
        n = 1;
        while (n < sz) n <<= 1;
        seg.assign(2*n, id());
    }

    // 配列初期化
    SegTree(const vector<S>& v) {
        int sz = v.size();

        n = 1;
        while (n < sz) n <<= 1;

        seg.assign(2*n, id());

        for (int i = 0; i < sz; i++)
            seg[n+i] = v[i];

        for (int i = n-1; i > 0; i--)
            seg[i] = combine(seg[i<<1], seg[i<<1|1]);
    }

    // ========================
    // 点更新
    // ========================
    void set(int i, S x) {
        i += n;

        seg[i] = x;

        while (i >>= 1)
            seg[i] = combine(seg[i<<1], seg[i<<1|1]);
    }

    // ========================
    // 区間取得 [l,r)
    // ========================
    S range(int l, int r) {
        S L = id(), R = id();

        for (l += n, r += n; l < r; l >>= 1, r >>= 1) {

            if (l & 1)
                L = combine(L, seg[l++]);

            if (r & 1)
                R = combine(seg[--r], R);
        }

        return combine(L, R);
    }

    S get(int i) {
        return seg[n+i];
    }

    S all() {
        return seg[1];
    }
};

void random_test() {
    random_device rd;
    mt19937 rng(rd());

    rep(tc, 0, 10000) {

        int N = rng() % 20 + 1;

        vector<ll> v(N);

        rep(i, 0, N)
            v[i] = (ll)(rng() % 200) - 100;

        SegTree seg(v);

        vector<ll> naive = v;

        rep(q, 0, 1000) {

            // 0:set
            // 1:range max
            int type = rng() % 2;

            // ========================
            // 点更新
            // ========================
            if (type == 0) {

                int idx = rng() % N;

                ll x = (ll)(rng() % 200) - 100;

                seg.set(idx, x);

                naive[idx] = x;
            }

            // ========================
            // 区間max
            // ========================
            else {

                int l = rng() % N;
                int r = rng() % N;

                if (l > r)
                    swap(l, r);

                r++;

                ll ans1 = seg.range(l, r);

                ll ans2 = -(1LL << 60);

                rep(i, l, r)
                    ans2 = max(ans2, naive[i]);

                if (ans1 != ans2) {

                    cout << "WA\n";

                    cout << "N = " << N << '\n';

                    cout << "array:\n";

                    for (auto x : naive)
                        cout << x << ' ';

                    cout << '\n';

                    cout << "l r = " << l << ' ' << r << '\n';

                    cout << "seg = " << ans1 << '\n';

                    cout << "naive = " << ans2 << '\n';

                    exit(0);
                }
            }
        }
    }

    cout << "AC\n";
}
