struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
 * SortedArrayHelper
 * -----------------
 * ソート済み配列に対して二分探索を用いた個数クエリを高速に行う構造体。
 *
 * 【用途】
 * - x 以下 / 未満 / 以上 / より大きい の個数取得
 * - 区間 [l, r] に含まれる要素数
 *
 * 【前提】
 * - 内部配列は昇順ソートされている
 *
 * 【計算量】
 * - 各クエリ: O(log N)
 *
 * 【コンストラクタ】
 * - 第2引数 sorted_flag:
 *      true  -> すでにソート済みとして扱う（そのまま使用）
 *      false -> 内部で sort する
 */
template<class T>
struct SortedArrayHelper {
    vector<T> a;

    /*
     * コンストラクタ
     * v: 元の配列
     * sorted_flag: ソート済みかどうか
     */
    SortedArrayHelper(const vector<T>& v, bool sorted_flag) {
        a = v;
        if (!sorted_flag) sort(a.begin(), a.end());
    }

    /*
     * x 以下の要素数 (<= x)
     */
    int count_le(const T& x) const {
        return upper_bound(a.begin(), a.end(), x) - a.begin();
    }

    /*
     * x 未満の要素数 (< x)
     */
    int count_lt(const T& x) const {
        return lower_bound(a.begin(), a.end(), x) - a.begin();
    }

    /*
     * x 以上の要素数 (>= x)
     */
    int count_ge(const T& x) const {
        return a.end() - lower_bound(a.begin(), a.end(), x);
    }

    /*
     * x より大きい要素数 (> x)
     */
    int count_gt(const T& x) const {
        return a.end() - upper_bound(a.begin(), a.end(), x);
    }

    /*
     * 閉区間 [l, r] に含まれる要素数
     */
    int count_range(const T& l, const T& r) const {
        return count_le(r) - count_lt(l);
    }

    // k 番目 (0-indexed)
    T kth(int k) const {
        return a[k];
    }

    // 存在判定
    bool exists(const T& x) const {
        auto it = lower_bound(a.begin(), a.end(), x);
        return it != a.end() && *it == x;
    }
};

template<class T>
void random_test_sorted_array_helper() {
    mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());

    for (int test = 0; test < 1000; test++) {
        int n = rng() % 50 + 1;

        vector<T> v(n);
        for (int i = 0; i < n; i++) {
            v[i] = rng() % 20 - 10; // 小さめの値で重複あり
        }

        // ソート済み・未ソート両方試す
        for (int flag = 0; flag < 2; flag++) {
            vector<T> input = v;
            if (flag) sort(input.begin(), input.end());

            SortedArrayHelper<T> helper(input, flag);

            // 正解用（常にソート）
            vector<T> sorted = v;
            sort(sorted.begin(), sorted.end());

            for (int q = 0; q < 200; q++) {
                T x = rng() % 20 - 10;
                T l = rng() % 20 - 10;
                T r = rng() % 20 - 10;
                if (l > r) swap(l, r);

                // --- 愚直計算 ---
                int le = 0, lt = 0, ge = 0, gt = 0, range = 0;
                bool ex = false;

                for (auto val : sorted) {
                    if (val <= x) le++;
                    if (val < x) lt++;
                    if (val >= x) ge++;
                    if (val > x) gt++;
                    if (l <= val && val <= r) range++;
                    if (val == x) ex = true;
                }

                // --- 検証 ---
                assert(helper.count_le(x) == le);
                assert(helper.count_lt(x) == lt);
                assert(helper.count_ge(x) == ge);
                assert(helper.count_gt(x) == gt);
                assert(helper.count_range(l, r) == range);
                assert(helper.exists(x) == ex);
            }

            // kth の検証
            for (int i = 0; i < n; i++) {
                assert(helper.kth(i) == sorted[i]);
            }
        }
    }

    cout << "All tests passed!" << endl;
}

void solv(){
    random_test_sorted_array_helper<ll>();
    return;
}
