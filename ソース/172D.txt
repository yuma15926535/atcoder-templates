vector<int> makepl(int n){
    vector<bool> rec(n+1,true);
    vector<int> ret;
    erep(i,2,n)if(rec[i]){
        ret.push_back(i);
        for(ll j=2;i*j<=n;j++)rec[i*j]=false;
    }
    return ret;
}

int main(){
    auto pl=makepl(1e7);
    int N;
    cin>>N;
    //区間の素因数分解はエラトステネスの篩っぽいアルゴリズムが使える
    vector<vector<int>>v(N+1);
    for(int p:pl){
        if(p>N)break;
        int idx=p;
        v[idx].push_back(2);
        while(idx+p<=N){
            idx+=p;
            ll c=1,n=idx;
            while(n%p==0){
                n/=p;
                c++;
            }
            v[idx].push_back(c);
        }
    }
    ll ans=0;
    erep(i,1,N){
        ll cnt=1;
        for(int j:v[i])cnt*=j;
        ans+=i*cnt;
    }
    cout<<ans<<'\n';
    return 0;
}
