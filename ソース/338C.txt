#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int N;
vector<ll> Q,A,B;
bool check(ll a){
    bool ret=true;
    rep(i,0,N)if(Q[i]<A[i]*a)ret=false;
    return ret;
}
ll bcount(ll a){
    ll ret=1e9;
    vector<ll> R=Q;
    rep(i,0,N)R[i]-=a*A[i];
    rep(i,0,N)if(B[i])ret=min(ret,R[i]/B[i]);
    return ret;
}

int main()
{
    cin>>N;
    Q.resize(N);
    A.resize(N);
    B.resize(N);
    rep(i,0,N)cin>>Q[i];
    rep(i,0,N)cin>>A[i];
    rep(i,0,N)cin>>B[i];
    ll ans=0;
    //Nが十分に小さいのでAの個数を固定して全探索可能
    erep(i,0,1e6){
        if(check(i))ans=max(ans,i+bcount(i));
        else break;
    }
    cout<<ans<<"\n";
    return 0;
}
