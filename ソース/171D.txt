#define next_p(v) next_permutation(v.begin(),v.end())

struct repl{
    map<int,int> m;
    ll ans;
    repl(int n){
        ans=0;
        rep(i,0,n){
            int a;
            cin>>a;if(m.count(a))m[a]++;
            else m[a]=1;
            ans+=a;
        }
    }
    ll query(int b,int c){
        if(m.count(b)){
            ans+=(ll)c*m[b]-(ll)b*m[b];
            if(m.count(c))m[c]+=m[b];
            else m[c]=m[b];
            m.erase(b);
        }
        return ans;
    }
};

int main()
{
    cin.tie(nullptr);
    ios::sync_with_stdio(false);
    int N;
    cin>>N;
    repl r=repl(N);
    int Q;
    cin>>Q;
    rep(i,0,Q){
        int B,C;
        cin>>B>>C;
        cout<<r.query(B,C)<<"\n";
    }
    return 0;
}
