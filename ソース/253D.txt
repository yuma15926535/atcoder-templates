ll N;
//N以下のxの倍数の和
ll fb(ll x){
    ll m=N/x;
    ll ret=m*(m+1)/2*x;
    return ret;
}

int main()
{
    ll A,B;
    cin>>N>>A>>B;
    ll ans=N*(N+1)/2;
    ll C=lcm(A,B);
    ans-=fb(A);
    ans-=fb(B);
    ans+=fb(C);
    cout<<ans<<"\n";
    return 0;
}
