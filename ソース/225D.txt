struct train{
    //前後を記録
    struct pn{
        int p;
        int n;
    };
    vector<pn> v;
    train(int n){
        v.resize(n+1,{0,0});
    }
    void query1(int x,int y){
        v[x].n=y;
        v[y].p=x;
    }
    void query2(int x,int y){
        v[x].n=0;
        v[y].p=0;
    }
    //出力時の制限から全体を数えて出力しても間に合う
    void query3(int x){
        while(v[x].n)x=v[x].n;
        int m=1;
        while(v[x].p){
            x=v[x].p;
            m++;
        }
        cout<<m<<" ";
        while(v[x].n){
            cout<<x<<" ";
            x=v[x].n;
        }
        cout<<x<<"\n";
    }
};

int main()
{
    cin.tie(nullptr);
    ios::sync_with_stdio(false);
    int N,Q;
    cin>>N>>Q;
    train t=train(N);
    rep(i,0,Q){
        int q;
        cin>>q;
        if(q==1){
            int x,y;
            cin>>x>>y;
            t.query1(x,y);
        }else if(q==2){
            int x,y;
            cin>>x>>y;
            t.query2(x,y);
        }else{
            int x;
            cin>>x;
            t.query3(x);
        }
    }
    return 0;
}
