#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    int N;
    cin>>N;
    vector<int> P(N);
    rep(i,0,N)cin>>P[i];
    //prev_permutationを使って解くことも可能
    // prev_permutation(nall(P));
    // rep(i,0,N)cout<<P[i]<<" ";
    // cout<<"\n";
    //https://atcoder.jp/contests/abc276/editorial/5161 参照
    int j = N - 2;
    while (P[j] < P[j + 1]) {
        j -= 1;
    }
    int k = N - 1;
    while (P[j] < P[k]) {
        k -= 1;
    }
    swap(P[j], P[k]);
    reverse(begin(P) + j + 1, end(P));
    rep(i,0,N) {
        cout << P[i] << " \n"[i + 1 == N];
    }
    return 0;
}
