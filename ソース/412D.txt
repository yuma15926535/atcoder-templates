struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

vector<int> indices(ull mask){
    vector<int> res;
    for(int i=0;mask;i++){
        if(mask&1) res.push_back(i);
        mask>>=1;
    }
    return res;
}

struct CombinationBits {
    using ull = unsigned long long;

    int N, K;
    ull cur;
    ull limit;

    //n桁,popcount=k
    CombinationBits(int n, int k) : N(n), K(k) {
        cur = (1ULL << K) - 1;   // 最初: 下位Kbitが1
        limit = (1ULL << N);
    }

    bool has_next() const {
        return cur < limit;
    }

    ull next() {
        ull ret = cur;

        // --- Gosper's Hack ---
        ull c = cur & -cur;
        ull r = cur + c;
        cur = (((r ^ cur) >> 2) / c) | r;

        return ret;
    }
};

void solv(){
    int N,M;
    cin>>N>>M;
    auto G=GraphIO::read_unweighted(N,M);
    vector<vector<bool>>Gv(N,vector<bool>(N,false));
    rep(i,0,N)for(int j:G[i])Gv[i][j]=true;
    map<int,pair<int,int>>m;
    {
        int cnt=0;
        rep(i,0,N)rep(j,i+1,N)m[cnt]=make_pair(i,j),cnt++;
    }
    debug(m);
    CombinationBits gen(N*(N-1)/2,N);
    int ans=INF;
    while(gen.has_next()){
        auto mask=gen.next();
        set<pair<int,int>> edge;
        vector<vector<bool>>Gv2(N,vector<bool>(N,false));
        for(int e:indices(mask)){
            auto p=m[e];
            Gv2[p.first][p.second]=true;
        }
        int cnt=0;
        vector<int>degree(N,0);
        bool b=true;
        rep(i,0,N)rep(j,i+1,N){
            if(Gv2[i][j]){
                degree[i]++;
                degree[j]++;
                if(degree[i]>2)b=false;
                if(degree[j]>2)b=false;
            }
            if(Gv[i][j]^Gv2[i][j])cnt++;
        }
        if(b)chmin(ans,cnt);
    }
    cout<<ans<<'\n';
    return;
}
