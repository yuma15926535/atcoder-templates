struct S {
    ll sum;
    int len;
};

/*
=====================================================
★ 更新情報（問題ごとに変更）
=====================================================
*/

using F = ll;

/*
=====================================================
★ 演算定義
=====================================================
*/

// ノード結合
S combine(S a, S b) {
    return {a.sum + b.sum, a.len + b.len};
}

// 更新適用
S apply(F f, S x) {
    x.sum += f * x.len;
    return x;
}

/*
composition(f, g)

意味：
既存更新 g の後に
新しい更新 f を適用

つまり：

new_lazy = f ∘ g

加算なら単純に足す
*/
F compose(F f, F g) {
    return f + g;
}

S idS() { return {0,0}; }
F idF() { return 0; }

/*
=====================================================
Lazy SegTree 本体
=====================================================
*/

struct LazySegTree {
    int n;
    vector<S> seg;
    vector<F> lazy;

    LazySegTree(const vector<ll>& v) {
        int sz = v.size();
        n = 1;
        while (n < sz) n <<= 1;

        seg.assign(2*n, idS());
        lazy.assign(2*n, idF());

        for (int i = 0; i < sz; i++)
            seg[n+i] = {v[i],1};
        for (int i = sz; i < n; i++)
            seg[n+i] = {0,1};

        for (int i = n-1; i > 0; i--)
            seg[i] = combine(seg[i<<1], seg[i<<1|1]);
    }

    // ========================
    // lazy伝播
    // ========================
    void push(int k) {
        if (lazy[k] == idF()) return;

        apply_node(k<<1, lazy[k]);
        apply_node(k<<1|1, lazy[k]);

        lazy[k] = idF();
    }

    void apply_node(int k, F f) {
        seg[k] = apply(f, seg[k]);
        if (k < n)
            lazy[k] = compose(f, lazy[k]);
    }

    // ========================
    // 区間更新
    // ========================
    void range_apply(int l, int r, F f, int k, int a, int b) {
        if (r <= a || b <= l) return;
        if (l <= a && b <= r) {
            apply_node(k, f);
            return;
        }

        push(k);

        int mid = (a+b)/2;
        range_apply(l,r,f,k<<1,a,mid);
        range_apply(l,r,f,k<<1|1,mid,b);

        seg[k] = combine(seg[k<<1], seg[k<<1|1]);
    }

    void range_apply(int l, int r, F f) {
        range_apply(l,r,f,1,0,n);
    }

    // ========================
    // 区間取得
    // ========================
    S range(int l, int r, int k, int a, int b) {
        if (r <= a || b <= l) return idS();
        if (l <= a && b <= r) return seg[k];

        push(k);

        int mid = (a+b)/2;
        return combine(
            range(l,r,k<<1,a,mid),
            range(l,r,k<<1|1,mid,b)
        );
    }

    ll range(int l, int r) {
        return range(l,r,1,0,n).sum;
    }

    // ========================
    // prefix探索
    // ========================
    int ge(ll x, int k, ll acc, int a, int b) {
        if (b-a == 1) return a;

        push(k);

        int mid = (a+b)/2;

        if (acc + seg[k<<1].sum >= x)
            return ge(x,k<<1,acc,a,mid);
        else
            return ge(x,k<<1|1,acc+seg[k<<1].sum,mid,b);
    }

    int ge(ll x) {
        if (seg[1].sum < x) return n;
        return ge(x,1,0,0,n);
    }

    int gt(ll x){ return ge(x+1); }
    int le(ll x){ return ge(x+1)-1; }
    int lt(ll x){ return ge(x)-1; }
};

void solv(){
    cout << "=============================\n";
    cout << " LazySegTree 動作確認\n";
    cout << "=============================\n\n";

    vector<ll> v = {5,3,7,1,4};

    cout << "▼ 初期配列:\n";
    for(auto x: v) cout << x << " ";
    cout << "\n\n";

    LazySegTree seg(v);

    cout << "▼ 区間和 [0,5): " << seg.range(0,5) << "\n";
    cout << "▼ 区間和 [1,4): " << seg.range(1,4) << "\n\n";

    // ==========================
    // 区間更新テスト
    // ==========================
    cout << "=============================\n";
    cout << " 区間更新テスト\n";
    cout << "=============================\n";

    cout << "\n▼ [1,4) に +2\n";
    seg.range_apply(1,4,2);

    cout << "現在の値:\n";
    for(int i=0;i<5;i++)
        cout << seg.range(i,i+1) << " ";
    cout << "\n";

    cout << "\n▼ [0,3) に +5\n";
    seg.range_apply(0,3,5);

    cout << "現在の値:\n";
    for(int i=0;i<5;i++)
        cout << seg.range(i,i+1) << " ";
    cout << "\n";

    cout << "\n▼ 全区間和: " << seg.range(0,5) << "\n";

    // ==========================
    // lazy伝播確認
    // ==========================
    cout << "\n=============================\n";
    cout << " lazy伝播確認\n";
    cout << "=============================\n";

    cout << "一点ずつ取得:\n";
    for(int i=0;i<5;i++)
        cout << "index " << i << " = "
             << seg.range(i,i+1) << "\n";

    // ==========================
    // prefix探索
    // ==========================
    cout << "\n=============================\n";
    cout << " prefix探索テスト\n";
    cout << "=============================\n";

    vector<ll> tests = {1,10,20,30,50};

    for(auto x: tests){
        cout << "\nx = " << x << "\n";
        cout << " 累積 >= x : index " << seg.ge(x) << "\n";
        cout << " 累積 >  x : index " << seg.gt(x) << "\n";
        cout << " 累積 <= x : index " << seg.le(x) << "\n";
        cout << " 累積 <  x : index " << seg.lt(x) << "\n";
    }

    // ==========================
    // 最終状態
    // ==========================
    cout << "\n=============================\n";
    cout << " 最終状態\n";
    cout << "=============================\n";

    for(int i=0;i<5;i++)
        cout << seg.range(i,i+1) << " ";
    cout << "\n";

    cout << "\n✅ LazySegTree 動作確認完了\n";
    return;
}
