#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int N;
vector<ll> A;
vector<ll> rec;
//ソートされないかわりに検索、追加、削除をO(1)で行う
unordered_set<ll> s;
//sizeを引数で与えることで区別なしチーム分けの計算量を最低限にする
void dfs(int pos,int sz){
    if(pos==N){
        ll ans;
        rep(i,0,sz){
            if(i==0)ans=rec[0];
            else ans^=rec[i];
        }
        s.insert(ans);
    }else{
        erep(i,0,sz){
            int q=0;
            if(i==sz){
                rec[i]+=A[pos];
                q=1;
            }else rec[i]+=A[pos];
            dfs(pos+1,sz+q);
            rec[i]-=A[pos];
        }
    }
}

int main()
{
    cin>>N;
    A.resize(N);
    rep(i,0,N)cin>>A[i];
    rec.resize(N,0);
    dfs(0,0);
    cout<<s.size()<<"\n";
    return 0;
}
