#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    int N,Q;
    cin>>N>>Q;
    vector<vector<int>> G(N);
    rep(i,0,N-1){
        int a,b;
        cin>>a>>b;
        a--;
        b--;
        G[a].push_back(b);
        G[b].push_back(a);
    }
    vector<int> color(N,0);
    color[0]=1;
    queue<int> q;
    q.push(0);
    //木は二部グラフなのでBFSで二色彩色
    //必ず別の色に移動するので色が一致するか否かで判定可能
    while(!q.empty()){
        int p=q.front();
        q.pop();
        for(int n:G[p])if(color[n]==0){
            color[n]=color[p]*(-1);
            q.push(n);
        }
    }
    rep(i,0,Q){
        int c,d;
        cin>>c>>d;
        c--;
        d--;
        if(color[c]==color[d])cout<<"Town\n";
        else cout<<"Road\n";
    }
    return 0;
}
