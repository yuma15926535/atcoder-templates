struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// 順序付き重複集合
// 値ごとの個数を管理しながら、前後検索・最小最大取得を行う
struct PrevNextCounter{
    map<ll,ll> mp;
    ll sz = 0;

    static constexpr ll NOT_FOUND = (ll)4e18;

    // x を 1 個追加
    void insert(ll x){
        insert(x, 1);
    }

    // x を cnt 個追加
    void insert(ll x, ll cnt){
        if(cnt <= 0) return;
        mp[x] += cnt;
        sz += cnt;
    }

    // x を 1 個削除
    bool erase_one(ll x){
        return erase(x, 1);
    }

    // x を最大 cnt 個削除
    bool erase(ll x, ll cnt){
        auto it = mp.find(x);
        if(it == mp.end() || cnt <= 0) return false;

        ll d = std::min(cnt, it->second);
        it->second -= d;
        sz -= d;

        if(it->second == 0) mp.erase(it);
        return true;
    }

    // x を全削除
    ll erase_all(ll x){
        auto it = mp.find(x);
        if(it == mp.end()) return 0;

        ll cnt = it->second;
        sz -= cnt;
        mp.erase(it);
        return cnt;
    }

    // x が存在するか
    bool contains(ll x) const{
        return mp.find(x) != mp.end();
    }

    // x の個数
    ll count(ll x) const{
        auto it = mp.find(x);
        return (it == mp.end() ? 0 : it->second);
    }

    // 最小値
    ll min() const{
        return mp.begin()->first;
    }

    // 最大値
    ll max() const{
        return std::prev(mp.end())->first;
    }

    // x より大きい最小の値
    ll next(ll x) const{
        auto it = mp.upper_bound(x);
        return (it == mp.end() ? NOT_FOUND : it->first);
    }

    // x 以上の最小の値
    ll next_eq(ll x) const{
        auto it = mp.lower_bound(x);
        return (it == mp.end() ? NOT_FOUND : it->first);
    }

    // x 未満の最大の値
    ll prev(ll x) const{
        auto it = mp.lower_bound(x);
        if(it == mp.begin()) return NOT_FOUND;
        --it;
        return it->first;
    }

    // x 以下の最大の値
    ll prev_eq(ll x) const{
        auto it = mp.upper_bound(x);
        if(it == mp.begin()) return NOT_FOUND;
        --it;
        return it->first;
    }

    // 全要素数（重複込み）
    ll size() const{
        return sz;
    }

    // 異なる値の個数
    ll distinct_size() const{
        return (ll)mp.size();
    }

    // 空かどうか
    bool empty() const{
        return sz == 0;
    }

    // 全削除
    void clear(){
        mp.clear();
        sz = 0;
    }

    // 昇順の配列に変換（重複あり）
    vector<ll> vec() const{
        vector<ll> res;
        res.reserve(sz);

        for(auto &[x, cnt] : mp){
            rep(_, cnt){
                res.push_back(x);
            }
        }

        return res;
    }
};

void random_test_prev_next_counter(){
    mt19937 rng(
        chrono::steady_clock::now().time_since_epoch().count()
    );

    rep(_,10000){

        PrevNextCounter pc;
        vector<ll> vec;

        rep(__,1000){

            int op = rng()%11;
            ll x = rng()%20;
            ll cnt = rng()%5 + 1;

            // insert(x)
            if(op==0){
                pc.insert(x);

                vec.push_back(x);
            }

            // insert(x,cnt)
            if(op==1){
                pc.insert(x,cnt);

                rep(i,cnt){
                    vec.push_back(x);
                }
            }

            // erase_one(x)
            if(op==2){
                bool res1 = pc.erase_one(x);

                bool res2 = false;
                auto it = find(nall(vec),x);
                if(it!=vec.end()){
                    vec.erase(it);
                    res2 = true;
                }

                assert(res1==res2);
            }

            // erase(x,cnt)
            if(op==3){
                bool res1 = pc.erase(x,cnt);

                bool res2 = false;

                rep(i,cnt){
                    auto it = find(nall(vec),x);
                    if(it==vec.end()) break;

                    vec.erase(it);
                    res2 = true;
                }

                assert(res1==res2);
            }

            // erase_all(x)
            if(op==4){
                ll res1 = pc.erase_all(x);

                ll res2 = 0;
                vector<ll> nxt;

                for(ll y:vec){
                    if(y==x){
                        res2++;
                    }else{
                        nxt.push_back(y);
                    }
                }

                vec.swap(nxt);

                assert(res1==res2);
            }

            vector<ll> a = vec;
            sort(nall(a));

            // count
            {
                ll ans = count(nall(a),x);
                assert(pc.count(x)==ans);
            }

            // contains
            {
                bool ans = count(nall(a),x)>0;
                assert(pc.contains(x)==ans);
            }

            // next
            {
                ll ans = PrevNextCounter::NOT_FOUND;

                for(ll y:a){
                    if(y>x){
                        ans=y;
                        break;
                    }
                }

                assert(pc.next(x)==ans);
            }

            // next_eq
            {
                ll ans = PrevNextCounter::NOT_FOUND;

                for(ll y:a){
                    if(y>=x){
                        ans=y;
                        break;
                    }
                }

                assert(pc.next_eq(x)==ans);
            }

            // prev
            {
                ll ans = PrevNextCounter::NOT_FOUND;

                for(ll y:a){
                    if(y<x){
                        ans=y;
                    }
                }

                assert(pc.prev(x)==ans);
            }

            // prev_eq
            {
                ll ans = PrevNextCounter::NOT_FOUND;

                for(ll y:a){
                    if(y<=x){
                        ans=y;
                    }
                }

                assert(pc.prev_eq(x)==ans);
            }

            // vec
            {
                assert(pc.vec()==a);
            }

            // size
            {
                assert(pc.size()==(ll)a.size());
            }

            // distinct_size
            {
                set<ll> st(nall(a));
                assert(pc.distinct_size()==(ll)st.size());
            }

            // empty
            {
                assert(pc.empty()==a.empty());
            }

            // min / max
            if(!a.empty()){
                assert(pc.min()==a.front());
                assert(pc.max()==a.back());
            }
        }

        // clear
        pc.clear();

        assert(pc.empty());
        assert(pc.size()==0);
        assert(pc.distinct_size()==0);
        assert(pc.vec().empty());
    }

    cout<<"PrevNextCounter Random Test Passed\n";
}
