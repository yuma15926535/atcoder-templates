#define chmax(x,y) x = max(x,y)

int main(){
    int N,M;
    cin>>N>>M;
    vector<vector<int>>G(N);
    rep(i,0,M){
        int a,b;
        cin>>a>>b;
        a--;b--;
        G[a].push_back(b);
        G[b].push_back(a);
    }
    //BFSを使った最短経路の数え上げの典型
    //最短経路を更新する場合と最短経路を追加する場合で分けて考える
    ll m=1e9+7;
    vector<ll>dist(N,1e15);
    vector<ll>cnt(N,0);
    dist[0]=0;
    cnt[0]=1;
    queue<int>q;
    q.push(0);
    while(!q.empty()){
        int pos=q.front();
        q.pop();
        for(int nxt:G[pos]){
            if(dist[nxt]==1e15){
                q.push(nxt);
                dist[nxt]=dist[pos]+1;
                cnt[nxt]=cnt[pos];
            }else if(dist[nxt]==dist[pos]+1)cnt[nxt]+=cnt[pos],cnt[nxt]%=m;
        }
    }
    cout<<cnt[N-1]<<'\n';
    return 0;
}
