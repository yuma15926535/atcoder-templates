int main(){
    int N,A,B;
    string S;
    cin>>N>>A>>B>>S;
    vector<int>cntA(N+1,0),cntB(N+1,0);
    {
        int a=0,b=0;
        rep(i,0,N){
            if(S[i]=='a')a++;
            else b++;
            cntA[i+1]=a;
            cntB[i+1]=b;
        }
    }
    ll ans=0;
    //Aの個数とBの個数は独立している、かつ単調性がある
    int aidx=0,bidx=0;
    rep(i,0,N){
        if(aidx<i)aidx=i;
        if(bidx<i)bidx=i;
        while(aidx+1<N&&cntA[aidx+1]-cntA[i]<A)aidx++;
        while(bidx+1<N&&cntB[bidx+2]-cntB[i]<B)bidx++;
        if(aidx<=bidx&&cntA[aidx+1]-cntA[i]>=A&&cntB[bidx+1]-cntB[i]<B)ans+=bidx-aidx+1;
    }
    cout<<ans<<'\n';
    return 0;
}
