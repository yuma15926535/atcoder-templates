void solv(){
    int N;
    cin>>N;
    vector<int>P(N);
    rep(i,0,N)cin>>P[i],P[i]--;
    string S;
    cin>>S;
    ll left,right,m=998244353;
    rep(t,0,2){
        ll cnt=1;
        vector<bool>rec(N,true);
        rep(i,0,N){
            ll time;
            int idx=P[i];
            if(rec[idx]&&rec[(idx+1)%N]){
                if(t){
                    if(S[idx]=='L')time=0;
                    else{
                        time=1;
                        rec[(idx+1)%N]=false;
                    }
                }else{
                    if(S[idx]=='R')time=0;
                    else{
                        time=1;
                        rec[idx]=false;
                    }
                }
            }else if(rec[idx]){
                if(t==1)time=0;
                else{
                    if(S[idx]=='?')time=2;
                    else time=1;
                    rec[idx]=false;
                }
            }else if(rec[(idx+1)%N]){
                if(t==0)time=0;
                else{
                    if(S[idx]=='?')time=2;
                    else time=1;
                    rec[(idx+1)%N]=false;
                }
            }else time=0;
            cnt*=time;
            cnt%=m;
        }
        if(t)right=cnt;
        else left=cnt;
    }
    cout<<(left+right)%m<<'\n';
    return;
}

int main(){
    BUFF;
    int T=1;
    //cin>>T;
    while(T--)solv();
    return 0;
}
