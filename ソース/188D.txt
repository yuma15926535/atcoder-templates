void solv(){
    int N;
    ll C;
    cin>>N>>C;
    auto cmp=[](pair<int,int>a,pair<int,int>b){return a.first!=b.first?a.first>b.first:a.second>b.second;};
    priority_queue<pair<int,int>,vector<pair<int,int>>,decltype(cmp)> pq(cmp);
    vector<ll>cost(N);
    rep(i,0,N){
        int a,b;
        cin>>a>>b>>cost[i];
        pq.push(make_pair(a,i));
        pq.push(make_pair(b+1,i));
    }
    vector<bool>use(N,false);
    ll ans=0,sum=0,now=0;
    //状況が変わるのは加入、停止タイミングのみ
    //小さい順に現在のcostの和とCを比較しながらansへの加算を行う
    while(!pq.empty()){
        auto [time,idx]=pq.top();
        pq.pop();
        if(sum<C)ans+=sum*(time-now);
        else ans+=C*(time-now);
        now=time;
        use[idx]=use[idx]^true;
        if(use[idx])sum+=cost[idx];
        else sum-=cost[idx];
    }
    cout<<ans<<'\n';
    return;
}

int main(){
    BUFF;
    int T=1;
    //cin>>T;
    while(T--)solv();
    return 0;
