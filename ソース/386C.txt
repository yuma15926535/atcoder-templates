#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    int K;
    string S,T;
    cin>>K>>S>>T;
    string U;
    //0回で等しい場合
    if(S==T){
        YES;
        return 0;
    }
    if(abs((int)S.size()-(int)T.size())>1){
        NO;
        return 0;
    }
    if(S.size()!=T.size()){
        if(S.size()<T.size()){
            //1回文字の追加を行う
            //break;することで計算量を抑える
            rep(i,0,S.size())if(S[i]!=T[i]){
                U=S.substr(0,i)+T[i]+S.substr(i);
                if(U==T){
                    YES;
                    return 0;
                }
                break;
            }
            //最後尾に追加
            if(S+T[(int)T.size()-1]==T){
                YES;
                return 0;
            }
        }else{
            //1回文字の削除を行う
            //break;することで計算量を抑える
            rep(i,0,T.size())if(S[i]!=T[i]){
                U=S.substr(0,i)+S.substr(i+1);
                if(U==T){
                    YES;
                    return 0;
                }
                break;
            }
            //最後尾から削除
            if(S.substr(0,T.size())==T){
                YES;
                return 0;
            }
        }
    }else{
        //1回文字の変更を行う
        //break;することで計算量を抑える
        rep(i,0,S.size())if(S[i]!=T[i]){
            S[i]=T[i];
            if(S==T){
                YES;
                return 0;
            }
            break;
        }
    }
    NO;
    return 0;
}
