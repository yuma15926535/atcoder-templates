struct point{
    //点数の集合
    set<ll> s;
    //点数が複数存在する場合の個数を管理
    map<ll,int> m;
    //実際の点数の管理
    vector<ll> v;
    point(int n){
        v.resize(n,0);
        s.insert(0);
        if(n>=2)m[0]=n;
    }
    void solv(int a,int b){
        a--;
        if(m.count(v[a])){
            m[v[a]]--;
            if(m[v[a]]==1)m.erase(m.find(v[a]));
        }else s.erase(v[a]);
        v[a]+=b;
        if(s.count(v[a])){
            if(m.count(v[a]))m[v[a]]++;
            else m[v[a]]=2;
        }else s.insert(v[a]);
        cout<<s.size()<<"\n";
    }
};

int main()
{
    cin.tie(nullptr);
    ios::sync_with_stdio(false);
    int N,T;
    cin>>N>>T;
    point p=point(N);
    rep(i,0,T){
        int a,b;
        cin>>a>>b;
        p.solv(a,b);
    }
    return 0;
}
