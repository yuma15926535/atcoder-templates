template<class T> using PQ = priority_queue<T, vector<T>>;//大きい順

#define chmax(x,y) x = max(x,y)

int main(){
    int N,K;
    cin>>N>>K;
    vector<int>P(N);
    rep(i,0,N)cin>>P[i],P[i]--;
    vector<int>ans(N,-1),parent(N);
    set<int>omote;
    unordered_map<int,vector<int>>yama;
    erep(i,1,N){
        int card=P[i-1];
        auto it=omote.upper_bound(card);
        if(it==omote.end()){
            omote.insert(card);
            parent[card]=card;
        }else{
            parent[card]=parent[*it];
            omote.erase(it);
            omote.insert(card);
        }
        yama[parent[card]].push_back(card);
        if(yama[parent[card]].size()==K){
            for(int x:yama[parent[card]])ans[x]=i;
            omote.erase(card);
        }
    }
    rep(i,0,N)cout<<ans[i]<<'\n';
    return 0;
}
