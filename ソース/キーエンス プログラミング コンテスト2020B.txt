//区間スケジューリング問題
//区間が共通部分を持たないように区間を選ぶ時の、選べる区間数の最大値
//右端を小さい順に並べて重ならないように貪欲に選ぶのが最適
//https://take44444.github.io/Algorithm-Book/range/isp/main.html
int interval_scheduling_problem(vector<pair<ll,ll>>tasks){
    sort(nall(tasks),[](pair<ll,ll>p,pair<ll,ll>q){return p.second<q.second;});
    int ret=0;
    ll now=-1e18;
    for (auto p:tasks) {
        //この問題では開区間なので=をつけない
        if(p.first<now)continue;
        ret++;
        now=p.second;
    }
    return ret;
}


int main(){
    int N;
    cin>>N;
    vector<pair<ll,ll>>v(N);
    rep(i,0,N){
        int x,l;
        cin>>x>>l;
        v[i]={x-l,x+l};
    }
    cout<<interval_scheduling_problem(v)<<'\n';
    return 0;
}
