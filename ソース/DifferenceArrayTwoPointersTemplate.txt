/*
==================================================
 Difference Array（差分配列 / 1次元 Imos）
==================================================

【用途】
- 区間加算・区間操作を高速に処理したいとき
- すべての更新後に最終配列が欲しいとき

【計算量】
- add : O(1)
- restore : O(N)

【区間】
- [l, r)（l以上 r未満）
*/
template <class T>
struct DifferenceArray {
    int n;
    vector<T> diff;


    DifferenceArray(int n_) : n(n_), diff(n + 1, T(0)) {}

    // [l,r)にxを加算
    void add(int l, int r, T x) {
        diff[l] += x;
        diff[r] -= x;
    }

    // 差分から元の配列を復元
    vector<T> restore() const {
        vector<T> a(n);
        T cur = 0;
        for (int i = 0; i < n; i++) {
            cur += diff[i];
            a[i] = cur;
        }
        return a;
    }
};
