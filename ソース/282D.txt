template<size_t I,class...Ts>
void _tuple_out(ostream&os,const tuple<Ts...>&t){
    if constexpr(I<sizeof...(Ts)){
        if(I) os<<",";
        os<<get<I>(t);
        _tuple_out<I+1>(os,t);
    }
}
template<class...Ts>
ostream&operator<<(ostream&os,const tuple<Ts...>&t){
    _tuple_out<0>(os,t);
    return os;
}
template<class T>
std::enable_if_t<is_iter_v<T>,ostream&>
operator<<(ostream&os,const T&v){
    using V = std::decay_t<decltype(*begin(v))>;
    if constexpr(is_iter_v<V>){
        bool first_row=true;
        os<<"\n";
        for(auto&&row:v){
            if(!first_row) os<<"\n";
            first_row=false;
            bool f=0;
            for(auto&&x:row){
                if(f) os<<" ";
                os<<x;
                f=1;
            }
        }
    }
    else{
        os<<"[";
        bool f=0;
        for(auto&&x:v){
            if(f) os<<", ";
            os<<x;
            f=1;
        }
        os<<"]";
    }
    return os;
}
void _dbg(const char*){
    cerr<<DBG_COLOR<<"========================================\n"<<DBG_RESET;
}
template<class T,class...U>
void _dbg(const char*names,T&&x,U&&...y){
    const char*comma=strchr(names,',');
    if(!comma){
        cerr<<DBG_COLOR<<names<<" = "<<x<<"\n"<<DBG_RESET;
    }else{
        cerr<<DBG_COLOR;
        cerr.write(names,comma-names)<<" = "<<x<<" | ";
        _dbg(comma+1,y...);
    }
}

void solv(){
    int N,M;
    cin>>N>>M;
    vector<vector<int>>G(N);
    vector<unordered_set<int>>vs(N);
    rep(i,0,M){
        int u,v;
        cin>>u>>v;
        u--;v--;
        G[u].push_back(v);
        G[v].push_back(u);
        vs[u].insert(v);
        vs[v].insert(u);
    }
    vector<int>color(N,-1);
    bool b=true;
    //辺を追加し二部グラフになる頂点の対の数
    //すでに二部グラフでなければ不可
    //頂点iに対してiと同じ色または辺が既に存在する点は不可
    //上の条件では2回数えられるので2で割る
    auto dfs=[&](int pos,int col,auto dfs)->void{
        color[pos]=col;
        int ncol=col+1;
        if(ncol%2==0)ncol-=2;
        for(int nxt:G[pos]){
            if(color[nxt]==col)b=false;
            if(color[nxt]!=-1)continue;
            dfs(nxt,ncol,dfs);
        }
    };
    int base=0;
    rep(i,0,N)if(color[i]==-1){
        dfs(i,base,dfs);
        base+=2;
    }
    vector<ll>cnt(base,0);
    rep(i,0,N)cnt[color[i]]++;
    debug(color);
    debug(cnt);
    ll ans=0;
    if(b){
        rep(i,0,N){
            ll pnt=N-cnt[color[i]]-sizei(vs[i]);
            debug(i,pnt);
            ans+=pnt;
        }
        ans/=2;
    }
    cout<<ans<<'\n';
    return;
}

int main(){
    BUFF;
    int T=1;
    //cin>>T;
    while(T--)solv();
    return 0;
