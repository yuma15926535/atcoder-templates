struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*

using mint = modint998244353;

struct S {
    mint sum;
    int len;
};

/*
=====================================================
★ 更新情報
=====================================================
*/

struct F {
    mint a, b;
};

/*
=====================================================
★ 演算定義
=====================================================
*/

// ノード結合
S combine(S x, S y) {
    return {
        x.sum + y.sum,
        x.len + y.len
    };
}

// 更新適用
S apply(F f, S x) {
    return {
        f.a * x.sum + f.b * x.len,
        x.len
    };
}

// 更新合成
// compose(f,g) = f ∘ g
// 先に g を適用し、その後 f を適用
F compose(F f, F g) {
    return {
        f.a * g.a,
        f.a * g.b + f.b
    };
}

S idS() {
    return {0, 0};
}

F idF() {
    return {1, 0};
}

/*
=====================================================
Lazy SegTree 本体
=====================================================
*/

struct LazySegTree {
    int n;
    vector<S> seg;
    vector<F> lazy;

    LazySegTree(const vector<ll>& v) {
        int sz = v.size();
        n = 1;
        while (n < sz) n <<= 1;

        seg.assign(2 * n, idS());
        lazy.assign(2 * n, idF());

        for (int i = 0; i < sz; i++)
            seg[n + i] = {v[i], 1};

        for (int i = sz; i < n; i++)
            seg[n + i] = {0, 1};

        for (int i = n - 1; i > 0; i--)
            seg[i] = combine(seg[i << 1], seg[i << 1 | 1]);
    }

    void apply_node(int k, F f) {
        seg[k] = apply(f, seg[k]);

        if (k < n)
            lazy[k] = compose(f, lazy[k]);
    }

    void push(int k) {
        if (lazy[k].a == 1 && lazy[k].b == 0) return;

        apply_node(k << 1, lazy[k]);
        apply_node(k << 1 | 1, lazy[k]);

        lazy[k] = idF();
    }

    void range_apply(int l, int r, F f, int k, int a, int b) {
        if (r <= a || b <= l) return;

        if (l <= a && b <= r) {
            apply_node(k, f);
            return;
        }

        push(k);

        int mid = (a + b) / 2;

        range_apply(l, r, f, k << 1, a, mid);
        range_apply(l, r, f, k << 1 | 1, mid, b);

        seg[k] = combine(seg[k << 1], seg[k << 1 | 1]);
    }

    void range_apply(int l, int r, F f) {
        range_apply(l, r, f, 1, 0, n);
    }

    S range(int l, int r, int k, int a, int b) {
        if (r <= a || b <= l) return idS();

        if (l <= a && b <= r)
            return seg[k];

        push(k);

        int mid = (a + b) / 2;

        return combine(
            range(l, r, k << 1, a, mid),
            range(l, r, k << 1 | 1, mid, b)
        );
    }

    mint range_sum(int l, int r) {
        return range(l, r, 1, 0, n).sum;
    }
};

using mint = modint998244353;
mt19937 rng;
void random_test() {
    rng.seed(chrono::steady_clock::now().time_since_epoch().count());

    rep(_, 0, 1000) {
        int N = rng() % 20 + 1;
        int Q = rng() % 200 + 1;

        vector<ll> A(N);

        rep(i, 0, N)
            A[i] = rng() % 100;

        LazySegTree seg(A);

        vector<mint> naive(N);
        rep(i, 0, N)
            naive[i] = A[i];

        rep(q, 0, Q) {

            // 0 : affine update
            // 1 : range sum
            int t = rng() % 2;

            int l = rng() % N;
            int r = rng() % N;

            if (l > r) swap(l, r);
            r++;

            if (t == 0) {

                ll a = rng() % 10;
                ll b = rng() % 10;

                seg.range_apply(l, r, {a, b});

                rep(i, l, r)
                    naive[i] = a * naive[i] + b;
            }

            else {

                mint seg_ans = seg.range_sum(l, r);

                mint naive_ans = 0;

                rep(i, l, r)
                    naive_ans += naive[i];

                if (seg_ans != naive_ans) {

                    cout << "NG\n";

                    cout << "N = " << N << "\n";
                    cout << "Q = " << Q << "\n";

                    cout << "l = " << l << "\n";
                    cout << "r = " << r << "\n";

                    cout << "seg = " << seg_ans.val() << "\n";
                    cout << "naive = " << naive_ans.val() << "\n";

                    cout << "\narray\n";

                    rep(i, 0, N)
                        cout << naive[i].val() << " ";
                    cout << "\n";

                    assert(false);
                }
            }
        }
    }

    cout << "OK\n";
}
