struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
==================================================
 線形篩ベース 数論全部入りテンプレ（SPF）
==================================================

【概要】
最小素因数（SPF）をベースに、数論でよく使う情報をすべて前計算する構造体。

【前計算でできること（O(N)）】
- spf[i]     : i の最小素因数
- primes     : 素数列（昇順）
- is_prime   : 素数判定
- pi[i]      : i 以下の素数の個数

- phi[i]     : オイラーのトーシェント関数 φ(i)
- mu[i]      : メビウス関数 μ(i)

- Omega[i]   : 素因数の総数（重複あり）
- omega[i]   : 異なる素因数の個数

- d[i]       : 約数の個数
- sigma[i]   : 約数の総和

【計算量】
- 構築：O(N)
- クエリ：O(1) または O(log N)

【制約目安】
- N ≦ 1e7 程度（メモリに注意）

【用途】
- 素因数分解
- 約数系
- φ, μ
- 包除原理
- gcd畳み込み
など数論問題ほぼ全対応

【注意】
- 1-indexed（1から使う）
- x は必ず 1 <= x <= n の範囲で使うこと
*/
struct SPF {
    int n;
    vector<int> spf;
    vector<int> primes;
    vector<bool> is_prime;
    vector<int> pi;
    vector<int> phi;
    vector<int> mu;
    vector<int> Omega;
    vector<int> omega;
    vector<int> d;
    vector<long long> sigma;

    vector<int> cnt;
    vector<int> p_pow;
    vector<long long> sigma_p;

    SPF(int n): n(n), spf(n+1), is_prime(n+1, false), pi(n+1, 0), phi(n+1, 0), mu(n+1, 0) {
        build();
    }

    void build() {
        spf[0] = 0;
        spf[1] = 1;
        phi[1] = 1;
        mu[1] = 1;
        d.assign(n+1, 0);
        sigma.assign(n+1, 0);
        cnt.assign(n+1, 0);
        p_pow.assign(n+1, 0);
        sigma_p.assign(n+1, 0);
        d[1] = 1;
        sigma[1] = 1;
        cnt[1] = 0;
        p_pow[1] = 1;
        sigma_p[1] = 1;
        for (int i = 2; i <= n; i++) {
            if (spf[i] == 0) {
                spf[i] = i;
                primes.push_back(i);
                phi[i] = i - 1;
                mu[i] = -1;

                d[i] = 2;
                sigma[i] = 1 + i;

                cnt[i] = 1;
                p_pow[i] = i;
                sigma_p[i] = 1 + i;
            }
            for (int p : primes) {
                if (p > spf[i] || 1LL * i * p > n) break;
                spf[i * p] = p;
                if (p == spf[i]) {
                    phi[i * p] = phi[i] * p;
                    mu[i * p] = 0;

                    cnt[i * p] = cnt[i] + 1;
                    p_pow[i * p] = p_pow[i] * p;
                    sigma_p[i * p] = sigma_p[i] + p_pow[i * p];

                    d[i * p] = d[i] / (cnt[i] + 1) * (cnt[i * p] + 1);
                    sigma[i * p] = sigma[i] / sigma_p[i] * sigma_p[i * p];
                    break;
                } else {
                    phi[i * p] = phi[i] * (p - 1);
                    mu[i * p] = -mu[i];

                    cnt[i * p] = 1;
                    p_pow[i * p] = p;
                    sigma_p[i * p] = 1 + p;

                    d[i * p] = d[i] * 2;
                    sigma[i * p] = sigma[i] * (1 + p);
                }
            }
        }

        is_prime[0] = is_prime[1] = false;
        for (int i = 2; i <= n; i++) {
            is_prime[i] = (spf[i] == i);
        }

        for (int i = 1; i <= n; i++) {
            pi[i] = pi[i-1] + (is_prime[i] ? 1 : 0);
        }

        Omega.assign(n+1, 0);
        omega.assign(n+1, 0);
        Omega[1] = 0;
        omega[1] = 0;
        for (int i = 2; i <= n; i++) {
            int p = spf[i];
            int m = i / p;

            // 総数（重複込み）
            Omega[i] = Omega[m] + 1;

            // 異なる素因数数
            if (m % p == 0) {
                omega[i] = omega[m];
            } else {
                omega[i] = omega[m] + 1;
            }
        }
    }

    // 素因数分解
    // x を素因数分解し、(素数, 指数) のペアを返す
    // 例: 12 → {(2,2), (3,1)}
    // 計算量: O(log x)
    vector<pair<int,int>> factorize(int x) const {
        vector<pair<int,int>> res;
        while (x > 1) {
            int p = spf[x];
            int cnt = 0;
            while (x % p == 0) {
                x /= p;
                cnt++;
            }
            res.emplace_back(p, cnt);
        }
        return res;
    }

    // 約数列挙
    // x の約数をすべて列挙する（順序は未保証）
    // 計算量: O(約数の個数)
    vector<int> divisors(int x) const {
        auto f = factorize(x);

        vector<int> res = {1};
        for (auto [p, c] : f) {
            int sz = res.size();
            long long mul = 1;
            for (int e = 1; e <= c; e++) {
                mul *= p;
                for (int i = 0; i < sz; i++) {
                    res.push_back(res[i] * mul);
                }
            }
        }
        return res;
    }

    // 素数判定
    // x が素数なら true、そうでなければ false を返す
    // 計算量: O(1)
    bool isPrime(int x) const {
        return is_prime[x];
    }

    // 素数の個数
    // 1 以上 x 以下の素数の個数を返す
    // 計算量: O(1)
    int primeCount(int x) const {
        return pi[x];
    }

    // k 番目の素数
    // primes[k] を返す（0-index）
    // 例: k=0 → 2, k=1 → 3
    // 計算量: O(1)
    int kthPrime(int k) const {
        return primes[k]; // 0-index
    }

    // オイラーのトーシェント関数 φ(x)
    // 1 以上 x 以下で x と互いに素な整数の個数を返す
    // 計算量: O(1)
    int totient(int x) const {
        return phi[x];
    }

    // メビウス関数 μ(x)
    // - 平方因子あり → 0
    // - 素因数の個数が偶数 → 1
    // - 素因数の個数が奇数 → -1
    // 計算量: O(1)
    int mobius(int x) const {
        return mu[x];
    }

    // 約数の個数
    // x の約数の個数を返す
    // 計算量: O(1)
    int divisorCount(int x) const {
        return d[x];
    }

    // 約数の総和
    // x の約数の合計値を返す
    // 計算量: O(1)
    long long divisorSum(int x) const {
        return sigma[x];
    }

    // 異なる素因数の個数
    // x を素因数分解したときの異なる素数の個数を返す
    // 例: 12=2^2*3 → 2
    // 計算量: O(1)
    int omegaCount(int x) const {
        return omega[x];
    }

    // 素因数の総数（重複あり）
    // x を素因数分解したときの指数の合計を返す
    // 例: 12=2^2*3 → 3
    // 計算量: O(1)
    int OmegaCount(int x) const {
        return Omega[x];
    }

    // 最小素因数
    // x の最小の素因数を返す
    // 計算量: O(1)
    int lpf(int x) const {
        return spf[x];
    }

    // 区間の素数個数
    // [l, r] に含まれる素数の個数を返す
    // 計算量: O(1)
    int range_prime_count(int l, int r) const {
        if (l > r) return 0;
        l = max(l, 1);
        r = min(r, n);
        if (l > r) return 0;
        return pi[r] - pi[l-1];
    }

    // 平方因子を持たないか判定
    // x が平方因子を持たなければ true（square-free）
    // 計算量: O(1)
    bool isSquareFree(int x) const {
        return mu[x] != 0;
    }

    // 互いに素なペア数
    // 1 ≤ i, j ≤ n において gcd(i, j) = 1 となる組の数を返す
    // 計算量: O(n)
    long long count_coprime_pairs(int n) const {
        long long res = 0;
        for (int d = 1; d <= n; d++) {
            long long t = n / d;
            res += (long long)mu[d] * t * t;
        }
        return res;
    }
};

ll power(ll base,ll exp){
    ll result=1;
    while(exp>0){
        if(exp&1)result*=base;
        base*=base;
        exp>>=1;
    }
    return result;
}

void solv(){
    ll N;
    cin>>N;
    SPF spf(2000000);
    ll ans=0;
    for(ll p:spf.primes){
        ll num=power(p,8);
        if(num>N)break;
        ans++;
    }
    for(ll p:spf.primes){
        ll num1=power(p,2);
        if(num1>N)break;
        ans+=spf.range_prime_count(p+1,sqrt(N/num1));
    }
    cout<<ans<<'\n';
    return;
}
