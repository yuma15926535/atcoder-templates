#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int N,M,X;
vector<int> A,B;
vector<bool> rec;
//DP
void solv(){
    rec[0]=true;
    int a=0,b=0;
    while(a<=X){
        if(rec[a]){
            while(b<M&&B[b]<a)b++;
            if((b<M&&a!=B[b])||b>=M){
                for(int c:A)if(a+c<=X)rec[a+c]=true;
            }
        }
        a++;
    }
}

int main()
{
    cin>>N;
    A.resize(N);
    rep(i,0,N)cin>>A[i];
    cin>>M;
    B.resize(M);
    rep(i,0,M)cin>>B[i];
    cin>>X;
    rec.resize(X+1,false);
    solv();
    if(rec[X])YN;
    return 0;
}
