struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

template <class S, S (*op)(S, S), S (*e)()>
struct SegTree {
  public:
    SegTree() : SegTree(0) {}
    explicit SegTree(int n) : SegTree(std::vector<S>(n, e())) {}
    explicit SegTree(const std::vector<S>& v) : _n(int(v.size())) {
        size = 1;
        log = 0;
        while (size < _n) {
            size <<= 1;
            log++;
        }
        d = std::vector<S>(2 * size, e());
        for (int i = 0; i < _n; i++) d[size + i] = v[i];
        for (int i = size - 1; i >= 1; i--) {
            update(i);
        }
    }

    void set(int p, S x) {
        assert(0 <= p && p < _n);
        p += size;
        d[p] = x;
        for (int i = 1; i <= log; i++) update(p >> i);
    }

    S get(int p) const {
        assert(0 <= p && p < _n);
        return d[p + size];
    }

    S prod(int l, int r) const {
        assert(0 <= l && l <= r && r <= _n);
        S sml = e(), smr = e();
        l += size;
        r += size;

        while (l < r) {
            if (l & 1) sml = op(sml, d[l++]);
            if (r & 1) smr = op(d[--r], smr);
            l >>= 1;
            r >>= 1;
        }
        return op(sml, smr);
    }

    S all_prod() const { return d[1]; }

    template <class F> int max_right(int l, F f) const {
        assert(0 <= l && l <= _n);
        assert(f(e()));
        if (l == _n) return _n;
        l += size;
        S sm = e();
        do {
            while (l % 2 == 0) l >>= 1;
            if (!f(op(sm, d[l]))) {
                while (l < size) {
                    l = (2 * l);
                    if (f(op(sm, d[l]))) {
                        sm = op(sm, d[l]);
                        l++;
                    }
                }
                return l - size;
            }
            sm = op(sm, d[l]);
            l++;
        } while ((l & -l) != l);
        return _n;
    }

    template <class F> int min_left(int r, F f) const {
        assert(0 <= r && r <= _n);
        assert(f(e()));
        if (r == 0) return 0;
        r += size;
        S sm = e();
        do {
            r--;
            while (r > 1 && (r % 2)) r >>= 1;
            if (!f(op(d[r], sm))) {
                while (r < size) {
                    r = (2 * r + 1);
                    if (f(op(d[r], sm))) {
                        sm = op(d[r], sm);
                        r--;
                    }
                }
                return r + 1 - size;
            }
            sm = op(d[r], sm);
        } while ((r & -r) != r);
        return 0;
    }

  private:
    int _n, size, log;
    std::vector<S> d;

    void update(int k) { d[k] = op(d[2 * k], d[2 * k + 1]); }
};


/**
 * ACLのsegtree用の設定 (区間和)
 */
using ll = long long;
ll op(ll a, ll b) { return a + b; }
ll e() { return 0LL; }

/**
 * ACLの遅延なしsegtreeをラップし、二分探索クエリを追加したクラス
 */
class SumSegTree {
private:
    SegTree<ll, op, e> seg;
    int _n;

public:
    SumSegTree(int n) : seg(n), _n(n) {}
    SumSegTree(const vector<ll>& v) : seg(v), _n((int)v.size()) {}

    void set(int i, ll x) {
        assert(0 <= i && i < _n);
        seg.set(i, x);
    }

    ll get(int i) const {
        assert(0 <= i && i < _n);
        return seg.get(i);
    }

    ll prod(int l, int r) const {
        assert(0 <= l && l <= r && r <= _n);
        return seg.prod(l, r);
    }

    /**
     * sum[0, i) >= x となる最小の i を返す。
     * 存在しない場合は -1 を返す。
     * 
     * @param x 目標値
     * @return 最小のインデックス i (0-indexed, 範囲 [0, n])、存在しなければ -1
     */
    int prefix_ge(ll x) {
        return range_ge(0, x);
    }

    /**
     * sum[l, i) >= x となる最小の i (>= l) を返す。
     * 存在しない場合は -1 を返す。
     * 
     * @param l 開始インデックス
     * @param x 目標値
     * @return 最小のインデックス i (0-indexed, 範囲 [l, n])、存在しなければ -1
     */
    int range_ge(int l, ll x) {
        assert(0 <= l && l <= _n);
        if (x <= 0) return l;

        // max_right(l, f) は、f(prod(l, r)) が真である最大の r を返す。
        // f を 「合計が x 未満」 と定義すると、
        // 返り値 r は sum[l, r) < x を満たす最大の r。
        // つまり、次のインデックス r + 1 が sum[l, r + 1) >= x となる最初の i。
        
        int r = seg.max_right(l, [x](ll v) { return v < x; });
        
        if (r == _n) {
            // max_rightが n を返した場合、f(prod(l, n)) が真である（つまり合計が x 未満）。
            // したがって条件を満たす i は存在しない。
            return -1;
        }
        
        // r は 「x 未満である最大の右端」 なので、r + 1 が 「x 以上になる最小の右端」。
        // ACLの内部実装的に、f(prod(l, r)) = T, f(prod(l, r+1)) = F となる r が返るので
        // prod(l, r+1) >= x となり、求める i は r + 1。
        return r + 1;
    }

    int size() const { return _n; }
};

/**
 * ランダムテスト用関数
 */
void run_random_test() {
    const int N = 2000;
    const int Q = 20000;
    const ll MAX_VAL = 10000;
    
    mt19937 mt(42);
    // prefix_ge, range_geの二分探索は要素が非負であることを前提とする
    uniform_int_distribution<ll> val_dist(0, MAX_VAL);
    uniform_int_distribution<int> idx_dist(0, N - 1);
    
    vector<ll> a(N);
    for (int i = 0; i < N; ++i) a[i] = val_dist(mt);
    
    SumSegTree st(a);
    
    cout << "Starting random test (N=" << N << ", Q=" << Q << ")..." << endl;
    
    for (int q = 0; q < Q; ++q) {
        int type = mt() % 3;
        if (type == 0) { // Update
            int i = idx_dist(mt);
            ll x = val_dist(mt);
            a[i] = x;
            st.set(i, x);
        } else if (type == 1) { // prefix_ge
            ll x = uniform_int_distribution<ll>(0, (ll)N * MAX_VAL + 100)(mt);
            int res = st.prefix_ge(x);
            
            // ナイーブな実装で検証
            int expected = -1;
            ll sum = 0;
            if (sum >= x) expected = 0;
            else {
                for (int i = 1; i <= N; ++i) {
                    sum += a[i - 1];
                    if (sum >= x) {
                        expected = i;
                        break;
                    }
                }
            }
            if (res != expected) {
                cerr << "Mismatch at query " << q << " (prefix_ge): x=" << x 
                     << ", got " << res << ", expected " << expected << endl;
                exit(1);
            }
        } else { // range_ge
            int l = uniform_int_distribution<int>(0, N)(mt);
            ll x = uniform_int_distribution<ll>(0, (ll)N * MAX_VAL + 100)(mt);
            int res = st.range_ge(l, x);
            
            // ナイーブな実装で検証
            int expected = -1;
            ll sum = 0;
            if (sum >= x) expected = l;
            else {
                for (int i = l + 1; i <= N; ++i) {
                    sum += a[i - 1];
                    if (sum >= x) {
                        expected = i;
                        break;
                    }
                }
            }
            if (res != expected) {
                cerr << "Mismatch at query " << q << " (range_ge): l=" << l << ", x=" << x 
                     << ", got " << res << ", expected " << expected << endl;
                exit(1);
            }
        }
    }
    
    cout << "Random test passed successfully!" << endl;
}

void solv(){
    run_random_test();
    // 使用例
    cout << "\nExample usage:" << endl;
    vector<ll> v = {10, 20, 30, 40, 50};
    SumSegTree st(v);
    
    cout << "Array: {10, 20, 30, 40, 50}" << endl;
    cout << "prefix_ge(25): " << st.prefix_ge(25) << " (expected 2: 10+20=30 >= 25)" << endl;
    cout << "prefix_ge(60): " << st.prefix_ge(60) << " (expected 3: 10+20+30=60 >= 60)" << endl;
    cout << "range_ge(2, 50): " << st.range_ge(2, 50) << " (expected 4: 30+40=70 >= 50)" << endl;
    cout << "range_ge(0, 200): " << st.range_ge(0, 200) << " (expected -1: total sum is 150)" << endl;
    return;
}
