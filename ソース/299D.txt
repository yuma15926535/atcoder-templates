//インタラクティブな問題
//printfでは出力のフラッシュが行われないのでcoutを使う
int main()
{
    int N;
    cin>>N;
    int left=1,right=N;
    while(right-left>1){
        int mid=(right+left)/2;
        cout<<"? "<<mid<<"\n";
        //endlでもフラッシュされるらしい
        cout.flush();
        int a;
        cin>>a;
        if(a)right=mid;
        else left=mid;
    }
    cout<<"! "<<left<<"\n";
    cout.flush();
    return 0;
}
