// a^b mod m を計算（繰り返し二乗法）
ll mod_pow(ll a, ll b, ll m) {
    ll ret = 1;
    a %= m;
    while (b) {
        if (b & 1) ret = ret * a % m;  // b が奇数ならば ret に a を掛ける
        a = a * a % m;  // a を二乗
        b >>= 1;        // b を2で割る
    }
    return ret;
}

int main(){
    ll N,P;
    cin>>N>>P;
    ll m=1e9+7;
    ll ans=P-1;
    ans%=m;
    ans*=mod_pow(P-2,N-1,m);
    ans%=m;
    cout<<ans<<'\n';
    return 0;
}
