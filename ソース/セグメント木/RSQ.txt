//https://judge.u-aizu.ac.jp/onlinejudge/description.jsp?id=DSL_2_B&lang=ja
//値の更新(加算)
//区間和の集計
struct SegTree{
    //区間集約とは無関係な値
    const ll INF=0;//★
    //葉の数
    ll sz;
    //完全二分木の配列
    vector<ll> seg;
    //引数の大きさで初期化
    void init(vector<ll> &v){
        int n=v.size();
        sz=1;
        while(sz<n)sz*=2;
        seg.resize(sz*2-1,INF);
        //葉の初期化
        rep(i,0,n)seg[sz-1+i]=v[i];
        //葉以外のノードの初期化
        rrep(i,sz-1,0)seg[i]=seg[i*2+1]+seg[i*2+2];//★
    }
    //値の更新
    void update(int pos,ll x){
        pos+=sz-1;
        seg[pos]+=x;//★
        while(pos>0){
            pos=(pos-1)/2;
            seg[pos]=seg[pos*2+1]+seg[pos*2+2];//★
        }
    }
    //[a,b)を集計
    ll query(int a,int b){
        return sub_query(a,b,0,sz,0);
    }
    //k:[l,r)を表すノードのindex
    ll sub_query(int a,int b,int l,int r,int k){
        if(r<=a||b<=l)return INF;
        if(a<=l&&r<=b)return seg[k];
        ll vl=sub_query(a,b,l,(l+r)/2,k*2+1);
        ll vr=sub_query(a,b,(l+r)/2,r,k*2+2);
        return vl+vr;//★
    }
    void output(){
        rep(i,0,sz*2-1)cout<<seg[i]<<' ';
        cout<<'\n';
    }
};

int main(){
    BUFF;
    int N,Q;
    cin>>N>>Q;
    vector<ll>v(N,0);
    SegTree st;
    st.init(v);
    rep(i,0,Q){
        int q;
        ll x,y;
        cin>>q>>x>>y;
        if(q==0)st.update(x-1,y);
        else cout<<st.query(x-1,y)<<'\n';
    }
    return 0;
}
