struct hebi{
    vector<ll> v;
    int nuke;
    int next;
    hebi(){
        //累積和のために初期化
        v.push_back(0);
        nuke=0;
        next=0;
    }
    void query1(int l){
        //頭の座標なので加算がずれる
        v.push_back(v[(int)v.size()-1]+next);
        next=l;
    }
    void query2(){
        //削除はせずにindexの参照をずらす
        nuke++;
    }
    ll query3(int k){
        return v[k+nuke]-v[nuke+1];
    }
};

        else{

        }
    }
    return 0;
}
