struct SubarrayAll {
    // ===== 最大（空あり）=====
    long long max_allow_sum;
    int max_allow_l, max_allow_r;

    // ===== 最大（空なし）=====
    long long max_force_sum;
    int max_force_l, max_force_r;

    // ===== 最小（空あり）=====
    long long min_allow_sum;
    int min_allow_l, min_allow_r;

    // ===== 最小（空なし）=====
    long long min_force_sum;
    int min_force_l, min_force_r;

    SubarrayAll(const vector<long long>& a) {
        int n = a.size();

        // ---------- 最大（空あり）----------
        {
            long long cur = 0;
            int cur_l = 0;
            max_allow_sum = 0;
            max_allow_l = max_allow_r = 0;

            for (int i = 0; i < n; i++) {
                cur += a[i];
                if (cur > max_allow_sum) {
                    max_allow_sum = cur;
                    max_allow_l = cur_l;
                    max_allow_r = i + 1;
                }
                if (cur < 0) {
                    cur = 0;
                    cur_l = i + 1;
                }
            }
        }

        // ---------- 最大（空なし）----------
        {
            long long cur = a[0];
            int cur_l = 0;
            max_force_sum = a[0];
            max_force_l = 0;
            max_force_r = 1;

            for (int i = 1; i < n; i++) {
                if (cur < 0) {
                    cur = a[i];
                    cur_l = i;
                } else {
                    cur += a[i];
                }
                if (cur > max_force_sum) {
                    max_force_sum = cur;
                    max_force_l = cur_l;
                    max_force_r = i + 1;
                }
            }
        }

        // ---------- 最小（空あり）----------
        {
            long long cur = 0;
            int cur_l = 0;
            min_allow_sum = 0;
            min_allow_l = min_allow_r = 0;

            for (int i = 0; i < n; i++) {
                cur += a[i];
                if (cur < min_allow_sum) {
                    min_allow_sum = cur;
                    min_allow_l = cur_l;
                    min_allow_r = i + 1;
                }
                if (cur > 0) {
                    cur = 0;
                    cur_l = i + 1;
                }
            }
        }

        // ---------- 最小（空なし）----------
        {
            long long cur = a[0];
            int cur_l = 0;
            min_force_sum = a[0];
            min_force_l = 0;
            min_force_r = 1;

            for (int i = 1; i < n; i++) {
                if (cur > 0) {
                    cur = a[i];
                    cur_l = i;
                } else {
                    cur += a[i];
                }
                if (cur < min_force_sum) {
                    min_force_sum = cur;
                    min_force_l = cur_l;
                    min_force_r = i + 1;
                }
            }
        }
    }
};

void solv(){
    int N;
    string S;
    cin >> N >> S;

    vector<ll> a(N);
    for(int i = 0; i < N; i++){
        if(S[i] == '1') a[i] = +1;
        else a[i] = -1;
    }

    SubarrayAll kadane(a);

    // 元の 1 の個数
    ll ones = 0;
    for(char c : S) if(c == '1') ones++;

    ll ans = ones - kadane.max_allow_sum;
    cout << ans << '\n';
    return;
}

    while(T--)solv();
