#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

vector<int> A;
vector<bool> rec;
int c=0;
void solv(int pos){
    if(pos==1)return;
    else if(rec[pos]){
        c=-1;
        return;
    }else{
        c++;
        rec[pos]=true;
        solv(A[pos]);
    }
}

int main()
{
    int N;
    cin>>N;
    A.resize(N);
    rep(i,0,N){
        int a;
        cin>>a;
        a--;
        A[i]=a;
    }
    //resizeはサイズを伸ばす場合は元の値を残して延長(第二引数で初期値の指定も可能)
    //resizeでサイズを減らす場合も元の値は残る
    //assignは元の値が残らない
    rec.assign(N,false);
    solv(0);
    cout<<c<<"\n";
    return 0;
}
