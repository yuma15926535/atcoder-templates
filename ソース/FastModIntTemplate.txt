struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// ============================================================================
//  (1) FastModInt : mod がコンパイル時定数（素数）の modint
// ============================================================================
//  ・int / long long と同じ感覚で使える（暗黙変換・全演算子・cout 直接出力）
//  ・内部は「モンゴメリ乗算 + 遅延簡約」。__int128 を使わないので MSVC でも動く
//  ・atcoder::static_modint の機能はすべて持っている（raw / mod / val / pow / inv）
//
//  速度（g++ -O2 / Xeon(Skylake) / mod=998244353 で実測。数値は「何倍速いか」）
//                         現行の FastModInt(Barrett) 比 | ACL static_modint 比
//      x = x*y+c の直列       1.46 倍速                 |  1.44 倍速
//      内積（独立な乗算）     1.28 倍速                 |  1.06 倍速
//      O(n^2) 畳み込み        1.24 倍速                 |  1.00 倍速
//      ※ 現行 FastModInt は定数 mod なのに Barrett を使っており、
//        コンパイラの定数除算最適化（＝ACL）より遅くなっていました。
//
//  使い方:
//      using mint = FastModInt<998244353>;
//      mint a = 5, b = -3;  cout << a * b + 1 << "\n";   // .val() 不要
//
//  内部表現メモ:
//      値 x は「モンゴメリ表現 x * 2^32 mod MOD」を _v に持つ。
//      MOD < 2^30 のときは _v を [0, 2*MOD) の非正規形のまま持ち回り（遅延簡約）、
//      乗算から条件分岐（cmov）を 1 個削っている。これが速さの主因。
//      MOD >= 2^30 では [0, MOD) の正規形、MOD が偶数（=2）のときは
//      モンゴメリが使えないので素直な % 実装に if constexpr で切り替わる。
// ============================================================================
template <unsigned int MOD> struct FastModInt {
  private:
    using mint = FastModInt;
    using u32 = unsigned int;
    using u64 = unsigned long long;
    using i64 = long long;

    static_assert(MOD >= 1, "MOD は 1 以上");
    static_assert(MOD < (1u << 31), "MOD は 2^31 未満（int に収まる範囲）");

    // ---- コンパイル時に決まるパラメータ ------------------------------------
    static constexpr bool MONT = (MOD & 1u) != 0;                 // モンゴメリを使うか
    static constexpr bool LAZY = MONT && (MOD < (1u << 30));      // 遅延簡約を使うか
    static constexpr u32 VMAX = LAZY ? MOD * 2u : MOD;            // 内部表現の上限 _v < VMAX

    static constexpr u32 calc_ninv() {   // -MOD^{-1} mod 2^32 （ニュートン法）
        u32 x = MOD;                     // MOD が奇数なら MOD*MOD ≡ 1 (mod 8)
        for (int i = 0; i < 5; i++) x *= 2u - MOD * x;
        return (u32)(0u - x);
    }
    static constexpr u32 NINV = MONT ? calc_ninv() : 0u;
    static constexpr u32 R2 = MONT ? (u32)((((u64)(-1)) % MOD) + 1) % MOD : 0u;  // 2^64 mod MOD
    static constexpr u32 ONE_I = MONT ? (u32)((((u64)1) << 32) % MOD) : (1u % MOD);  // 1 の内部表現

    static constexpr u64 cpow(u64 a, u64 n, u64 m) {   // a,m < 2^31 前提
        u64 r = 1 % m;
        a %= m;
        while (n) { if (n & 1) r = r * a % m; a = a * a % m; n >>= 1; }
        return r;
    }
    static constexpr bool calc_is_prime(u32 n) {       // 決定的ミラーラビン（n < 2^32）
        if (n <= 1) return false;
        if (n == 2 || n == 7 || n == 61) return true;
        if (n % 2 == 0) return false;
        u32 d = n - 1;
        while (d % 2 == 0) d /= 2;
        const u32 bases[3] = {2u, 7u, 61u};
        for (int i = 0; i < 3; i++) {
            u32 a = bases[i], t = d;
            u64 y = cpow(a, t, n);
            while (t != n - 1 && y != 1 && y != n - 1) { y = y * y % n; t <<= 1; }
            if (y != n - 1 && t % 2 == 0) return false;
        }
        return true;
    }

    // 拡張ユークリッド: {gcd(a,b), x}  ただし a*x ≡ gcd (mod b), 0 <= x < b/gcd
    static constexpr std::pair<i64, i64> inv_gcd(i64 a, i64 b) {
        a %= b;
        if (a < 0) a += b;
        if (a == 0) return {b, 0};
        i64 s = b, t = a, m0 = 0, m1 = 1;
        while (t) {
            i64 u = s / t;
            s -= t * u; m0 -= m1 * u;
            i64 tmp = s; s = t; t = tmp;
            tmp = m0; m0 = m1; m1 = tmp;
        }
        if (m0 < 0) m0 += b / s;
        return {s, m0};
    }

    // ---- モンゴメリ簡約 -----------------------------------------------------
    static u32 mr_raw(u64 t) {              // 戻り値 < 2*MOD  （t < MOD*2^32 が前提）
        u32 q = (u32)t * NINV;
        return (u32)((t + (u64)q * MOD) >> 32);
    }
    static u32 mul_internal(u32 a, u32 b) { // 内部表現同士の乗算
        if constexpr (MONT) {
            u32 r = mr_raw((u64)a * b);
            if constexpr (!LAZY) { if (r >= MOD) r -= MOD; }
            return r;
        } else {
            return (u32)((u64)a * b % MOD);
        }
    }
    static u32 to_internal(u32 x) {         // 0 <= x < MOD → 内部表現
        if constexpr (MONT) return mul_internal(x, R2);
        else return x;
    }
    u32 norm() const {                      // 内部表現の正規形（== 用）
        if constexpr (LAZY) return _v >= MOD ? _v - MOD : _v;
        else return _v;
    }

    u32 _v;

  public:
    static constexpr bool IS_PRIME = calc_is_prime(MOD);

    // ---- 構築 --------------------------------------------------------------
    constexpr FastModInt() : _v(0) {}

    template <class T, std::enable_if_t<std::is_integral_v<T> && std::is_signed_v<T>>* = nullptr>
    FastModInt(T x) {
        if constexpr (sizeof(T) <= 4) {          // int 以下は 32bit 除算で十分
            int y = (int)x % (int)MOD;
            if (y < 0) y += (int)MOD;
            _v = to_internal((u32)y);
        } else {
            i64 y = (i64)x % (i64)MOD;
            if (y < 0) y += (i64)MOD;
            _v = to_internal((u32)y);
        }
    }
    template <class T, std::enable_if_t<std::is_integral_v<T> && !std::is_signed_v<T>>* = nullptr>
    FastModInt(T x) {
        if constexpr (sizeof(T) <= 4) _v = to_internal((u32)x % MOD);
        else _v = to_internal((u32)((u64)x % MOD));
    }

    // 0 <= x < mod がわかっているときの高速構築（剰余算を省く）
    static mint raw(int x) {
        assert(0 <= x && (u32)x < MOD);
        mint r;
        r._v = to_internal((u32)x);
        return r;
    }

    // ---- 取得 --------------------------------------------------------------
    int val() const {
        if constexpr (MONT) {
            u32 r = mr_raw(_v);
            return (int)(r >= MOD ? r - MOD : r);
        } else {
            return (int)_v;
        }
    }
    static constexpr int mod() { return (int)MOD; }
    explicit operator bool() const { return norm() != 0; }

    // ---- 加減乗除 ----------------------------------------------------------
    mint& operator+=(const mint& o) { _v += o._v; if (_v >= VMAX) _v -= VMAX; return *this; }
    mint& operator-=(const mint& o) { if (_v < o._v) _v += VMAX; _v -= o._v; return *this; }
    mint& operator*=(const mint& o) { _v = mul_internal(_v, o._v); return *this; }
    mint& operator/=(const mint& o) { return *this *= o.inv(); }

    mint operator+() const { return *this; }
    mint operator-() const { mint r; r._v = _v ? VMAX - _v : 0; return r; }

    mint& operator++() { _v += ONE_I; if (_v >= VMAX) _v -= VMAX; return *this; }
    mint& operator--() { if (_v < ONE_I) _v += VMAX; _v -= ONE_I; return *this; }
    mint operator++(int) { mint t = *this; ++*this; return t; }
    mint operator--(int) { mint t = *this; --*this; return t; }

    friend mint operator+(mint a, const mint& b) { return a += b; }
    friend mint operator-(mint a, const mint& b) { return a -= b; }
    friend mint operator*(mint a, const mint& b) { return a *= b; }
    friend mint operator/(mint a, const mint& b) { return a /= b; }

    // ---- 比較（int と同じ感覚で書けるように。値 val() での比較）------------
    friend bool operator==(const mint& a, const mint& b) { return a.norm() == b.norm(); }
    friend bool operator!=(const mint& a, const mint& b) { return a.norm() != b.norm(); }
    friend bool operator<(const mint& a, const mint& b) { return a.val() < b.val(); }
    friend bool operator>(const mint& a, const mint& b) { return a.val() > b.val(); }
    friend bool operator<=(const mint& a, const mint& b) { return a.val() <= b.val(); }
    friend bool operator>=(const mint& a, const mint& b) { return a.val() >= b.val(); }

    // ---- pow / inv ---------------------------------------------------------
    // 負の指数もOK（逆元を使う）。static_modint は n>=0 のみなので上位互換。
    mint pow(long long n) const {
        mint x = *this, r = 1;
        unsigned long long e;
        if (n < 0) { x = x.inv(); e = (unsigned long long)(-(n + 1)) + 1ULL; }
        else e = (unsigned long long)n;
        while (e) { if (e & 1) r *= x; x *= x; e >>= 1; }
        return r;
    }
    mint inv() const {
        if constexpr (IS_PRIME) {
            assert(norm() != 0 && "FastModInt::inv() : 0 の逆元は存在しません");
            return pow((long long)MOD - 2);
        } else {   // MOD が素数でない定数でも正しく動く（static_modint と同じ挙動）
            auto eg = inv_gcd((i64)val(), (i64)MOD);
            assert(eg.first == 1 && "FastModInt::inv() : 逆元が存在しません（gcd != 1）");
            return mint(eg.second);
        }
    }

    // ---- 入出力 ------------------------------------------------------------
    friend std::ostream& operator<<(std::ostream& os, const mint& a) { return os << a.val(); }
    friend std::istream& operator>>(std::istream& is, mint& a) { long long x; is >> x; a = x; return is; }
};

struct Ref {
    long long m;
    explicit Ref(long long m) : m(m) {}
    long long nrm(long long x) const { x %= m; if (x < 0) x += m; return x; }
    long long add(long long a, long long b) const { return (a + b) % m; }
    long long sub(long long a, long long b) const { return ((a - b) % m + m) % m; }
    long long mul(long long a, long long b) const { return a * b % m; }   // a,b < 2^31 なので安全
    long long pw(long long a, long long e) const {
        long long r = 1 % m, x = a % m;
        while (e) { if (e & 1) r = mul(r, x); x = mul(x, x); e >>= 1; }
        return r;
    }
    long long gcd(long long a, long long b) const { while (b) { long long t = a % b; a = b; b = t; } return a; }
};

template <unsigned int MOD> void random_test_one(int iters, std::mt19937_64& rng) {
    using M = FastModInt<MOD>;
    Ref ref((long long)MOD);

    CHECK(M::mod() == (int)MOD, "mod()");
    CHECK(M().val() == 0, "default ctor");

    // 境界値を必ず含める
    std::vector<long long> edge = {0, 1, 2, (long long)MOD - 1, (long long)MOD - 2, (long long)MOD,
                                   (long long)MOD + 1, -1, -2, -(long long)MOD, -(long long)MOD - 1};
    std::vector<long long> cand;
    for (long long e : edge) cand.push_back(e);

    for (int it = 0; it < iters; it++) {
        long long ra, rb;
        if (it < (int)cand.size() * (int)cand.size()) {
            ra = cand[it / cand.size()];
            rb = cand[it % cand.size()];
        } else {
            ra = (long long)(rng() % MOD);
            rb = (long long)(rng() % MOD);
            if (it % 7 == 0) ra = -(long long)(rng() % 1000000000000000000ULL);
            if (it % 11 == 0) rb = (long long)(rng() % 1000000000000000000ULL);
        }
        M a = ra, b = rb;
        long long x = ref.nrm(ra), y = ref.nrm(rb);

        CHECK(a.val() == (int)x, "ctor / val");
        CHECK(b.val() == (int)y, "ctor / val");
        CHECK(0 <= a.val() && a.val() < (int)MOD, "val range");

        CHECK((a + b).val() == (int)ref.add(x, y), "operator+");
        CHECK((a - b).val() == (int)ref.sub(x, y), "operator-");
        CHECK((a * b).val() == (int)ref.mul(x, y), "operator*");
        CHECK((-a).val() == (int)ref.sub(0, x), "unary -");
        CHECK((+a).val() == (int)x, "unary +");

        // 複合代入
        { M t = a; t += b; CHECK(t.val() == (int)ref.add(x, y), "+="); }
        { M t = a; t -= b; CHECK(t.val() == (int)ref.sub(x, y), "-="); }
        { M t = a; t *= b; CHECK(t.val() == (int)ref.mul(x, y), "*="); }

        // 整数との混在（左右両方向）
        CHECK((a + 5).val() == (int)ref.add(x, ref.nrm(5)), "mint + int");
        CHECK((5 + a).val() == (int)ref.add(ref.nrm(5), x), "int + mint");
        CHECK((a - 7LL).val() == (int)ref.sub(x, ref.nrm(7)), "mint - ll");
        CHECK((7LL - a).val() == (int)ref.sub(ref.nrm(7), x), "ll - mint");
        CHECK((a * 3u).val() == (int)ref.mul(x, ref.nrm(3)), "mint * uint");
        CHECK((3u * a).val() == (int)ref.mul(ref.nrm(3), x), "uint * mint");

        // 比較
        CHECK((a == b) == (x == y), "operator==");
        CHECK((a != b) == (x != y), "operator!=");
        CHECK((a < b) == (x < y), "operator<");
        CHECK((a >= b) == (x >= y), "operator>=");
        CHECK((bool)a == (x != 0), "operator bool");

        // ++ / --
        { M t = a; CHECK((++t).val() == (int)ref.add(x, 1 % MOD), "pre ++");
                   CHECK(t.val() == (int)ref.add(x, 1 % MOD), "pre ++ result"); }
        { M t = a; CHECK((t++).val() == (int)x, "post ++");
                   CHECK(t.val() == (int)ref.add(x, 1 % MOD), "post ++ result"); }
        { M t = a; CHECK((--t).val() == (int)ref.sub(x, 1 % MOD), "pre --"); }
        { M t = a; CHECK((t--).val() == (int)x, "post --");
                   CHECK(t.val() == (int)ref.sub(x, 1 % MOD), "post -- result"); }

        // pow
        long long e = (long long)(rng() % 1000000);
        CHECK(a.pow(e).val() == (int)ref.pw(x, e), "pow");
        CHECK(a.pow(0).val() == (int)(1 % MOD), "pow 0");

        // inv / 除算（逆元があるときのみ）
        if (ref.gcd(y, (long long)MOD) == 1) {
            M ib = b.inv();
            CHECK((b * ib).val() == (int)(1 % MOD), "inv");
            CHECK((a / b).val() == (int)ref.mul(x, ib.val()), "operator/");
            CHECK(((a / b) * b).val() == (int)x, "div then mul");
            { M t = a; t /= b; CHECK(t.val() == (int)ref.mul(x, ib.val()), "/="); }
            CHECK(b.pow(-1) == ib, "pow(-1)");
            CHECK(b.pow(-3) == ib * ib * ib, "pow(-3)");
        }

        // raw
        CHECK(M::raw((int)x).val() == (int)x, "raw");

        // 入出力
        { std::ostringstream os; os << a; CHECK(os.str() == std::to_string(x), "operator<<"); }
        { std::istringstream is(std::to_string(ra)); M t; is >> t; CHECK(t.val() == (int)x, "operator>>"); }
    }

    // 結合律・分配律をランダムに（3 項）
    for (int it = 0; it < iters; it++) {
        M a = (long long)(rng() % MOD), b = (long long)(rng() % MOD), c = (long long)(rng() % MOD);
        CHECK((a + b) + c == a + (b + c), "add assoc");
        CHECK((a * b) * c == a * (b * c), "mul assoc");
        CHECK(a * (b + c) == a * b + a * c, "distrib");
        CHECK(a - b == a + (-b), "sub == add neg");
        CHECK(a + M(0) == a && a * M(1 % MOD) == a, "identity");
    }
}

void demo() {
    using namespace std;
    using SM = FastModInt<1000000007>;

    cout << "[2] 動作確認 : FastModInt<1000000007>\n";
    cout << "=============================\n";

    SM a = 123456789;
    SM b = 987654321;
    cout << "a = " << a << "\n";
    cout << "b = " << b << "\n";

    cout << "\n[基本演算]\n";
    cout << "a + b = " << a + b << "\n";
    cout << "a - b = " << a - b << "\n";
    cout << "a * b = " << a * b << "\n";
    cout << "a / b = " << a / b << "\n";
    cout << "-a    = " << -a << "\n";

    cout << "\n[pow / inv]\n";
    cout << "pow(a,10)  = " << a.pow(10) << "\n";
    cout << "pow(a,-1)  = " << a.pow(-1) << "   (負の指数も可)\n";
    cout << "inv(a)     = " << a.inv() << "\n";
    cout << "a*inv(a)   = " << a * a.inv() << "\n";

    cout << "\n[++ / --]\n";
    SM x = a;
    cout << "初期 x = " << x << "\n";
    cout << "前置++ = " << ++x << "\n";
    cout << "後置++ = " << x++ << " → 現在 " << x << "\n";
    cout << "前置-- = " << --x << "\n";
    cout << "後置-- = " << x-- << " → 現在 " << x << "\n";

    cout << "\n[整数混在]\n";
    cout << "a + 5   = " << a + 5 << "\n";
    cout << "5 + a   = " << 5 + a << "\n";
    cout << "a - 100 = " << a - 100 << "\n";
    cout << "a * 3   = " << a * 3 << "\n";
    cout << "a / 7   = " << a / 7 << "\n";

    cout << "\n[チェーン演算]\n";
    cout << "(a+3)*(b-2)/5 = " << (a + 3) * (b - 2) / 5 << "\n";

    cout << "\n[正規化確認]\n";
    cout << "SM(-123)  = " << SM(-123) << "\n";
    cout << "SM(1e18)  = " << SM(1000000000000000000LL) << "\n";
    cout << "SM(1e18の符号なし) = " << SM(18446744073709551615ULL) << "\n";

    cout << "\n[比較・コンテナ]\n";
    cout << "(a == a)  = " << (a == a) << "\n";
    cout << "(a < b)   = " << (a < b) << "   (val() での比較)\n";
    vector<SM> v = {SM(5), SM(2), SM(9), SM(2)};
    sort(v.begin(), v.end());
    cout << "sort 後   = "; for (auto& e : v) cout << e << " "; cout << "\n";

    cout << "\n[cin からの読み込み]\n";
    { istringstream is("-5 1000000008"); SM p, q; is >> p >> q;
      cout << "\"-5\" → " << p << " , \"1000000008\" → " << q << "\n"; }

    cout << "\n[静的情報]\n";
    cout << "SM::mod()      = " << SM::mod() << "\n";
    cout << "SM::IS_PRIME   = " << SM::IS_PRIME << "\n";
    cout << "SM::raw(42)    = " << SM::raw(42) << "   (0<=x<mod のときの高速構築)\n";

    cout << "\n[実用例] フィボナッチ / 二項係数\n";
    {
        SM f0 = 0, f1 = 1;
        for (int i = 0; i < 1000000; i++) { SM t = f0 + f1; f0 = f1; f1 = t; }
        cout << "fib(1000000) = " << f0 << "\n";
    }
    {
        const int N = 200000;
        vector<SM> fac(N + 1), ifac(N + 1);
        fac[0] = 1;
        for (int i = 1; i <= N; i++) fac[i] = fac[i - 1] * i;
        ifac[N] = fac[N].inv();
        for (int i = N; i >= 1; i--) ifac[i - 1] = ifac[i] * i;
        auto C = [&](int n, int r) { return (r < 0 || r > n) ? SM(0) : fac[n] * ifac[r] * ifac[n - r]; };
        cout << "C(200000,100000) = " << C(200000, 100000) << "\n";
        cout << "C(10,3)          = " << C(10, 3) << "   (= 120)\n";
    }

    cout << "\n=============================\n";
    cout << "動作確認 完了\n\n";
}

// ============================================================================
//  簡易ベンチ（あくまで目安。タイトなループはコード配置の影響で
//  同じ処理でも 1〜2 割ぶれることがあります）（速度の目安。提出時は不要）
// ============================================================================
void benchmark() {
    using namespace std;
    using namespace std::chrono;
    cout << "[3] 簡易ベンチ (mod = 998244353)\n";
    using M = FastModInt<998244353>;

    {   // 直列依存ループ（レイテンシ律速：DP などの実戦形）
        M x = 12345, y = 123456789, c = 7;
        auto t0 = steady_clock::now();
        for (int i = 0; i < 30000000; i++) x = x * y + c;
        auto t1 = steady_clock::now();
        cout << "    x = x*y+c を 3*10^7 回 : "
             << duration<double, milli>(t1 - t0).count() << " ms   (結果 " << x << ")\n";
    }
    {   // 畳み込み（スループット律速）
        const int n = 2000;
        vector<M> a(n), b(n), c(2 * n);
        mt19937 rng(1);
        for (int i = 0; i < n; i++) { a[i] = (long long)rng(); b[i] = (long long)rng(); }
        auto t0 = steady_clock::now();
        for (int i = 0; i < n; i++) for (int j = 0; j < n; j++) c[i + j] += a[i] * b[j];
        auto t1 = steady_clock::now();
        cout << "    2000x2000 の畳み込み   : "
             << duration<double, milli>(t1 - t0).count() << " ms   (結果 " << c[n] << ")\n";
    }
    cout << "\n";
}
