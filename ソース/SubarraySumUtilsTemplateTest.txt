/*
==================================================
 連続部分列和ユーティリティ（hash版・非円環・完全版）
 - 負の数あり対応
 - 区間数 + 最短 / 最長区間 + 最大 / 最小部分列和
 - 全て O(N)
==================================================
*/
struct SubarraySumUtils {
    int n;

    // ==============================
    // 和 = K
    // ==============================
    ll count_eqK;
    bool exist_eqK;
    int minK_l, minK_r, maxK_l, maxK_r;

    // ==============================
    // 最大 / 最小部分列和
    // ==============================
    ll max_sum, min_sum;
    ll max_count, min_count;
    int max_short_l, max_short_r, max_long_l, max_long_r;
    int min_short_l, min_short_r, min_long_l, min_long_r;

    SubarraySumUtils(const vector<ll>& a, ll K) {
        n = a.size();

        // ------------------------------
        // 1. 和 = K (hash版)
        // ------------------------------
        count_eqK = 0;
        exist_eqK = false;
        minK_l = minK_r = maxK_l = maxK_r = 0;
        int min_len = INT_MAX, max_len = -1;

        unordered_map<ll, deque<int>> mp;
        mp[0].push_back(0);

        ll sum = 0;
        for (int r = 1; r <= n; r++) {
            sum += a[r - 1];
            ll need = sum - K;
            if (mp.count(need)) {
                for (int l : mp[need]) {
                    int len = r - l;
                    if (len < min_len) { min_len = len; minK_l = l; minK_r = r; }
                    if (len > max_len) { max_len = len; maxK_l = l; maxK_r = r; }
                    count_eqK++;
                }
            }
            mp[sum].push_back(r);
        }
        exist_eqK = (count_eqK > 0);

        // ------------------------------
        // 2. 最大 / 最小部分列和 (Kadane版)
        // ------------------------------
        max_sum = LLONG_MIN; min_sum = LLONG_MAX;
        max_count = min_count = 0;

        max_short_l = max_short_r = max_long_l = max_long_r = 0;
        min_short_l = min_short_r = min_long_l = min_long_r = 0;

        ll cur_max = a[0], cur_min = a[0];
        int cur_max_l = 0, cur_min_l = 0;

        max_sum = a[0]; min_sum = a[0];
        max_short_l = max_long_l = 0; max_short_r = max_long_r = 1;
        min_short_l = min_long_l = 0; min_short_r = min_long_r = 1;
        max_count = min_count = 1;

        for (int i = 1; i < n; i++) {
            // 最大部分列和
            if (cur_max < 0) { cur_max = a[i]; cur_max_l = i; } else { cur_max += a[i]; }
            if (cur_max > max_sum) {
                max_sum = cur_max; max_count = 1;
                max_short_l = max_long_l = cur_max_l;
                max_short_r = max_long_r = i + 1;
            } else if (cur_max == max_sum) {
                max_count++;
                int len = i + 1 - cur_max_l;
                if (len < max_short_r - max_short_l) { max_short_l = cur_max_l; max_short_r = i + 1; }
                if (len > max_long_r - max_long_l) { max_long_l = cur_max_l; max_long_r = i + 1; }
            }

            // 最小部分列和
            if (cur_min > 0) { cur_min = a[i]; cur_min_l = i; } else { cur_min += a[i]; }
            if (cur_min < min_sum) {
                min_sum = cur_min; min_count = 1;
                min_short_l = min_long_l = cur_min_l;
                min_short_r = min_long_r = i + 1;
            } else if (cur_min == min_sum) {
                min_count++;
                int len = i + 1 - cur_min_l;
                if (len < min_short_r - min_short_l) { min_short_l = cur_min_l; min_short_r = i + 1; }
                if (len > min_long_r - min_long_l) { min_long_l = cur_min_l; min_long_r = i + 1; }
            }
        }
    }
};

/*
==================================================
 円環 連続部分列和ユーティリティ（hash版・完全版）
 - 最大 / 最小部分列和を O(N) Kadane 円環版で復元
==================================================
*/
struct CircularSubarraySumUtils {
    int n;

    // ==============================
    // 和 = K
    // ==============================
    ll count_eqK;
    bool exist_eqK;
    int minK_l, minK_r, maxK_l, maxK_r;

    // ==============================
    // 最大 / 最小部分列和
    // ==============================
    ll max_sum, min_sum;
    ll max_count, min_count;
    int max_short_l, max_short_r, max_long_l, max_long_r;
    int min_short_l, min_short_r, min_long_l, min_long_r;

    CircularSubarraySumUtils(const vector<ll>& a, ll K) {
        n = a.size();

        // ------------------------------
        // 1. 和 = K (hash版)
        // ------------------------------
        count_eqK = 0;
        exist_eqK = false;
        minK_l = minK_r = maxK_l = maxK_r = 0;
        int min_len = INT_MAX, max_len = -1;

        vector<ll> P(2*n + 1, 0);
        for(int i=0;i<2*n;i++) P[i+1]=P[i]+a[i%n];

        unordered_map<ll, deque<int>> mp;
        mp[0].push_back(0);

        for(int r=1;r<=2*n;r++){
            ll need = P[r]-K;
            if(mp.count(need)){
                for(int l: mp[need]){
                    int len = r-l;
                    if(len>n) continue;
                    count_eqK++;
                    if(len<min_len){ min_len=len; minK_l=l%n; minK_r=r%n; }
                    if(len>max_len){ max_len=len; maxK_l=l%n; maxK_r=r%n; }
                }
            }
            mp[P[r]].push_back(r);
            auto& dq=mp[P[r]];
            while(!dq.empty() && dq.front()<r-n) dq.pop_front();
        }
        exist_eqK = (count_eqK>0);

        // ------------------------------
        // 2. 最大 / 最小部分列和 (円環 Kadane O(N))
        // ------------------------------
        ll total = accumulate(a.begin(),a.end(),0LL);

        // 非円環 Kadane
        ll cur_max=a[0], cur_min=a[0];
        int cur_max_l=0, cur_min_l=0;
        ll best_max=a[0], best_min=a[0];
        int best_max_l=0, best_max_r=1, best_min_l=0, best_min_r=1;

        for(int i=1;i<n;i++){
            // 最大
            if(cur_max<0){ cur_max=a[i]; cur_max_l=i; } else cur_max+=a[i];
            if(cur_max>best_max){ best_max=cur_max; best_max_l=cur_max_l; best_max_r=i+1; }
            // 最小
            if(cur_min>0){ cur_min=a[i]; cur_min_l=i; } else cur_min+=a[i];
            if(cur_min<best_min){ best_min=cur_min; best_min_l=cur_min_l; best_min_r=i+1; }
        }

        // 円環最大
        if(best_max>=total-best_min){
            max_sum=best_max; max_short_l=max_long_l=best_max_l; max_short_r=max_long_r=best_max_r;
        }else{
            max_sum=total-best_min; max_short_l=max_long_l=best_min_r%n; max_short_r=max_long_r=best_min_l%n;
        }

        // 円環最小
        if(best_min<=total-best_max){
            min_sum=best_min; min_short_l=min_long_l=best_min_l; min_short_r=min_long_r=best_min_r;
        }else{
            min_sum=total-best_max; min_short_l=min_long_l=best_max_r%n; min_short_r=min_long_r=best_max_l%n;
        }

        // 最大 / 最小の区間数 (O(N))
        max_count=min_count=0;
        for(int i=0;i<n;i++){
            ll s=0;
            for(int len=1;len<=n;len++){
                s+=a[(i+len-1)%n];
                int r=(i+len)%n;
                // 最大
                if(s==max_sum){
                    max_count++;
                    if(len<max_short_r-max_short_l){ max_short_l=i; max_short_r=r; }
                    if(len>max_long_r-max_long_l){ max_long_l=i; max_long_r=r; }
                }
                // 最小
                if(s==min_sum){
                    min_count++;
                    if(len<min_short_r-min_short_l){ min_short_l=i; min_short_r=r; }
                    if(len>min_long_r-min_long_l){ min_long_l=i; min_long_r=r; }
                }
            }
        }
    }
};

void solv(){
    vector<vector<ll>> tests = {
        {1, -2, 3, -2},       // 既存: 正負混合
        {5, -3, 5},            // 既存: 正負混合
        {-3, -2, -3},          // 既存: 全負
        {2, -1, 2, -1, 2},     // 既存: 正負交互
        {1, -1, 1, -1, 1},     // 既存: 0になる部分列

        // 追加テスト
        {-1, -2, -3, -4, -5},  // 全負, 折り返しが最大区間になる
        {-1, -2, -3},          // 全負, 和=Kが存在しないケース
        {3, 1, 2, 1, 2},       // 正数だけ, 折り返しが最長区間になる
        {10, -1, -1, 2, -1},   // 正負混合, 和=Kが複数存在
        {1, 2, 3, 4, 5, -15}   // 大負数で折り返し部分列になる
    };

    vector<ll> Ks = {
        3, 5, -5, 2, 0,
        -5, -10, 6, 3, 5
    };

    for(int t=0;t<tests.size();t++){
        const auto& a = tests[t];
        ll K = Ks[t];
        int n = a.size();

        cout << "=============================" << endl;
        cout << "テスト " << t+1 << endl;
        cout << "配列: ";
        for(ll x:a) cout << x << " ";
        cout << endl << "K = " << K << endl;

        // ------------------------------
        // 非円環
        SubarraySumUtils nonCircular(a,K);

        cout << endl << "[非円環]" << endl;
        cout << "和 = K の区間数: " << nonCircular.count_eqK << endl;
        cout << "最短区間: [" << nonCircular.minK_l << "," << nonCircular.minK_r << ")";
        if(!nonCircular.exist_eqK) cout << " (存在しません)";
        cout << endl;
        cout << "最長区間: [" << nonCircular.maxK_l << "," << nonCircular.maxK_r << ")";
        if(!nonCircular.exist_eqK) cout << " (存在しません)";
        cout << endl;

        cout << "最大部分列和: " << nonCircular.max_sum << endl;
        cout << "最大和 区間数: " << nonCircular.max_count << endl;
        cout << "最大和 最短区間: [" << nonCircular.max_short_l << "," << nonCircular.max_short_r << ")" << endl;
        cout << "最大和 最長区間: [" << nonCircular.max_long_l << "," << nonCircular.max_long_r << ")" << endl;

        cout << "最小部分列和: " << nonCircular.min_sum << endl;
        cout << "最小和 区間数: " << nonCircular.min_count << endl;
        cout << "最小和 最短区間: [" << nonCircular.min_short_l << "," << nonCircular.min_short_r << ")" << endl;
        cout << "最小和 最長区間: [" << nonCircular.min_long_l << "," << nonCircular.min_long_r << ")" << endl;

        // ------------------------------
        // 円環
        CircularSubarraySumUtils circular(a,K);

        cout << endl << "[円環]" << endl;
        cout << "和 = K の区間数: " << circular.count_eqK << endl;
        cout << "最短区間: [" << circular.minK_l << "," << circular.minK_r << ")";
        if(!circular.exist_eqK) cout << " (存在しません)";
        // 折り返し判定
        else if(circular.minK_l > circular.minK_r) cout << " (折り返し)";
        cout << endl;

        cout << "最長区間: [" << circular.maxK_l << "," << circular.maxK_r << ")";
        if(!circular.exist_eqK) cout << " (存在しません)";
        else if(circular.maxK_l > circular.maxK_r) cout << " (折り返し)";
        cout << endl;

        cout << "最大部分列和: " << circular.max_sum << endl;
        cout << "最大和 区間数: " << circular.max_count << endl;
        cout << "最大和 最短区間: [" << circular.max_short_l << "," << circular.max_short_r << ")";
        if(circular.max_short_l > circular.max_short_r) cout << " (折り返し)";
        cout << endl;
        cout << "最大和 最長区間: [" << circular.max_long_l << "," << circular.max_long_r << ")";
        if(circular.max_long_l > circular.max_long_r) cout << " (折り返し)";
        cout << endl;

        cout << "最小部分列和: " << circular.min_sum << endl;
        cout << "最小和 区間数: " << circular.min_count << endl;
        cout << "最小和 最短区間: [" << circular.min_short_l << "," << circular.min_short_r << ")";
        if(circular.min_short_l > circular.min_short_r) cout << " (折り返し)";
        cout << endl;
        cout << "最小和 最長区間: [" << circular.min_long_l << "," << circular.min_long_r << ")";
        if(circular.min_long_l > circular.min_long_r) cout << " (折り返し)";
        cout << endl;

        cout << "=============================" << endl << endl;
    }
    return;
}
