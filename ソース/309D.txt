vector<vector<int>> G;
int N1,N2,M;
vector<int> d;
//BFSを2回行う
int bfs(int pos){
    int ret=0;
    d[pos]=0;
    queue<int> q;
    q.push(pos);
    while(!q.empty()){
        int p=q.front();
        q.pop();
        int dn=d[p]+1;
        for(int r:G[p])if(d[r]==-1){
            ret=dn;
            d[r]=dn;
            q.push(r);
        }
    }
    return ret;
}

    }

}
