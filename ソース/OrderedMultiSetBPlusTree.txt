struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/**
 * @brief OrderedMultiSet (B+Treeベースの多重順序集合)
 * * 【データ構造の概要】
 * 独自のB+Tree（Bプラス木）をベースに実装された、重複要素を許可する順序付き多重集合（マルチセット）です。
 * C++標準の `std::multiset` の機能に加え、「k番目の要素の取得」や「特定値未満の要素数（ランク）の取得」などの
 * インデックスによるアクセスを高速に行うことができます。
 * * 【内部実装の概要】
 * - B+Treeベース: 各ノードは最大 B 個のキー/子ノードを持ち、木の高さが低く抑えられるため探索が極めて高速です。
 * - 葉ノードの連結リスト化: すべての葉ノード（Leaf）が双方向連結リスト（next_leaf, prev_leaf）で接続されており、
 * 全要素の走査（vec() など）を線形時間で効率的に行うことができます。
 * - 重複要素の許可: 同一の値を複数保持することが可能です。
 * - 部分木情報の集約: 各ノードが自身の実質的な要素数（total_weight）と要素の総和（total_sum）を保持しています。
 * これにより、全体のサイズや総和を O(1)、kth や rank などの高度なクエリを O(log N) で処理します。
 * * 【特徴】
 * - 非常に高速な k 番目の要素へのアクセスおよび二分探索クエリ。
 * - 集合全体の要素の総和（sum）を O(1) で取得可能。
 * - 競技プログラミングにおいて、インデックスアクセスが可能な平衡二分探索木（平衡木）の上位互換として強力に機能します。
 * * 【計算量】
 * insert(x)       : O(log N)
 * erase_one(x)    : O(log N)
 * erase_all(x)    : O(k log N) (※ k は削除される重複要素の個数)
 * size()          : O(1)
 * empty()         : O(1)
 * sum()           : O(1)
 * min()           : O(log N)
 * max()           : O(log N)
 * rank(x)         : O(log N)
 * count_lt(x)     : O(log N)
 * count_le(x)     : O(log N)
 * count_gt(x)     : O(log N)
 * count_ge(x)     : O(log N)
 * count(x)        : O(log N)
 * sum_lt(x)       : O(log N)
 * sum_le(x)       : O(log N)
 * sum_gt(x)       : O(log N)
 * sum_ge(x)       : O(log N)
 * sum_range(l, r) : O(log N)
 * contains(x)     : O(log N)
 * kth(k)          : O(log N)
 * kth_smallest(k) : O(log N)
 * kth_largest(k)  : O(log N)
 * next(x)         : O(log N)
 * next_eq(x)      : O(log N)
 * prev(x)         : O(log N)
 * prev_eq(x)      : O(log N)
 * vec()           : O(N)
 * * 【提供関数一覧】
 * - 挿入・削除: insert, erase_one, erase_all
 * - 状態取得: size, empty, sum
 * - 極値取得: min, max
 * - 順位・カウント: rank, count_lt, count_le, count_gt, count_ge, count, contains
 * - 条件付き総和・区間和: sum_lt, sum_le, sum_gt, sum_ge, sum_range
 * - インデックスアクセス: kth, kth_smallest, kth_largest
 * - 前後要素探索(optional): next, next_eq, prev, prev_eq
 * - 全要素走査: vec
 */
template <int B = 64>
struct OrderedMultiSet {
    struct Node {
        bool is_leaf;
        vector<long long> keys;
        vector<Node*> children;
        long long total_weight = 0;
        long long total_sum = 0;
        Node* next_leaf = nullptr;
        Node* prev_leaf = nullptr;

        Node(bool leaf) : is_leaf(leaf) {}
        ~Node() {
            for (Node* child : children) delete child;
        }
    };

    Node* root = nullptr;

    ~OrderedMultiSet() { delete root; }

    // --- 内部ヘルパー ---
    void update_node(Node* node) {
        node->total_weight = 0;
        node->total_sum = 0;
        if (node->is_leaf) {
            node->total_weight = node->keys.size();
            for (long long x : node->keys) node->total_sum += x;
        } else {
            for (Node* child : node->children) {
                node->total_weight += child->total_weight;
                node->total_sum += child->total_sum;
            }
        }
    }

    long long get_min_key(Node* node) {
        while (node && !node->is_leaf) node = node->children[0];
        return node->keys.empty() ? 0 : node->keys[0];
    }

    void fix_keys(Node* node) {
        if (node->is_leaf) return;
        node->keys.resize(node->children.size() - 1);
        for (size_t i = 0; i < node->keys.size(); ++i) {
            node->keys[i] = get_min_key(node->children[i + 1]);
        }
    }

    // --- Insert ---
    pair<Node*, long long> insert_rec(Node* node, long long x) {
        if (node->is_leaf) {
            auto it = lower_bound(node->keys.begin(), node->keys.end(), x);
            node->keys.insert(it, x);
            
            if (node->keys.size() > B) {
                Node* right = new Node(true);
                int mid = node->keys.size() / 2;
                right->keys.assign(node->keys.begin() + mid, node->keys.end());
                node->keys.erase(node->keys.begin() + mid, node->keys.end());
                
                right->next_leaf = node->next_leaf;
                if (right->next_leaf) right->next_leaf->prev_leaf = right;
                node->next_leaf = right;
                right->prev_leaf = node;
                
                update_node(node);
                update_node(right);
                return {right, right->keys[0]};
            }
            update_node(node);
            return {nullptr, 0};
        } else {
            int idx = upper_bound(node->keys.begin(), node->keys.end(), x) - node->keys.begin();
            auto [new_child, sep] = insert_rec(node->children[idx], x);
            
            if (new_child) {
                node->keys.insert(node->keys.begin() + idx, sep);
                node->children.insert(node->children.begin() + idx + 1, new_child);
                
                // 内部ノードの分割（children基準で判定）
                if (node->children.size() > B) {
                    Node* right = new Node(false);
                    int mid = node->children.size() / 2;
                    
                    right->children.assign(node->children.begin() + mid, node->children.end());
                    node->children.erase(node->children.begin() + mid, node->children.end());
                    
                    // fix_keysを用いて安全に再構築
                    fix_keys(node);
                    fix_keys(right);
                    long long up_sep = get_min_key(right->children[0]);
                    
                    update_node(node);
                    update_node(right);
                    return {right, up_sep};
                }
            }
            update_node(node);
            return {nullptr, 0};
        }
    }

    // --- Rebalance ---
    void maintain(Node* node, int idx) {
        Node* child = node->children[idx];
        
        // Underflow判定をLeaf/Internalで分離
        if (child->is_leaf && child->keys.size() >= B / 2) return;
        if (!child->is_leaf && child->children.size() >= B / 2) return;

        // 1. 左から Borrow
        if (idx > 0) {
            Node* left = node->children[idx - 1];
            bool can_borrow = left->is_leaf ? (left->keys.size() > B / 2) : (left->children.size() > B / 2);
            if (can_borrow) {
                if (child->is_leaf) {
                    child->keys.insert(child->keys.begin(), left->keys.back());
                    left->keys.pop_back();
                } else {
                    child->children.insert(child->children.begin(), left->children.back());
                    left->children.pop_back();
                    fix_keys(left);
                    fix_keys(child);
                }
                update_node(left); update_node(child);
                return;
            }
        }

        // 2. 右から Borrow
        if (idx + 1 < node->children.size()) {
            Node* right = node->children[idx + 1];
            bool can_borrow = right->is_leaf ? (right->keys.size() > B / 2) : (right->children.size() > B / 2);
            if (can_borrow) {
                if (child->is_leaf) {
                    child->keys.push_back(right->keys.front());
                    right->keys.erase(right->keys.begin());
                } else {
                    child->children.push_back(right->children.front());
                    right->children.erase(right->children.begin());
                    fix_keys(right);
                    fix_keys(child);
                }
                update_node(right); update_node(child);
                return;
            }
        }

        // 3. Merge
        if (idx > 0) {
            Node* left = node->children[idx - 1];
            if (child->is_leaf) {
                left->keys.insert(left->keys.end(), child->keys.begin(), child->keys.end());
                left->next_leaf = child->next_leaf;
                if (left->next_leaf) left->next_leaf->prev_leaf = left;
            } else {
                left->children.insert(left->children.end(), child->children.begin(), child->children.end());
                child->children.clear();
                fix_keys(left);
            }
            node->keys.erase(node->keys.begin() + idx - 1);
            node->children.erase(node->children.begin() + idx);
            delete child;
            update_node(left);
        } else {
            Node* right = node->children[idx + 1];
            if (child->is_leaf) {
                child->keys.insert(child->keys.end(), right->keys.begin(), right->keys.end());
                child->next_leaf = right->next_leaf;
                if (child->next_leaf) child->next_leaf->prev_leaf = child;
            } else {
                child->children.insert(child->children.end(), right->children.begin(), right->children.end());
                right->children.clear();
                fix_keys(child);
            }
            node->keys.erase(node->keys.begin() + idx);
            node->children.erase(node->children.begin() + idx + 1);
            delete right;
            update_node(child);
        }
    }

    // --- Erase ---
    pair<long long, long long> erase_rec(Node* node, long long x, bool all) {
        if (node->is_leaf) {
            auto it = lower_bound(node->keys.begin(), node->keys.end(), x);
            if (it == node->keys.end() || *it != x) return {0, 0};
            
            long long count = 0;
            if (all) {
                auto it2 = upper_bound(node->keys.begin(), node->keys.end(), x);
                count = distance(it, it2);
                node->keys.erase(it, it2);
            } else {
                count = 1;
                node->keys.erase(it);
            }
            update_node(node);
            return {count, x * count};
        } else {
            int idx = upper_bound(node->keys.begin(), node->keys.end(), x) - node->keys.begin();
            auto res = erase_rec(node->children[idx], x, all);
            
            if (res.first > 0) {
                maintain(node, idx);
                fix_keys(node);
                update_node(node);
            }
            return res;
        }
    }

    void trim_root() {
        if (root && root->keys.empty() && root->is_leaf) {
            delete root;
            root = nullptr;
        } else if (root && !root->is_leaf && root->children.size() == 1) {
            Node* old = root;
            root = root->children[0];
            old->children.clear();
            delete old;
        }
    }

    // --- パブリックメソッド群 ---

    // 値 x を集合に挿入する（重複を許可する）
    // 計算量: O(log N)
    void insert(long long x) {
        if (!root) { root = new Node(true); }
        auto [new_node, sep] = insert_rec(root, x);
        if (new_node) {
            Node* new_root = new Node(false);
            new_root->keys.push_back(sep);
            new_root->children.push_back(root);
            new_root->children.push_back(new_node);
            update_node(new_root);
            root = new_root;
        }
    }

    // 値 x を集合から「1つだけ」削除する
    // 削除に成功した場合は true、値が存在しなかった場合は false を返す
    // 計算量: O(log N)
    bool erase_one(long long x) {
        if (!root) return false;
        auto res = erase_rec(root, x, false);
        trim_root();
        return res.first > 0;
    }

    // 値 x を集合から「すべて」削除する
    // 実際に削除された要素の個数を返す
    // 計算量: O(k log N) (kは削除された要素数)
    long long erase_all(long long x) {
        if (!root) return 0;
        long long total = 0;
        while (true) {
            auto res = erase_rec(root, x, true);
            trim_root();
            if (res.first == 0) break;
            total += res.first;
            if (!root) break;
        }
        return total;
    }

    // 集合に含まれる全要素数を返す
    // 計算量: O(1)
    long long size() { return root ? root->total_weight : 0; }

    // 集合が空であるかを判定する
    // 計算量: O(1)
    bool empty() { return size() == 0; }

    // 集合に含まれる全要素の総和を返す
    // 計算量: O(1)
    long long sum() { return root ? root->total_sum : 0; }

    // 集合内の最小値を返す
    // 注意: 集合が空の状態で呼び出すと runtime_error 例外を投げる
    // 計算量: O(log N)
    long long min() {
        if (!root) throw runtime_error("Empty set");
        Node* curr = root;
        while (!curr->is_leaf) curr = curr->children[0];
        return curr->keys[0];
    }

    // 集合内の最大値を返す
    // 注意: 集合が空の状態で呼び出すと runtime_error 例外を投げる
    // 計算量: O(log N)
    long long max() {
        if (!root) throw runtime_error("Empty set");
        Node* curr = root;
        while (!curr->is_leaf) curr = curr->children.back();
        return curr->keys.back();
    }

    // 値 x 未満の要素の個数（0-indexedでの最初の位置）を返す
    // 計算量: O(log N)
    long long rank(long long x) {
        if (!root) return 0;
        Node* curr = root;
        long long res = 0;
        while (!curr->is_leaf) {
            int idx = lower_bound(curr->keys.begin(), curr->keys.end(), x) - curr->keys.begin();
            for (int i = 0; i < idx; ++i) res += curr->children[i]->total_weight;
            curr = curr->children[idx];
        }
        res += distance(curr->keys.begin(), lower_bound(curr->keys.begin(), curr->keys.end(), x));
        return res;
    }

    // 値 x 未満（strict <）の要素の個数を返す (rankと同等)
    // 計算量: O(log N)
    long long count_lt(long long x) { return rank(x); }

    // 値 x 以下（<=）の要素の個数を返す
    // 計算量: O(log N)
    long long count_le(long long x) {
        if (!root) return 0;
        Node* curr = root;
        long long res = 0;
        while (!curr->is_leaf) {
            int idx = upper_bound(curr->keys.begin(), curr->keys.end(), x) - curr->keys.begin();
            for (int i = 0; i < idx; ++i) res += curr->children[i]->total_weight;
            curr = curr->children[idx];
        }
        res += distance(curr->keys.begin(), upper_bound(curr->keys.begin(), curr->keys.end(), x));
        return res;
    }

    // 値 x より大きい（strict >）要素の個数を返す
    // 計算量: O(log N)
    long long count_gt(long long x) { return size() - count_le(x); }

    // 値 x 以上（>=）の要素の個数を返す
    // 計算量: O(log N)
    long long count_ge(long long x) { return size() - count_lt(x); }

    // 値 x と等しい要素の個数を返す
    // 計算量: O(log N)
    long long count(long long x) { return count_le(x) - count_lt(x); }

    // 値 x 未満 (strict <) の全要素の総和を返す
    // 計算量: O(log N)
    long long sum_lt(long long x) {
        if (!root) return 0;
        Node* curr = root;
        long long res = 0;
        while (!curr->is_leaf) {
            int idx = lower_bound(curr->keys.begin(), curr->keys.end(), x) - curr->keys.begin();
            for (int i = 0; i < idx; ++i) res += curr->children[i]->total_sum;
            curr = curr->children[idx];
        }
        auto it = lower_bound(curr->keys.begin(), curr->keys.end(), x);
        for (auto i = curr->keys.begin(); i != it; ++i) res += *i;
        return res;
    }

    // 値 x 以下 (<=) の全要素の総和を返す
    // 計算量: O(log N)
    long long sum_le(long long x) {
        if (!root) return 0;
        Node* curr = root;
        long long res = 0;
        while (!curr->is_leaf) {
            int idx = upper_bound(curr->keys.begin(), curr->keys.end(), x) - curr->keys.begin();
            for (int i = 0; i < idx; ++i) res += curr->children[i]->total_sum;
            curr = curr->children[idx];
        }
        auto it = upper_bound(curr->keys.begin(), curr->keys.end(), x);
        for (auto i = curr->keys.begin(); i != it; ++i) res += *i;
        return res;
    }

    // 値 x より大きい (strict >) 全要素の総和を返す
    // 計算量: O(log N)
    long long sum_gt(long long x) {
        return sum() - sum_le(x);
    }

    // 値 x 以上 (>=) の全要素の総和を返す
    // 計算量: O(log N)
    long long sum_ge(long long x) {
        return sum() - sum_lt(x);
    }

    // 半開区間 [l, r) に含まれる全要素の総和を返す
    // 注意: l >= r の場合は 0 を返す
    // 計算量: O(log N)
    long long sum_range(long long l, long long r) {
        if (l >= r) return 0;
        return sum_lt(r) - sum_lt(l);
    }

    // 値 x が集合に含まれているかを判定する
    // 計算量: O(log N)
    bool contains(long long x) { return count(x) > 0; }

    // 集合内で小さい方から k 番目（0-indexed）の値を返す
    // 注意: k が範囲外（k < 0 または k >= size()）の場合は runtime_error 例外を投げる
    // 計算量: O(log N)
    long long kth(long long k) {
        if (!root || k < 0 || k >= root->total_weight) throw runtime_error("OOB");
        Node* curr = root;
        while (!curr->is_leaf) {
            for (size_t i = 0; i < curr->children.size(); ++i) {
                long long w = curr->children[i]->total_weight;
                if (k < w) { curr = curr->children[i]; break; }
                k -= w;
            }
        }
        return curr->keys[k];
    }

    // 集合内で小さい方から k 番目（0-indexed）の値を返す (kthと同等)
    // 注意: k が範囲外の場合は runtime_error 例外を投げる
    // 計算量: O(log N)
    long long kth_smallest(long long k) {
        return kth(k);
    }

    // 集合内で大きい方から k 番目（0-indexed）の値を返す
    // 注意: k が範囲外の場合は runtime_error 例外を投げる
    // 計算量: O(log N)
    long long kth_largest(long long k) {
        return kth(size() - 1 - k);
    }

    // 値 x より大きい（strict >）最小の要素を返す
    // 存在しない場合は std::nullopt を返す
    // 計算量: O(log N)
    optional<long long> next(long long x) {
        long long r = count_le(x);
        if (r >= size()) return nullopt;
        return kth(r);
    }

    // 値 x 以上（>=）の最小の要素を返す
    // 存在しない場合は std::nullopt を返す
    // 計算量: O(log N)
    optional<long long> next_eq(long long x) {
        long long r = count_lt(x);
        if (r >= size()) return nullopt;
        return kth(r);
    }

    // 値 x より小さい（strict <）最大の要素を返す
    // 存在しない場合は std::nullopt を返す
    // 計算量: O(log N)
    optional<long long> prev(long long x) {
        long long r = count_lt(x);
        if (r == 0) return nullopt;
        return kth(r - 1);
    }

    // 値 x 以下（<=）の最大の要素を返す
    // 存在しない場合は std::nullopt を返す
    // 計算量: O(log N)
    optional<long long> prev_eq(long long x) {
        long long r = count_le(x);
        if (r == 0) return nullopt;
        return kth(r - 1);
    }

    // 集合内のすべての要素を昇順に並べた vector を返す
    // 計算量: O(N)
    vector<long long> vec() {
        vector<long long> res;
        if (!root) return res;
        Node* curr = root;
        while (!curr->is_leaf) curr = curr->children[0];
        while (curr) {
            res.insert(res.end(), curr->keys.begin(), curr->keys.end());
            curr = curr->next_leaf;
        }
        return res;
    }
};

// --- テスト群 ---
template <int B>
void run_tests(int num_queries, int val_range, const string& test_name) {
    OrderedMultiSet<B> bst;
    multiset<long long> ms;
    long long current_sum = 0;
    mt19937_64 rng(1337);
    
    cout << "Starting " << test_name << " (B=" << B << ", " << num_queries << " queries)..." << endl;

    for (int i = 0; i < num_queries; ++i) {
        int op = rng() % 100;
        long long val = rng() % val_range; 

        if (op < 45) { // insert
            bst.insert(val);
            ms.insert(val);
            current_sum += val;
        } 
        else if (op < 60) { // erase_one
            bool b_erased = bst.erase_one(val);
            auto it = ms.find(val);
            bool ms_erased = (it != ms.end());
            assert(b_erased == ms_erased);
            if (ms_erased) {
                ms.erase(it);
                current_sum -= val;
            }
        } 
        else if (op < 70) { // erase_all
            long long b_erased = bst.erase_all(val);
            long long ms_erased = ms.count(val);
            assert(b_erased == ms_erased);
            if (ms_erased > 0) {
                ms.erase(val);
                current_sum -= val * ms_erased;
            }
        }

        assert(bst.size() == ms.size());
        assert(bst.empty() == ms.empty());
        assert(bst.sum() == current_sum);

        // 各種カウント・探索クエリのテスト（10回に1回程度実行して軽量化）
        if (!ms.empty() && rng() % 10 == 0) {
            long long test_val = rng() % (val_range + 10);
            
            // prev/next tests
            auto it_next = ms.upper_bound(test_val);
            if (it_next == ms.end()) assert(!bst.next(test_val).has_value());
            else assert(bst.next(test_val).value() == *it_next);

            auto it_next_eq = ms.lower_bound(test_val);
            if (it_next_eq == ms.end()) assert(!bst.next_eq(test_val).has_value());
            else assert(bst.next_eq(test_val).value() == *it_next_eq);
            
            auto it_prev = ms.lower_bound(test_val);
            if (it_prev == ms.begin()) assert(!bst.prev(test_val).has_value());
            else assert(bst.prev(test_val).value() == *std::prev(it_prev));

            auto it_prev_eq = ms.upper_bound(test_val);
            if (it_prev_eq == ms.begin()) assert(!bst.prev_eq(test_val).has_value());
            else assert(bst.prev_eq(test_val).value() == *std::prev(it_prev_eq));

            // count/rank tests
            assert(bst.count(test_val) == ms.count(test_val));
            assert(bst.count_lt(test_val) == distance(ms.begin(), ms.lower_bound(test_val)));
            assert(bst.count_le(test_val) == distance(ms.begin(), ms.upper_bound(test_val)));
            assert(bst.count_gt(test_val) == distance(ms.upper_bound(test_val), ms.end()));
            assert(bst.count_ge(test_val) == distance(ms.lower_bound(test_val), ms.end()));

            // kth test
            long long k = rng() % ms.size();
            auto it_k = ms.begin();
            advance(it_k, k);
            assert(bst.kth(k) == *it_k);
        }
    }
    
    vector<long long> bst_vec = bst.vec();
    vector<long long> ms_vec(ms.begin(), ms.end());
    assert(bst_vec == ms_vec);

    cout << test_name << " passed successfully!" << endl;
}

void stress_exhaustive_test() {
    cout << "Running Exhaustive Small Stress Test..." << endl;
    for (int seed = 0; seed < 100; ++seed) {
        OrderedMultiSet<4> bst;
        multiset<long long> ms;
        mt19937 rng(seed);

        for (int i = 0; i < 500; ++i) {
            int val = rng() % 10;
            if (rng() % 2) {
                bst.insert(val);
                ms.insert(val);
            } else if (!ms.empty()) {
                // 削除対象をランダム選択
                auto it = ms.begin();
                advance(it, rng() % ms.size());
                bst.erase_one(*it);
                ms.erase(it);
            }
            
            // 検証
            assert(bst.size() == ms.size());
            if (!ms.empty()) {
                assert(bst.min() == *ms.begin());
                assert(bst.max() == *ms.rbegin());
                
                // 全要素走査の一致
                vector<long long> bst_v = bst.vec();
                vector<long long> ms_v(ms.begin(), ms.end());
                assert(bst_v == ms_v);
            }
        }
    }
    cout << "Exhaustive stress test passed!" << endl;
}

    cout << "Running Rank/Kth Exhaustive Test..." << endl;

    OrderedMultiSet<4> bst;
    multiset<long long> ms;

    mt19937 rng(0);

    for(int step = 0; step < 5000; step++) {

        int op = rng() % 3;
        int x = rng() % 10;

        if(op == 0) {
            bst.insert(x);
            ms.insert(x);
        }
        else if(op == 1) {
            bst.erase_one(x);
            auto it = ms.find(x);
            if(it != ms.end()) ms.erase(it);

            bst.erase_all(x);
            ms.erase(x);

        for(int q = -2; q <= 12; q++) {

            );

            assert(
                bst.count_le(q) ==

            );

            assert(
                bst.count_gt(q) ==
                distance(ms.upper_bound(q), ms.end())
            );

            assert(
                bst.count_ge(q) ==
                distance(ms.lower_bound(q), ms.end())
            );
        }

        // kth全探索
        for(size_t k = 0; k < v.size(); k++) {
            assert(bst.kth(k) == v[k]);
        }
    }

    cout << "Rank/Kth Exhaustive Test passed!" << endl;
}

void random_test() {
    cout << "--- 総和系関数のランダムテストを開始します ---" << endl;
    
    // 木の分割・マージを頻発させるため、あえて B=4 と小さくしてテスト
    OrderedMultiSet<4> st;
    multiset<long long> ms;
    
    mt19937_64 rng(42); // 固定シード
    
    for (int i = 0; i < 2000; ++i) {
        int op = rng() % 6;
        if (op == 0) {
            // 挿入
            long long x = rng() % 100;
            st.insert(x);
            ms.insert(x);
        } else if (op == 1) {
            // 削除
            if (!ms.empty()) {
                auto it = ms.begin();
                advance(it, rng() % ms.size());
                long long x = *it;
                st.erase_one(x);
                ms.erase(ms.find(x));
            }
        } else {
            // 各種総和関数のテストクエリ
            long long x = rng() % 100;
            
            // 1. sum_lt の愚直計算
            long long naive_sum_lt = 0;
            for (long long v : ms) if (v < x) naive_sum_lt += v;
            assert(st.sum_lt(x) == naive_sum_lt);
            
            // 2. sum_le の愚直計算
            long long naive_sum_le = 0;
            for (long long v : ms) if (v <= x) naive_sum_le += v;
            assert(st.sum_le(x) == naive_sum_le);
            
            // 3. sum_gt の愚直計算
            long long naive_sum_gt = 0;
            for (long long v : ms) if (v > x) naive_sum_gt += v;
            assert(st.sum_gt(x) == naive_sum_gt);
            
            // 4. sum_ge の愚直計算
            long long naive_sum_ge = 0;
            for (long long v : ms) if (v >= x) naive_sum_ge += v;
            assert(st.sum_ge(x) == naive_sum_ge);
            
            // 5. sum_range の愚直計算 (半開区間 [l, r))
            long long l = rng() % 100;
            long long r = rng() % 100;
            if (l > r) swap(l, r);
            long long naive_sum_range = 0;
            for (long long v : ms) if (v >= l && v < r) naive_sum_range += v;
            assert(st.sum_range(l, r) == naive_sum_range);
        }
    }
    cout << "ランダムテストが正常に完了しました（すべてのクエリで一致）。" << endl;
}

    cout << "現在の要素の総和: " << st.sum() << endl;

    // すべて削除する検証
    long long erased_all_count = st.erase_all(20);
    cout << "20 をすべて削除しました。削除個数 [erase_all(20)]: " << erased_all_count << " 個" << endl;
    print_vec(); // 20が完全に消滅していることを確認
    cout << "現在の要素の総和: " << st.sum() << endl;

    // 存在しない要素の削除
    bool erased_none = st.erase_one(99);
    cout << "存在しない 99 を1つ削除しようとしました [erase_one(99)]: " << (erased_none ? "成功" : "失敗") << endl;

    return;
}
