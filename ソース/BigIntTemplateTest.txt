/*
============================================================
  BigInt（多倍長整数）テンプレート
============================================================

■ 概要
  任意精度整数を扱うクラス。
  四則演算（+,-,*,/,%）、比較、gcd/lcm、pow に対応。

■ 内部表現
  - vector<int> a : 下位桁から保持（little endian）
  - 各要素は BASE = 1e9（10^9）進数
  - bool neg : 負数フラグ

  例：
    1234567890123
    → a = [890123, 234567, 1]（イメージ）

■ 設計方針
  - 符号と絶対値を完全分離
  - 演算は基本「絶対値で処理 → 符号を後付け」
  - normalize() によって常に正規形を維持

------------------------------------------------------------

■ 不変条件（重要）
  - a に leading zero は存在しない
    → つまり a.empty() ⇔ 値は0
  - 0 のとき必ず neg = false

------------------------------------------------------------

■ 基本関数

◆ normalize()
  - 末尾の0を削除
  - a.empty() の場合 neg=false にする
  → すべての演算の最後に必ず呼ぶこと

◆ cmp_abs()
  - 絶対値同士の比較
  - 符号は見ない

------------------------------------------------------------

■ 四則演算

◆ 加算・減算
  - add_abs / sub_abs を内部で使用
  - sub_abs は「|a| >= |b|」前提（重要）

  ロジック：
    同符号 → 加算
    異符号 → 大きい方から小さい方を引く

◆ 乗算
  - O(n^2) の単純実装
  - long long で中間計算（オーバーフロー防止）

◆ 除算（重要）
  - Knuth風の正規化割り算を使用
  - 計算量：O(n^2)

  手順：
    1. norm = BASE / (b.back()+1) で正規化
    2. 上位2桁で商を推定
    3. 必要なら補正（最大2回程度）
    4. 最後に余りを逆正規化

------------------------------------------------------------

■ 剰余（%）
  - C++と同じ定義
    → 余りは「被除数と同符号」

  例：
    -7 % 3 = -1

------------------------------------------------------------

■ gcd / lcm

◆ gcd
  - ユークリッドの互除法
  - 必ず絶対値で処理

◆ lcm
  - lcm(a,b) = a/gcd(a,b) * b
  - オーバーフロー回避のため順序に注意

------------------------------------------------------------

■ pow
  - 二分累乗（O(log n)）
  - 指数は long long

------------------------------------------------------------

■ 計算量まとめ

  +, -        : O(n)
  *           : O(n^2)
  /, %        : O(n^2)
  gcd         : O(n^2 log n)
  pow         : O(n^2 log exp)

------------------------------------------------------------

■ 注意点（超重要）

1. normalize忘れ
   → ほぼ確実にバグる

2. 0の扱い
   → a.empty() && neg=false を維持

3. sub_absの前提
   → |a| >= |b| を必ず守る

4. オーバーフロー
   → 掛け算・割り算は long long を使う

5. 除算の符号
   → 最後に neg ^ b.neg で決める

6. % の符号
   → 被除数と同符号（C++仕様）

7. lcmの計算順
   → (a / gcd) * b にする

8. absの実装
   → 符号だけ消す（値は触らない）

------------------------------------------------------------

■ 拡張候補

  - 高速化（Karatsuba / FFT）
  - mod付き演算
  - 文字列I/O最適化

------------------------------------------------------------

■ 想定用途（競プロ）

  - 桁数制限のない整数計算
  - 高精度計算
  - 特殊問題（巨大数、桁DPなど）

  ※ 通常は modint の方が速いので使い分けること

============================================================
*/
struct BigInt {
    static const int BASE = 1000000000;
    vector<int> a; // 下位桁から
    bool neg;

    // ===== コンストラクタ =====
    BigInt(long long x = 0) {
        neg = x < 0;
        if (x < 0) x = -x;
        while (x) {
            a.push_back(x % BASE);
            x /= BASE;
        }
        normalize();
    }

    BigInt(const string& s) {
        neg = false;
        a.clear();

        int pos = 0;
        if (!s.empty() && s[0] == '-') {
            neg = true;
            pos = 1;
        }

        for (int i = (int)s.size(); i > pos; i -= 9) {
            int l = max(pos, i - 9);
            int len = i - l;
            int x = stoi(s.substr(l, len));
            a.push_back(x);
        }

        normalize();
    }

    string to_string() const {
        if (a.empty()) return "0";

        string s = neg ? "-" : "";
        s += std::to_string(a.back());

        for (int i = (int)a.size() - 2; i >= 0; i--) {
            string t = std::to_string(a[i]);
            s += string(9 - t.length(), '0') + t;
        }

        return s;
    }

    // ===== 正規化 =====
    void normalize() {
        while (!a.empty() && a.back() == 0) a.pop_back();
        if (a.empty()) neg = false; // 0は正
    }

    // ===== 絶対値比較 =====
    int cmp_abs(const BigInt& b) const {
        if (a.size() != b.a.size())
            return a.size() < b.a.size() ? -1 : 1;
        for (int i = (int)a.size() - 1; i >= 0; i--) {
            if (a[i] != b.a[i])
                return a[i] < b.a[i] ? -1 : 1;
        }
        return 0;
    }

    BigInt abs() const {
        BigInt r = *this;
        r.neg = false;
        return r;
    }

    // ===== 比較 =====
    bool operator<(const BigInt& b) const {
        if (neg != b.neg) return neg;
        int c = cmp_abs(b);
        return neg ? (c > 0) : (c < 0);
    }

    bool operator==(const BigInt& b) const {
        return neg == b.neg && a == b.a;
    }

    bool operator!=(const BigInt& b) const {
        return !(*this == b);
    }

    bool operator<=(const BigInt& b) const {
        return !(*this > b);
    }

    bool operator>(const BigInt& b) const {
        return b < *this;
    }

    bool operator>=(const BigInt& b) const {
        return !(*this < b);
    }

    static BigInt add_abs(const BigInt& a, const BigInt& b) {
        BigInt res;
        res.neg = false;

        int n = a.a.size(), m = b.a.size();
        int sz = max(n, m);
        res.a.resize(sz);

        long long carry = 0;

        for (int i = 0; i < sz; i++) {
            long long cur = carry;
            if (i < n) cur += a.a[i];
            if (i < m) cur += b.a[i];

            res.a[i] = cur % BASE;
            carry = cur / BASE;
        }

        if (carry) res.a.push_back(carry);

        res.normalize();
        return res;
    }

    static BigInt sub_abs(const BigInt& a, const BigInt& b) {
        BigInt res;
        res.neg = false;

        int n = a.a.size(), m = b.a.size();
        res.a.resize(n);

        long long borrow = 0;

        for (int i = 0; i < n; i++) {
            long long cur = (long long)a.a[i] - borrow;
            if (i < m) cur -= b.a[i];

            if (cur < 0) {
                cur += BASE;
                borrow = 1;
            } else {
                borrow = 0;
            }

            res.a[i] = (int)cur;
        }

        res.normalize();
        return res;
    }

    BigInt operator+(const BigInt& b) const {
        if (neg == b.neg) {
            BigInt res = add_abs(*this, b);
            res.neg = neg;
            return res;
        } else {
            int c = cmp_abs(b);
            if (c == 0) {
                return BigInt(0);
            } else if (c > 0) {
                BigInt res = sub_abs(*this, b);
                res.neg = neg;
                return res;
            } else {
                BigInt res = sub_abs(b, *this);
                res.neg = b.neg;
                return res;
            }
        }
    }

    BigInt operator-(const BigInt& b) const {
        BigInt nb = b;
        nb.neg = !b.neg;
        return *this + nb;
    }

    BigInt operator*(const BigInt& b) const {
        BigInt res;
        res.neg = neg ^ b.neg;

        int n = a.size(), m = b.a.size();
        res.a.assign(n + m, 0);

        for (int i = 0; i < n; i++) {
            long long carry = 0;
            for (int j = 0; j < m || carry; j++) {
                long long cur = res.a[i + j]
                    + (long long)a[i] * (j < m ? b.a[j] : 0)
                    + carry;

                res.a[i + j] = (int)(cur % BASE);
                carry = cur / BASE;
            }
        }

        res.normalize();
        return res;
    }

    BigInt operator/(int v) const {
        assert(v != 0);

        BigInt res;
        res.neg = (neg ^ (v < 0));

        long long dv = v >= 0 ? v : -(long long)v;

        res.a.resize(a.size());

        long long rem = 0;

        for (int i = (int)a.size() - 1; i >= 0; i--) {
            long long cur = a[i] + rem * BASE;
            res.a[i] = (int)(cur / dv);
            rem = cur % dv;
        }

        res.normalize();
        return res;
    }

    int operator%(int v) const {
        assert(v != 0);

        long long dv = v >= 0 ? v : -(long long)v;
        long long rem = 0;

        for (int i = (int)a.size() - 1; i >= 0; i--) {
            long long cur = a[i] + rem * BASE;
            rem = cur % dv;
        }

        return neg ? -rem : rem;
    }

    // 返り値: {quotient, remainder} （どちらも非負）
    static pair<BigInt, BigInt> divmod_abs(BigInt a, BigInt b) {
        assert(!b.a.empty()); // b != 0

        if (a.cmp_abs(b) < 0) {
            return {BigInt(0), a};
        }

        int n = a.a.size();
        int m = b.a.size();

        // --- 正規化 ---
        long long norm = BASE / ((long long)b.a.back() + 1);
        a = a * (int)norm;
        b = b * (int)norm;

        BigInt q;
        q.a.assign(n - m + 1, 0);

        BigInt r;
        r.a.assign(a.a.begin(), a.a.end());
        r.a.push_back(0); // 番兵

        for (int i = n - m; i >= 0; i--) {
            long long r2 = ((long long)r.a[i + m] * BASE + r.a[i + m - 1]);
            long long d = r2 / b.a[m - 1];

            if (d >= BASE) d = BASE - 1;

            // --- b * d を引く ---
            BigInt bd = b * (int)d;

            // shift
            bd.a.insert(bd.a.begin(), i, 0);

            while (r.cmp_abs(bd) < 0) {
                d--;
                bd = b * (int)d;
                bd.a.insert(bd.a.begin(), i, 0);
            }

            r = r - bd;
            q.a[i] = (int)d;
        }

        q.normalize();

        // --- 余りを戻す ---
        r = r / (int)norm;
        r.normalize();

        return {q, r};
    }

    BigInt operator%(const BigInt& b) const {
        assert(!b.a.empty());

        auto [q, r] = divmod_abs(abs(), b.abs());
        r.neg = neg;
        r.normalize();
        return r;
    }

    BigInt& operator+=(const BigInt& b) {
        *this = *this + b;
        return *this;
    }

    BigInt& operator-=(const BigInt& b) {
        *this = *this - b;
        return *this;
    }

    BigInt& operator*=(const BigInt& b) {
        *this = *this * b;
        return *this;
    }

    BigInt& operator/=(int v) {
        *this = *this / v;
        return *this;
    }

    BigInt operator/(const BigInt& b) const {
        assert(!b.a.empty());

        auto [q, r] = divmod_abs(abs(), b.abs());
        q.neg = neg ^ b.neg;
        q.normalize();
        return q;
    }

    static BigInt gcd(BigInt a, BigInt b) {
        a.neg = b.neg = false; // 絶対値でやる

        while (!b.a.empty()) {
            BigInt r = a % b;
            a = b;
            b = r;
        }
        return a;
    }

    static BigInt lcm(const BigInt& a, const BigInt& b) {
        if (a.a.empty() || b.a.empty()) return BigInt(0);

        BigInt g = gcd(a, b);
        BigInt res = (a / g) * b;

        res.neg = false; // lcmは常に非負
        return res;
    }

    static BigInt pow(BigInt base, long long exp) {
        assert(exp >= 0);

        BigInt res(1);

        while (exp > 0) {
            if (exp & 1) res = res * base;
            base = base * base;
            exp >>= 1;
        }

        return res;
    }
};

long long to_ll(const BigInt& x) {
    long long res = 0;
    for (int i = (int)x.a.size() - 1; i >= 0; i--) {
        res = res * BigInt::BASE + x.a[i];
    }
    return x.neg ? -res : res;
}

void test_step1() {
    mt19937_64 rng(0);

    for (int t = 0; t < 100000; t++) {
        long long x = (long long)(rng() % 2000000000LL) - 1000000000LL;
        long long y = (long long)(rng() % 2000000000LL) - 1000000000LL;

        BigInt a(x), b(y);

        // < の検証
        if ((a < b) != (x < y)) {
            cout << "Error <\n";
            cout << x << " " << y << "\n";
            exit(0);
        }

        // == の検証
        if ((a == b) != (x == y)) {
            cout << "Error ==\n";
            cout << x << " " << y << "\n";
            exit(0);
        }

        // 対称性チェック
        if ((a < b) && (b < a)) {
            cout << "Error antisymmetry\n";
            exit(0);
        }
    }

    cout << "Step1 OK\n";
}

void test_string_constructor() {
    mt19937_64 rng(0);

    for (int t = 0; t < 100000; t++) {
        long long x = (long long)(rng() % 2000000000LL) - 1000000000LL;

        string s = std::to_string(x);

        BigInt a(x);
        BigInt b(s);

        if (!(a == b)) {
            cout << "Error constructor mismatch\n";
            cout << "x = " << x << "\n";
            cout << "a = " << a.to_string() << "\n";
            cout << "b = " << b.to_string() << "\n";
            exit(0);
        }

        // 逆変換チェック
        if (b.to_string() != s) {
            cout << "Error to_string mismatch\n";
            cout << "original = " << s << "\n";
            cout << "got = " << b.to_string() << "\n";
            exit(0);
        }
    }

    cout << "String constructor OK\n";
}

        // 加算
        {
            BigInt c = a + b;
            long long expected = x + y;

            if (to_ll(c) != expected) {
                cout << "Error +\n";
                cout << "x = " << x << " y = " << y << "\n";
                cout << "got = " << c.to_string() << "\n";
                cout << "expected = " << expected << "\n";
                exit(0);
            }
        }

        // 減算
        {
            BigInt c = a - b;
            long long expected = x - y;

            if (to_ll(c) != expected) {
                cout << "Error -\n";
                cout << "x = " << x << " y = " << y << "\n";
                cout << "got = " << c.to_string() << "\n";
                cout << "expected = " << expected << "\n";
                exit(0);
            }
        }
    }

    cout << "Step3 OK\n";
}

void test_minus_equal() {

        long long y = (long long)(rng() % 2000000000LL) - 1000000000LL;

        BigInt c = a;
        c -= b;

        BigInt d = BigInt(x) - BigInt(y);

            cout << "Error -= mismatch\n";
            cout << "x = " << x << " y = " << y << "\n";
            cout << "c = " << c.to_string() << "\n";
            cout << "d = " << d.to_string() << "\n";
            exit(0);
        }

        // long longとの比較
        long long expected = x - y;
        if (to_ll(c) != expected) {
            cout << "Error -= value\n";
            cout << "x = " << x << " y = " << y << "\n";
            cout << "got = " << c.to_string() << "\n";
            cout << "expected = " << expected << "\n";
            exit(0);
        }
    }

    cout << "operator-= OK\n";
}

void test_step5A() {
    mt19937_64 rng(0);

    for (int t = 0; t < 200000; t++) {
        long long x = (long long)(rng() % 1000000000000LL) - 500000000000LL;
        int v = (int)(rng() % 1000000) - 500000;
        if (v == 0) v = 1;

        BigInt a(x);

        BigInt q = a / v;
        int r = a % v;

        long long expected_q = x / v;
        long long expected_r = x % v;

        if (to_ll(q) != expected_q) {
            cout << "Error / int\n";
            cout << "x=" << x << " v=" << v << "\n";
            cout << "got=" << q.to_string() << "\n";
            cout << "expected=" << expected_q << "\n";
            exit(0);
        }

        if (r != expected_r) {
            cout << "Error % int\n";
            cout << "x=" << x << " v=" << v << "\n";
            cout << "got=" << r << "\n";
            cout << "expected=" << expected_r << "\n";
            exit(0);
        }

        // 検算: q*v + r == x
        BigInt check = q * BigInt(v) + BigInt(r);
        if (to_ll(check) != x) {
            cout << "Error reconstruction\n";
            cout << "x=" << x << " v=" << v << "\n";
            exit(0);
        }
    }

    cout << "Step5A OK\n";
}

void test_all() {
    mt19937_64 rng(0);

    for (int t = 0; t < 200000; t++) {
        long long x = (long long)(rng() % 1000000000LL) - 500000000LL;
        long long y = (long long)(rng() % 1000000000LL) - 500000000LL;

        if (y == 0) y = 1;

        BigInt a(x), b(y);

        // ===== 四則演算 =====
        BigInt add = a + b;
        BigInt sub = a - b;
        BigInt mul = a * b;
        BigInt div = a / b;
        BigInt mod = a % b;

        if (to_ll(add) != x + y) {
            cout << "Error +\n"; exit(0);
        }
        if (to_ll(sub) != x - y) {
            cout << "Error -\n"; exit(0);
        }
        if (to_ll(mul) != x * y) {
            cout << "Error *\n"; exit(0);
        }
        if (to_ll(div) != x / y) {
            cout << "Error /\n"; exit(0);
        }
        if (to_ll(mod) != x % y) {
            cout << "Error %\n"; exit(0);
        }

        // ===== 恒等式 =====
        if (!((div * b + mod) == a)) {
            cout << "Error divmod identity\n"; exit(0);
        }

        if (!(((a + b) - b) == a)) {
            cout << "Error add-sub identity\n"; exit(0);
        }

        if (!(((a - b) + b) == a)) {
            cout << "Error sub-add identity\n"; exit(0);
        }

        if (y != 0) {
            if (!(((a * b) / b) == a)) {
                cout << "Error mul-div identity\n"; exit(0);
            }
        }

        // ===== gcd =====
        BigInt g = BigInt::gcd(a, b);
        long long expected_g = std::gcd(x, y);

        if (to_ll(g) != expected_g) {
            cout << "Error gcd\n"; exit(0);
        }

        // 割り切れるか
        if (!(a % g == BigInt(0) && b % g == BigInt(0))) {
            cout << "Error gcd divides\n"; exit(0);
        }

        // ===== lcm =====
        BigInt l = BigInt::lcm(a, b);
        long long expected_l = std::lcm(x, y);

        if (to_ll(l) != expected_l) {
            cout << "Error lcm\n"; exit(0);
        }

        // lcm * gcd = |a*b|
        BigInt lhs = l * g;
        BigInt rhs = a * b;
        rhs.neg = false;

        if (!(lhs == rhs)) {
            cout << "Error lcm*gcd identity\n"; exit(0);
        }

        // ===== pow =====
        int e1 = rng() % 10;
        int e2 = rng() % 10;

        BigInt p1 = BigInt::pow(a, e1);
        BigInt p2 = BigInt::pow(a, e2);
        BigInt p3 = BigInt::pow(a, e1 + e2);

        if (!(p3 == p1 * p2)) {
            cout << "Error pow identity\n"; exit(0);
        }
    }

    cout << "ALL TEST OK\n";
}

void test_compare_all() {
    mt19937_64 rng(0);

    for (int t = 0; t < 200000; t++) {
        long long x = (long long)(rng() % 2000000000LL) - 1000000000LL;
        long long y = (long long)(rng() % 2000000000LL) - 1000000000LL;

        BigInt a(x), b(y);

        if ((a < b) != (x < y)) {
            cout << "Error <\n"; exit(0);
        }
        if ((a > b) != (x > y)) {
            cout << "Error >\n"; exit(0);
        }
        if ((a <= b) != (x <= y)) {
            cout << "Error <=\n"; exit(0);
        }
        if ((a >= b) != (x >= y)) {
            cout << "Error >=\n"; exit(0);
        }
        if ((a == b) != (x == y)) {
            cout << "Error ==\n"; exit(0);
        }
        if ((a != b) != (x != y)) {
            cout << "Error !=\n"; exit(0);
        }

        // 追加：整合性チェック
        if ((a < b) && (a >= b)) {
            cout << "Error consistency 1\n"; exit(0);
        }
        if ((a > b) && (a <= b)) {
            cout << "Error consistency 2\n"; exit(0);
        }
        if ((a == b) && (a != b)) {
            cout << "Error consistency 3\n"; exit(0);
        }
    }

    cout << "compare operators OK\n";
}
