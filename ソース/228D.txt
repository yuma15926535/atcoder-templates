const int m=1<<20;
struct SegTree{
    //区間集約とは無関係な値
    const ll INF=1e15;//★
    //葉の数
    ll sz;
    //完全二分木の配列
    vector<ll> seg;
    //引数の大きさで初期化
    void init(vector<ll> &v){
        int n=v.size();
        sz=1;
        while(sz<n)sz*=2;
        seg.resize(sz*2-1,INF);
        //葉の初期化
        rep(i,0,n)seg[sz-1+i]=v[i];
        //葉以外のノードの初期化
        rrep(i,sz-1,0)seg[i]=min(seg[i*2+1],seg[i*2+2]);//★
    }
    //値の更新
    void update(int pos,ll x){
        pos+=sz-1;
        seg[pos]=x;//★
        while(pos>0){
            pos=(pos-1)/2;
            seg[pos]=min(seg[pos*2+1],seg[pos*2+2]);//★
        }
    }
    //[a,b)を集計
    ll query(int a,int b){
        return sub_query(a,b,0,sz,0);
    }
    //k:[l,r)を表すノードのindex
    ll sub_query(int a,int b,int l,int r,int k){
        if(r<=a||b<=l)return INF;
        if(a<=l&&r<=b)return seg[k];
        ll vl=sub_query(a,b,l,(l+r)/2,k*2+1);
        ll vr=sub_query(a,b,(l+r)/2,r,k*2+2);
        return min(vl,vr);//★
    }
    void output(){
        rep(i,0,sz*2-1)cout<<seg[i]<<' ';
        cout<<'\n';
    }
};

int main(){
    BUFF;
    vector<ll>A(2*m);
    iota(nall(A),0);
    SegTree st;
    st.init(A);
    for(auto &a:A)a=-1;
    int Q;
    cin>>Q;
    rep(_,0,Q){
        int t;
        ll x;
        cin>>t>>x;
        if(t==1){
            ll idx=st.query(x%m,2*m)%m;
            st.update(idx,1e15);
            st.update(idx+m,1e15);
            A[idx]=x;
        }else cout<<A[x%m]<<'\n';
    }
    return 0;
}
