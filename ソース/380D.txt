int main(){
    cin.tie(nullptr);
    ios::sync_with_stdio(false);
    string S;
    cin>>S;
    ll sz=S.size();
    vector<ll>v(62,0);
    v[1]=1;
    rep(i,2,62)v[i]=v[i-1]*2;
    auto hanten=[&](char c)->char{
        if('a'<=c&&c<='z')return (char)(c-'a'+'A');
        return (char)(c-'A'+'a');
    };
    auto solv=[&](auto solv,ll pos)->char{
        if(pos<sz)return S[pos];
        int left=0,right=61;
        while(right-left>1){
            int mid=(right+left)/2;
            if(v[mid]<=pos/sz)left=mid;
            else right=mid;
        }
        return hanten(solv(solv,pos-sz*v[left]));
    };
    int Q;
    cin>>Q;
    rep(i,0,Q){
        ll k;
        cin>>k;
        k--;
        cout<<solv(solv,k)<<' ';
    }
    cout<<'\n';
    return 0;
}
