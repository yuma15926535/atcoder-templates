/**
 * 2^61 - 1 を法とするローリングハッシュ用のユーティリティ
 */
struct RollingHash {
    static const uint64_t MOD = (1ULL << 61) - 1;
    uint64_t base;
    vector<uint64_t> power;

    RollingHash(uint64_t n, uint64_t b = 0) {
        if (b == 0) {
            mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());
            uniform_int_distribution<uint64_t> dist(128, MOD - 1);
            base = dist(rng);
        } else {
            base = b;
        }
        power.assign(n + 1, 1);
        for (int i = 0; i < n; i++) {
            power[i + 1] = mul(power[i], base);
        }
    }

    static uint64_t add(uint64_t a, uint64_t b) {
        if ((a += b) >= MOD) a -= MOD;
        return a;
    }

    static uint64_t mul(uint64_t a, uint64_t b) {
        __uint128_t ans = (__uint128_t)a * b;
        uint64_t res = (uint64_t)(ans >> 61) + (uint64_t)(ans & MOD);
        if (res >= MOD) res -= MOD;
        return res;
    }
};

// グローバルなRollingHash設定 (必要に応じて初期化)
RollingHash* rh_ptr = nullptr;

struct S {
    uint64_t hash;
    int len;
};

S op(S a, S b) {
    if (a.len == 0) return b;
    if (b.len == 0) return a;
    return {RollingHash::add(RollingHash::mul(a.hash, rh_ptr->power[b.len]), b.hash), a.len + b.len};
}

S e() {
    return {0, 0};
}

#include <vector>
#include <algorithm>
#include <cassert>

template <class S, S (*op)(S, S), S (*e)()>
struct SegTree {
  public:
    SegTree() : SegTree(0) {}
    explicit SegTree(int n) : SegTree(std::vector<S>(n, e())) {}
    explicit SegTree(const std::vector<S>& v) : _n(int(v.size())) {
        size = 1;
        log = 0;
        while (size < _n) {
            size <<= 1;
            log++;
        }
        d = std::vector<S>(2 * size, e());
        for (int i = 0; i < _n; i++) d[size + i] = v[i];
        for (int i = size - 1; i >= 1; i--) {
            update(i);
        }
    }

    void set(int p, S x) {
        assert(0 <= p && p < _n);
        p += size;
        d[p] = x;
        for (int i = 1; i <= log; i++) update(p >> i);
    }

    S get(int p) const {
        assert(0 <= p && p < _n);
        return d[p + size];
    }

    S prod(int l, int r) const {
        assert(0 <= l && l <= r && r <= _n);
        S sml = e(), smr = e();
        l += size;
        r += size;

        while (l < r) {
            if (l & 1) sml = op(sml, d[l++]);
            if (r & 1) smr = op(d[--r], smr);
            l >>= 1;
            r >>= 1;
        }
        return op(sml, smr);
    }

    S all_prod() const { return d[1]; }

    template <class F> int max_right(int l, F f) const {
        assert(0 <= l && l <= _n);
        assert(f(e()));
        if (l == _n) return _n;
        l += size;
        S sm = e();
        do {
            while (l % 2 == 0) l >>= 1;
            if (!f(op(sm, d[l]))) {
                while (l < size) {
                    l = (2 * l);
                    if (f(op(sm, d[l]))) {
                        sm = op(sm, d[l]);
                        l++;
                    }
                }
                return l - size;
            }
            sm = op(sm, d[l]);
            l++;
        } while ((l & -l) != l);
        return _n;
    }

    template <class F> int min_left(int r, F f) const {
        assert(0 <= r && r <= _n);
        assert(f(e()));
        if (r == 0) return 0;
        r += size;
        S sm = e();
        do {
            r--;
            while (r > 1 && (r % 2)) r >>= 1;
            if (!f(op(d[r], sm))) {
                while (r < size) {
                    r = (2 * r + 1);
                    if (f(op(d[r], sm))) {
                        sm = op(d[r], sm);
                        r--;
                    }
                }
                return r + 1 - size;
            }
            sm = op(d[r], sm);
        } while ((r & -r) != r);
        return 0;
    }

  private:
    int _n, size, log;
    std::vector<S> d;

    void update(int k) { d[k] = op(d[2 * k], d[2 * k + 1]); }
};

/**
 * RollingHashをSegTreeで管理するクラス
 */
class RollingHashSegTree {
    int n;
    SegTree<S, op, e> seg;

public:
    RollingHashSegTree(const string& s) : n(s.size()) {
        if (rh_ptr == nullptr || (int)rh_ptr->power.size() < n + 1) {
            if (rh_ptr) delete rh_ptr;
            rh_ptr = new RollingHash(n);
        }
        vector<S> initial(n);
        for (int i = 0; i < n; i++) initial[i] = { (uint64_t)s[i], 1 };
        seg = SegTree<S, op, e>(initial);
    }

    void set(int i, char c) {
        seg.set(i, { (uint64_t)c, 1 });
    }

    uint64_t get_hash(int l, int r) {
        return seg.prod(l, r).hash;
    }

    bool equal(int l1, int r1, int l2, int r2) {
        if (r1 - l1 != r2 - l2) return false;
        return get_hash(l1, r1) == get_hash(l2, r2);
    }

    int lcp(int l1, int l2) {
        int len = min(n - l1, n - l2);
        int low = 0, high = len + 1;
        while (high - low > 1) {
            int mid = (low + high) / 2;
            if (get_hash(l1, l1 + mid) == get_hash(l2, l2 + mid)) low = mid;
            else high = mid;
        }
        return low;
    }

    int compare(int l1, int r1, int l2, int r2) {
        int len1 = r1 - l1;
        int len2 = r2 - l2;
        int common = lcp(l1, l2);
        if (common >= len1 || common >= len2) {
            if (len1 == len2) return 0;
            return len1 < len2 ? -1 : 1;
        }
        char c1 = (char)seg.get(l1 + common).hash;
        char c2 = (char)seg.get(l2 + common).hash;
        return c1 < c2 ? -1 : 1;
    }
};

/**
 * RollingHashSegTreeのランダムテスト
 */
void test_rolling_hash() {
    int n = 1000;
    string s = "";
    mt19937 rng(1337);
    for (int i = 0; i < n; i++) s += (char)('a' + rng() % 3);

    RollingHashSegTree rh_seg(s);

    for (int q = 0; q < 5000; q++) {
        int type = rng() % 5;
        if (type == 0) { // set
            int i = rng() % n;
            char c = (char)('a' + rng() % 3);
            s[i] = c;
            rh_seg.set(i, c);
        } else if (type == 1) { // equal
            int l1 = rng() % n, r1 = l1 + rng() % (n - l1 + 1);
            int l2 = rng() % n, r2 = l2 + rng() % (n - l2 + 1);
            bool res = rh_seg.equal(l1, r1, l2, r2);
            bool expected = s.substr(l1, r1 - l1) == s.substr(l2, r2 - l2);
            assert(res == expected);
        } else if (type == 2) { // lcp
            int l1 = rng() % n, l2 = rng() % n;
            int res = rh_seg.lcp(l1, l2);
            int expected = 0;
            while (l1 + expected < n && l2 + expected < n && s[l1 + expected] == s[l2 + expected]) expected++;
            assert(res == expected);
        } else if (type == 3) { // compare
            int l1 = rng() % n, r1 = l1 + rng() % (n - l1 + 1);
            int l2 = rng() % n, r2 = l2 + rng() % (n - l2 + 1);
            int res = rh_seg.compare(l1, r1, l2, r2);
            string s1 = s.substr(l1, r1 - l1);
            string s2 = s.substr(l2, r2 - l2);
            int expected = (s1 == s2 ? 0 : (s1 < s2 ? -1 : 1));
            assert(res == expected);
        }
    }
    cout << "RollingHashSegTree test passed!" << endl;
}
