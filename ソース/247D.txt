struct Pipe{
    vector<pair<int,int>> v;
    int index=0;
    void query1(int x,int c){
        v.push_back({x,c});
    }
    ll query2(int c){
        ll ret=0;
        while(c){
            if(c>=v[index].second){
                c-=v[index].second;
                ret+=(ll)v[index].first*v[index].second;
                v[index].second=0;
                index++;
            }else{
                ret+=(ll)v[index].first*c;
                v[index].second-=c;
                c=0;
            }
        }
        return ret;
    }
};

        }
    }

}
