struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
SmallKSumPQ

【概要】
動的集合に対して「小さい順k個」を管理するデータ構造
add / erase に対応しつつ以下を高速に取得できる

- kth（小さい順k番目）
- sum_k_smallest（小さい順k個の和）

【内部構造】
- small：小さい方k個（max-heap）
- large：残り（min-heap）
- sum_small：smallの総和
- lazy削除（unordered_map）でerase対応
- 論理サイズ(sz_small, sz_large)で管理

【計算量】
- add / erase / kth / sum：すべて O(log N)

【不変条件】
- smallは常に「小さい方k個」
- sz_small == min(k, 全体サイズ)
- smallの最大値 = kth

【実装のポイント】
- sumは「smallに入る/出るときだけ更新」
- lazy削除ではsumを触らない（重要）
- top参照前に必ずprune
- rebalanceはsz_small基準（heap.size()は使わない）

【使いどころ】
- 小さい順k個の和を動的に管理
- スライド最小k和、貪欲＋データ構造系

【注意】
- kthやsumは要素数 < k のときは無効（-1など返す）
*/
struct SmallKSumPQ {
    int k;
    int sz_small = 0;
    int sz_large = 0;
    ll sum_small = 0;

    // 小さい方k個（最大ヒープ）
    priority_queue<ll> small;

    // 残り（最小ヒープ）
    priority_queue<ll, vector<ll>, greater<ll>> large;

    unordered_map<ll,int> del_small, del_large;

    SmallKSumPQ(int _k) : k(_k) {}

    // ============================
    // ゴミ削除
    // ============================
    void prune_small() {
        while (!small.empty()) {
            ll x = small.top();
            if (del_small[x]) {
                del_small[x]--;
                small.pop();
            } else break;
        }
    }

    void prune_large() {
        while (!large.empty()) {
            ll x = large.top();
            if (del_large[x]) {
                del_large[x]--;
                large.pop();
            } else break;
        }
    }

    // ============================
    // バランス調整
    // ============================
    void rebalance() {
        prune_small();
        prune_large();

        while (sz_small > k) {
            ll x = small.top(); small.pop();
            sum_small -= x;
            large.push(x);
            sz_small--;
            sz_large++;
        }

        while (sz_small < k && sz_large > 0) {
            ll x = large.top(); large.pop();
            sum_small += x;
            small.push(x);
            sz_small++;
            sz_large--;
        }

        prune_small();
        prune_large();
    }

    // ============================
    // add
    // ============================
    void add(ll x) {
        if ((int)small.size() < k) {
            small.push(x);
            sum_small += x;
            sz_small++;
        } else {
            prune_small();
            if (!small.empty() && x <= small.top()) {
                small.push(x);
                sum_small += x;
                sz_small++;
            } else {
                large.push(x);
                sz_large++;
            }
        }
        rebalance();
    }

    // ============================
    // erase（1個削除）
    // ============================
    void erase(ll x) {
        prune_small();
        if (!small.empty() && x <= small.top()) {
            del_small[x]++;
            sum_small -= x;
            sz_small--;
            if (x == small.top()) prune_small();
        } else {
            del_large[x]++;
            sz_large--;
            if (!large.empty() && x == large.top()) prune_large();
        }
        rebalance();
    }

    // ============================
    // kth（小さい順k番目）
    // ============================
    ll kth() {
        prune_small();
        if (sz_small < k) return -1;
        return small.top();
    }

    ll sum_k_smallest() {
        if (sz_small < k) return -1;
        return sum_small;
    }
};

void random_test_SmallKSumPQ_step0() {
    mt19937 rng(0);

    for (int tc = 0; tc < 1000; tc++) {
        int k = uniform_int_distribution<int>(1, 5)(rng);

        SmallKSumPQ pq(k);
        vector<ll> v;

        for (int step = 0; step < 100; step++) {
            ll x = uniform_int_distribution<int>(0, 20)(rng);

            pq.add(x);
            v.push_back(x);

            if ((int)v.size() >= k) {
                vector<ll> tmp = v;
                sort(tmp.begin(), tmp.end());

                ll expected = tmp[k-1];
                ll actual = pq.kth();

                if (expected != actual) {
                    cout << "ERROR\n";
                    cout << "k = " << k << "\n";
                    cout << "expected = " << expected << "\n";
                    cout << "actual = " << actual << "\n";

                    cout << "v = ";
                    for (auto x : v) cout << x << " ";
                    cout << "\n";

                    return;
                }
            }
        }
    }
    cout << "OK\n";
}

void random_test_SmallKSumPQ_step1() {
    mt19937 rng(0);

    for (int tc = 0; tc < 1000; tc++) {
        int k = uniform_int_distribution<int>(1, 5)(rng);

        SmallKSumPQ pq(k);
        multiset<ll> st;

        for (int step = 0; step < 100; step++) {
            int op = uniform_int_distribution<int>(0, 1)(rng);

            if (op == 0 || st.empty()) {
                // add
                ll x = uniform_int_distribution<int>(0, 20)(rng);
                pq.add(x);
                st.insert(x);
            } else {
                // erase
                int idx = uniform_int_distribution<int>(0, (int)st.size()-1)(rng);
                auto it = st.begin();
                advance(it, idx);
                ll x = *it;

                pq.erase(x);
                st.erase(it);
            }

            if ((int)st.size() >= k) {
                vector<ll> v(st.begin(), st.end());
                sort(v.begin(), v.end());

                ll expected = v[k-1];
                ll actual = pq.kth();

                    cout << "ERROR\n";
                    cout << "k = " << k << "\n";

                    cout << "v = ";
                    for (auto x : v) cout << x << " ";
                    cout << "\n";

                    return;
                }
            }
        }
    }
    cout << "OK\n";
}
