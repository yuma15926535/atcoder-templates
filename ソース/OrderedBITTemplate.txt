    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }

/**
 * @brief Ordered BIT (個数管理・順序統計 Fenwick Tree)
 * 
 * @details 値の追加・削除、k番目の値の取得、指定値未満の個数取得などを高速に行うデータ構造。
 * 競技プログラミングにおける std::multiset の高速な代替（値の範囲が有限の整数 [0, MAX) である場合）として機能する。
 * 
 * 注意点:
 * - 管理する値は 0 以上 MAX 未満の整数に限られる。
 * - 範囲外の値が指定された場合、安全に無視またはエラー値 (-1, false, 0 など) を返す。
 * 
 * 用途:
 * - 動的な多重集合 (multiset) のシミュレーション
 * - 順位や値に応じた統計情報の取得 (k-th element)
 * 
 * 計算量:
 * - 構築: O(MAX)
 * - 追加 (insert): O(log MAX)
 * - 削除 (erase, erase_all): O(log MAX)
 * - 個数取得 (count, count_range 等): O(log MAX)
 * - 順位統計 (kth, rank, min, max 等): O(log MAX)
 * - 前後探索 (prev, next, lower_bound, upper_bound): O(log MAX)
 * - 最小値・最大値取り出し (pop_min, pop_max): O(log MAX)
 * - 全要素取得 (values, unique_values): O(MAX + size())
 * - クリア (clear): O(MAX)
 * 
 * API一覧:
 * - OrderedBIT(MAX)
 * - insert(x), erase(x), erase_all(x)
 * - count(x), contains(x)
 * - count_lt(x), count_le(x), count_gt(x), count_ge(x), count_range(l, r)
 * - rank(x), kth(k), min(), max()
 * - prev(x), next(x), lower_bound(x), upper_bound(x)
 * - pop_min(), pop_max()
 * - values(), unique_values()
 * - size(), empty(), clear()
 */
class OrderedBIT {
private:
    int max_val;
    int size_;
    std::vector<int> tree;

    // 内部用の Fenwick Tree add
    void add(int i, int x) {
        for (int idx = i + 1; idx <= max_val; idx += idx & -idx) {
            tree[idx] += x;
        }
    }

    // 内部用の Fenwick Tree sum [0, r)
    int sum(int r) const {
        int s = 0;
        for (int idx = r; idx > 0; idx -= idx & -idx) {
            s += tree[idx];
        }
        return s;
    }

public:
    /**
     * @brief コンストラクタ
     * @param MAX 管理する値の上限。値の範囲は [0, MAX) となる。
     * @note 計算量: O(MAX)
     */
    explicit OrderedBIT(int MAX) : max_val(MAX), size_(0), tree(MAX + 1, 0) {}

    /**
     * @brief 要素数を取得
     * @return 集合に含まれる全要素数
     * @note 計算量: O(1)
     */
    int size() const {
        return size_;
    }

    /**
     * @brief 空であるか判定
     * @return 空ならtrue、そうでなければfalse
     * @note 計算量: O(1)
     */
    bool empty() const {
        return size_ == 0;
    }

    /**
     * @brief 集合をクリアして空にする
     * @note 計算量: O(MAX)
     */
    void clear() {
        std::fill(tree.begin(), tree.end(), 0);
        size_ = 0;
    }

    /**
     * @brief 値 x を1つ追加する
     * @param x 追加する値 (0-index, [0, MAX))
     * @return 追加に成功したらtrue、範囲外ならfalse
     * @note 計算量: O(log MAX)
     */
    bool insert(int x) {
        if (x < 0 || x >= max_val) return false;
        add(x, 1);
        size_++;
        return true;
    }

    /**
     * @brief 値 x を1つ削除する
     * @param x 削除する値 (0-index)
     * @return 削除に成功したらtrue、存在しない・範囲外ならfalse
     * @note 計算量: O(log MAX)
     */
    bool erase(int x) {
        if (x < 0 || x >= max_val) return false;
        if (count(x) == 0) return false;
        add(x, -1);
        size_--;
        return true;
    }

    /**
     * @brief 値 x をすべて削除する
     * @param x 削除する値 (0-index)
     * @return 実際に削除された個数
     * @note 計算量: O(log MAX)
     */
    int erase_all(int x) {
        if (x < 0 || x >= max_val) return 0;
        int cnt = count(x);
        if (cnt > 0) {
            add(x, -cnt);
            size_ -= cnt;
        }
        return cnt;
    }

    /**
     * @brief 値 x の個数を取得する
     * @param x 個数を確認する値 (0-index)
     * @return 値 x の個数。範囲外なら0
     * @note 計算量: O(log MAX)
     */
    int count(int x) const {
        if (x < 0 || x >= max_val) return 0;
        // x の個数は sum(x + 1) - sum(x)
        int s = tree[x + 1];
        int l = x & (x + 1);
        for (int idx = x; idx > l; idx -= idx & -idx) {
            s -= tree[idx];
        }
        return s;
    }

    /**
     * @brief 値 x が集合に存在するか判定する
     * @param x 判定する値 (0-index)
     * @return 存在すればtrue、そうでなければfalse
     * @note 計算量: O(log MAX)
     */
    bool contains(int x) const {
        return count(x) > 0;
    }

    /**
     * @brief x 未満の要素数を取得する
     * @param x 境界値
     * @return x 未満の要素数
     * @note 計算量: O(log MAX)
     */
    int count_lt(int x) const {
        if (x <= 0) return 0;
        if (x >= max_val) return size_;
        return sum(x);
    }

    /**
     * @brief x 以下の要素数を取得する
     * @param x 境界値
     * @return x 以下の要素数
     * @note 計算量: O(log MAX)
     */
    int count_le(int x) const {
        if (x < 0) return 0;
        if (x >= max_val - 1) return size_;
        return sum(x + 1);
    }

    /**
     * @brief x より大きい要素数を取得する
     * @param x 境界値
     * @return x より大きい要素数
     * @note 計算量: O(log MAX)
     */
    int count_gt(int x) const {
        return size_ - count_le(x);
    }

    /**
     * @brief x 以上の要素数を取得する
     * @param x 境界値
     * @return x 以上の要素数
     * @note 計算量: O(log MAX)
     */
    int count_ge(int x) const {
        return size_ - count_lt(x);
    }

    /**
     * @brief 半開区間 [l, r) に含まれる要素数を取得する
     * @param l 左端 (0-index)
     * @param r 右端 (0-index)
     * @return 区間に含まれる要素数
     * @note 計算量: O(log MAX)
     */
    int count_range(int l, int r) const {
        if (l >= r) return 0;
        return count_lt(r) - count_lt(l);
    }

    /**
     * @brief x より小さい要素数を取得する (count_lt と同じ)
     * @param x 境界値
     * @return x より小さい要素数
     * @note 計算量: O(log N)
     */
    int rank(int x) const {
        return count_lt(x);
    }

    /**
     * @brief 集合内の 0-index で k 番目に小さい値を取得する
     * @param k インデックス (0-index)
     * @return k 番目に小さい値。存在しなければ -1
     * @note 計算量: O(log MAX)
     */
    int kth(int k) const {
        if (k < 0 || k >= size_) return -1;
        int idx = 0;
        int val = k + 1;
        int step = 1;
        while (step * 2 <= max_val) step *= 2;
        for (; step > 0; step /= 2) {
            if (idx + step <= max_val && tree[idx + step] < val) {
                val -= tree[idx + step];
                idx += step;
            }
        }
        return idx;
    }

    /**
     * @brief 集合内の最小値を取得する
     * @return 最小値。空なら -1
     * @note 計算量: O(log MAX)
     */
    int min() const {
        return kth(0);
    }

    /**
     * @brief 集合内の最大値を取得する
     * @return 最大値。空なら -1
     * @note 計算量: O(log MAX)
     */
    int max() const {
        return kth(size_ - 1);
    }

    /**
     * @brief x 未満で最大の要素を取得する
     * @param x 境界値
     * @return x 未満最大の値。存在しなければ -1
     * @note 計算量: O(log MAX)
     */
    int prev(int x) const {
        int cnt = count_lt(x);
        return kth(cnt - 1);
    }

    /**
     * @brief x より大きい要素で最小のものを取得する
     * @param x 境界値
     * @return x より大きい最小の値。存在しなければ -1
     * @note 計算量: O(log MAX)
     */
    int next(int x) const {
        int cnt = count_le(x);
        return kth(cnt);
    }

    /**
     * @brief x 以上で最小の要素を取得する
     * @param x 境界値
     * @return x 以上最小の値。存在しなければ -1
     * @note 計算量: O(log MAX)
     */
    int lower_bound(int x) const {
        int cnt = count_lt(x);
        return kth(cnt);
    }

    /**
     * @brief x より大きい要素で最小のものを取得する (nextと同じ)
     * @param x 境界値
     * @return x より大きい最小の値。存在しなければ -1
     * @note 計算量: O(log MAX)
     */
    int upper_bound(int x) const {
        return next(x);
    }

    /**
     * @brief 集合内の最小値を削除して返す
     * @return 削除された最小値。空なら -1
     * @note 計算量: O(log MAX)
     */
    int pop_min() {
        int val = min();
        if (val != -1) {
            erase(val);
        }
        return val;
    }

    /**
     * @brief 集合内の最大値を削除して返す
     * @return 削除された最大値。空なら -1
     * @note 計算量: O(log MAX)
     */
    int pop_max() {
        int val = max();
        if (val != -1) {
            erase(val);
        }
        return val;
    }

    /**
     * @brief 集合内の全要素を昇順に並べた vector を取得する
     * @return 重複を含む全要素の昇順 vector
     * @note 計算量: O(MAX + size())
     */
    std::vector<int> values() const {
        std::vector<int> res;
        res.reserve(size_);
        for (int i = 0; i < max_val; ++i) {
            int cnt = count(i);
            for (int j = 0; j < cnt; ++j) {
                res.push_back(i);
            }
        }
        return res;
    }

    /**
     * @brief 集合内に存在する一意な値の昇順 vector を取得する
     * @return 重複を除いた要素の昇順 vector
     * @note 計算量: O(MAX)
     */
    std::vector<int> unique_values() const {
        std::vector<int> res;
        for (int i = 0; i < max_val; ++i) {
            if (contains(i)) {
                res.push_back(i);
            }
        }
        return res;
    }
};

    std::mt19937 mt(42);
    int MAX = 500;

    OrderedBIT bit(MAX);
    std::multiset<int> naive;

    std::uniform_int_distribution<int> val_dist(-50, MAX + 50); // 範囲外も含む

    for (int q = 0; q < queries; ++q) {
        int op = op_dist(mt);
        int x = val_dist(mt);

        if (op == 0) {
            // insert
            bool b_res = bit.insert(x);
            bool n_res = (x >= 0 && x < MAX);
            assert(b_res == n_res);
            if (n_res) naive.insert(x);
        } else if (op == 1) {
            // erase
            bool b_res = bit.erase(x);
            bool n_res = false;
            auto it = naive.find(x);
            if (it != naive.end()) {
                naive.erase(it);
                n_res = true;
            }
            assert(b_res == n_res);
        } else if (op == 2) {
            // erase_all
            int b_res = bit.erase_all(x);
            int n_res = naive.count(x);
            naive.erase(x);
            assert(b_res == n_res);
        } else if (op == 3) {
            // count, contains
            assert(bit.count(x) == (int)naive.count(x));
            assert(bit.contains(x) == (naive.count(x) > 0));
        } else if (op == 4) {
            // min, max, kth
            if (naive.empty()) {
                assert(bit.min() == -1);
                assert(bit.max() == -1);
                assert(bit.kth(0) == -1);
            } else {
                assert(bit.min() == *naive.begin());
                assert(bit.max() == *naive.rbegin());
                
                std::uniform_int_distribution<int> k_dist(0, (int)naive.size() - 1);
                int k = k_dist(mt);
                auto it = naive.begin();
                std::advance(it, k);
                assert(bit.kth(k) == *it);
            }
        } else if (op == 5) {
            // count_lt, count_le, count_gt, count_ge, count_range
            int lt = 0, le = 0, gt = 0, ge = 0;
            for (int v : naive) {
                if (v < x) lt++;
                if (v <= x) le++;
                if (v > x) gt++;
                if (v >= x) ge++;
            }
            assert(bit.count_lt(x) == lt);
            assert(bit.count_le(x) == le);
            assert(bit.count_gt(x) == gt);
            assert(bit.count_ge(x) == ge);
            assert(bit.rank(x) == lt);

            int y = val_dist(mt);
            int l = x, r = y;
            if (l > r) std::swap(l, r);
            int rng = 0;
            for (int v : naive) {
                if (v >= l && v < r) rng++;
            }
            assert(bit.count_range(l, r) == rng);
        } else if (op == 6) {
            // prev, next, lower_bound, upper_bound
            int naive_prev = -1;
            int naive_next = -1;
            int naive_lb = -1;
            int naive_ub = -1;

            auto it_lb = naive.lower_bound(x);
            if (it_lb != naive.end()) naive_lb = *it_lb;

            auto it_ub = naive.upper_bound(x);
            if (it_ub != naive.end()) naive_ub = *it_ub;
            naive_next = naive_ub;

            // prev
            auto it_prev = naive.lower_bound(x);
            if (it_prev != naive.begin()) {
                naive_prev = *(--it_prev);
            }

            assert(bit.prev(x) == naive_prev);
            assert(bit.next(x) == naive_next);
            assert(bit.lower_bound(x) == naive_lb);
            assert(bit.upper_bound(x) == naive_ub);
        } else {
            // size, empty, values, unique_values
            assert(bit.size() == (int)naive.size());
            assert(bit.empty() == naive.empty());
            
            std::vector<int> b_vals = bit.values();
            std::vector<int> n_vals(naive.begin(), naive.end());
            assert(b_vals == n_vals);

            std::vector<int> b_uvals = bit.unique_values();
            std::vector<int> n_uvals;
            for (int v : naive) {
                if (n_uvals.empty() || n_uvals.back() != v) {
                    n_uvals.push_back(v);
                }
            }
            assert(b_uvals == n_uvals);
        }
    }

    std::cout << "OK" << std::endl;
    std::cout << "--- random_test_OrderedBIT 終了 ---" << std::endl << std::endl;
}
