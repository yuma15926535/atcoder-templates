int main()
{
    ll A,B,N;
    cin>>A>>B>>N;
    ll x;
    //floor(x/y)=(x-x%y)/yを使って式変形
    //floor(床関数、切り下げ)と割り算では典型の式変形
    //f((A*x)/B)-A*f(x/B)
    //=(A*x-(A*x)%B)/B-A*((x-x%B)/B)
    //=(A*(x%B)-(A(x%B))%B)/B
    //A%Bは定数、A>=A%Bなのでx%Bの最大値を取るxを使えばいい
    if(N<B)x=N;
    else x=B-1;
    cout<<(A*x)/B-A*(x/B)<<"\n";
    return 0;
}
