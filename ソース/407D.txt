#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())
template <typename T> constexpr auto ssize(const T& container) -> std::ptrdiff_t {return static_cast<std::ptrdiff_t>(container.size());}

int main(){
    int H,W;
    cin>>H>>W;
    vector A(H,vector<ll>(W));
    rep(i,0,H)rep(j,0,W)cin>>A[i][j];
    vector covered(H,vector<bool>(W,false));
    ll ans=0;
    //再帰関数をラムダ式で定義
    //再帰関数のラムダ式は自身を引数に追加
    //ドミノの置き方を全探索
    auto f=[&](auto f,vector<vector<bool>>covered,int i,int j)->void{
        //端にきていたら縦方向に移動
        if(j==W)i++,j=0;
        //縦方向に端が来ていれば遷移を終えてスコアを計算
        if(i==H){
            ll now=0;
            rep(ni,0,H)rep(nj,0,W)if(!covered[ni][nj])now^=A[ni][nj];
            ans=max(ans,now);
            return;
        }
        //すでに置かれて入れば何も置かず遷移
        if(covered[i][j])f(f,covered,i,j+1);
        else{
            //何も置かずに遷移
            f(f,covered,i,j+1)
            //横方向にドミノを置ける場合;
            if(j+1<W&&!covered[i][j+1]){
                //ドミノを横に置き遷移、遷移後置いていない状態に戻す
                covered[i][j]=covered[i][j+1]=true;
                f(f,covered,i,j+1);
                covered[i][j]=covered[i][j+1]=false;
            }
            //縦方向にドミノを置ける場合
            if(i+1<H&&!covered[i+1][j]){
                //ドミノを盾に置き遷移、遷移後置いていない状態に戻す
                covered[i][j]=covered[i+1][j]=true;
                f(f,covered,i,j+1);
                covered[i][j]=covered[i+1][j]=false;
            }
        }
    };
    f(f,covered,0,0);
    cout<<ans<<"\n";
    return 0;
}
