    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }

        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;

            cin >> u >> v >> w;

            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }

struct Treap {
    struct Node {
        int key;
        uint32_t pri;
        int sz;
        Node *l, *r;

        Node(int key, uint32_t pri)
            : key(key), pri(pri), sz(1), l(nullptr), r(nullptr) {}
    };

    using NP = Node*;

    // RNG（高速・衝突少）
    mt19937 rng;

    Treap() : rng(chrono::steady_clock::now().time_since_epoch().count()) {}

    // ノード生成（priorityは乱数）
    NP new_node(int key) {
        return new Node(key, rng());
    }

    // 部分木サイズ取得（nullは0）
    int size(NP t) const {
        return t ? t->sz : 0;
    }

    // 部分木サイズを更新（子から再計算）
    void update(NP t) {
        if (!t) return;
        t->sz = 1 + size(t->l) + size(t->r);
    }

    // 2つのTreapをマージ（lの全要素 <= rの全要素が前提）
    NP merge(NP l, NP r) {
        if (!l || !r) return l ? l : r;

        if (l->pri > r->pri) {
            l->r = merge(l->r, r);
            update(l);
            return l;
        } else {
            r->l = merge(l, r->l);
            update(r);
            return r;
        }
    }

    // key未満 / key以上 に分割
    // 戻り値: {< key, >= key}
    pair<NP, NP> split(NP t, int key) {
        if (!t) return {nullptr, nullptr};

        if (t->key < key) {
            auto [l, r] = split(t->r, key);
            t->r = l;
            update(t);
            return {t, r};
        } else {
            auto [l, r] = split(t->l, key);
            t->l = r;
            update(t);
            return {l, t};
        }
    }

    // 要素keyを1つ追加（multiset）
    // split→mergeで安全に挿入
    NP insert(NP root, int key) {
        auto [l, r] = split(root, key);
        return merge(merge(l, new_node(key)), r);
    }

    // 要素keyを1つ削除（multiset）
    // 存在すれば1つだけ削除
    NP erase(NP root, int key) {
        auto [l, r] = split(root, key);       // < key | >= key
        auto [m, r2] = split(r, key + 1);     // == key | > key

        // mの中から1つだけ削除
        if (m) {
            m = merge(m->l, m->r);
        }

        return merge(l, merge(m, r2));
    }

    // k番目の値を取得（1-index）
    // k=1が最小
    int kth(NP t, int k) {
        if (!t) return -1; // 異常（テストで弾く前提でもOK）

        int left_size = size(t->l);

        if (k <= left_size) {
            return kth(t->l, k);
        } else if (k == left_size + 1) {
            return t->key;
        } else {
            return kth(t->r, k - left_size - 1);
        }
    }

    // key未満の要素数
    int order_of_key(NP t, int key) {
        if (!t) return 0;

        if (key <= t->key) {
            // 全部左にある
            return order_of_key(t->l, key);
        } else {
            // 左 + 自分 + 右の一部
            return size(t->l) + 1 + order_of_key(t->r, key);
        }
    }

    // keyの出現回数
    int count(NP t, int key) {
        return order_of_key(t, key + 1) - order_of_key(t, key);
    }

    // key以上で最小の値（lower_bound）
    // 存在しなければ -1
    int lower_bound(NP t, int key) {
        int k = order_of_key(t, key) + 1;
        if (k > size(t)) return -1;
        return kth(t, k);
    }

    // keyより大きい最小の値（upper_bound）
    // 存在しなければ -1
    int upper_bound(NP t, int key) {
        int k = order_of_key(t, key + 1) + 1;
        if (k > size(t)) return -1;
        return kth(t, k);
    }

    // key未満で最大の値（前駆）
    // 存在しなければ -1
    int prev(NP t, int key) {
        int k = order_of_key(t, key);
        if (k == 0) return -1;
        return kth(t, k);
    }

    // keyより大きい最小の値（後継）
    // 存在しなければ -1
    int next(NP t, int key) {
        int k = order_of_key(t, key + 1) + 1;
        if (k > size(t)) return -1;
        return kth(t, k);
    }

    // 中順走査（昇順配列に変換）
    void inorder(NP t, vector<int>& v) const {
        if (!t) return;
        inorder(t->l, v);
        v.push_back(t->key);
        inorder(t->r, v);
    }
};

/*
=====================================================
★ CountSet（順序付きmultiset）
=====================================================

【できること】
・insert(x)        : xを追加
・erase(x)         : xを1個削除（存在しなければfalse）
・count_lt(x)      : x未満の個数
・count_le(x)      : x以下の個数
・count_gt(x)      : xより大きい個数
・count_ge(x)      : x以上の個数
・count_eq(x)      : xの個数
・contains(x)      : xが存在するか
・kth(k)           : k番目の値を取得（0-indexed）
・lower_bound(x)   : x以上の最小値
・upper_bound(x)   : xより大きい最小値
・prev(x)          : x未満で最大の値
・next(x)          : xより大きい最小値
・min()            : 最小値
・max()            : 最大値
・size()           : 要素数
・empty()          : 空かどうか
・to_vector()      : 昇順配列化

【計算量】
・insert / erase / count / kth
    -> 平均 O(log N)

【内部実装】
・Treap
・multiset対応（重複可）

【注意】
・lower_bound / upper_bound / prev / next
    は存在しない場合 -1 を返す
・kth(k) は 0-indexed
・空で min/max/kth は呼ばない

=====================================================
*/
struct CountSet {
private:
    struct Treap {
        struct Node {
            int key;
            uint32_t pri;
            int sz;
            Node *l, *r;

            Node(int key, uint32_t pri)
                : key(key), pri(pri), sz(1), l(nullptr), r(nullptr) {}
        };

        using NP = Node*;

        mt19937 rng;

        Treap() : rng(chrono::steady_clock::now().time_since_epoch().count()) {}

        NP root = nullptr;

        NP new_node(int key) {
            return new Node(key, rng());
        }

        int size(NP t) const {
            return t ? t->sz : 0;
        }

        void update(NP t) {
            if (!t) return;
            t->sz = 1 + size(t->l) + size(t->r);
        }

        NP merge(NP l, NP r) {
            if (!l || !r) return l ? l : r;

            if (l->pri > r->pri) {
                l->r = merge(l->r, r);
                update(l);
                return l;
            } else {
                r->l = merge(l, r->l);
                update(r);
                return r;
            }
        }

        pair<NP, NP> split(NP t, int key) {
            if (!t) return {nullptr, nullptr};

            if (t->key < key) {
                auto [l, r] = split(t->r, key);
                t->r = l;
                update(t);
                return {t, r};
            } else {
                auto [l, r] = split(t->l, key);
                t->l = r;
                update(t);
                return {l, t};
            }
        }

        NP insert(NP root, int key) {
            auto [l, r] = split(root, key);
            return merge(merge(l, new_node(key)), r);
        }

        NP erase(NP root, int key) {
            auto [l, r] = split(root, key);
            auto [m, r2] = split(r, key + 1);

            if (m) {
                m = merge(m->l, m->r);
            }

            return merge(l, merge(m, r2));
        }

        int kth(NP t, int k) const {
            int left_size = size(t->l);

            if (k <= left_size) {
                return kth(t->l, k);
            } else if (k == left_size + 1) {
                return t->key;
            } else {
                return kth(t->r, k - left_size - 1);
            }
        }

        int order_of_key(NP t, int key) const {
            if (!t) return 0;

            if (key <= t->key) {
                return order_of_key(t->l, key);
            } else {
                return size(t->l) + 1 + order_of_key(t->r, key);
            }
        }

        int count(NP t, int key) const {
            return order_of_key(t, key + 1)
                 - order_of_key(t, key);
        }
    };

    Treap tr;

public:
    CountSet() {}

    // 要素数
    int size() const {
        return tr.size(tr.root);
    }

    bool empty() const {
        return size() == 0;
    }

    // 挿入
    void insert(int x) {
        tr.root = tr.insert(tr.root, x);
    }

    // 1個削除
    bool erase(int x) {
        if (!count_eq(x)) return false;
        tr.root = tr.erase(tr.root, x);
        return true;
    }

    // x未満
    int count_lt(int x) const {
        return tr.order_of_key(tr.root, x);
    }

    // x以下
    int count_le(int x) const {
        return tr.order_of_key(tr.root, x + 1);
    }

    // xより大
    int count_gt(int x) const {
        return size() - count_le(x);
    }

    // x以上
    int count_ge(int x) const {
        return size() - count_lt(x);
    }

    // xの個数
    int count_eq(int x) const {
        return tr.count(tr.root, x);
    }

    // 存在判定
    bool contains(int x) const {
        return count_eq(x);
    }

    // k番目 (0-indexed)
    int kth(int k) const {
        return tr.kth(tr.root, k + 1);
    }

    // x以上の最初のindex
    int lower_bound_idx(int x) const {
        return count_lt(x);
    }

    // xより大きい最初のindex
    int upper_bound_idx(int x) const {
        return count_le(x);
    }

    // 最小値
    int min() const {
        return kth(0);
    }

    // 最大値
    int max() const {
        return kth(size() - 1);
    }

    // x以上の最小値
    int lower_bound(int x) const {
        int k = count_lt(x);

        if (k == size()) return -1;

        return kth(k);
    }

    // xより大きい最小値
    int upper_bound(int x) const {
        int k = count_le(x);

        if (k == size()) return -1;

        return kth(k);
    }

    // x未満で最大
    int prev(int x) const {
        int k = count_lt(x);

        if (k == 0) return -1;

        return kth(k - 1);
    }

    // xより大きい最小
    int next(int x) const {
        int k = count_le(x);

        if (k == size()) return -1;

        return kth(k);
    }

    // 昇順配列化
    vector<int> to_vector() const {
        vector<int> v;

        auto dfs = [&](auto&& self, Treap::NP t) -> void {
            if (!t) return;
            self(self, t->l);
            v.push_back(t->key);
            self(self, t->r);
        };

        dfs(dfs, tr.root);

        return v;
    }
};

void random_test() {
    mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());

    CountSet st;
    multiset<int> ms;

    for (int t = 1; t <= 100000; t++) {

        int op = rng() % 8;
        int x = rng() % 20;

        // insert
        if (op == 0) {
            st.insert(x);
            ms.insert(x);
        }

        // erase
        else if (op == 1) {
            bool a = st.erase(x);

            bool b = false;
            auto it = ms.find(x);

            if (it != ms.end()) {
                ms.erase(it);
                b = true;
            }

            assert(a == b);
        }

        // count_lt
        else if (op == 2) {
            int a = st.count_lt(x);

            int b = 0;
            for (int v : ms) {
                if (v < x) b++;
            }

            assert(a == b);
        }

        // count_le
        else if (op == 3) {
            int a = st.count_le(x);

            int b = 0;
            for (int v : ms) {
                if (v <= x) b++;
            }

            assert(a == b);
        }

        // count_gt
        else if (op == 4) {
            int a = st.count_gt(x);

            int b = 0;
            for (int v : ms) {
                if (v > x) b++;
            }

            assert(a == b);
        }

        // count_ge
        else if (op == 5) {
            int a = st.count_ge(x);

            int b = 0;
            for (int v : ms) {
                if (v >= x) b++;
            }

            assert(a == b);
        }

        // count_eq
        else if (op == 6) {
            int a = st.count_eq(x);
            int b = ms.count(x);

            assert(a == b);
        }

        // kth
        else {
            if (ms.empty()) continue;

            int k = rng() % ms.size();

            int a = st.kth(k);

            auto it = ms.begin();
            advance(it, k);

            int b = *it;

            assert(a == b);
        }

        // size
        assert(st.size() == (int)ms.size());

        // to_vector
        vector<int> a = st.to_vector();
        vector<int> b(ms.begin(), ms.end());

        assert(a == b);
    }

    cout << "Random Test OK" << endl;
}

                b = *it;
            }

            assert(a == b);
        }

        // next
        else {
            int a = st.next(x);

            auto it = ms.upper_bound(x);

            int b = (it == ms.end() ? -1 : *it);

            assert(a == b);
        }

        // size
        assert(st.size() == (int)ms.size());

        // empty
        assert(st.empty() == ms.empty());

        // to_vector
        vector<int> a = st.to_vector();
        vector<int> b(ms.begin(), ms.end());

        assert(a == b);

        // min
        if (!ms.empty()) {
            assert(st.min() == *ms.begin());
        }

        // max
        if (!ms.empty()) {
            assert(st.max() == *prev(ms.end()));
        }
    }

    cout << "Random Test2 OK" << endl;
}
