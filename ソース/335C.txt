#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor

struct xy{
    int x;
    int y;
    xy operator+=(const xy &right){
        x+=right.x;
        y+=right.y;
        return *this;
    }
};
map<char,xy> m;
deque<xy> deq;
struct solv{
    solv(int n){
        erep(i,1,n)deq.push_back({(int)i,0});
        m['R']={1,0};
        m['L']={-1,0};
        m['U']={0,1};
        m['D']={0,-1};
    }
    void print(int pos){
        cout<<deq[pos-1].x<<" "<<deq[pos-1].y<<"\n";
    }
    void move(char c){
        xy p=deq[0];
        p+=m[c];
        deq.push_front(p);
        deq.pop_back();
    }
};

            s.move(c);
        }
    }
    return 0;
}
