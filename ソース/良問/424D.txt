#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define BUFF cin.tie(nullptr);ios::sync_with_stdio(false)
#define next_p(v) next_permutation(v.begin(),v.end())
template<class T> using PQ = priority_queue<T, vector<T>>;//大きい順
template<class T> using PQ_g = priority_queue<T, vector<T>, greater<T>>;//小さい順
#define chmax(x,y) x = max(x,y)
#define chmin(x,y) x = min(x,y)
#define popcount __builtin_popcountll

int main(){
    BUFF;
    int T;
    cin>>T;
    //bitDP:部分集合の全探索
    rep(_,0,T){
        int H,W;
        cin>>H>>W;
        int k=1<<W;
        vector<string>S(H);
        rep(j,0,H)cin>>S[j];

        //2行の状態を事前計算
        //(2^7)^2*7=10^5なのでT:100回計算しても間に合う
        vector<vector<bool> >ok(k,vector<bool>(k,true));
        rep(i,0,k){
            rep(j,0,k){
                rep(ii,0,W-1){
                    //3:2進法で11
                    if((((i>>ii)&3)==3)&&(((j>>ii)&3)==3)){
                        ok[i][j]=false;
                        break;
                    }
                }
            }
        }

        //各行の状態がiの時に条件を満たしていればそれまでに(上の行を含む)追加した黒の最小値
        //全部白である0をコスト0にしておいて初期化
        vector<int>dp(k,1e9);
        dp[0]=0;
        //行分繰り返す
        rep(i,0,H){
            //state:i行目の状態(j列目が黒であれば2^jの位に1を立てる)
            int state=0;
            rep(j,0,W)if(S[i][j]=='#')state+=(1<<j);
            //dpはi-1行目、dp2はi行目
            vector<int>dp2(k,1e9);
            //i行目を全通り試す
            rep(j,0,k){
                //jがi行目の黒を全部含んでいるときのみ更新する
                if((j|state)==state){
                    //i-1行目を全通り試す
                    rep(jj,0,k){
                        //i-1行目とi行目が条件を満たすならば更新
                        //(i行目の状態j)と(i-1行目の状態jjにi行目をstateからjにへこうするコスト)の最小値
                        //排他的論理和は変更のあるbitのみ1が立つ
                        //jがi行目の黒を全部含んでいるため追加した黒のみ残る
                        if(ok[jj][j])chmin(dp2[j],dp[jj]+popcount(((unsigned int)(j^state))));
                    }
                }
            }
            //dpをi行目に更新
            dp=dp2;
        }

        int ans=1e9;
        rep(i,0,k)chmin(ans,dp[i]);
        cout<<ans<<'\n';
    }
    return 0;
}
