#define chmax(x,y) x = max(x,y)
#define chmin(x,y) x = min(x,y)

//操作は区間から偶数の区間を削除することと同値
//区間の削除はDP()典型
int main(){
    int N;
    cin>>N;
    vector<ll>A(N);
    rep(i,0,N)cin>>A[i];
    //範囲外にならないように大きめに定義
    vector<ll>dp(N+5,0);
    rep(i,0,N){
        //i+1の更新のみなので追加前に削除する心配がない
        //初手は追加しかできない
        dp[i+1]=dp[i]+A[i];
        ///追加したものと削除したもののうち最大値のみを採用
        if(i>0)chmax(dp[i+1],dp[i-1]);
    }
    cout<<dp[N]<<'\n';
    return 0;
}
