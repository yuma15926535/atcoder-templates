struct Interval{
    vector<bool>rec;
    int ret;
    Interval(int n){
        //rec[i]:0,N+1に白を追加し、iとi+1が異なる場合にのみtrue
        rec.resize(n+1,false);
        ret=0;
    }
    int upd(int pos){
        //pos-1とpos+1の色が同じ場合にのみretが変化する
        bool b=rec[pos-1]==rec[pos];
        rec[pos-1]=rec[pos-1]^true;
        rec[pos]=rec[pos]^true;
        if(b){
            if(rec[pos])ret+=2;
            else ret-=2;
        }
        return ret/2;
    }
};

int main(){
    cin.tie(nullptr);
    ios::sync_with_stdio(false);
    int N,Q;
    cin>>N>>Q;
    vector<int>A(Q);
    rep(i,0,Q)cin>>A[i];
    Interval intv=Interval(N);
    rep(i,0,Q)cout<<intv.upd(A[i])<<"\n";
    return 0;
}
