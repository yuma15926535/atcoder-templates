//最大部分配列問題
//Kadane's algorithm
//引数の連続部分列の最大値を求め、[最大値,開始index,終了index]を返却
tuple<ll, ll, ll> kadane(const vector<ll>& arry) {
    ll max_cur = arry[0];
    ll max_glb = arry[0];
    ll start = 0, end = 0, temp_start = 0;

    rep(i,1,arry.size()) {
        if (arry[i] > max_cur + arry[i]) {
            max_cur = arry[i];
            temp_start = i;
        } else max_cur += arry[i];

        if (max_cur > max_glb) {
            max_glb = max_cur;
            start = temp_start;
            end = i;
        }
    }

    return {max_glb, start, end};
}
//引数の連続部分列の最小値を求め、[最小値,開始index,終了index]を返却
tuple<ll, ll, ll> kadane2(const vector<ll>& arry) {
    ll min_cur = arry[0];
    ll min_glb = arry[0];
    ll start = 0, end = 0, temp_start = 0;

    rep(i,1,arry.size()) {
        if (arry[i] < min_cur + arry[i]) {
            min_cur = arry[i];
            temp_start = i;
        } else min_cur += arry[i];

        if (min_cur < min_glb) {
            min_glb = min_cur;
            start = temp_start;
            end = i;
        }
    }

    return {min_glb, start, end};
}
