struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
=====================================================
★ モノイド定義（括弧列SegTree）
=====================================================

sum      : '(' を +1, ')' を -1 した総和
min_pref : prefix の最小値

正しい括弧列 ⇔
sum == 0 && min_pref >= 0
=====================================================
*/

struct S {
    int sum;
    int min_pref;
};

// ★区間結合
S combine(S L, S R) {
    return {
        L.sum + R.sum,
        min(L.min_pref, L.sum + R.min_pref)
    };
}

// ★単位元
S id() {
    return {0, 0};
}

/*
=====================================================
SegTree 本体
=====================================================
*/

struct SegTree {
    int n;
    vector<S> seg;

    SegTree(int sz) {
        n = 1;
        while (n < sz) n <<= 1;
        seg.assign(2*n, id());
    }

    // 文字列から初期化
    SegTree(const string& s) {

        int sz = s.size();

        n = 1;
        while (n < sz) n <<= 1;

        seg.assign(2*n, id());

        for (int i = 0; i < sz; i++) {

            if (s[i] == '(')
                seg[n+i] = {1, 0};

            else
                seg[n+i] = {-1, -1};
        }

        for (int i = n-1; i > 0; i--)
            seg[i] = combine(seg[i<<1], seg[i<<1|1]);
    }

    // ========================
    // 点更新
    // ========================
    void set(int i, char c) {

        i += n;

        if (c == '(')
            seg[i] = {1, 0};

        else
            seg[i] = {-1, -1};

        while (i >>= 1)
            seg[i] = combine(seg[i<<1], seg[i<<1|1]);
    }

    // ========================
    // 区間取得 [l,r)
    // ========================
    S range(int l, int r) {

        S L = id(), R = id();

        for (l += n, r += n; l < r; l >>= 1, r >>= 1) {

            if (l & 1)
                L = combine(L, seg[l++]);

            if (r & 1)
                R = combine(seg[--r], R);
        }

        return combine(L, R);
    }

    // ========================
    // 区間が正しい括弧列か
    // ========================
    bool ok(int l, int r) {

        S x = range(l, r);

        return x.sum == 0 && x.min_pref >= 0;
    }

    // ========================
    // 1文字取得
    // ========================
    char get(int i) {

        S x = seg[n+i];

        return (x.sum == 1 ? '(' : ')');
    }

    // ========================
    // 全体が正しい括弧列か
    // ========================
    bool all_ok() {
        return ok(0, n);
    }
};

mt19937 rng;
void random_test() {

    rng.seed(chrono::steady_clock::now().time_since_epoch().count());

    rep(_, 0, 1000) {

        int N = rng() % 20 + 1;
        int Q = rng() % 200 + 1;

        string s(N, '(');

        rep(i, 0, N) {
            if (rng() % 2)
                s[i] = '(';
            else
                s[i] = ')';
        }

        SegTree seg(s);

        string naive = s;

        rep(q, 0, Q) {

            // 0 : 点更新
            // 1 : 区間判定
            int t = rng() % 2;

            if (t == 0) {

                int i = rng() % N;

                char c;

                if (rng() % 2)
                    c = '(';
                else
                    c = ')';

                seg.set(i, c);
                naive[i] = c;
            }

            else {

                int l = rng() % N;
                int r = rng() % N;

                if (l > r) swap(l, r);

                r++;

                bool seg_ans = seg.ok(l, r);

                // naive 判定
                int sum = 0;
                int mn = 0;

                rep(i, l, r) {

                    if (naive[i] == '(')
                        sum++;
                    else
                        sum--;

                    mn = min(mn, sum);
                }

                bool naive_ans = (sum == 0 && mn >= 0);

                if (seg_ans != naive_ans) {

                    cout << "NG\n";

                    cout << "文字列 : " << naive << "\n";

                    cout << "l = " << l << "\n";
                    cout << "r = " << r << "\n";

                    cout << "seg = " << seg_ans << "\n";
                    cout << "naive = " << naive_ans << "\n";

                    assert(false);
                }
            }
        }
    }

    cout << "OK\n";
}
