#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    int N;
    cin>>N;
    vector<vector<ll>> dp(2,vector<ll>(9,1));
    int m=998244353;
    //末尾の数字を持つ桁DP
    //直前の状態しか参照しないので余りをindexに使う
    //メモリの削減が利点
    rep(i,1,N)rep(j,0,9){
        if(j==0)dp[i%2][j]=(dp[(i-1)%2][j]+dp[(i-1)%2][j+1])%m;
        else if(j==8)dp[i%2][j]=(dp[(i-1)%2][j]+dp[(i-1)%2][j-1])%m;
        else dp[i%2][j]=(dp[(i-1)%2][j-1]+dp[(i-1)%2][j]+dp[(i-1)%2][j+1])%m;
    }
    ll ans=0;
    rep(i,0,9)ans+=dp[(N-1)%2][i];
    cout<<ans%m<<"\n";
    return 0;
}
