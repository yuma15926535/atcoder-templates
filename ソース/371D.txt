using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"

#define next_p(v) next_permutation(v.begin(),v.end())

int N;
vector<int> where;
//二分探索
vector<ll> how;
ll getl(int l){
    if(l<=where[0])return 0;
    int left=0,right=N-1;
    while(right-left>1){
        int mid=(right+left)/2;
        if(l<=where[mid])right=mid;
        else left=mid;
    }
    return how[left];
}
ll getr(int r){
    if(r>=where[N-1])return how[N-1];
    int left=0,right=N-1;
    while(right-left>1){
        int mid=(right+left)/2;
        if(r>=where[mid])left=mid;
        else right=mid;
    }
    return how[left];
}
ll solv(int l,int r){
    //区間に入っていない場合は別で処理
    if(r<where[0]||l>where[N-1])return 0;
    return getr(r)-getl(l);
}

int main()
{
    //範囲が広い累積和
    //vectorで全体を持つことができないので、存在する部分だけ持ち、両方で二分探索
    cin.tie(nullptr);
    ios::sync_with_stdio(false);
    cin>>N;
    where.resize(N);
    how.resize(N);
    rep(i,0,N)cin>>where[i];
    rep(i,0,N)cin>>how[i];
    rep(i,1,N)how[i]+=how[i-1];
    int Q;
    cin>>Q;
    rep(i,0,Q){
        int l,r;
        cin>>l>>r;
        cout<<solv(l,r)<<"\n";
    }
    return 0;
}
