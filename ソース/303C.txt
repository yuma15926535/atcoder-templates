int N,M,H,K;
string S;
struct xy{
    int x;
    int y;
    xy operator+=(const xy &right){
        x+=right.x;
        y+=right.y;
        return *this;
    }
    //setの要素に構造体を用いる場合は比較演算子を定義
    bool operator<( const xy& right ) const {
        if(x==right.x)return y<right.y;
        return x<right.x;
    }
};
set<xy> s;
bool solv(){
    int index=0;
    map<char,xy> m;
    m['R']={1,0};
    m['L']={-1,0};
    m['U']={0,1};
    m['D']={0,-1};
    xy now={0,0};
    while(index<N){
        H--;
        now+=m[S[index]];
        if(H<0)return false;
        if(s.count(now)&&H<K){
            H=K;
            s.erase(now);
        }
        index++;
    }
    return true;
}

int main()
{

    rep(i,0,M){
        int x,y;
        cin>>x>>y;
        s.insert({x,y});
    }
    if(solv())YN;
    return 0;
}
