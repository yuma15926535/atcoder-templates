struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// =============================================================================
//  modint
//  （作り直し版：元ファイル 425E.txt）
// -----------------------------------------------------------------------------
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
//modint:「剰余演算」を効率的かつ簡潔に扱うための構造体(剰余が自動計算される)
//加減乗除、累乗(pow 引数乗を返却)、逆元(inv)
// ======================= ここからテンプレート本体（コピー範囲） =======================
// ACL の modint（mod を実行時に決める）: modint::set_mod(m); で mod を設定（m は素数でなくてよい）
//   modint x = 5; x *= 3; x.val() で値（0 以上 m 未満）。m が素数でないと x.inv() は使えないことがある
// パスカルの三角形で nCr の表を作る（逆元を使わないので mod が素数でなくてよい）  O(N^2) 時間・メモリ
//   auto C = pascal_table<modint>(N);   C[n][r] = nCr（0 ≤ r ≤ n ≤ N。r > n は 0）
//   T は modint / modint998244353 / long long（n ≤ 66 まで溢れない）など何でもよい
template<class T> vector<vector<T>> pascal_table(int N) {
    vector<vector<T>> C(N + 1, vector<T>(N + 1, T(0)));
    for (int n = 0; n <= N; n++) {
        C[n][0] = T(1);
        for (int r = 1; r <= n; r++) C[n][r] = C[n - 1][r - 1] + C[n - 1][r];
    }
    return C;
}
// ======================= ここまでテンプレート本体 =======================
// ---- 以下 ABC425 E の解答（元のテンプレート部分）----
const int M = 5000;
modint C[M+1][M+1];

void solve_one() {
    int n;
    cin >> n;
    int s = 0;
    modint ans = 1;
    rep(i,0,n) {
        int c;
        cin >> c;
        s += c;
        ans *= C[s][c];
    }
    cout << ans.val() << '\n';
}



// -----------------------------------------------------------------------------
//  実戦での使用例（ABC425 E）… 標準入力を読むので main からは呼ばない
// -----------------------------------------------------------------------------

int solve_ABC425E(){
    BUFF;
    int T, mod;
    cin >> T >> mod;
    //modintのmodを指定
    modint::set_mod(mod);
    //modが総数とは限らない場合は逆元が存在しない場合がある
    //Cの和が小さいのでパスカルの三角形で計算
    //構築は2乗オーダーになる
    C[0][0] = 1;
    rep(i,0,M)rep(j,0,i+1) {
        C[i+1][j] += C[i][j];
        C[i+1][j+1] += C[i][j];
    }
    rep(ti,0,T) solve_one();
    return 0;
}

void demo(){
    cout << "===== modint（実行時 mod）と pascal_table デモ =====\n";
    modint::set_mod(12);                       // 素数でない mod
    auto C = pascal_table<modint>(10);
    cout << "mod 12: C[10][3] = " << C[10][3].val() << "   // 120 mod 12 = 0\n";
    cout << "mod 12: C[7][2]  = " << C[7][2].val() << "   // 21 mod 12 = 9\n";
    cout << "mod 12: C[3][5]  = " << C[3][5].val() << "   // r > n は 0\n";
    auto D = pascal_table<long long>(60);      // 厳密な値（long long）
    cout << "long long: C[60][30] = " << D[60][30] << "\n";
}
