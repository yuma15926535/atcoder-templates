    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
=====================================================
★ ノード情報（区間代入 + Kadane）
=====================================================
*/

struct S {

    ll sum;   // 区間和

    ll pref;  // 最大 prefix 和

    ll suff;  // 最大 suffix 和

    ll best;  // 最大部分列和

    int len;  // 区間長
};

/*
=====================================================
★ 更新情報
=====================================================
*/

using F = ll;

/*
=====================================================
★ 演算定義
=====================================================
*/

// ノード結合
S op(S l, S r) {

    // dummy node 対策
    if(l.len == 0)
        return r;

    if(r.len == 0)
        return l;

    return {

        // sum
        l.sum + r.sum,

        // pref
        max(l.pref, l.sum + r.pref),

        // suff
        max(r.suff, r.sum + l.suff),

        // best
        max({
            l.best,
            r.best,
            l.suff + r.pref
        }),

        // len
        l.len + r.len
    };
}

// 更新適用
S mapping(F f, S x) {

    if(f == -LINF)
        return x;

    ll total = f * x.len;

    ll val;

    // 空部分列不許可
    if(f > 0)
        val = total;
    else
        val = f;

    return {
        total,
        val,
        val,
        val,
        x.len
    };
}

// 更新の合成
F composition(F f, F g) {

    return (f != -LINF ? f : g);
}

S e() {

    return {
        0,
        -LINF,
        -LINF,
        -LINF,
        0
    };
}

F id() { return -LINF; }

/*
=====================================================
Lazy SegTree 本体
=====================================================
*/

struct LazyKadaneSegTree {

    int n;

    vector<S> seg;

    vector<F> lazy;

    /*
    =====================================================
    ★ 値 → ノード
    =====================================================
    */

    S make_node(ll x) {

        return {
            x,
            x,
            x,
            x,
            1
        };
    }

    /*
    =====================================================
    ★ construct
    =====================================================
    */

    LazyKadaneSegTree(const vector<ll>& v) {

        int sz = v.size();

        n = 1;

        while(n < sz)
            n <<= 1;

        seg.assign(2 * n, e());

        lazy.assign(2 * n, id());

        for(int i = 0; i < sz; i++)
            seg[n + i] = make_node(v[i]);

        /*
        =============================================
        ★ dummy leaf
        =============================================
        */

        for(int i = sz; i < n; i++) {

            seg[n + i] = {
                0,
                -LINF,
                -LINF,
                -LINF,
                1
            };
        }

        for(int i = n - 1; i > 0; i--)
            seg[i] = op(seg[i << 1], seg[i << 1 | 1]);
    }

    /*
    =====================================================
    ★ apply node
    =====================================================
    */

    void apply_node(int k, F f) {

        seg[k] = mapping(f, seg[k]);

        if(k < n)
            lazy[k] = composition(f, lazy[k]);
    }

    /*
    =====================================================
    ★ push
    =====================================================
    */

    void push(int k) {

        if(lazy[k] == id())
            return;

        apply_node(k << 1, lazy[k]);

        apply_node(k << 1 | 1, lazy[k]);

        lazy[k] = id();
    }

    // ---- 非再帰の区間作用・区間取得（ACL の lazy_segtree と同じやり方。再帰版より 2〜3 倍速い）----
    void apply_range(int l, int r, F f) {
        if (l >= r) return;
        const int lg = __builtin_ctz((unsigned)n);
        l += n; r += n;
        for (int i = lg; i >= 1; i--) {               // 上から遅延を下ろす
            if (((l >> i) << i) != l) push(l >> i);
            if (((r >> i) << i) != r) push((r - 1) >> i);
        }
        for (int a = l, b = r; a < b; a >>= 1, b >>= 1) {
            if (a & 1) apply_node(a++, f);
            if (b & 1) apply_node(--b, f);
        }
        for (int i = 1; i <= lg; i++) {               // 下から集約し直す
            if (((l >> i) << i) != l) { int k = l >> i; seg[k] = op(seg[k << 1], seg[k << 1 | 1]); }
            if (((r >> i) << i) != r) { int k = (r - 1) >> i; seg[k] = op(seg[k << 1], seg[k << 1 | 1]); }
        }
    }
    S prod_range(int l, int r) {
        if (l >= r) return e();
        const int lg = __builtin_ctz((unsigned)n);
        l += n; r += n;
        for (int i = lg; i >= 1; i--) {
            if (((l >> i) << i) != l) push(l >> i);
            if (((r >> i) << i) != r) push((r - 1) >> i);
        }
        S sml = e(), smr = e();
        for (; l < r; l >>= 1, r >>= 1) {
            if (l & 1) sml = op(sml, seg[l++]);
            if (r & 1) smr = op(seg[--r], smr);
        }
        return op(sml, smr);
    }

    /*
    =====================================================
    ★ range apply
    =====================================================
    */

    void apply(int l, int r, F f) {

        apply_range(l, r, f);
    }

    /*
    =====================================================
    ★ range query
    =====================================================
    */

    // 区間 [l, r) の集約値 S を返す（ACL の prod と同じ）

    S prod(int l, int r) { return prod_range(l, r); }

    /*
    =====================================================
    ★ 最大部分列和
    =====================================================
    */

    ll range_max_subarray(int l, int r) {

        return prod_range(l, r).best;
    }

    /*
    =====================================================
    ★ 全体最大部分列和
    =====================================================
    */

    ll all_max_subarray() {

        return seg[1].best;
    }
};
// ======================= ここまでテンプレート本体 =======================

void random_test_lazy_kadane() {

    random_device rd;

    mt19937 rng(rd());

    rep(_, 0, 10000) {

        /*
        =====================================================
        ★ 配列生成
        =====================================================
        */

        int N = uniform_int_distribution<int>(1, 20)(rng);

        vector<ll> A(N);

        rep(i, 0, N) {

            A[i] = uniform_int_distribution<ll>(-20, 20)(rng);
        }

        LazyKadaneSegTree seg(A);

        /*
        =====================================================
        ★ ランダムクエリ
        =====================================================
        */

        rep(q, 0, 200) {

            int type = uniform_int_distribution<int>(0, 1)(rng);

            /*
            =================================================
            ★ 区間代入更新
            =================================================
            */

            if(type == 0) {

                int l =
                    uniform_int_distribution<int>(0, N - 1)(rng);

                int r =
                    uniform_int_distribution<int>(l + 1, N)(rng);

                ll x =
                    uniform_int_distribution<ll>(-20, 20)(rng);

                /*
                =============================================
                ★ brute
                =============================================
                */

                rep(i, l, r)
                    A[i] = x;

                /*
                =============================================
                ★ segtree
                =============================================
                */

                seg.apply(l, r, x);
            }

            /*
            =================================================
            ★ 区間最大部分列和
            =================================================
            */

            else {

                int l =
                    uniform_int_distribution<int>(0, N - 1)(rng);

                int r =
                    uniform_int_distribution<int>(l + 1, N)(rng);

                /*
                =============================================
                ★ brute
                =============================================
                */

                ll brute = -LINF;

                rep(i, l, r) {

                    ll sum = 0;

                    rep(j, i, r) {

                        sum += A[j];

                        brute = max(brute, sum);
                    }
                }

                /*
                =============================================
                ★ segtree
                =============================================
                */

                ll seg_ans =
                    seg.range_max_subarray(l, r);

                /*
                =============================================
                ★ verify
                =============================================
                */

                if(brute != seg_ans) {

                    cerr << "WA detected\n";

                    cerr << "N = " << N << "\n";

                    cerr << "A : ";

                    for(auto x : A)
                        cerr << x << " ";

                    cerr << "\n";

                    cerr << "l r = "
                         << l << " "
                         << r << "\n";

                    cerr << "brute = "
                         << brute << "\n";

                    cerr << "seg   = "
                         << seg_ans << "\n";

                    assert(false);
                }
            }
        }
    }

    cerr << "Lazy Kadane SegTree Random Test Passed\n";
}

void demo(){
    cout << "===== 区間代入・区間の最大部分配列和 デモ =====\n";
    vector<ll> a = {2, -5, 3, 4, -1, 2};
    LazyKadaneSegTree seg(a);                       // 配列から構築 O(N)
    cout << "a = {2, -5, 3, 4, -1, 2}\n";
    cout << "seg.range_max_subarray(0, 6) = " << seg.range_max_subarray(0, 6) << "   // 3+4-1+2 → 8\n";
    seg.apply(0, 2, -1);                  // a[0..1] = -1 に代入 → {-1,-1,3,4,-1,2}
    cout << "seg.apply(0, 2, -1) 後: seg.range_max_subarray(0, 2) = " << seg.range_max_subarray(0, 2) << "   // 空にしないので → -1\n";
    cout << "seg.all_max_subarray() = " << seg.all_max_subarray() << "   // → 8\n";
}

void solv(){
    random_test_lazy_kadane();
    demo();
    return;
}
