struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

//  配列で作る双方向連結リスト
//  （作り直し版：元ファイル 455D.txt）
// -----------------------------------------------------------------------------
//  説明
//    - 順序を持つ要素の「つなぎ替え」
//    - 区間を切り出して別の場所へ移動
//    - 前後関係のみが重要（ランダムアクセス不要）
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
// ======================= ここからテンプレート本体（コピー範囲） =======================
/*
==================================================
 配列で作る双方向連結リスト
==================================================

【用途】
・順序を持つ要素の「つなぎ替え」
・区間を切り出して別の場所へ移動
・前後関係のみが重要（ランダムアクセス不要）

【計算量】
・link / cut / insert：すべて O(1)

【注意】
・番兵（ダミー）を使うとバグりにくい
==================================================
*/
struct LinkedList {
    int n;
    vector<int> nxt, prv;
    int H, T;   // 番兵：H = 先頭の前、T = 末尾の後ろ（push_back / push_front / to_vector などで使う 1 本のリスト）

    // 頂点 0〜n を使える。-1 は「なし」
    LinkedList(int n) : n(n), nxt(n+3, -1), prv(n+3, -1), H(n+1), T(n+2) {
        nxt[H] = T;
        prv[T] = H;
    }

    // ===== 番兵つきのリスト（H と T の間）=====
    void push_back(int x)  { insert_after(prv[T], x); }   // 末尾に x
    void push_front(int x) { insert_after(H, x); }        // 先頭に x
    int front() const { return nxt[H] == T ? -1 : nxt[H]; }   // 空なら -1
    int back()  const { return prv[T] == H ? -1 : prv[T]; }
    bool empty() const { return nxt[H] == T; }
    // 先頭から順に並べた vector
    vector<int> to_vector() const { return walk(nxt[H]); }
    // start から nxt をたどった列（-1 か番兵で止まる）。番兵を使わないリストにも使える
    vector<int> walk(int start) const {
        vector<int> res;
        for (int v = start; v != -1 && v != T; v = nxt[v]) if (v != H) res.push_back(v);
        return res;
    }

    // x の前に y を挿入（x の前がある＝番兵つきのリストか、先頭でないこと）
    void insert_before(int x, int y) { insert_after(prv[x], y); }

    // x の後ろに y をつなぐ（x -> y）
    void link(int x, int y) {
        nxt[x] = y;
        if (y != -1) prv[y] = x;
    }

    // x をリストから外す
    void cut(int x) {
        int p = prv[x];
        int q = nxt[x];
        if (p != -1) nxt[p] = q;
        if (q != -1) prv[q] = p;
        prv[x] = nxt[x] = -1;
    }

    // x の後ろに y を挿入（y は単体 or 別リストでもOK）
    void insert_after(int x, int y) {
        int z = nxt[x];
        link(x, y);
        link(y, z);
    }

    // 区間 [l, r] を切り出す（l と r は同一リスト内）
    // 戻り値：(前のノード, 次のノード)
    pair<int,int> cut_segment(int l, int r) {
        int p = prv[l];
        int q = nxt[r];
        if (p != -1) nxt[p] = q;
        if (q != -1) prv[q] = p;
        prv[l] = nxt[r] = -1;
        return {p, q};
    }

    // 区間 [l, r] を x の後ろに挿入
    void insert_segment(int x, int l, int r) {
        int z = nxt[x];
        link(x, l);
        link(r, z);
    }
};

// -----------------------------------------------------------------------------
//  実戦での使用例（ABC455 D）… 標準入力を読むので main からは呼ばない
// -----------------------------------------------------------------------------

void solve_ABC455D(){
    int N,Q;
    cin>>N>>Q;
    LinkedList llist(2*N);
    rep(i,0,N)llist.nxt[N+i]=i,llist.prv[i]=N+i;
    rep(i,0,Q){
        int C,P;
        cin>>C>>P;
        C--;P--; 
        int p=llist.prv[C];
        llist.prv[C]=P;
        llist.nxt[P]=C;
        llist.nxt[p]=-1;
    }
    rep(i,0,N){
        int x=N+i,ans=0;
        while(llist.nxt[x]!=-1){
            x=llist.nxt[x];
            ans++;
        }
        cout<<ans<<' ';
    }
    cout<<'\n';
    return;
}

void demo(){
    cout << "===== 配列で作る双方向連結リスト デモ =====\n";
    LinkedList L(6);                       // 頂点 0〜6（n+1 個）。-1 は「なし」
    for (int i = 1; i < 5; i++) L.link(i, i + 1);     // 1-2-3-4-5（番兵を使わず自分でつなぐ）
    auto print = [&](int head){ for (int v : L.walk(head)) cout << v << " "; cout << "\n"; };
    cout << "最初            : "; print(1);
    L.cut(3);                              // 3 を外す
    cout << "cut(3)          : "; print(1);
    L.insert_after(1, 3);                  // 1 の後ろに 3
    cout << "insert_after(1,3): "; print(1);
    auto [p, q] = L.cut_segment(4, 5);     // 区間 4-5 を切り出す（前後の頂点が返る）
    cout << "cut_segment(4,5) : "; print(1); cout << "  （前 = " << p << ", 次 = " << q << "）\n";
    L.insert_segment(1, 4, 5);             // 1 の後ろに 4-5 を入れる
    cout << "insert_segment(1,4,5): "; print(1);

    // 番兵つきで使う（先頭・末尾の管理を任せる）
    LinkedList M(5);
    for (int v : {2, 4, 1}) M.push_back(v);   // 2 4 1
    M.push_front(5);                          // 5 2 4 1
    M.insert_before(4, 3);                    // 5 2 3 4 1
    M.cut(2);                                 // 5 3 4 1（先頭・末尾でも同じように外せる）
    cout << "番兵つき: to_vector() = "; for (int v : M.to_vector()) cout << v << " ";
    cout << "  front() = " << M.front() << ", back() = " << M.back() << ", empty() = " << M.empty() << "\n";
}
