struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
=====================================================
★ モノイド定義（問題ごとに変更）
=====================================================

S : ノードの値
op : 区間結合
id : 単位元

※ prefix探索を使う場合は
累積が単調になる演算であること
=====================================================
*/

using S = ll;

// ★区間結合
S op(S a, S b) {
    return a + b; // 例：区間和
}

// ★単位元
S e() {
    return 0;
}

/*
=====================================================
SegTree 本体
=====================================================
*/

struct SegTree {
    int n;
    vector<S> seg;

    SegTree(int sz) {
        n = 1;
        while (n < sz) n <<= 1;
        seg.assign(2*n, e());
    }

    // 配列初期化
    SegTree(const vector<S>& v) {
        int sz = v.size();
        n = 1;
        while (n < sz) n <<= 1;
        seg.assign(2*n, e());

        for (int i = 0; i < sz; i++)
            seg[n+i] = v[i];

        for (int i = n-1; i > 0; i--)
            seg[i] = op(seg[i<<1], seg[i<<1|1]);
    }

    // ========================
    // 点更新
    // ========================
    void set(int i, S x) {
        i += n;
        seg[i] = x;

        while (i >>= 1)
            seg[i] = op(seg[i<<1], seg[i<<1|1]);
    }

    // 和専用の便利関数
    void add(int i, S x) {
        set(i, get(i) + x);
    }

    // ========================
    // 区間取得 [l,r)
    // ========================
    S prod(int l, int r) {
        S L = e(), R = e();
        for (l += n, r += n; l < r; l >>= 1, r >>= 1) {
            if (l & 1) L = op(L, seg[l++]);
            if (r & 1) R = op(seg[--r], R);
        }
        return op(L, R);
    }

    S get(int i) {
        return seg[n+i];
    }

    S all_prod() {
        return seg[1];
    }

    // ========================
    // prefix探索
    // ========================
    // 累積 >= x の最小 index
    int ge(S x) {
        if (all_prod() < x) return n;

        int node = 1;
        S acc = e();

        while (node < n) {
            S nxt = op(acc, seg[node<<1]);
            if (nxt >= x) {
                node <<= 1;
            } else {
                acc = nxt;
                node = node<<1 | 1;
            }
        }
        return node - n;
    }

    // 累積 > x
    int gt(S x) { return ge(x+1); }

    // 累積 ≤ x
    int le(S x) {
        int pos = ge(x+1);
        return pos - 1;
    }

    // 累積 < x
    int lt(S x) {
        int pos = ge(x);
        return pos - 1;
    }

    // ========================
    // デバッグ
    // ========================
    void dump(int sz) {
        for (int i = 0; i < sz; i++)
            cout << get(i) << " ";
        cout << "\n";
    }
};
// ======================= ここまでテンプレート本体 =======================

void solv(){
    cout<<"=============================\n";
    cout<<" SegTree 動作確認\n";
    cout<<"=============================\n\n";

    vector<ll> v={5,3,7,1,4};
    SegTree seg(v);

    cout<<"▼ 初期配列:\n";
    seg.dump(v.size());

    cout<<"\n▼ get(2): index2 の値 = "
        <<seg.get(2)<<"\n";

    cout<<"\n▼ prod(1,4): 区間[1,4) の和 = "
        <<seg.prod(1,4)<<"\n";

    cout<<"\n▼ all_prod(): 全体の和 = "
        <<seg.all_prod()<<"\n";

    cout<<"\n▼ set(1,10): index1 を 10 に更新\n";
    seg.set(1,10);
    seg.dump(v.size());

    cout<<"\n▼ add(2,5): index2 に +5\n";
    seg.add(2,5);
    seg.dump(v.size());

    cout<<"\n▼ prod(0,5): 全区間和 = "
        <<seg.prod(0,5)<<"\n";

    cout<<"\n=============================\n";
    cout<<" prefix探索テスト\n";
    cout<<"=============================\n";

    for(int x=1;x<=25;x+=5){
        cout<<"\nx="<<x<<"\n";

        cout<<" 累積 >= x : index "
            <<seg.ge(x)<<"\n";

        cout<<" 累積 > x : index "
            <<seg.gt(x)<<"\n";

        cout<<" 累積 <= x : index "
            <<seg.le(x)<<"\n";

        cout<<" 累積 < x : index "
            <<seg.lt(x)<<"\n";
    }

    cout<<"\n=============================\n";
    cout<<" 最終状態\n";
    cout<<"=============================\n";

    seg.dump(v.size());

    cout<<"\n✅ 動作確認完了\n";
    return;
}
