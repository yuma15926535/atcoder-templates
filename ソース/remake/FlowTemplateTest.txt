struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
========================================
 Dinic 法（最大フロー）
========================================
 - 最大フローを高速に解く標準アルゴリズム
 - 二部マッチングにもそのまま使える
*/
struct Dinic {
    struct Edge {
        int to;
        ll cap;
        int rev;
        ll orig_cap;   // 復元用：元の容量
    };

    int N;
    vector<vector<Edge>> G;
    vector<int> level, iter;

    Dinic(int n) : N(n), G(n), level(n), iter(n) {}

    // 辺を追加（逆辺も自動で追加）
    void add_edge(int from, int to, ll cap) {
        Edge fwd = {to, cap, (int)G[to].size(), cap};
        Edge rev = {from, 0, (int)G[from].size(), 0};
        G[from].push_back(fwd);
        G[to].push_back(rev);
    }

    // BFS でレベルグラフを作る
    void bfs(int s) {
        fill(level.begin(), level.end(), -1);
        queue<int> q;
        level[s] = 0;
        q.push(s);
        while (!q.empty()) {
            int v = q.front(); q.pop();
            for (auto &e : G[v]) {
                if (e.cap > 0 && level[e.to] < 0) {
                    level[e.to] = level[v] + 1;
                    q.push(e.to);
                }
            }
        }
    }

    // DFS でフローを流す
    ll dfs(int v, int t, ll f) {
        if (v == t) return f;
        for (int &i = iter[v]; i < (int)G[v].size(); i++) {
            auto &e = G[v][i];
            if (e.cap > 0 && level[v] < level[e.to]) {
                ll d = dfs(e.to, t, min(f, e.cap));
                if (d > 0) {
                    e.cap -= d;
                    G[e.to][e.rev].cap += d;
                    return d;
                }
            }
        }
        return 0;
    }

    // 最大フローを計算
    ll max_flow(int s, int t) {
        ll flow = 0;
        while (true) {
            bfs(s);
            if (level[t] < 0) return flow;
            fill(iter.begin(), iter.end(), 0);
            ll f;
            while ((f = dfs(s, t, LLONG_MAX)) > 0) {
                flow += f;
            }
        }
    }
};

/*
========================================
 最小費用流（ポテンシャル法）
========================================
 - 一定量のフローを最小コストで流す
 - 二部マッチング（コスト付き）にも対応
*/
struct MinCostFlow {
    struct Edge {
        int to, rev;
        ll cap, cost;
        ll orig_cap;   // 復元用
    };

    int N;
    vector<vector<Edge>> G;
    vector<ll> h, dist;
    vector<int> prevv, preve;

    MinCostFlow(int n) : N(n), G(n), h(n), dist(n), prevv(n), preve(n) {}

    // 辺を追加（逆辺も追加）
    void add_edge(int from, int to, ll cap, ll cost) {
        G[from].push_back({to, (int)G[to].size(), cap, cost, cap});
        G[to].push_back({from, (int)G[from].size() - 1, 0, -cost, 0});
    }

    // s → t に f 流す最小コスト
    ll min_cost_flow(int s, int t, ll f) {
        ll res = 0;
        fill(h.begin(), h.end(), 0);

        while (f > 0) {
            priority_queue<pair<ll,int>, vector<pair<ll,int>>, greater<>> pq;
            fill(dist.begin(), dist.end(), LLONG_MAX);
            dist[s] = 0;
            pq.push({0, s});

            while (!pq.empty()) {
                auto [d, v] = pq.top(); pq.pop();
                if (dist[v] < d) continue;
                for (int i = 0; i < (int)G[v].size(); i++) {
                    auto &e = G[v][i];
                    if (e.cap > 0) {
                        ll nd = dist[v] + e.cost + h[v] - h[e.to];
                        if (dist[e.to] > nd) {
                            dist[e.to] = nd;
                            prevv[e.to] = v;
                            preve[e.to] = i;
                            pq.push({nd, e.to});
                        }
                    }
                }
            }

            if (dist[t] == LLONG_MAX) return -1;

            for (int v = 0; v < N; v++) if (dist[v] != LLONG_MAX) h[v] += dist[v];  // 届かない頂点は足さない（溢れ防止）

            ll d = f;
            for (int v = t; v != s; v = prevv[v]) {
                d = min(d, G[prevv[v]][preve[v]].cap);
            }

            f -= d;
            res += d * h[t];
            for (int v = t; v != s; v = prevv[v]) {
                auto &e = G[prevv[v]][preve[v]];
                e.cap -= d;
                G[v][e.rev].cap += d;
            }
        }
        return res;
    }
};

void random_test_bruteforce(){ mt19937 rng(12); int bad=0;
 for(int it=0;it<3000;it++){ int n=rng()%6+2, m=rng()%12; Dinic d(n); mf_graph<ll> g(n); MinCostFlow c(n); mcf_graph<ll,ll> h(n);
  for(int i=0;i<m;i++){ int u=rng()%n,v=rng()%n; if(u==v)continue; ll cap=rng()%10, cost=rng()%10; d.add_edge(u,v,cap); g.add_edge(u,v,cap); c.add_edge(u,v,cap,cost); h.add_edge(u,v,cap,cost); }
  int s=0,t=n-1; ll f1=d.max_flow(s,t), f2=g.flow(s,t); if(f1!=f2)bad++;
  ll want=rng()%15; auto [fl,co]=h.flow(s,t,want); ll r=c.min_cost_flow(s,t,want);
  if(fl<want){ if(r!=-1)bad++; } else if(r!=co)bad++;
 } 
    cout << "random_test_bruteforce（愚直解とのランダム比較）: " << (((bad)==0) ? "OK" : "NG") << "\n";
    assert((bad)==0);
}

void solv(){
    // --- Dinic 単体動作確認 ---
    cout << "=== Dinic 最大フロー 単体動作確認 ===\n";
    {
        Dinic mf(4);
        mf.add_edge(0, 1, 2);
        mf.add_edge(0, 2, 1);
        mf.add_edge(1, 3, 1);
        mf.add_edge(2, 3, 2);

        cout << "最大フロー量 = " << mf.max_flow(0, 3) << "\n\n";
    }

    // --- Dinic 二部マッチング（復元付き） ---
    cout << "=== Dinic 二部マッチング（復元付き） ===\n";
    {
        int L = 2, R = 2;
        int S = 4, T = 5;
        Dinic mf(6);

        for (int i = 0; i < L; i++) mf.add_edge(S, i, 1);
        for (int j = 0; j < R; j++) mf.add_edge(L + j, T, 1);

        mf.add_edge(0, 2, 1);
        mf.add_edge(0, 3, 1);
        mf.add_edge(1, 3, 1);

        cout << "最大マッチング数 = " << mf.max_flow(S, T) << "\n";
        cout << "マッチング結果:\n";

        for (int i = 0; i < L; i++) {
            for (auto &e : mf.G[i]) {
                if (e.to >= L && e.to < L + R &&
                    e.orig_cap == 1 && e.cap == 0) {
                    cout << "左 " << i << " - 右 " << (e.to - L) << "\n";
                }
            }
        }
        cout << "\n";
    }

    // --- 最小費用流 単体動作確認 ---
    cout << "=== 最小費用流 単体動作確認 ===\n";
    {
        MinCostFlow mcf(4);
        mcf.add_edge(0, 1, 1, 1);
        mcf.add_edge(0, 2, 1, 2);
        mcf.add_edge(1, 3, 1, 1);
        mcf.add_edge(2, 3, 1, 0);

        cout << "流量2を流す最小コスト = "
             << mcf.min_cost_flow(0, 3, 2) << "\n\n";
    }

    // --- 最小費用流 二部マッチング（復元付き） ---
    cout << "=== 最小費用流 二部マッチング（復元付き） ===\n";
    {
        int L = 2, R = 2;
        int S = 4, T = 5;
        MinCostFlow mcf(6);

        for (int i = 0; i < L; i++) mcf.add_edge(S, i, 1, 0);
        for (int j = 0; j < R; j++) mcf.add_edge(L + j, T, 1, 0);

        mcf.add_edge(0, 2, 1, 3);
        mcf.add_edge(0, 3, 1, 1);
        mcf.add_edge(1, 2, 1, 1);
        mcf.add_edge(1, 3, 1, 2);

        cout << "最小コスト = " << mcf.min_cost_flow(S, T, 2) << "\n";
        cout << "マッチング結果:\n";

        for (int i = 0; i < L; i++) {
            for (auto &e : mcf.G[i]) {
                if (e.to >= L && e.to < L + R &&
                    e.orig_cap == 1 && e.cap == 0) {
                    cout << "左 " << i << " - 右 " << (e.to - L) << "\n";
                }
            }
        }
    }
    random_test_bruteforce();
    return;
}
