struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// =============================================================================
//  拡張ユークリッド互除法
//  mod逆元（合成数mod対応）
//  一次不定方程式 ax + by = c を解く（1解）
//  ax ≡ b (mod m) の最小非負解
//  中国剰余定理（CRT）
//  オイラーのφ関数
//  （作り直し版：元ファイル NumberTheoryTemplateTest.txt）
// -----------------------------------------------------------------------------
//  説明
//    - ax + by = gcd(a,b)を解く
//  メモ
//    - O(log m)
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
// ------------------------------------------------------------
// 安全版 拡張ユークリッド互除法
// ax + by = gcd(a,b) を満たす (x,y) を求める
// 戻り値: gcd(a,b)（常に非負）
// 計算量: O(log min(|a|,|b|))
// ------------------------------------------------------------
// ======================= ここからテンプレート本体（コピー範囲） =======================
long long ext_gcd(long long a, long long b,
                      long long &x, long long &y) {
    if (b == 0) {
        x = (a >= 0 ? 1 : -1);
        y = 0;
        return llabs(a);
    }
    long long x1, y1;
    long long g = ext_gcd(b, a % b, x1, y1);
    x = y1;
    y = x1 - (long long)((__int128)(a / b) * y1);
    return g;
}

// ------------------------------------------------------------
// mod逆元（合成数mod対応）
// gcd(a,m)=1 のときのみ存在
// 存在しない場合は -1
// 計算量: O(log m)
// ------------------------------------------------------------
long long mod_inv(long long a, long long m) {
    a %= m;
    if (a < 0) a += m;
    long long x, y;
    long long g = ext_gcd(a, m, x, y);
    if (g != 1) return -1;
    x %= m;
    if (x < 0) x += m;
    return x;
}

// ------------------------------------------------------------
// 一次不定方程式 ax + by = c を解く（1解）
// 解が存在すれば true
// 解の1つを (x,y) に格納
// 計算量: O(log min(|a|,|b|))
// ------------------------------------------------------------
bool linear_diophantine(long long a, long long b, long long c,
                  long long &x, long long &y) {
    long long x0, y0;
    long long g = ext_gcd(a, b, x0, y0);
    if (c % g != 0) return false;
    long long mul = c / g;
    x = (long long)((__int128)x0 * mul);
    y = (long long)((__int128)y0 * mul);
    return true;
}

// ------------------------------------------------------------
// ★ ax ≡ b (mod m) の最小非負解を求める
// 戻り値: {解 r, 法 m'}  (解は 0 ≤ r < m')
// 解なしなら {0, -1}
// 計算量: O(log m)
// ------------------------------------------------------------
pair<long long,long long>
linear_congruence(long long a, long long b, long long m) {

    long long x, y;
    long long g = ext_gcd(a, m, x, y);

    if (b % g != 0) return {0, -1};

    long long a2 = a / g;
    long long b2 = b / g;
    long long m2 = m / g;

    long long inv = mod_inv(a2 % m2 + m2, m2);
    long long r = (long long)((__int128)(b2 % m2 + m2) * inv % m2);

    if (r < 0) r += m2;
    return {r, m2};   // 最小非負解
}

// ------------------------------------------------------------
// 安全版 CRT（2式）
// x ≡ r1 (mod m1)
// x ≡ r2 (mod m2)
// 戻り値: {解 r, lcm}
// 解なしなら {0, -1}
// 計算量: O(log max(m1,m2))
// ------------------------------------------------------------
pair<long long,long long>
crt_merge(long long r1,long long m1,
          long long r2,long long m2) {

    r1 = (r1 % m1 + m1) % m1;
    r2 = (r2 % m2 + m2) % m2;

    long long x, y;
    long long g = ext_gcd(m1, m2, x, y);

    if ((r2 - r1) % g != 0) return {0, -1};

    long long lcm =
        (long long)((__int128)m1 / g * m2);

    long long m2g = m2 / g;
    long long tmp = (r2 - r1) / g;
    tmp = (tmp % m2g + m2g) % m2g;

    x = (x % m2g + m2g) % m2g;

    long long k =
        (long long)((__int128)tmp * x % m2g);

    long long r =
        (long long)((__int128)m1 * k + r1);

    r = (r % lcm + lcm) % lcm;

    return {r, lcm};
}

// ------------------------------------------------------------
// 安全版 複数CRT
// r[i], m[i]
// 戻り値: {解 r, lcm}
// 解なしなら {0, -1}
// 計算量: O(N log M)
// ------------------------------------------------------------
pair<long long,long long>
crt_all(const vector<long long>& r,
         const vector<long long>& m) {

    long long r0 = 0;
    long long m0 = 1;

    for (int i = 0; i < (int)r.size(); i++) {
        auto res = crt_merge(r0, m0, r[i], m[i]);
        if (res.second == -1) return {0, -1};
        r0 = res.first;
        m0 = res.second;
    }
    return {r0, m0};
}

// =====================================================
// オイラーのφ関数（単発）
// φ(n) を O(√n) で求める
// n <= 1e12 くらいまでOK
// =====================================================
long long euler_phi(long long n) {
    long long result = n;

    for (long long p = 2; p * p <= n; p++) {
        if (n % p == 0) {
            // p は素因数
            while (n % p == 0) {
                n /= p;
            }
            result -= result / p;  // φ公式
        }
    }

    // 最後に素因数が残っている場合
    if (n > 1) {
        result -= result / n;
    }

    return result;
}

// =====================================================
// トーシェント篩
// 1〜N の φ を全て求める
// 計算量: O(N log log N)
// =====================================================
vector<long long> phi_table(int N) {
    vector<long long> phi(N + 1);

    // 初期化
    for (int i = 0; i <= N; i++) {
        phi[i] = i;
    }

    for (int i = 2; i <= N; i++) {
        if (phi[i] == i) {  // iは素数
            for (int j = i; j <= N; j += i) {
                phi[j] -= phi[j] / i;
            }
        }
    }

    return phi;
}

void random_test_bruteforce(){ mt19937 rng(3); int bad=0;
 for(int it=0;it<20000;it++){
  ll a=(ll)(rng()%41)-20, m=rng()%30+1, b=(ll)(rng()%41)-20;
  // modinv
  ll inv=mod_inv(a,m); ll want=-1; for(ll x=0;x<m;x++) if((((a%m)+m)%m*x)%m==1%m){want=x;break;}
  if(inv!=want)bad++;
  // linear congruence
  auto [r,mm]=linear_congruence(a,b,m); ll first=-1; for(ll x=0;x<m;x++) if(((a*x-b)%m+m)%m==0){first=x;break;}
  if(first==-1){ if(mm!=-1)bad++; } else { if(mm==-1||r!=first)bad++; }
  // crt
  ll m1=rng()%12+1,m2=rng()%12+1,r1=(ll)(rng()%30)-10,r2=(ll)(rng()%30)-10;
  auto [cr,cm]=crt_merge(r1,m1,r2,m2); ll L=m1/__gcd(m1,m2)*m2; ll f2=-1; for(ll x=0;x<L;x++) if(((x-r1)%m1+m1)%m1==0&&((x-r2)%m2+m2)%m2==0){f2=x;break;}
  if(f2==-1){ if(cm!=-1)bad++; } else { if(cm!=L||cr!=f2)bad++; }
  // diophantine
  ll A=(ll)(rng()%21)-10,B=(ll)(rng()%21)-10,C=(ll)(rng()%41)-20; ll x,y;
  if(A!=0||B!=0){ bool ok=linear_diophantine(A,B,C,x,y); ll g=__gcd(llabs(A),llabs(B)); bool ex=(C%g==0);
   if(ok!=ex)bad++; else if(ok && A*x+B*y!=C)bad++; }
 }
 for(ll n=1;n<=2000;n++){ ll c=0; for(ll k=1;k<=n;k++) if(__gcd(n,k)==1)c++; if(euler_phi(n)!=c)bad++; }
 auto ph=phi_table(2000); for(int n=1;n<=2000;n++) if(ph[n]!=euler_phi(n))bad++;
 
    cout << "random_test_bruteforce（愚直解とのランダム比較）: " << (((bad)==0) ? "OK" : "NG") << "\n";
    assert((bad)==0);
}
