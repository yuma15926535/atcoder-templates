struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// =============================================================================
//  PrefixMinMax（pair版・完成形）
//  （作り直し版：元ファイル 263D.txt）
// -----------------------------------------------------------------------------
//  説明
//    - コンストラクタで左右からの要素のmin/maxを構成
//    - 左右からの要素のmin/maxを返却
//  メモ
//    - 前計算:O(N)
//    - クエリ:O(1)
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
// ======================= ここからテンプレート本体（コピー範囲） =======================
/*
==================================================
 PrefixMinMax（pair版・完成形）
 - 前計算 O(N)
 - クエリ O(1)
 - min / max 両対応
 - 値 + 位置(index) を保持
 - 空区間も安全に扱える
==================================================

【想定用途】
- Left Right Operation
- 1点除外の最適値
- 貪欲法の復元
- 区間最適化テンプレ

【仕様】
- 0-index
- 空区間：
    min -> {INF, -1}
    max -> {-INF, -1}
*/
struct PrefixMinMax {
    using pli = pair<long long,int>;
    const long long INF = (1LL<<60);

    int N;
    vector<pli> minL, minR;
    vector<pli> maxL, maxR;

    PrefixMinMax(const vector<long long>& a) {
        N = a.size();
        minL.resize(N);
        minR.resize(N);
        maxL.resize(N);
        maxR.resize(N);
        if(N == 0) return;              // 空配列でも落ちないように

        // ===== 左から =====
        minL[0] = {a[0], 0};
        maxL[0] = {a[0], 0};

        for(int i = 1; i < N; i++){
            // min
            if(a[i] < minL[i-1].first){
                minL[i] = {a[i], i};
            }else{
                minL[i] = minL[i-1];
            }
            // max
            if(a[i] > maxL[i-1].first){
                maxL[i] = {a[i], i};
            }else{
                maxL[i] = maxL[i-1];
            }
        }

        // ===== 右から =====
        minR[N-1] = {a[N-1], N-1};
        maxR[N-1] = {a[N-1], N-1};

        for(int i = N-2; i >= 0; i--){
            // min
            if(a[i] < minR[i+1].first){
                minR[i] = {a[i], i};
            }else{
                minR[i] = minR[i+1];
            }
            // max
            if(a[i] > maxR[i+1].first){
                maxR[i] = {a[i], i};
            }else{
                maxR[i] = maxR[i+1];
            }
        }
    }

    // ------------------------------
    // 左側 [0, idx]
    // ------------------------------
    pli left_min(int idx) const {
        if(idx < 0) return {INF, -1};   // 空区間
        return minL[idx];
    }

    pli left_max(int idx) const {
        if(idx < 0) return {-INF, -1};  // 空区間
        return maxL[idx];
    }

    // ------------------------------
    // 右側 [idx, N-1]
    // ------------------------------
    pli right_min(int idx) const {
        if(idx >= N) return {INF, -1};  // 空区間
        return minR[idx];
    }

    pli right_max(int idx) const {
        if(idx >= N) return {-INF, -1}; // 空区間
        return maxR[idx];
    }

    // ------------------------------
    // 左右を合わせた取得
    //   i を除外したときの最適値
    // ------------------------------
    pli except_min(int i) const {
        return min(left_min(i-1), right_min(i+1));
    }

    pli except_max(int i) const {
        return max(left_max(i-1), right_max(i+1));
    }
};
// ======================= ここまでテンプレート本体 =======================


// -----------------------------------------------------------------------------
//  実戦での使用例（ABC263 D）… 標準入力を読むので main からは呼ばない
// -----------------------------------------------------------------------------

void solve_ABC263D(){
    int N;
    ll L, R;
    cin >> N >> L >> R;

    vector<ll> A(N);
    for(int i = 0; i < N; i++) cin >> A[i];

    // 元の総和
    ll base_sum = accumulate(A.begin(), A.end(), 0LL);

    // 左側を L にしたときの「差分累積」
    vector<ll> left_diff(N);
    {
        ll cur = 0;
        for(int i = 0; i < N; i++){
            cur += L - A[i];
            left_diff[i] = cur;
        }
    }

    // 右側を R にしたときの「差分累積」
    vector<ll> right_diff(N);
    {
        ll cur = 0;
        for(int i = N-1; i >= 0; i--){
            cur += R - A[i];
            right_diff[i] = cur;
        }
    }

    // PrefixMinMax を構築
    PrefixMinMax PL(left_diff);
    PrefixMinMax PR(right_diff);

    // right_diff の suffix min
    vector<ll> suf_min(N+1, 0);
    suf_min[N] = 0;  // 右を選ばない
    for(int i = N-1; i >= 0; i--){
        suf_min[i] = min(suf_min[i+1], right_diff[i]);
    }

    ll ans = base_sum;

    // i は「境界」
    //範囲外ではINFを返却するので明示的な回避が必要
    for(int i = -1; i < N; i++){
        ll addL = (i >= 0 ? PL.left_min(i).first : 0);
        ll addR = (i+1 < N ? PR.right_min(i+1).first : 0);
        ans = min(ans, base_sum + addL + addR);
    }

    cout << ans << "\n";
    return;
}

//  使い方デモ（入力なしで動く）
// -----------------------------------------------------------------------------
void demo(){
    cout << "===== PrefixMinMax デモ =====\n";
    vector<ll> a = {5, 2, 8, 2, 9, 1};
    PrefixMinMax pm(a);                       // 前計算 O(N)
    cout << "a = {5, 2, 8, 2, 9, 1}\n";
    auto p = pm.left_min(3);                  // [0, 3] の最小値と位置（同じ値なら左の位置）
    cout << "pm.left_min(3)  = {" << p.first << ", " << p.second << "}   // [0,3] の最小 2 は位置 1\n";
    p = pm.right_max(2);                      // [2, N-1] の最大値と位置
    cout << "pm.right_max(2) = {" << p.first << ", " << p.second << "}   // [2,5] の最大 9 は位置 4\n";
    p = pm.except_max(4);                     // a[4] を除いたときの最大値と位置
    cout << "pm.except_max(4) = {" << p.first << ", " << p.second << "}   // 9 を除くと 8（位置 2）\n";
    p = pm.left_min(-1);                      // 空区間は {INF, -1}
    cout << "pm.left_min(-1) = {" << p.first << ", " << p.second << "}   // 空区間 → {INF, -1}\n";
}
