struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

//  ExtendedDeque（dequeのmax/min/count方向への拡張）
//  （作り直し版：元ファイル SuperDequeTemplateTest.txt）
// -----------------------------------------------------------------------------
//  説明
//    - ExtendedDeque
//    - deque+min/max/count(pop_min/max)
//    - OrderStatisticDeque
//    - k番目アクセス
//    - range(sum/min/max)
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける

/*
========================================================
 ExtendedDeque（dequeのmax/min/count方向への拡張）
========================================================
【機能】
- push_front / push_back
- pop_front / pop_back
- front / back 取得
- get_min / get_max
- pop_min / pop_max
- count(x) : 値 x の個数
========================================================
*/
template<class T>
struct ExtendedDeque {
    // 値を「位置番号つき」で deque に並べ、最小・最大はヒープ（vector）で持つ。
    // 途中の要素の削除（pop_min / pop_max）は「削除済み」の印を付けるだけにして、両端に来たら捨てる。
    // list + multiset と違って要素ごとの確保がないので速い。各操作 O(log N)（ならし）
    deque<T> buf;                 // 位置番号 p の値は buf[p - off]
    deque<long long> stamp;       // その位置に入れたときの通し番号（削除済みなら -1）
    long long off = 0;            // buf[0] の位置番号
    long long next_stamp = 0;
    using E = tuple<T, long long, long long>;           // (値, 位置番号, 通し番号)
    priority_queue<E, vector<E>, greater<E>> mnq;       // 値の小さい順（同じ値なら前にあるもの）
    priority_queue<E> mxq;                               // 値の大きい順（位置番号は -p で持つ）
    map<T, int> freq;             // 値の個数
    int cnt = 0;

    bool alive(long long p, long long st) const {
        return off <= p && p < off + (long long)buf.size() && stamp[p - off] == st;
    }
    void trim() {  // 両端の削除済みを捨てる
        while (!stamp.empty() && stamp.front() == -1) { stamp.pop_front(); buf.pop_front(); off++; }
        while (!stamp.empty() && stamp.back() == -1) { stamp.pop_back(); buf.pop_back(); }
    }
    void push_at(long long p, const T& x) {
        long long st = next_stamp++;
        if (p < off) { buf.push_front(x); stamp.push_front(st); off--; }
        else { buf.push_back(x); stamp.push_back(st); }
        mnq.push({x, p, st});
        mxq.push({x, -p, st});
        freq[x]++;
        cnt++;
    }
    void remove_at(long long p) {
        const T x = buf[p - off];
        stamp[p - off] = -1;
        if (--freq[x] == 0) freq.erase(x);
        cnt--;
        trim();
    }

    //============================
    // 基本操作
    //============================

    void push_front(const T& x) { push_at(off - 1, x); }
    void push_back(const T& x) { push_at(off + (long long)buf.size(), x); }
    void pop_front() { if (cnt) remove_at(off); }                              // 空なら何もしない
    void pop_back() { if (cnt) remove_at(off + (long long)buf.size() - 1); }

    //============================
    // front / back（空でないこと）
    //============================

    const T& front() const { return buf.front(); }
    const T& back() const { return buf.back(); }

    //============================
    // min / max（空でないこと）。同じ値が複数あるときは前にあるものが対象
    //============================

    const T& get_min() {
        while (!alive(get<1>(mnq.top()), get<2>(mnq.top()))) mnq.pop();
        return buf[get<1>(mnq.top()) - off];
    }
    const T& get_max() {
        while (!alive(-get<1>(mxq.top()), get<2>(mxq.top()))) mxq.pop();
        return buf[-get<1>(mxq.top()) - off];
    }
    T pop_min() {
        get_min();
        auto [x, p, st] = mnq.top(); mnq.pop();
        remove_at(p);
        return x;
    }
    T pop_max() {
        get_max();
        auto [x, np, st] = mxq.top(); mxq.pop();
        remove_at(-np);
        return x;
    }

    //============================

    int count(const T& x) const {
        auto it = freq.find(x);
        return (it == freq.end() ? 0 : it->second);
    }

    int size() const { return cnt; }
    bool empty() const { return cnt == 0; }

    // 前から順に並べた vector（確認用。O(N)）
    vector<T> to_vector() const {
        vector<T> res;
        for (size_t i = 0; i < buf.size(); i++) if (stamp[i] != -1) res.push_back(buf[i]);
        return res;
    }
};

template<class T>
struct OrderStatisticDeque {

    struct Node {
        T val, sum, mn, mx;
        int prio, sz;
        Node *l, *r;

        Node(T v)
            : val(v), sum(v), mn(v), mx(v),
              prio(rand()), sz(1), l(nullptr), r(nullptr) {}
    };

    // ノードはまとめて確保する（1 個ずつ new するより速い）
    static Node* alloc_node(T v) {
        static vector<Node*> blocks;
        static int used = 1 << 14;
        if (used == (1 << 14)) { blocks.push_back(static_cast<Node*>(::operator new(sizeof(Node) << 14))); used = 0; }
        return new (blocks.back() + used++) Node(v);
    }

    using np = Node*;
    np root = nullptr;

    //============================
    // 補助関数
    //============================

    int sz(np t){ return t ? t->sz : 0; }
    T sum(np t){ return t ? t->sum : 0; }
    T mn(np t){ return t ? t->mn : numeric_limits<T>::max(); }
    T mx(np t){ return t ? t->mx : numeric_limits<T>::lowest(); }

    void upd(np t){
        if(!t) return;
        t->sz = 1 + sz(t->l) + sz(t->r);
        t->sum = t->val + sum(t->l) + sum(t->r);
        t->mn = min({t->val, mn(t->l), mn(t->r)});
        t->mx = max({t->val, mx(t->l), mx(t->r)});
    }

    // split [0,k) | [k,n)
    void split(np t, np &a, np &b, int k){
        if(!t){ a=b=nullptr; return; }

        if(sz(t->l) >= k){
            split(t->l, a, t->l, k);
            b = t;
        }else{
            split(t->r, t->r, b, k - sz(t->l) - 1);
            a = t;
        }
        upd(t);
    }

    np merge(np a, np b){
        if(!a||!b) return a?a:b;
        if(a->prio > b->prio){
            a->r = merge(a->r, b);
            upd(a);
            return a;
        }else{
            b->l = merge(a, b->l);
            upd(b);
            return b;
        }
    }

    //============================
    // 基本deque操作
    //============================

    void push_front(T x){
        root = merge(alloc_node(x), root);
    }

    void push_back(T x){
        root = merge(root, alloc_node(x));
    }

    void pop_front(){
        np a,b;
        split(root,a,b,1);
        root=b;
    }

    void pop_back(){
        np a,b;
        split(root,a,b,sz(root)-1);
        root=a;
    }

    //============================
    // insert / erase
    //============================

    void insert(int k, T x){
        np a,b;
        split(root,a,b,k);
        root = merge(merge(a,alloc_node(x)),b);
    }

    void erase(int k){
        np a,b,c;
        split(root,a,b,k);
        split(b,b,c,1);
        root = merge(a,c);
    }

    //============================
    // k番目アクセス
    //============================

    T& kth(np t,int k){
        int left=sz(t->l);
        if(k<left) return kth(t->l,k);
        if(k==left) return t->val;
        return kth(t->r,k-left-1);
    }

    T& operator[](int k){
        return kth(root,k);
    }

    //============================
    // range query
    //============================

    // [l, r) の (和, 最小, 最大)。l < r であること
    tuple<T,T,T> range_query(int l,int r){
        np a,b,c;
        split(root,a,b,l);
        split(b,b,c,r-l);

        T S=sum(b);
        T MN=mn(b);
        T MX=mx(b);

        root=merge(a,merge(b,c));
        return {S,MN,MX};
    }

    // [l, r] の (和, 最小, 最大)（閉区間版）
    tuple<T,T,T> range_query_closed(int l,int r){ return range_query(l, r + 1); }

    //============================

    int size(){ return sz(root); }
};

void solv(){

    //乱数をランダムに設定
    //srand(0);などで固定できる
    srand(time(NULL));

    cout<<"==============================\n";
    cout<<"ExtendedDeque 動作確認\n";
    cout<<"==============================\n";

    ExtendedDeque<int> ed;

    auto show_ed = [&](){
        cout<<"現在の状態: [ ";
        for(auto &x:ed.to_vector()) cout<<x<<" ";
        cout<<"]\n";
    };

    cout<<"\n--- push テスト ---\n";
    ed.push_back(5);
    ed.push_front(3);
    ed.push_back(7);
    ed.push_back(3);
    show_ed();

    cout<<"front = "<<ed.front()<<"\n";
    cout<<"back = "<<ed.back()<<"\n";
    cout<<"min = "<<ed.get_min()<<"\n";
    cout<<"max = "<<ed.get_max()<<"\n";
    cout<<"count(3) = "<<ed.count(3)<<"\n";

    cout<<"\n--- pop_min ---\n";
    cout<<"削除値 = "<<ed.pop_min()<<"\n";
    show_ed();

    cout<<"\n--- pop_max ---\n";
    cout<<"削除値 = "<<ed.pop_max()<<"\n";
    show_ed();

    cout<<"\n--- pop_front/back ---\n";
    ed.pop_front();
    ed.pop_back();
    show_ed();

    cout<<"\n==============================\n";
    cout<<"OrderStatisticDeque 動作確認\n";
    cout<<"==============================\n";

    OrderStatisticDeque<long long> osd;

    auto show_osd = [&](){
        cout<<"現在の状態: [ ";
        for(int i=0;i<osd.size();i++) cout<<osd[i]<<" ";
        cout<<"]\n";
    };

    cout<<"\n--- push ---\n";
    for(int x:{10,20,30,40,50}) osd.push_back(x);
    show_osd();

    cout<<"\n--- insert ---\n";
    osd.insert(2,99);
    show_osd();

    cout<<"\n--- erase ---\n";
    osd.erase(3);
    show_osd();

    cout<<"\n--- kth アクセス確認 ---\n";
    for(int i=0;i<osd.size();i++)
        cout<<"index "<<i<<" = "<<osd[i]<<"\n";

    cout<<"\n--- range query [1,4)（= 閉区間 [1,3]） ---\n";
    auto [s,mn,mx]=osd.range_query(1,4);
    cout<<"sum="<<s<<" min="<<mn<<" max="<<mx<<"\n";
    auto [s2,mn2,mx2]=osd.range_query_closed(1,3);
    cout<<"range_query_closed(1,3): sum="<<s2<<" min="<<mn2<<" max="<<mx2<<"\n";

    cout<<"\n--- pop_front/back ---\n";
    osd.pop_front();
    osd.pop_back();
    show_osd();

    cout<<"\nテスト完了\n";
    return;
}
