        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す

#ifndef SPLITMIX_HASH_DEFINED
#define SPLITMIX_HASH_DEFINED
// ハックされにくいハッシュ（splitmix64 + 実行ごとに変わる乱数）。整数はそのまま、それ以外は std::hash の結果を混ぜる
//   unordered_map<ll, int, SplitMixHash> のように使う（HashSetTemplate.txt と同じもの）
struct SplitMixHash {
    static uint64_t splitmix64(uint64_t x) {
        x += 0x9e3779b97f4a7c15;
        x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9;
        x = (x ^ (x >> 27)) * 0x94d049bb133111eb;
        return x ^ (x >> 31);
    }
    template<class U> size_t operator()(const U& x) const {
        static const uint64_t R = chrono::steady_clock::now().time_since_epoch().count();
        if constexpr (is_integral_v<U>) return splitmix64((uint64_t)x + R);
        else return splitmix64((uint64_t)std::hash<U>{}(x) + R);
    }
};
#endif
/*
==================================================
 連続部分列和ユーティリティ（hash版・非円環 / 円環・完全版）
 - 負の数あり対応。空の部分列は数えない
 - 和 = K の区間数 + 最短 / 最長区間
 - 最大 / 最小部分列和 + その和になる区間数 + 最短 / 最長区間
 - 全て O(N)（unordered_map を使うので平均。ハッシュは SplitMixHash なのでハックされにくい）
 - 区間は半開区間 [l, r)。円環版は r が n を超えることがある（要素は a[i % n]、長さ r - l）
==================================================
*/
// ------------------------------------------------------------
// 共通: 和がちょうど T になる連続部分列（空でない）の個数と、最短・最長の 1 つ  O(N)（平均）
//   区間は半開 [l, r)。見つからなければ cnt = 0 で区間は 0
// ------------------------------------------------------------
inline void subarray_sum_eq_stats(const vector<ll>& a, ll T, ll& cnt, int& sl, int& sr, int& ll_, int& lr) {
    int n = a.size();
    cnt = 0; sl = sr = ll_ = lr = 0;
    int best_short = INT_MAX, best_long = -1;
    struct Info { ll c; int first, last; };
    unordered_map<ll, Info, SplitMixHash> mp;
    mp.reserve(n * 2 + 2);
    mp[0] = {1, 0, 0};
    ll s = 0;
    for (int r = 1; r <= n; r++) {
        s += a[r - 1];
        auto it = mp.find(s - T);
        if (it != mp.end()) {
            cnt += it->second.c;
            int L1 = it->second.last;            // 一番右の左端 → 最短
            if (r - L1 < best_short) { best_short = r - L1; sl = L1; sr = r; }
            int L2 = it->second.first;           // 一番左の左端 → 最長
            if (r - L2 > best_long) { best_long = r - L2; ll_ = L2; lr = r; }
        }
        auto jt = mp.find(s);
        if (jt == mp.end()) mp[s] = {1, r, r};
        else { jt->second.c++; jt->second.last = r; }
    }
}

// ------------------------------------------------------------
// 共通（円環）: 和がちょうど T になる円環の連続部分列（長さ 1〜n）の個数と、最短・最長の 1 つ  O(N)（平均）
//   区間は [l, r)（0 ≤ l < n、長さ r - l。r は n を超えることがあり、要素は a[i % n]）
//   全体（長さ n）はどこから始めても同じなので 1 個と数える
// ------------------------------------------------------------
inline void circular_subarray_sum_eq_stats(const vector<ll>& a, ll T, ll& cnt, int& sl, int& sr, int& ll_, int& lr) {
    int n = a.size();
    cnt = 0; sl = sr = ll_ = lr = 0;
    if (n == 0) return;
    int best_short = INT_MAX, best_long = -1;
    vector<ll> P(2 * n + 1, 0);
    for (int i = 0; i < 2 * n; i++) P[i + 1] = P[i] + a[i % n];
    // 長さ 1〜n-1：右端 j について、左端 i は [j-n+1, j-1] かつ [0, n-1]
    unordered_map<ll, deque<int>, SplitMixHash> pos;
    int lo = 0;
    for (int j = 1; j <= 2 * n - 2; j++) {
        if (j - 1 <= n - 1) pos[P[j - 1]].push_back(j - 1);
        while (lo < j - n + 1) { if (lo <= n - 1) pos[P[lo]].pop_front(); lo++; }
        auto it = pos.find(P[j] - T);
        if (it == pos.end() || it->second.empty()) continue;
        auto& dq = it->second;
        cnt += dq.size();
        if (j - dq.back() < best_short) { best_short = j - dq.back(); sl = dq.back(); sr = j; }
        if (j - dq.front() > best_long) { best_long = j - dq.front(); ll_ = dq.front(); lr = j; }
    }
    // 長さ n（全体）
    if (P[n] == T) {
        cnt++;
        if (n < best_short) { best_short = n; sl = 0; sr = n; }
        if (n > best_long) { best_long = n; ll_ = 0; lr = n; }
    }
}
// 円環の最大・最小部分列和（空でない）  O(N)
//   全体ではない部分列（a[0] か a[n-1] を含まない）を除いた残りが「またぐ」部分列になる
inline pair<ll,ll> circular_max_min_subarray(const vector<ll>& a) {
    int n = a.size();
    auto kad = [&](int l, int r, bool mx) {          // [l, r) の空でない最大 / 最小部分列和
        ll best = mx ? LLONG_MIN : LLONG_MAX, cur = 0; bool first = true;
        for (int i = l; i < r; i++) {
            if (first) { cur = a[i]; first = false; }
            else cur = mx ? max(a[i], cur + a[i]) : min(a[i], cur + a[i]);
            best = mx ? max(best, cur) : min(best, cur);
        }
        return best;
    };
    ll total = accumulate(a.begin(), a.end(), 0LL);
    ll mx = kad(0, n, true), mn = kad(0, n, false);
    if (n >= 2) {
        ll min_proper = min(kad(1, n, false), kad(0, n - 1, false));
        ll max_proper = max(kad(1, n, true), kad(0, n - 1, true));
        mx = max(mx, total - min_proper);
        mn = min(mn, total - max_proper);
    }
    return {mx, mn};
}

struct SubarraySumUtils {
    int n;
    // ===== 和 = K =====
    ll count_eqK;                      // 区間数
    bool exist_eqK;                    // 1 つでもあるか
    int minK_l, minK_r, maxK_l, maxK_r;   // 最短 / 最長の区間の 1 つ
    // ===== 最大 / 最小部分列和 =====
    ll max_sum, min_sum;
    ll max_count, min_count;           // その和になる区間の数
    int max_short_l, max_short_r, max_long_l, max_long_r;
    int min_short_l, min_short_r, min_long_l, min_long_r;
    SubarraySumUtils(const vector<ll>& a, ll K) {
        n = a.size();
        subarray_sum_eq_stats(a, K, count_eqK, minK_l, minK_r, maxK_l, maxK_r);
        exist_eqK = (count_eqK > 0);
        max_sum = min_sum = 0;
        if (n == 0) { max_count = min_count = 0; max_short_l = max_short_r = max_long_l = max_long_r = min_short_l = min_short_r = min_long_l = min_long_r = 0; return; }
        // Kadane（空でない）
        ll cmx = a[0], cmn = a[0];
        max_sum = min_sum = a[0];
        for (int i = 1; i < n; i++) {
            cmx = max(a[i], cmx + a[i]); max_sum = max(max_sum, cmx);
            cmn = min(a[i], cmn + a[i]); min_sum = min(min_sum, cmn);
        }
        // その和になる区間を数え直す（Kadane だけでは個数・最短・最長は正しく出ない）
        subarray_sum_eq_stats(a, max_sum, max_count, max_short_l, max_short_r, max_long_l, max_long_r);
        subarray_sum_eq_stats(a, min_sum, min_count, min_short_l, min_short_r, min_long_l, min_long_r);
    }
};
struct CircularSubarraySumUtils {
    int n;
    // ===== 和 = K =====
    ll count_eqK;
    bool exist_eqK;
    int minK_l, minK_r, maxK_l, maxK_r;
    // ===== 最大 / 最小部分列和 =====
    ll max_sum, min_sum;
    ll max_count, min_count;
    int max_short_l, max_short_r, max_long_l, max_long_r;
    int min_short_l, min_short_r, min_long_l, min_long_r;
    CircularSubarraySumUtils(const vector<ll>& a, ll K) {
        n = a.size();
        circular_subarray_sum_eq_stats(a, K, count_eqK, minK_l, minK_r, maxK_l, maxK_r);
        exist_eqK = (count_eqK > 0);
        max_sum = min_sum = 0;
        if (n == 0) { max_count = min_count = 0; max_short_l = max_short_r = max_long_l = max_long_r = min_short_l = min_short_r = min_long_l = min_long_r = 0; return; }
        tie(max_sum, min_sum) = circular_max_min_subarray(a);
        circular_subarray_sum_eq_stats(a, max_sum, max_count, max_short_l, max_short_r, max_long_l, max_long_r);
        circular_subarray_sum_eq_stats(a, min_sum, min_count, min_short_l, min_short_r, min_long_l, min_long_r);
    }
};

// ---- ランダムテスト用: 円環も扱える区間和 [l, r)（l > r なら折り返し）----
ll csum_test(const vector<ll>&a,int l,int r){ int n=a.size(); ll s=0; if(l<r){for(int i=l;i<r;i++)s+=a[i%n];} else if(l>r){for(int i=l;i<n;i++)s+=a[i]; for(int i=0;i<r;i++)s+=a[i];} return s; }

void random_test_subarray(){ mt19937 rng(3); int bad[10]={0};
 for(int it=0;it<20000;it++){ int n=rng()%6+1; vector<ll>a(n); for(auto&x:a)x=(ll)(rng()%7)-3; ll K=(ll)(rng()%7)-3;
  ll cnt=0; int mnl=INT_MAX,mxl=-1; ll mx=LLONG_MIN,mn=LLONG_MAX;
  for(int l=0;l<n;l++){ll s=0; for(int r=l;r<n;r++){s+=a[r]; if(s==K){cnt++; mnl=min(mnl,r-l+1); mxl=max(mxl,r-l+1);} mx=max(mx,s); mn=min(mn,s);} }
  ll mxc=0,mnc=0; int mxs=INT_MAX,mxlg=-1,mns=INT_MAX,mnlg=-1;
  for(int l=0;l<n;l++){ll s=0; for(int r=l;r<n;r++){s+=a[r]; if(s==mx){mxc++; mxs=min(mxs,r-l+1); mxlg=max(mxlg,r-l+1);} if(s==mn){mnc++; mns=min(mns,r-l+1); mnlg=max(mnlg,r-l+1);} } }
  SubarraySumUtils u(a,K);
  if(u.count_eqK!=cnt)bad[0]++;
  if(u.max_sum!=mx)bad[1]++; if(u.min_sum!=mn)bad[2]++;
  if(cnt && (u.minK_r-u.minK_l!=mnl || u.maxK_r-u.maxK_l!=mxl || csum_test(a,u.minK_l,u.minK_r)!=K || csum_test(a,u.maxK_l,u.maxK_r)!=K))bad[3]++;
  if(u.max_count!=mxc)bad[4]++; if(u.min_count!=mnc)bad[5]++;
  if(u.max_short_r-u.max_short_l!=mxs || u.max_long_r-u.max_long_l!=mxlg)bad[6]++;
  if(u.min_short_r-u.min_short_l!=mns || u.min_long_r-u.min_long_l!=mnlg)bad[7]++;
 } int tot=0; for(int i=0;i<8;i++) tot+=bad[i];
 cout << "random_test_subarray: " << (tot==0 ? "OK" : "NG") << "（愚直解と 20000 回比較）\n"; assert(tot==0);
}


void random_test_circular_subarray(){ mt19937 rng(5); int bad[10]={0};
 for(int it=0;it<20000;it++){ int n=rng()%6+1; vector<ll>a(n); for(auto&x:a)x=(ll)(rng()%7)-3; ll K=(ll)(rng()%7)-3;
  // circular subarrays: start s, length 1..n, but full-length counted once
  ll cnt=0; int mnl=INT_MAX,mxl=-1; ll mx=LLONG_MIN,mn=LLONG_MAX;
  vector<tuple<ll,int>> all;
  for(int s=0;s<n;s++){ll t=0; for(int len=1;len<=n;len++){ t+=a[(s+len-1)%n]; if(len==n&&s>0)continue; all.push_back({t,len}); } }
  for(auto [t,len]:all){ if(t==K){cnt++; mnl=min(mnl,len); mxl=max(mxl,len);} mx=max(mx,t); mn=min(mn,t);}
  ll mxc=0,mnc=0; int mxs=INT_MAX,mxlg=-1,mns=INT_MAX,mnlg=-1;
  for(auto [t,len]:all){ if(t==mx){mxc++; mxs=min(mxs,len); mxlg=max(mxlg,len);} if(t==mn){mnc++; mns=min(mns,len); mnlg=max(mnlg,len);} }
  CircularSubarraySumUtils u(a,K);
  if(u.count_eqK!=cnt)bad[0]++;
  if(u.max_sum!=mx)bad[1]++; if(u.min_sum!=mn)bad[2]++;
  auto L=[&](int l,int r){ return r-l; };
  if(cnt && (L(u.minK_l,u.minK_r)!=mnl || L(u.maxK_l,u.maxK_r)!=mxl))bad[3]++;
  if(u.max_count!=mxc)bad[4]++; if(u.min_count!=mnc)bad[5]++;
  if(L(u.max_short_l,u.max_short_r)!=mxs || L(u.max_long_l,u.max_long_r)!=mxlg)bad[6]++;
  if(L(u.min_short_l,u.min_short_r)!=mns || L(u.min_long_l,u.min_long_r)!=mnlg)bad[7]++;
 } int tot=0; for(int i=0;i<8;i++) tot+=bad[i];
 cout << "random_test_circular_subarray: " << (tot==0 ? "OK" : "NG") << "（愚直解と 20000 回比較）\n"; assert(tot==0);
}

    vector<vector<ll>> tests = {

        // 追加テスト
        {-1, -2, -3, -4, -5},  // 全負, 折り返しが最大区間になる
        {-1, -2, -3},          // 全負, 和=Kが存在しないケース
        {3, 1, 2, 1, 2},       // 正数だけ, 折り返しが最長区間になる
        {10, -1, -1, 2, -1},   // 正負混合, 和=Kが複数存在
        {1, 2, 3, 4, 5, -15}   // 大負数で折り返し部分列になる
    };

    vector<ll> Ks = {
        3, 5, -5, 2, 0,
        -5, -10, 6, 3, 5
    };

    for(int t=0;t<tests.size();t++){
        const auto& a = tests[t];
        ll K = Ks[t];
        int n = a.size();

        cout << "=============================" << endl;
        cout << "テスト " << t+1 << endl;
        cout << "配列: ";
        for(ll x:a) cout << x << " ";
        cout << endl << "K = " << K << endl;

        // ------------------------------
        // 非円環
        SubarraySumUtils nonCircular(a,K);

        cout << endl << "[非円環]" << endl;
        cout << "和 = K の区間数: " << nonCircular.count_eqK << endl;
        cout << "最短区間: [" << nonCircular.minK_l << "," << nonCircular.minK_r << ")";
        if(!nonCircular.exist_eqK) cout << " (存在しません)";
        cout << endl;
        cout << "最長区間: [" << nonCircular.maxK_l << "," << nonCircular.maxK_r << ")";
        if(!nonCircular.exist_eqK) cout << " (存在しません)";
        cout << endl;

        cout << "最大部分列和: " << nonCircular.max_sum << endl;
        cout << "最大和 区間数: " << nonCircular.max_count << endl;
        cout << "最大和 最短区間: [" << nonCircular.max_short_l << "," << nonCircular.max_short_r << ")" << endl;
        cout << "最大和 最長区間: [" << nonCircular.max_long_l << "," << nonCircular.max_long_r << ")" << endl;

        cout << "最小部分列和: " << nonCircular.min_sum << endl;
        cout << "最小和 区間数: " << nonCircular.min_count << endl;
        cout << "最小和 最短区間: [" << nonCircular.min_short_l << "," << nonCircular.min_short_r << ")" << endl;
        cout << "最小和 最長区間: [" << nonCircular.min_long_l << "," << nonCircular.min_long_r << ")" << endl;

        // ------------------------------
        // 円環
        CircularSubarraySumUtils circular(a,K);

        cout << endl << "[円環]" << endl;
        cout << "和 = K の区間数: " << circular.count_eqK << endl;
        cout << "最短区間: [" << circular.minK_l << "," << circular.minK_r << ")";
        if(!circular.exist_eqK) cout << " (存在しません)";
        // 折り返し判定
        else if(circular.minK_l > circular.minK_r) cout << " (折り返し)";
        cout << endl;

        cout << "最長区間: [" << circular.maxK_l << "," << circular.maxK_r << ")";
        if(!circular.exist_eqK) cout << " (存在しません)";
        else if(circular.maxK_l > circular.maxK_r) cout << " (折り返し)";
        cout << endl;

        cout << "最大部分列和: " << circular.max_sum << endl;
        cout << "最大和 区間数: " << circular.max_count << endl;
        cout << "最大和 最短区間: [" << circular.max_short_l << "," << circular.max_short_r << ")";
        if(circular.max_short_l > circular.max_short_r) cout << " (折り返し)";
        cout << endl;
        cout << "最大和 最長区間: [" << circular.max_long_l << "," << circular.max_long_r << ")";
        if(circular.max_long_l > circular.max_long_r) cout << " (折り返し)";
        cout << endl;

        cout << "最小部分列和: " << circular.min_sum << endl;
        cout << "最小和 区間数: " << circular.min_count << endl;
        cout << "最小和 最短区間: [" << circular.min_short_l << "," << circular.min_short_r << ")";
        if(circular.min_short_l > circular.min_short_r) cout << " (折り返し)";
        cout << endl;
        cout << "最小和 最長区間: [" << circular.min_long_l << "," << circular.min_long_r << ")";
        if(circular.min_long_l > circular.min_long_r) cout << " (折り返し)";
        cout << endl;

        cout << "=============================" << endl << endl;
    }
    random_test_subarray();
    random_test_circular_subarray();
    return;
}
