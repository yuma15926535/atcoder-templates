struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// =============================================================================
//  Grundy数(NIM)
//  （作り直し版：元ファイル 368F.txt）
// -----------------------------------------------------------------------------
//  説明
//    - grandy数:複数の山に対して二人が同一の操作を行うことができ、操作を行えなくなったら負け。かつ操作の状態を頂点、操作を有効辺と見たときDAGになっている。場合に使える。遷移先のMEXで求める。(遷移の末端から計算する)→NIMにおける石の個数になる。NIMの勝敗:xorを取り0なら(先手が)負け、非0なら勝ち。xorが0になるように渡すのが最善。xor結果の先頭の1と同位置に1がある山に対して、その山とxor結果とのxorを取りその数になるように操作をすると勝てる。
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
//Grundy数(NIM)
// ======================= ここからテンプレート本体（コピー範囲） =======================
// DAG（ゲームの状態遷移）の Grundy 数
//   add_edge(u, v)：状態 u から v へ 1 手で移れる
//   calc(v)       ：状態 v の Grundy 数（0 なら手番の人の負け）。グラフは DAG が前提
//   非再帰（深い DAG でもスタックが溢れない）。mex は vector<char> で取る
struct Grundy {
    int N;
    vector<vector<int>> G;
    vector<int> memo;

    Grundy(int n) : N(n), G(n), memo(n, -1) {}

    void add_edge(int u, int v){
        G[u].push_back(v);
    }

    int calc(int s){
        if(memo[s] != -1) return memo[s];
        vector<pair<int,int>> st;   // (頂点, 次に見る遷移の番号)
        vector<char> seen;
        st.push_back({s, 0});
        while(!st.empty()){
            auto& [v, i] = st.back();
            if(i < (int)G[v].size()){
                int to = G[v][i++];
                if(memo[to] == -1) st.push_back({to, 0});   // 先に遷移先を計算する
                continue;
            }
            // 遷移先がすべて計算済み → mex を取る（Grundy 数は遷移先の数以下）
            int k = G[v].size();
            seen.assign(k + 1, 0);
            for(int to : G[v]) if(memo[to] <= k) seen[memo[to]] = 1;
            int g = 0;
            while(seen[g]) g++;
            memo[v] = g;
            st.pop_back();
        }
        return memo[s];
    }
};
// ======================= ここまでテンプレート本体 =======================


// -----------------------------------------------------------------------------
//  実戦での使用例（ABC368 F）… 標準入力を読むので main からは呼ばない
// -----------------------------------------------------------------------------

void solve_ABC368F(){
    const int M = 100005;
    Grundy gr(M);
    for (int i = 1; i < M; i++) {
        for (int j = i*2; j < M; j += i)gr.add_edge(j,i);
    }
    int N;
    cin>>N;
    int x=0;
    rep(_,0,N){
        int a;
        cin>>a;
        x^=gr.calc(a);
    }
    if (x==0)cout<<"Bruno"<<endl;
    else cout<<"Anna"<<endl;
    return;
}

//  使い方デモ（入力なしで動く）

void demo(){
    cout << "===== Grundy 数デモ =====\n";
    // 石取り：山の石が x 個のとき 1〜3 個取れる。状態 x → x-1, x-2, x-3 の辺を張る
    int M = 10;
    Grundy gr(M + 1);
    for (int x = 1; x <= M; x++) for (int t = 1; t <= 3 && t <= x; t++) gr.add_edge(x, x - t);
    cout << "gr.calc(x) (x = 0..10): ";
    for (int x = 0; x <= M; x++) cout << gr.calc(x) << " ";
    cout << "  // x mod 4 になる\n";
    // 複数の山は Grundy 数の XOR が 0 なら先手負け
    int g = gr.calc(5) ^ gr.calc(6) ^ gr.calc(7);
    cout << "山 {5, 6, 7}: XOR = " << g << " → " << (g ? "先手勝ち" : "後手勝ち") << "\n";
}
