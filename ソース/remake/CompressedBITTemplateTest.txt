// ======================= ここからテンプレート本体（コピー範囲） =======================
/**
 * 座標圧縮 + BIT（Fenwick Tree）テンプレート
 * 
 * 使い方：
 *  1. コンストラクタで全ての値の配列を渡す
 *     第2引数で初期化方法を選べる（0: 0初期化, 1: 配列初期化）
 *  2. add / erase / count系 / kth などは値ベースで呼び出す
 * 
 * 将来的にLISや転倒数の計算も自然に追加可能
 */
struct CompressedBIT {
    using ll = long long;

    int n;                     // 圧縮後のサイズ
    vector<ll> bit;            // BIT本体
    vector<ll> data;           // 元の値の集計（個数）
    vector<ll> values;         // 圧縮前の値（ソート済み、重複なし）
    ll none = -1;              // 値が見つからないときに返す値（kth / first_ge など）。負の値も入れるなら LLONG_MIN などにする

    // 値 -> 圧縮後 index（登録していない値は -1）。二分探索なのでハッシュより速いことが多く、ハックにも強い
    int index_of(ll v) const {
        int i = lower_bound(values.begin(), values.end(), v) - values.begin();
        return (i < n && values[i] == v) ? i : -1;
    }

    // ----------------------------
    // コンストラクタ / build
    // ----------------------------
    // all_values: 全ての値の配列
    // init_type: 0=0初期化, 1=配列どおりに初期化
    CompressedBIT(const vector<ll>& all_values, int init_type=0) {
        // 重複削除 + ソート
        values = all_values;
        sort(values.begin(), values.end());
        values.erase(unique(values.begin(), values.end()), values.end());

        n = (int)values.size();
        bit.assign(n+1, 0);
        data.assign(n, 0);

        if (init_type == 1) {
            for (ll v : all_values) add(v, 1);
        }
    }

    // ----------------------------
    // BIT操作（内部）
    // ----------------------------
    void internal_add(int i, ll x) {
        i++; // BITは1-indexed
        while (i <= n) {
            bit[i] += x;
            i += i & -i;
        }
    }

    ll internal_sum(int i) const {
        ll res = 0;
        i++; // BITは1-indexed
        while (i > 0) {
            res += bit[i];
            i -= i & -i;
        }
        return res;
    }

    ll internal_sum(int l, int r) const {
        if (l > r) return 0;
        return internal_sum(r) - (l ? internal_sum(l-1) : 0);
    }

    // =============================
    // 内部：BIT上でk以上になる最小indexを探す
    // （0-indexedで返す）
    // 戻り値は「prefix和がk未満である最大index」
    // → kth(k) では lower_bound_bit(k + 1) で「0 始まりで k 番目の位置」を求める
    // =============================
    int lower_bound_bit(ll k) const {
        if (k <= 0) return 0;
        int idx = 0;
        int bit_mask = 1;

        // 最大の2冪を探す
        while ((bit_mask << 1) <= n) bit_mask <<= 1;

        for (int d = bit_mask; d > 0; d >>= 1) {
            int next = idx + d;
            if (next <= n && bit[next] < k) {
                k -= bit[next];
                idx = next;
            }
        }
        return idx; // ここは「prefixがk未満の最大位置」
    }

    // 値 v を個数 cnt だけ追加
    // ・事前にコンストラクタで登録した値のみ追加可能
    // ・存在しない値は false を返す
    bool add(ll v, ll cnt=1) {
        int i = index_of(v);
        if (i == -1) return false; // 事前にbuildしていない値は不可
        data[i] += cnt;
        internal_add(i, cnt);
        return true;
    }

    // 値 v を個数 cnt だけ削除
    // ・存在する分だけ削除（不足していれば全部削除）
    // ・cnt以上削除できた場合のみ true
    bool erase(ll v, ll cnt=1) {
        int i = index_of(v);
        if (i == -1) return false; // 事前にbuildしていない値は不可
        ll removed = min(cnt, data[i]);
        data[i] -= removed;
        internal_add(i, -removed);
        return removed == cnt; // cnt以上消せたらtrue, 足りなければfalse
    }

    // ----------------------------
    // 集計系
    // ----------------------------
    ll count(ll v) const {
        int i = index_of(v);
        if (i == -1) return 0;
        return data[i];
    }

    // v未満の個数（strict）
    // lower_boundを使うため、vが存在しなくてもOK
    ll count_lt(ll v) const { // v未満の個数
        auto it = lower_bound(values.begin(), values.end(), v);
        int i = it - values.begin();
        return internal_sum(i-1);
    }

    // v以下の個数
    // upper_boundを使うため、vが存在しなくてもOK
    ll count_le(ll v) const { // v以下の個数
        auto it = upper_bound(values.begin(), values.end(), v);
        int i = it - values.begin();
        return internal_sum(i-1);
    }

    ll count_gt(ll v) const {
        return count_all() - count_le(v);
    }

    ll count_ge(ll v) const {
        return count_all() - count_lt(v);
    }

    ll count_range(ll l, ll r) const { // [l, r) の個数（l >= r なら 0）
        if (l >= r) return 0;
        return count_lt(r) - count_lt(l);
    }

    ll count_range_closed(ll l, ll r) const { // [l, r] の個数（l > r なら 0）
        if (l > r) return 0;
        return count_le(r) - count_lt(l);
    }

    ll count_all() const { // 全体の個数
        return internal_sum(n-1);
    }

    // k番目に小さい値（0-indexed）
    // ・k=0 が最小
    // ・存在しない場合は none（既定 -1）
    ll kth(ll k) const {
        if (k < 0 || k >= count_all()) return none;
        int i = lower_bound_bit(k + 1);
        return values[i];
    }

    // ----------------------------
    // 順位（0-indexed）
    // 「vより小さい要素数」（= count_lt(v)）
    // ----------------------------
    ll rank(ll v) const {
        return count_lt(v);
    }

    // v以上で最小の値（lower_bound的）
    // ・存在しなければ none（既定 -1）
    ll first_ge(ll v) const {
        ll k = count_lt(v); // v以上の最初 = (0 始まりで v未満の個数番目)
        if (k >= count_all()) return none;
        return kth(k);
    }

    // vより大きい値の最小（upper_bound的）
    // ・存在しなければ none（既定 -1）
    ll first_gt(ll v) const {
        ll k = count_le(v); // v以下の次
        if (k >= count_all()) return none;
        return kth(k);
    }

    // LIS（最長増加部分列）
    // ・この構造体の座圧を使って O(N log N) で計算
    // ・strict増加（同値は不可）
    ll LIS(const vector<ll>& seq) {
        vector<ll> xs = seq;
        sort(xs.begin(), xs.end());
        xs.erase(unique(xs.begin(), xs.end()), xs.end());

        int m = xs.size();
        vector<ll> bit(m+1, 0);

        auto get_idx = [&](ll v){
            return lower_bound(xs.begin(), xs.end(), v) - xs.begin();
        };

        auto update = [&](int i, ll x){
            i++;
            while(i <= m){
                bit[i] = max(bit[i], x);
                i += i & -i;
            }
        };

        auto query = [&](int i){
            ll res = 0;
            i++;
            while(i > 0){
                res = max(res, bit[i]);
                i -= i & -i;
            }
            return res;
        };

        ll res = 0;
        for(ll v : seq){
            int i = get_idx(v);
            ll best = query(i-1);
            update(i, best+1);
            res = max(res, best+1);
        }
        return res;
    }

    // 転倒数
    // ・valuesを使って別インスタンスで計算
    // ・O(N log N)
    // 転倒数（i < j かつ seq[i] > seq[j] の組の数）。seq の値で座標圧縮し直すので、作ったときの値と関係なく使える
    ll inversion_count(const vector<ll>& seq) {
        CompressedBIT tmp(seq);
        ll res = 0;
        for (ll v : seq) {
            res += tmp.count_gt(v); // vより大きい要素数
            tmp.add(v, 1);
        }
        return res;
    }
};

void random_test(){
    mt19937_64 rng(0);

    for(int tc = 0; tc < 5000; tc++){
        int Q = 200;

        // =========================
        // 座圧は全集合で固定（ここが重要）
        // =========================
        vector<ll> vals;
        for(int i = 0; i < 20; i++){
            vals.push_back(i);
        }

        CompressedBIT bit(vals);
        multiset<ll> st;

        for(int q = 0; q < Q; q++){
            int type = rng()%9;
            ll x = rng()%20;

            // ------------------------
            // add
            // ------------------------
            if(type == 0){
                bit.add(x);
                st.insert(x);
            }

            // ------------------------
            // erase
            // ------------------------
            if(type == 1){
                bit.erase(x);
                auto it = st.find(x);
                if(it != st.end()) st.erase(it);
            }

            // ------------------------
            // count_lt
            // ------------------------
            if(type == 2){
                ll a = bit.count_lt(x);
                ll b = distance(st.begin(), st.lower_bound(x));
                if(a != b){
                    cout << "count_lt error\n";
                    cout << "x=" << x << "\n";
                    cout << a << " vs " << b << "\n";
                    return;
                }
            }

            // ------------------------
            // count_le
            // ------------------------
            if(type == 3){
                ll a = bit.count_le(x);
                ll b = distance(st.begin(), st.upper_bound(x));
                if(a != b){
                    cout << "count_le error\n";
                    cout << "x=" << x << "\n";
                    cout << a << " vs " << b << "\n";
                    return;
                }
            }

            // ------------------------
            // count_gt
            // ------------------------
            if(type == 4){
                ll a = bit.count_gt(x);
                ll b = distance(st.upper_bound(x), st.end());
                if(a != b){
                    cout << "count_gt error\n";
                    cout << "x=" << x << "\n";
                    cout << a << " vs " << b << "\n";
                    return;
                }
            }

            // ------------------------
            // count_ge
            // ------------------------
            if(type == 5){
                ll a = bit.count_ge(x);
                ll b = distance(st.lower_bound(x), st.end());
                if(a != b){
                    cout << "count_ge error\n";
                    cout << "x=" << x << "\n";
                    cout << a << " vs " << b << "\n";
                    return;
                }
            }

            // ------------------------
            // kth
            // ------------------------
            if(type == 6){
                if(st.empty()) continue;

                int k = rng()%st.size();

                ll a = bit.kth(k);

                auto it = st.begin();
                advance(it, k);
                ll b = *it;

                if(a != b){
                    cout << "kth error\n";
                    cout << "k=" << k << "\n";
                    cout << a << " vs " << b << "\n";
                    return;
                }
            }

            // ------------------------
            // first系
            // ------------------------
            if(type == 7){
                ll a = bit.first_ge(x);
                ll b = -1;
                auto it = st.lower_bound(x);
                if(it != st.end()) b = *it;

                if(a != b){
                    cout << "first_ge error\n";
                    cout << "x=" << x << "\n";
                    cout << a << " vs " << b << "\n";
                    return;
                }

                ll c = bit.first_gt(x);
                ll d = -1;
                auto it2 = st.upper_bound(x);
                if(it2 != st.end()) d = *it2;

                if(c != d){
                    cout << "first_gt error\n";
                    cout << "x=" << x << "\n";
                    cout << c << " vs " << d << "\n";
                    return;
                }
            }

            // ------------------------
            // rank（0-indexed）
            // ------------------------
            if(type == 8){
                ll a = bit.rank(x);
                ll b = distance(st.begin(), st.lower_bound(x));

                if(a != b){
                    cout << "rank error\n";
                    cout << "x=" << x << "\n";
                    cout << a << " vs " << b << "\n";
                    return;
                }
            }
        }
    }

    cout << "OK\n";
}
