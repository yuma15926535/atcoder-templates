struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// RollingHash (テンプレート)
// ======================= ここからテンプレート本体（コピー範囲） =======================
struct RollingHash {
    // ★ 必要なら変更（基本は 2^61-1）
    static const unsigned long long mod = (1ULL<<61) - 1;

    // ★ main でランダムに決める（衝突対策）
    //mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());
    //RollingHash::base = uniform_int_distribution<unsigned long long>(100000000, 1000000000)(rng);
    static unsigned long long base;

    // power[i] = base^i
    static vector<unsigned long long> power;

    vector<unsigned long long> h;   // prefix hash

    // ---- コンストラクタ ----
    // 文字列 s の prefix hash を構築
    RollingHash(const string &s){
        int n = s.size();
        ensure_power(n);
        h.assign(n+1, 0);
        for(int i=0;i<n;i++){
            h[i+1] = add( mul(h[i], base), (unsigned long long)(s[i] + 1) );
        }
    }

    // ---- 部分文字列 [l, r) のハッシュ ----
    unsigned long long get(int l, int r) const{
        // s[l..r)
        return add(h[r], mod - mul(h[l], power[r-l]));
    }

    // ---- 長さ指定取得 ----
    // s[l .. l+len)
    unsigned long long get_len(int l, int len) const{
        return get(l, l+len);
    }

    // ---- 文字列全体 ----
    unsigned long long get_from(int l=0) const{
        return h.back();
    }

    int size() const { return (int)h.size()-1; }

    // ---- ハッシュ連結 ----
    // h1 + h2（len2 は h2 の長さ）
    static unsigned long long concat(
        unsigned long long h1,
        unsigned long long h2,
        int len2
    ){
        ensure_power(len2);
        return add( mul(h1, power[len2]), h2 );
    }

    // ---- LCP: s[i..] と t[j..] の最長共通接頭辞長 ----
    static int lcp(
        const RollingHash &a, int i,
        const RollingHash &b, int j
    ){
        int low = 0, high = min(a.size()-i, b.size()-j) + 1;
        while(high - low > 1){
            int mid = (low + high) / 2;
            if(a.get(i, i+mid) == b.get(j, j+mid)) low = mid;
            else high = mid;
        }
        return low;
    }

    // ---- 2^61-1 乗算 ----
    static unsigned long long mul(unsigned long long a, unsigned long long b){
        __uint128_t t = (__uint128_t)a * b;
        t = (t >> 61) + (t & mod);
        if(t >= mod) t -= mod;
        return (unsigned long long)t;
    }

    static unsigned long long add(unsigned long long a, unsigned long long b){
        a += b;
        if(a >= mod) a -= mod;
        return a;
    }

    // ---- power を伸ばす ----
    // base^n まで事前計算
    static void ensure_power(int n){
        while((int)power.size() <= n){
            power.push_back( mul(power.back(), base) );
        }
    }

    // ---- 辞書順比較 ----
    // s[i..] < t[j..] ?
    static bool less_than(
        const RollingHash &a, int i,
        const RollingHash &b, int j
    ){
        int l = lcp(a, i, b, j);
        if(i + l == a.size() || j + l == b.size())
            return (a.size() - i) < (b.size() - j);
        return a.get(i+l, i+l+1) < b.get(j+l, j+l+1);
    }

};

// 静的メンバ定義
unsigned long long RollingHash::base;
vector<unsigned long long> RollingHash::power = {1};
// ======================= ここまでテンプレート本体 =======================

void solv(){
    mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());
    RollingHash::base =
        uniform_int_distribution<unsigned long long>(100000000, 1000000000)(rng);

    // ----------------------------
    // 1. 部分文字列一致テスト
    // ----------------------------
    string s = "abracadabra";
    string t = "cad";
    RollingHash rh(s), rt(t);

    // "cad" は s[4..7)
    cout << (rh.get(4,7) == rt.get(0,3)) << "\n"; // 1

    // ----------------------------
    // 2. 回文判定テスト
    // ----------------------------
    string p = "racecar";
    string rp = p;
    reverse(rp.begin(), rp.end());

    RollingHash rhp(p), rrhp(rp);
    cout << (rhp.get(0,p.size()) == rrhp.get(0,p.size())) << "\n"; // 1

    // ----------------------------
    // 3. concat テスト
    // ----------------------------
    RollingHash ra("a"), rb("b"), rc("c");

    unsigned long long h = RollingHash::concat(
        ra.get_from(), rb.get_from(), 1
    );
    h = RollingHash::concat(h, rc.get_from(), 1);

    RollingHash rabc("abc");
    cout << (h == rabc.get_from()) << "\n"; // 1

    // ----------------------------
    // 4. LCP テスト
    // ----------------------------
    string x = "banana";
    string y = "bandana";
    RollingHash rx(x), ry(y);

    // "ban" まで一致 → 2
    cout << RollingHash::lcp(rx,0,ry,0) << "\n"; // 3

    // ----------------------------
    // 5. 途中挿入テスト
    // "abde" に "c" を挿入 → "abcde"
    // ----------------------------
    RollingHash rabde("abde"), rc2("c");

    unsigned long long left  = rabde.get(0,2); // "ab"
    unsigned long long right = rabde.get(2,4); // "de"

    unsigned long long ins = RollingHash::concat(
        RollingHash::concat(left, rc2.get_from(), 1),
        right, 2
    );

    RollingHash rabcde("abcde");
    cout << (ins == rabcde.get_from()) << "\n"; // 1

    // ----------------------------
    // 6. 辞書順比較テスト
    // ----------------------------
    string a = "apple";
    string b = "apricot";
    RollingHash r1(a), r2(b);

    // a < b ?
    cout << RollingHash::less_than(r1, 0, r2, 0) << "\n"; // 1

    // b < a ?
    cout << RollingHash::less_than(r2, 0, r1, 0) << "\n"; // 0

    // 部分文字列同士
    // "ple" < "ric" ?
    cout << RollingHash::less_than(r1, 2, r2, 2) << "\n"; // 1

}

int main(){
    BUFF;
    int T=1;
    //cin>>T;
    while(T--)solv();
    return 0;
}
