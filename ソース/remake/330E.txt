struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// =============================================================================
//  MEX(座標圧縮なし+ll管理)
//  （作り直し版：元ファイル 330E.txt）
// -----------------------------------------------------------------------------
//  説明
//    - 種類数の最大値を指定することで座圧なしのll管理が可能
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
//https://atcoder.jp/contests/abc330/tasks/abc330_e

// add       O(logN)
// erase     O(logN)
// erase_all O(logN)
// mex       O(1)
// ======================= ここからテンプレート本体（コピー範囲） =======================
// 個数つきの MEX（値の範囲は long long 全体。負の値・巨大な値も入れてよい）
//   mex に関係するのは 0〜n の値だけなので、それ以外の値は map で個数だけ数える（mex() は O(log n)）
struct Mex{
    int kind;
    set<ll> s;            // 0〜kind のうち 1 個も無い値
    vector<ll>cnt;        // 0〜kind の値の個数
    map<ll,ll> other;     // 範囲外（負・kind より大きい）の値の個数

    //同時に管理する種類数の最大値（答えの mex は n+1 を超えない）
    Mex(int n){
        kind=n;
        erep(i,kind)s.insert(i);
        cnt.resize(kind+1,0);
    }

    //numを1個追加
    void add(ll num){ add(num,1); }

    //numをk個追加
    void add(ll num,ll k){
        if(k<=0)return;
        if(num<0||num>kind){ other[num]+=k; return; }
        bool was_zero=(cnt[num]==0);
        cnt[num]+=k;
        if(was_zero)s.erase(num);
    }

    //numを1個削除
    void erase(ll num){ erase(num,1); }

    //numをk個削除（足りなければ全部）
    void erase(ll num,ll k){
        if(k<=0)return;
        if(num<0||num>kind){
            auto it=other.find(num);
            if(it==other.end())return;
            if(it->second>k) it->second-=k;
            else other.erase(it);
            return;
        }
        if(cnt[num]==0)return;
        if(cnt[num]>k){
            cnt[num]-=k;
            return;
        }
        cnt[num]=0;
        s.insert(num);
    }

    //numを全て削除
    void erase_all(ll num){
        erase(num,count(num));
    }

    //MEXの取得（0〜kind が全部あるときは kind+1）
    int mex(){
        return s.empty()?kind+1:*s.begin();
    }
    //numの個数（範囲外の値も数える）
    ll count(ll num){
        if(num<0||num>kind){
            auto it=other.find(num);
            return it==other.end()?0:it->second;
        }
        return cnt[num];
    }
};
// ======================= ここまでテンプレート本体 =======================


// -----------------------------------------------------------------------------
//  実戦での使用例（ABC330 E）… 標準入力を読むので main からは呼ばない
// -----------------------------------------------------------------------------

void solve_ABC330E(){
    int N,Q;
    cin>>N>>Q;
    vector<int>A(N);
    cin>>A;
    Mex mex(N);
    rep(i,N)mex.add(A[i]);
    rep(i,Q){
        int idx,x;
        cin>>idx>>x;
        idx--;
        mex.erase(A[idx]);
        A[idx]=x;
        mex.add(A[idx]);
        cout<<mex.mex()<<'\n';
    }
    return;
}

// -----------------------------------------------------------------------------
//  使い方デモ（入力なしで動く）
// -----------------------------------------------------------------------------
void demo(){
    cout << "===== Mex（座標圧縮なし・個数つき）デモ =====\n";
    Mex mx(5);                         // 0〜5 の値を管理（要素数 N なら Mex(N) で十分）
    for (ll x : {0, 1, 1, 3, 100}) mx.add(x);   // 範囲外（100）も入れてよい（mex には関係しない）
    cout << "mx.mex() = " << mx.mex() << "   // {0,1,1,3} → 2\n";
    mx.add(2);
    cout << "mx.add(2) 後 mx.mex() = " << mx.mex() << "   // → 4\n";
    mx.erase(1);
    cout << "mx.erase(1) 後 mx.count(1) = " << mx.count(1) << ", mx.mex() = " << mx.mex() << "   // 1 はまだ 1 個残る → 4\n";
    mx.erase_all(0);
    cout << "mx.erase_all(0) 後 mx.mex() = " << mx.mex() << "   // → 0\n";
    mx.add(4, 3);                      // 4 を 3 個まとめて追加
    cout << "mx.add(4, 3) 後 mx.count(4) = " << mx.count(4) << "\n";
    mx.add(-7); mx.add(-7);
    cout << "mx.count(100) = " << mx.count(100) << ", mx.count(-7) = " << mx.count(-7) << "   // 範囲外の値も個数は数える\n";
    mx.erase(-7);
    cout << "mx.erase(-7) 後 mx.count(-7) = " << mx.count(-7) << ", mx.mex() = " << mx.mex() << "\n";
}
