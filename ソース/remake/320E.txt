struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

//  比較関数の渡し方

// -----------------------------------------------------------------------------
//  説明
//    - priority_queue/map/set
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
// ======================= ここからテンプレート本体（コピー範囲） =======================
// ---- 比較関数（ラムダ）の渡し方 ------------------------------------------------
//   sort / set / map        : cmp(a, b) が true なら a が「前」
//   priority_queue          : cmp(a, b) が true なら a が「後ろ」（top から遠い）
//                             → a < b なら top は最大、a > b なら top は最小
//   auto cmp = [](ll a, ll b){ return a > b; };
//   priority_queue<ll, vector<ll>, decltype(cmp)> pq(cmp);   // (cmp) をコンストラクタに渡す
//   set<ll, decltype(cmp)> st(cmp);  map<ll, int, decltype(cmp)> mp(cmp);  sort(v.begin(), v.end(), cmp);
// ---- 比較関数オブジェクト（構造体）：型だけで渡せるので (cmp) を書かなくてよい ----------
// 例: first は昇順、first が同じなら second は降順
struct CmpFirstAscSecondDesc {
    template<class P> bool operator()(const P& a, const P& b) const {
        if (a.first != b.first) return a.first < b.first;
        return a.second > b.second;
    }
};
// ======================= ここまでテンプレート本体 =======================
// 問題固有の構造体（ABC320 E で使用）
struct tws{
    //T:時刻
    //W:量
    //S:抜ける期間
    private:ll T,W,S;
    public:tws(ll t,ll w,ll s):T(t),W(w),S(s){
    }
    ll getT(){
        return T;
    }
    ll getW(){
        return W;
    }
    ll getS(){
        return S;
    }
};



// -----------------------------------------------------------------------------
//  実戦での使用例（ABC320 E）… 標準入力を読むので main からは呼ばない
// -----------------------------------------------------------------------------

int solve_ABC320E(){
    auto cmp=[](ll a,ll b){return a<b;}; // 比較関数
    // 比較関数の渡し方
    //priority_queueに比較関数を渡す
    //priority_queueはtopで値を取得するのでtrueが後から取得される
    priority_queue<ll,vector<ll>,decltype(cmp)> pq(cmp); // コンストラクタで cmp を渡す
    //mapに比較関数を渡す
    //sortの順序と同じ
    map<ll, int,decltype(cmp)> mp(cmp); // コンストラクタで cmp を渡す
    //setに比較関数を渡す
    //sortの順序と同じ
    set<ll,decltype(cmp)> st(cmp); // コンストラクタで cmp を渡す
    int N,M;
    cin>>N>>M;
    //そうめんイベント
    vector<tws>v;
    rep(i,0,M){
        ll t,w,s;
        cin>>t>>w>>s;
        tws e(t,w,s);
        v.emplace_back(e);
    }
    //列にいる人
    PQ_g<int>retsu;
    erep(i,1,N)retsu.push(i);
    //戻ってくる時刻、人
    PQ_g<pair<ll,int>>modori;
    vector<ll>ans(N+1,0);
    int idx=0;
    while(idx<M){
        if(!modori.empty()&&v[idx].getT()>=modori.top().first){
            retsu.push(modori.top().second);
            modori.pop();
        }else{
            if(retsu.empty())goto throwevent;
            ans[retsu.top()]+=v[idx].getW();
            modori.emplace(v[idx].getT()+v[idx].getS(),retsu.top());
            retsu.pop();
            throwevent:
            idx++;
        }
    }
    erep(i,1,N)cout<<ans[i]<<'\n';
    return 0;
}

void demo(){
    cout << "===== 比較関数の渡し方 デモ =====\n";
    auto cmp = [](ll a, ll b){ return a > b; };       // 「大きい方が前」
    vector<ll> v = {3, 1, 4, 1, 5};
    sort(v.begin(), v.end(), cmp);
    cout << "sort(v, a>b)            : "; for(ll x : v) cout << x << " "; cout << "  // 降順\n";
    set<ll, decltype(cmp)> st(cmp); for(ll x : {3, 1, 4}) st.insert(x);
    cout << "set<ll, decltype(cmp)>  : "; for(ll x : st) cout << x << " "; cout << "  // 降順に並ぶ\n";
    priority_queue<ll, vector<ll>, decltype(cmp)> pq(cmp); for(ll x : {3, 1, 4}) pq.push(x);
    cout << "priority_queue(a>b).top : " << pq.top() << "  // 逆になるので最小が top\n";
    vector<pair<ll,ll>> p = {{2, 1}, {1, 5}, {2, 9}, {1, 7}};
    sort(p.begin(), p.end(), CmpFirstAscSecondDesc());
    cout << "CmpFirstAscSecondDesc   : "; for(auto [a, b] : p) cout << "(" << a << "," << b << ") "; cout << "  // first 昇順・second 降順\n";
    set<pair<ll,ll>, CmpFirstAscSecondDesc> st2(p.begin(), p.end());   // 構造体なら (cmp) は不要
    cout << "set<..., CmpFirstAscSecondDesc>.begin() = (" << st2.begin()->first << "," << st2.begin()->second << ")\n";
}
