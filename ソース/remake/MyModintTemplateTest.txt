struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// ======================= ここからテンプレート本体（コピー範囲） =======================
template<uint32_t MOD>
struct FastModInt {

    using u32 = uint32_t;
    using u64 = uint64_t;

    struct Barrett {
        u32 m;
        u64 im;
        Barrett(u32 m):m(m),im((u64)(-1)/m+1){}
        u32 reduce(u64 z) const{
            u64 x=(__uint128_t(z)*im)>>64;
            u32 v=z-x*m;
            if(v>=m) v+=m;
            return v;
        }
    };

    static Barrett& bt(){
        static Barrett b(MOD);
        return b;
    }

    u32 v;

    FastModInt():v(0){}
    template<class T>
    FastModInt(T x){
        long long y=x%(long long)MOD;
        if(y<0)y+=MOD;
        v=y;
    }

    u32 val() const{return v;}

    FastModInt& operator+=(const FastModInt& o){
        v+=o.v; if(v>=MOD)v-=MOD; return *this;
    }
    FastModInt& operator-=(const FastModInt& o){
        if(v<o.v)v+=MOD; v-=o.v; return *this;
    }
    FastModInt& operator*=(const FastModInt& o){
        v=bt().reduce((u64)v*o.v); return *this;
    }
    FastModInt& operator/=(const FastModInt& o){
        return *this*=o.inv();
    }

    template<class T> FastModInt& operator+=(T x){return *this+=FastModInt(x);}
    template<class T> FastModInt& operator-=(T x){return *this-=FastModInt(x);}
    template<class T> FastModInt& operator*=(T x){return *this*=FastModInt(x);}
    template<class T> FastModInt& operator/=(T x){return *this/=FastModInt(x);}

    FastModInt pow(long long e) const{
        FastModInt r=1,x=*this;
        while(e){
            if(e&1) r*=x;
            x*=x;
            e>>=1;
        }
        return r;
    }

    FastModInt inv() const{
        return pow(MOD-2);
    }

    FastModInt& operator++(){*this+=1;return *this;}
    FastModInt operator++(int){auto t=*this;++*this;return t;}
    FastModInt& operator--(){*this-=1;return *this;}
    FastModInt operator--(int){auto t=*this;--*this;return t;}

    friend FastModInt operator+(FastModInt a,const FastModInt& b){return a+=b;}
    friend FastModInt operator-(FastModInt a,const FastModInt& b){return a-=b;}
    friend FastModInt operator*(FastModInt a,const FastModInt& b){return a*=b;}
    friend FastModInt operator/(FastModInt a,const FastModInt& b){return a/=b;}

    friend ostream& operator<<(ostream& os,const FastModInt& x){
        return os<<x.v;
    }
};

// DynBarrettModInt (動的mod)
//////////////////////////////////////////////////////////////
struct DynBarrettModInt {

    using u32=uint32_t;
    using u64=uint64_t;

    struct Barrett{
        u32 m; u64 im;
        void set(u32 mod){m=mod;im=(u64)(-1)/mod+1;}
        u32 reduce(u64 z) const{
            u64 x=(__uint128_t(z)*im)>>64;
            u32 v=z-x*m;
            if(v>=m)v+=m;
            return v;
        }
    };

    static Barrett& bt(){static Barrett b;return b;}
    static void set_mod(u32 m){bt().set(m);}
    static u32 mod(){return bt().m;}

    u32 v;

    DynBarrettModInt():v(0){}
    template<class T>
    DynBarrettModInt(T x){
        long long y=x%(long long)mod();
        if(y<0)y+=mod();
        v=y;
    }

    u32 val() const{return v;}

    DynBarrettModInt& operator+=(const DynBarrettModInt& o){
        v+=o.v;if(v>=mod())v-=mod();return *this;
    }
    DynBarrettModInt& operator-=(const DynBarrettModInt& o){
        if(v<o.v)v+=mod();v-=o.v;return *this;
    }
    DynBarrettModInt& operator*=(const DynBarrettModInt& o){
        v=bt().reduce((u64)v*o.v);return *this;
    }
    DynBarrettModInt& operator/=(const DynBarrettModInt& o){
        return *this*=o.inv();
    }

    template<class T> DynBarrettModInt& operator+=(T x){return *this+=DynBarrettModInt(x);}
    template<class T> DynBarrettModInt& operator-=(T x){return *this-=DynBarrettModInt(x);}
    template<class T> DynBarrettModInt& operator*=(T x){return *this*=DynBarrettModInt(x);}
    template<class T> DynBarrettModInt& operator/=(T x){return *this/=DynBarrettModInt(x);}

    DynBarrettModInt pow(long long e) const{
        DynBarrettModInt r=1,x=*this;
        while(e){
            if(e&1) r*=x;
            x*=x;
            e>>=1;
        }
        return r;
    }

    DynBarrettModInt inv() const{
        return pow(mod()-2);
    }

    DynBarrettModInt& operator++(){*this+=1;return *this;}
    DynBarrettModInt operator++(int){auto t=*this;++*this;return t;}
    DynBarrettModInt& operator--(){*this-=1;return *this;}
    DynBarrettModInt operator--(int){auto t=*this;--*this;return t;}

    friend DynBarrettModInt operator+(DynBarrettModInt a,const DynBarrettModInt& b){return a+=b;}
    friend DynBarrettModInt operator-(DynBarrettModInt a,const DynBarrettModInt& b){return a-=b;}
    friend DynBarrettModInt operator*(DynBarrettModInt a,const DynBarrettModInt& b){return a*=b;}
    friend DynBarrettModInt operator/(DynBarrettModInt a,const DynBarrettModInt& b){return a/=b;}

    friend ostream& operator<<(ostream& os,const DynBarrettModInt& x){
        return os<<x.v;
    }
};

void solv(){
    cout<<"=============================\n";
    cout<<"  modint 全演算 動作確認\n";
    cout<<"=============================\n\n";

    //------------------------------------------------
    // 静的mod（FastModInt）
    //------------------------------------------------

    cout<<"--- FastModInt (静的mod) ---\n";

    using SM = FastModInt<1000000007>;

    SM a = 123456789;
    SM b = 987654321;

    cout<<"a = "<<a<<"\n";
    cout<<"b = "<<b<<"\n";

    cout<<"\n[基本演算]\n";
    cout<<"a + b = "<<a+b<<"\n";
    cout<<"a - b = "<<a-b<<"\n";
    cout<<"a * b = "<<a*b<<"\n";
    cout<<"a / b = "<<a/b<<"\n";

    cout<<"\n[pow / inv]\n";
    cout<<"pow(a,10) = "<<a.pow(10)<<"\n";
    cout<<"inv(a) = "<<a.inv()<<"\n";

    cout<<"\n[++ / --]\n";
    SM x = a;
    cout<<"初期 x = "<<x<<"\n";
    cout<<"前置++ = "<<++x<<"\n";
    cout<<"後置++ = "<<x++<<" → 現在 "<<x<<"\n";
    cout<<"前置-- = "<<--x<<"\n";
    cout<<"後置-- = "<<x--<<" → 現在 "<<x<<"\n";

    cout<<"\n[整数混在]\n";
    cout<<"a + 5 = "<<a+5<<"\n";
    cout<<"5 + a = "<<5+a<<"\n";
    cout<<"a - 100 = "<<a-100<<"\n";
    cout<<"a * 3 = "<<a*3<<"\n";
    cout<<"a / 7 = "<<a/7<<"\n";

    cout<<"\n[チェーン演算]\n";
    cout<<"(a+3)*(b-2)/5 = "<<(a+3)*(b-2)/5<<"\n";

    cout<<"\n[正規化確認]\n";
    cout<<"SM(-123) = "<<SM(-123)<<"\n";
    cout<<"SM(1e18) = "<<SM(1000000000000000000LL)<<"\n";


    //------------------------------------------------
    // 動的mod（DynBarrettModInt）
    //------------------------------------------------

    cout<<"\n--- DynBarrettModInt (動的mod) ---\n";

    DynBarrettModInt::set_mod(1000000007);

    DynBarrettModInt p = 555555555;

    cout<<"p = "<<p<<"\n";
    cout<<"q = "<<q<<"\n";

    cout<<"\n[基本演算]\n";
    cout<<"p + q = "<<p+q<<"\n";
    cout<<"p - q = "<<p-q<<"\n";
    cout<<"p * q = "<<p*q<<"\n";
    cout<<"p / q = "<<p/q<<"\n";

    cout<<"\n[pow / inv]\n";
    cout<<"pow(p,10) = "<<p.pow(10)<<"\n";
    cout<<"inv(p) = "<<p.inv()<<"\n";

    cout<<"\n[++ / --]\n";
    DynBarrettModInt y = p;
    cout<<"初期 y = "<<y<<"\n";
    cout<<"前置++ = "<<++y<<"\n";
    cout<<"後置++ = "<<y++<<" → 現在 "<<y<<"\n";
    cout<<"前置-- = "<<--y<<"\n";
    cout<<"後置-- = "<<y--<<" → 現在 "<<y<<"\n";

    cout<<"\n[整数混在]\n";
    cout<<"p + 7 = "<<p+7<<"\n";
    cout<<"7 + p = "<<7+p<<"\n";
    cout<<"p - 50 = "<<p-50<<"\n";
    cout<<"p * 9 = "<<p*9<<"\n";
    cout<<"p / 11 = "<<p/11<<"\n";

    cout<<"\n[チェーン演算]\n";
    cout<<"(p+8)*(q-3)/6 = "<<(p+8)*(q-3)/6<<"\n";

    cout<<"\n[正規化確認]\n";
