struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// =============================================================================
//  状態付きグリッドのグラフ化BFS
//  （作り直し版：元ファイル 453D.txt）
// -----------------------------------------------------------------------------
//  説明
//    - 遷移関数を外部で定義
//    - 経路復元付き
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
// ======================= ここからテンプレート本体（コピー範囲） =======================
// 状態付きグリッドの BFS（状態 = 向き・残り回数・フラグなど 0..S-1）
//   dist / prev は 1 次元配列（番号 id = (i * W + j) * S + s）。キャッシュに乗りやすく速い
struct GridStateBFS {
    int H, W, S; // S = 状態数
    vector<string> grid;

    vector<int> dist;   // dist[id(i, j, s)]：スタートからの手数（届かなければ -1）
    vector<int> prev;   // prev[id]：1 つ前の id（スタートは -1）

    int dh[4] = {1,-1,0,0};
    int dw[4] = {0,0,1,-1};
    char dc[4] = {'D','U','R','L'};

    GridStateBFS(int h, int w, int s, const vector<string>& g)
        : H(h), W(w), S(s), grid(g), dist((size_t)h * w * s, -1), prev((size_t)h * w * s, -1) {}

    int id(int i, int j, int s) const { return (i * W + j) * S + s; }
    bool in(int i,int j) const {
        return (0<=i && i<H && 0<=j && j<W);
    }

    // (i, j) に状態 s でいるときの手数（届かなければ -1）
    int dist_at(int i, int j, int s) const { return dist[id(i, j, s)]; }
    // (i, j) のどれかの状態に着く最小手数（届かなければ -1）
    int min_dist(int i, int j) const {
        int best = -1;
        for (int s = 0; s < S; s++) {
            int d = dist[id(i, j, s)];
            if (d != -1 && (best == -1 || d < best)) best = d;
        }
        return best;
    }
    bool reachable(int i, int j) const { return min_dist(i, j) != -1; }

    // ========================
    // BFS
    // ========================
    // transition(i, j, s) は「(i, j) に状態 s でいるとき 1 手で行ける (ni, nj, ns) の列」を返す
    // starts：スタートの (i, j, s) の列（複数可）。呼ぶたびに dist / prev は作り直す
    template<class F>
    void bfs_multi(const vector<tuple<int,int,int>>& starts, F transition){
        fill(dist.begin(), dist.end(), -1);
        fill(prev.begin(), prev.end(), -1);
        vector<int> q;
        q.reserve(dist.size());
        for(auto [si, sj, s] : starts){
            int k = id(si, sj, s);
            if(dist[k] != -1) continue;
            dist[k] = 0;
            q.push_back(k);
        }
        for(size_t head = 0; head < q.size(); head++){
            int k = q[head];
            int s = k % S, i = k / S / W, j = k / S % W;
            // 遷移を外から渡す
            for(auto [ni,nj,ns] : transition(i,j,s)){
                if(!in(ni,nj)) continue;
                if(grid[ni][nj]=='#') continue;
                int nk = id(ni, nj, ns);
                if(dist[nk]!=-1) continue;
                dist[nk] = dist[k] + 1;
                prev[nk] = k;
                q.push_back(nk);
            }
        }
    }
    // スタートの状態が自由（すべての状態を距離 0 にする。向きが自由な問題向け）
    template<class F>
    void bfs(int si,int sj, F transition){
        vector<tuple<int,int,int>> starts;
        for(int s=0;s<S;s++) starts.push_back({si,sj,s});
        bfs_multi(starts, transition);
    }
    // スタートの状態を s0 に決める
    template<class F>
    void bfs(int si,int sj,int s0, F transition){
        bfs_multi({{si,sj,s0}}, transition);
    }

    // ========================
    // 経路復元
    // ========================
    // ゴール (gi, gj) の状態のうち距離が最小のものまでの (i, j, s) の列（スタートを含む）。届かなければ空
    vector<tuple<int,int,int>> restore_states(int gi,int gj){
        int best = -1;
        for(int s=0;s<S;s++){
            int d = dist[id(gi,gj,s)];
            if(d!=-1 && (best==-1 || d < dist[id(gi,gj,best)])) best = s;
        }
        if(best==-1) return {};
        vector<tuple<int,int,int>> res;
        for(int k = id(gi,gj,best); k != -1; k = prev[k]) res.push_back({k / S / W, k / S % W, k % S});
        reverse(res.begin(),res.end());
        return res;
    }
    // 移動の向きの列（'D','U','R','L'）。1 マスずつ動かない遷移は文字にならない
    // ※ 届かないときも、スタート = ゴールのときも空なので、区別は reachable(gi, gj) で行う
    vector<char> restore(int gi,int gj){
        auto st = restore_states(gi, gj);
        vector<char> res;
        for(size_t t = 1; t < st.size(); t++){
            auto [pi,pj,ps] = st[t-1];
            auto [i,j,s] = st[t];
            for(int d=0;d<4;d++) if(pi + dh[d] == i && pj + dw[d] == j) res.push_back(dc[d]);
        }
        return res;
    }
};
// ======================= ここまでテンプレート本体 =======================


// -----------------------------------------------------------------------------
//  実戦での使用例（ABC453 D）… 標準入力を読むので main からは呼ばない
// -----------------------------------------------------------------------------

void solve_ABC453D(){
    int H,W;
    cin>>H>>W;
    vector<string>S(H);
    cin>>S;
    int sh,sw,gh,gw;
    rep(i,0,H)rep(j,0,W){
        if(S[i][j]=='S'){
            sh=i;
            sw=j;
        }else if(S[i][j]=='G'){
            gh=i;
            gw=j;
        }
    }

    GridStateBFS bfs(H,W,4,S);

    auto transition = [&](int i,int j,int dir){
        vector<tuple<int,int,int>> res;

        for(int ndir=0; ndir<4; ndir++){
            int ni = i + bfs.dh[ndir];
            int nj = j + bfs.dw[ndir];

            char c = S[i][j];

            if(c=='o' && ndir!=dir) continue;
            if(c=='x' && ndir==dir) continue;

            res.push_back({ni,nj,ndir});
        }

        return res;
    };

    bfs.bfs(sh,sw,transition);

    auto path = bfs.restore(gh,gw);

    if(!bfs.reachable(gh,gw)){
        cout<<"No\n";
    }else{
        cout<<"Yes\n";
        for(char c:path) cout<<c;
        cout<<"\n";
    }
    return;
}

void demo(){
    cout << "===== GridStateBFS デモ（状態 = 直前に進んだ向き）=====\n";
    // 'o' のマスでは直進しかできない。'#' は壁
    vector<string> g = {"S.#",
                        ".o.",
                        "..G"};
    GridStateBFS bfs(3, 3, 4, g);                          // 状態数 4（D, U, R, L）
    auto transition = [&](int i, int j, int dir){
        vector<tuple<int,int,int>> res;
        for (int nd = 0; nd < 4; nd++) {
            if (g[i][j] == 'o' && nd != dir) continue;     // 'o' は直進のみ
            res.push_back({i + bfs.dh[nd], j + bfs.dw[nd], nd});
        }
        return res;
    };
    bfs.bfs(0, 0, transition);
    auto path = bfs.restore(2, 2);
    cout << "最短手数 = " << path.size() << ", 経路 = "; for (char c : path) cout << c; cout << "\n";
    cout << "bfs.min_dist(2, 2) = " << bfs.min_dist(2, 2) << ", bfs.reachable(2, 2) = " << bfs.reachable(2, 2) << "\n";
    cout << "bfs.dist_at(2, 2, 0) = " << bfs.dist_at(2, 2, 0) << "   // 状態 0（下向き）でゴールに着く手数\n";
    auto st = bfs.restore_states(2, 2);
    cout << "restore_states(2, 2) = "; for (auto [i, j, s] : st) cout << "(" << i << "," << j << ",状態" << s << ") "; cout << "\n";
    // スタートの状態を決める版：最初は右向き（状態 2）だけ
    bfs.bfs(0, 0, 2, transition);
    cout << "スタートを右向きに決めた場合: min_dist(2, 2) = " << bfs.min_dist(2, 2) << "\n";
    // 届かない例
    vector<string> g2 = {"S#", "#G"};
    GridStateBFS bfs2(2, 2, 1, g2);
    bfs2.bfs(0, 0, [&](int i, int j, int s){ vector<tuple<int,int,int>> r; for (int d = 0; d < 4; d++) r.push_back({i + bfs2.dh[d], j + bfs2.dw[d], 0}); return r; });
    cout << "壁で届かない: reachable(1, 1) = " << bfs2.reachable(1, 1) << ", restore(1, 1).size() = " << bfs2.restore(1, 1).size() << "\n";
}
