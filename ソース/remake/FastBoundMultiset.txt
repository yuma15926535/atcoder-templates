struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

//  FastBoundMultiset（イテレータを隠蔽した二分探索主軸のmultiset）
//  （作り直し版：元ファイル FastBoundMultiset.txt）
// -----------------------------------------------------------------------------
//  説明
//    - 引数の境界検索(以上、以下、より大きい、より小さい)
//    - min/maxの取得
//    - count
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
// ======================= ここからテンプレート本体（コピー範囲） =======================
/*
===========================================================
  FastBoundMultiset（イテレータを隠蔽した二分探索主軸のmultiset）
  iterator を隠蔽した高速 multiset ラッパー

  --- 境界検索 ---
  ge(x) : x以上の最小値
  gt(x) : xより大きい最小値
  le(x) : x以下の最大値
  lt(x) : xより小さい最大値

  --- 高速取得 ---
  get_min() : 最小値
  get_max() : 最大値

  --- カウント ---
  count(x) : x の個数 (O(log N))

  戻り値:
    成功 → true + out
    失敗 → false

  計算量:
    insert / erase / 検索 全て O(log N)
    min/max O(1)

  iterator は外に出ない
===========================================================
*/
template<class T>
struct FastBoundMultiset {
private:
    multiset<T> ms;

public:

    // =========================
    // 基本操作
    // =========================

    void insert(const T& x) {
        ms.insert(x);
    }

    void erase_one(const T& x) {
        auto it = ms.find(x);
        if (it != ms.end()) ms.erase(it);
    }

    int size() const { return ms.size(); }
    bool empty() const { return ms.empty(); }

    // =========================
    // x以上 (>=)
    // =========================
    bool ge(const T& x, T& out) const {
        auto it = ms.lower_bound(x);
        if (it == ms.end()) return false;
        out = *it;
        return true;
    }

    // =========================
    // xより大きい (>)
    // =========================
    bool gt(const T& x, T& out) const {
        auto it = ms.upper_bound(x);
        if (it == ms.end()) return false;
        out = *it;
        return true;
    }

    // =========================
    // x以下 (<=)
    // =========================
    bool le(const T& x, T& out) const {
        auto it = ms.upper_bound(x);
        if (it == ms.begin()) return false;
        --it;
        out = *it;
        return true;
    }

    // =========================
    // xより小さい (<)
    // =========================
    bool lt(const T& x, T& out) const {
        auto it = ms.lower_bound(x);
        if (it == ms.begin()) return false;
        --it;
        out = *it;
        return true;
    }

    // =========================
    // 最小値 O(1)
    // =========================
    bool get_min(T& out) const {
        if (ms.empty()) return false;
        out = *ms.begin();
        return true;
    }

    // =========================
    // 最大値 O(1)
    // =========================
    bool get_max(T& out) const {
        if (ms.empty()) return false;
        out = *ms.rbegin();
        return true;
    }

    // =========================
    // 高速 count
    // O(log N)
    // =========================
    int count(const T& x) const {
        auto l = ms.lower_bound(x);
        if (l == ms.end() || *l != x) return 0;
        auto r = ms.upper_bound(x);
        return distance(l, r);
    }

    // =========================
    // 高速 存在判定
    // O(log N)
    // =========================
    bool contains(const T& x) const {
        return ms.find(x) != ms.end();
    }

    // =========================
    // 高速 範囲要素存在判定
    // 判定範囲:[l,r)（半開区間。閉区間 [l,r] は contains_range_closed）
    // O(log N)
    // =========================
    bool contains_range(const T& l, const T& r) const {
        if (!(l < r)) return false;
        auto it = ms.lower_bound(l);
        return it != ms.end() && *it < r;
    }
    bool contains_range_closed(const T& l, const T& r) const {
        auto it = ms.lower_bound(l);
        return it != ms.end() && !(r < *it);
    }
};
// ======================= ここまでテンプレート本体 =======================

void solv(){
    cout << "=============================\n";
    cout << " FastBoundMultiset 動作確認\n";
    cout << "=============================\n\n";

    FastBoundMultiset<int> s;

    cout << "▼ 要素を追加\n";
    vector<int> init = {5, 10, 10, 3, 8};
    for(int x : init) {
        s.insert(x);
        cout << "  挿入: " << x << "\n";
    }
    cout << "\n";

    int v;

    cout << "▼ 最小値 / 最大値\n";
    if(s.get_min(v)) cout << "  最小値: " << v << "\n";
    else cout << "  最小値: なし\n";

    if(s.get_max(v)) cout << "  最大値: " << v << "\n";
    else cout << "  最大値: なし\n";

    cout << "\n";

    int q = 9;
    cout << "▼ 境界検索 (基準値 = " << q << ")\n";

    if(s.ge(q,v)) cout << "  " << q << "以上: " << v << "\n";
    else cout << "  " << q << "以上: なし\n";

    if(s.gt(q,v)) cout << "  " << q << "より大きい: " << v << "\n";
    else cout << "  " << q << "より大きい: なし\n";

    if(s.le(q,v)) cout << "  " << q << "以下: " << v << "\n";
    else cout << "  " << q << "以下: なし\n";

    if(s.lt(q,v)) cout << "  " << q << "より小さい: " << v << "\n";
    else cout << "  " << q << "より小さい: なし\n";

    cout << "\n";

    cout << "▼ 個数カウント\n";
    cout << "  値10の個数: " << s.count(10) << "\n";
    cout << "  値7の個数: " << s.count(7) << "\n";

    cout << "\n";

    cout << "▼ 要素削除\n";
    s.erase_one(10);
    cout << "  10を1個削除\n";
    cout << "  値10の個数: " << s.count(10) << "\n";

    cout << "\n";

    cout << "▼ 存在確認\n";
    cout << "  10の存在確認: " << (s.contains(10)?"存在します":"存在しません") << "\n";
    cout << "  7の存在確認: " << (s.contains(7)?"存在します":"存在しません") << "\n";

    cout << "\n";

    cout << "▼ 範囲要素存在判定確認\n";
    cout << "  [9,10]の要素存在確認（contains_range_closed）: " << (s.contains_range_closed(9,10)?"存在します":"存在しません") << "\n";
    cout << "  [9,10)の要素存在確認（contains_range）: " << (s.contains_range(9,10)?"存在します":"存在しません") << "\n";
    cout << "  [11,100]の要素存在確認: " << (s.contains_range_closed(11,100)?"存在します":"存在しません") << "\n";

    cout << "\n";

    cout << "=============================\n";
    cout << " 動作確認終了\n";
    cout << "=============================\n";
    return;
}

// 愚直解とのランダム比較テスト（contains_range / contains_range_closed）
void random_test_contains_range(){
    mt19937 rng(7);
    for(int it = 0; it < 3000; it++){
        FastBoundMultiset<int> s;
        multiset<int> ms;
        int n = rng() % 8;
        for(int i = 0; i < n; i++){ int x = rng() % 20; s.insert(x); ms.insert(x); }
        int l = rng() % 22 - 1, r = rng() % 22 - 1;
        bool h = false, c = false;
        for(int x : ms){ if(l <= x && x < r) h = true; if(l <= x && x <= r) c = true; }
        assert(s.contains_range(l, r) == h);
        assert(s.contains_range_closed(l, r) == c);
    }
    cout << "random_test_contains_range: OK\n";
}

int main(){
    random_test_contains_range();
    BUFF;
    int T=1;
    //cin>>T;
    while(T--)solv();
    return 0;
}
