struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

template<class T>
std::enable_if_t<is_iter_v<T>,ostream&>
operator<<(ostream&os,const T&v){
    using V = std::decay_t<decltype(*begin(v))>;
    if constexpr(is_iter_v<V>){
        bool first_row=true;
        os<<"\n";
        for(auto&&row:v){
            if(!first_row) os<<"\n";
            first_row=false;
            bool f=0;
            for(auto&&x:row){
                if(f) os<<" ";
                os<<x;
                f=1;
            }
        }
    }
    else{
        os<<"[";
        bool f=0;
        for(auto&&x:v){
            if(f) os<<", ";
            os<<x;
            f=1;
        }
        os<<"]";
    }
    return os;
}

template<class T,class...U>
void _dbg(const char*names,T&&x,U&&...y){
    const char*comma=strchr(names,',');
    if(!comma){
        cerr<<DBG_COLOR<<names<<" = "<<x<<"\n"<<DBG_RESET;
    }else{
        cerr<<DBG_COLOR;
        cerr.write(names,comma-names)<<" = "<<x<<" | ";
        _dbg(comma+1,y...);
    }
}

//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）

// ======================= ここからテンプレート本体（コピー範囲） =======================
// 小数を桁数を指定して出力する（「誤差 1e-6 以下なら正解」などの問題で使う）
//   cout << fixed << setprecision(15) << x << '\n';   ← Main_base の FIXED(15) を使って cout << FIXED(15) << x でも同じ
//   printf("%.15f\n", x);   long double は printf("%.15Lf\n", x);
//   ※ fixed を付けないと setprecision は「有効数字の桁数」になる（1e-5 などが指数表記になる）
//   print_fixed(x, 桁数) は出力したあと cout の設定（fixed・精度）を元に戻すので、後の出力に影響しない
template<class T> void print_fixed(const T& x, int digits = 15) {
    ios::fmtflags f = cout.flags();
    streamsize p = cout.precision();
    cout << fixed << setprecision(digits) << x << '\n';
    cout.flags(f);
    cout.precision(p);
}
// ======================= ここまでテンプレート本体 =======================
// -----------------------------------------------------------------------------
//  実戦での使用例（ABC126 C）… 標準入力を読むので main からは呼ばない
// -----------------------------------------------------------------------------
int solve_ABC126C()
{
    int N,K;
    cin>>N>>K;
    map<int,ll> m;
    int num=1,c=0;
    rrep(i,N+1,1){
        while(num*i<K){
            num*=2;
            c++;
        }
        if(m.count(c))m[c]++;
        else m[c]=1;
    }
    double ans=0;
    auto q=*m.begin();
    ll num2=num;
    rep(i,0,q.first)num/=2;
    for(auto p:m){
        ans+=p.second*num;
        num/=2;
    }
    //出力の桁数を指定
    cout<<setprecision(11)<<ans/(N*num2)<<"\n";
    //or printf("%.11f\n",ans/(N*num2));
    return 0;
}

    cout << "print_fixed(1.0/7, 3)   : "; print_fixed(x, 3);        // 0.143
    cout << "その後の普通の出力      : " << x << "   // print_fixed は設定を戻すので既定の表示（6 桁）のまま\n";
    cout << "cout << FIXED(10) << x  : " << FIXED(10) << x << "\n";
    cout << defaultfloat << setprecision(6);                         // 元に戻すとき
    cout << "setprecision だけ(1e-7) : " << setprecision(3) << 1e-7 << "   // fixed がないと指数表記\n";
    cout << flush;                                                   // BUFF（sync_with_stdio(false)）中に printf と混ぜるときは先に flush
    printf("printf(\"%%.5f\")        : %.5f\n", x);
    fflush(stdout);
}
