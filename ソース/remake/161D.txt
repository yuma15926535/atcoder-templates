struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// =============================================================================
//  状態遷移BFSテンプレート(K番目の値)
//  （作り直し版：元ファイル 161D.txt）
// -----------------------------------------------------------------------------
//  説明
//    - BFSで条件を満たす値を生成する
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
// ========================
// 状態遷移BFSテンプレート
// ========================
// 【変えるところ】
// - Stateの中身
// - next_states
// - 初期状態
// - 終了条件（K番目 or 条件）
// ======================= ここからテンプレート本体（コピー範囲） =======================
struct State {
    long long val;// 作っている数
    int last;// 末尾の桁
};
// ※問題によっては
// - 桁数
// - フラグ（条件達成済みなど）
// - mod
// などを持たせる

// 【任意】枝刈り例
// if (cur.val > N) continue;
vector<State> next_states(const State& s) {
    vector<State> res;
    for (int d = -1; d <= 1; d++) {
        int nd = s.last + d;
        if (0 <= nd && nd <= 9) {
            res.push_back({s.val * 10 + nd, nd});
        }
    }
    return res;
}
// BFS 本体：init から幅優先で状態を作っていく。visit(cur) が false を返したらそこで打ち切る
//   （状態が無限に増える問題では visit 側で打ち切る／枝刈りは next_states 側で）
template<class F>
void state_bfs(const vector<State>& init, F visit) {
    queue<State> q;
    for (const auto& s : init) q.push(s);
    while (!q.empty()) {
        State cur = q.front(); q.pop();
        if (!visit(cur)) return;
        for (const auto& nxt : next_states(cur)) q.push(nxt);
    }
}
// 同じ状態が何度も出る問題用：key(s) が同じ状態は最初の 1 回だけ訪れる（BFS なので最初＝最短手数）
//   key は set に入れられる型（long long / pair / tuple / string など）を返す
template<class F, class KeyF>
void state_bfs_unique(const vector<State>& init, F visit, KeyF key) {
    using Key = decay_t<decltype(key(declval<const State&>()))>;
    set<Key> seen;
    queue<State> q;
    for (const auto& s : init) if (seen.insert(key(s)).second) q.push(s);
    while (!q.empty()) {
        State cur = q.front(); q.pop();
        if (!visit(cur)) return;
        for (const auto& nxt : next_states(cur)) if (seen.insert(key(nxt)).second) q.push(nxt);
    }
}
// ======================= ここまでテンプレート本体 =======================


// -----------------------------------------------------------------------------
//  実戦での使用例（ABC161 D）… 標準入力を読むので main からは呼ばない
// -----------------------------------------------------------------------------

void solve_ABC161D(){
    int K;
    cin >> K;

    queue<State> q;

    // 【変える】初期状態
    for (int d = 1; d <= 9; d++) {
        q.push({d, d});
    }

    int cnt = 0;
    while (!q.empty()) {
        auto cur = q.front(); q.pop();

        // 【変える】処理
        cnt++;
        if (cnt == K) {
            cout << cur.val << endl;
            break;
        }

        // 遷移
        for (auto nxt : next_states(cur)) {
            q.push(nxt);
        }
    }
    return;
}

// -----------------------------------------------------------------------------
//  使い方デモ（入力なしで動く）
// -----------------------------------------------------------------------------
void demo(){
    cout << "===== 状態遷移 BFS デモ（隣り合う桁の差が 1 以下の数を小さい順に） =====\n";
    vector<State> init;
    for (int d = 1; d <= 9; d++) init.push_back({d, d});   // 【変える】初期状態
    int K = 15, cnt = 0;
    cout << "小さい順に 15 個: ";
    state_bfs(init, [&](const State& cur){
        cout << cur.val << " ";
        return ++cnt < K;                                  // false で打ち切り
    });
    cout << "  // 15 番目は 23（ABC161 D の入力例）\n";

    // 同じ状態を 2 回訪れない版：ここでは「末尾の桁」を状態の key にする
    //   → 末尾の桁ごとに、最初に（最も小さく）出てくる数だけを訪れる
    cout << "state_bfs_unique（key = 末尾の桁）: ";
    state_bfs_unique(init, [&](const State& cur){
        cout << cur.val << " ";
        return true;
    }, [](const State& s){ return s.last; });
    cout << "  // 1〜9 と、末尾が 0 で最初の 10 の 10 個\n";
}
