template<class T>
struct CoordinateCompress {
    vector<T> xs;

    //座標圧縮対象
    void add(const T& x) { xs.push_back(x); }
    void add(const vector<T>& v) { for (auto &x : v) xs.push_back(x); }

    //座標圧縮
    void build() {
        sort(xs.begin(), xs.end());
        xs.erase(unique(xs.begin(), xs.end()), xs.end());
    }

    //引数の座標圧縮後の値(0-index)
    int get(const T& x) const {
        return lower_bound(xs.begin(), xs.end(), x) - xs.begin();
    }

    //引数の座標圧縮後の値(0-index) 存在しなければ-1
    int get_or_minus1(const T& x) const {
        auto it = lower_bound(xs.begin(), xs.end(), x);
        if (it == xs.end() || *it != x) return -1;
        return it - xs.begin();
    }

    //実際の値から座標圧縮後の値
    bool contains(const T& x) const {
        return binary_search(xs.begin(), xs.end(), x);
    }

    //以上の最小の座標圧縮後の値
    int lower_idx(const T& x) const {
        return lower_bound(xs.begin(), xs.end(), x) - xs.begin();
    }

    //より大きい最小の座標圧縮後の値
    int upper_idx(const T& x) const {
        return upper_bound(xs.begin(), xs.end(), x) - xs.begin();
    }

    //座標圧縮後の値から実際の値
    T rev(int i) const { return xs[i]; }

    //座標圧縮後の値の範囲から実際の値の範囲長
    T range_len(int l, int r) const {
        return xs[r] - xs[l];
    }

    int size() const { return xs.size(); }
};

struct BIT {
    int n;
    vector<long long> bit;

    BIT(int n=0): n(n), bit(n+1,0) {}
    // すべての要素を init にして O(N) で作る（各ノードは「長さ lowbit(i) の区間の和」）
    BIT(int n, long long init): n(n), bit(n+1,0) {
        for(int i=1;i<=n;i++) bit[i] = init * (i & -i);
    }

    // index i に v を加算
    void add(int i, long long v){
        for(i++; i<=n; i+=i&-i) bit[i] += v;
    }

    // [0, r) の和（ALPCB.txt の BIT と同じ約束）
    long long sum(int r) const {
        long long s = 0;
        for(; r>0; r-=r&-r) s += bit[r];
        return s;
    }

    // 累積和 sum(i + 1) が k 以上になる最小の index i を返す（0-index。k >= 1 を想定。無ければ n）
    int kth(long long k) const {
        int idx = 0;
        long long cur = 0;
        int top = 1;
        while((top << 1) <= n) top <<= 1;   // n 以下の最大の 2 冪（n の大きさに制限なし）
        for(int d = top; d > 0; d >>= 1){
            if(idx + d <= n && cur + bit[idx + d] < k){
                cur += bit[idx + d];
                idx += d;
            }
        }
        return idx; // 0-index
    }
};

struct MEX {
    int N;                 // 管理する値は 0 〜 N-1
    vector<int> cnt;       // 各値の個数
    BIT bit;               // 欠番なら 1, 存在するなら 0
    long long sz = 0;      // 現在の要素数（重複込み）
    // 初期状態：すべて欠番
    MEX(int N): N(N), cnt(N,0), bit(N, 1) {}   // 全部 1（欠番）の BIT を O(N) で作る
    // x を 1 個追加（重複可。範囲外は無視）
    void add(int x){
        if(x < 0 || x >= N) return;
        if(cnt[x] == 0) bit.add(x, -1); // 欠番でなくなる
        cnt[x]++;
        sz++;
    }
    // x を 1 個削除（無ければ何もしない）
    void erase(int x){
        if(x < 0 || x >= N || cnt[x] == 0) return;
        cnt[x]--;
        if(cnt[x] == 0) bit.add(x, +1); // 再び欠番
        sz--;
    }
    // 現在の要素数（重複込み）
    long long size() const { return sz; }
    // x の個数 / x があるか / x が欠番か
    int count(int x) const { return (0 <= x && x < N) ? cnt[x] : 0; }
    bool contains(int x) const { return count(x) > 0; }
    bool is_missing(int x) const { return 0 <= x && x < N && cnt[x] == 0; }
    // 全体の欠番数
    long long missing_count() const { return bit.sum(N); }
    // [l, r) に含まれる欠番の個数
    long long count_missing(int l, int r) const {
        l = max(l, 0); r = min(r, N);
        if(l >= r) return 0;
        return bit.sum(r) - bit.sum(l);
    }
    // 最小の欠番（mex）。0 〜 N-1 が全部あるときは N
    int mex() const {
        if(missing_count() == 0) return N;
        return bit.kth(1);
    }
    // 最大の欠番（無ければ -1）
    int max_missing() const {
        long long tot = missing_count();
        if(tot == 0) return -1;
        return bit.kth(tot);
    }
    // 小さい方から k 番目の欠番（k は 0 始まり。無ければ -1）
    int kth_missing(long long k) const {
        if(k < 0 || missing_count() <= k) return -1;
        return bit.kth(k + 1);
    }
    // x より大きい最小の欠番（無ければ -1）。「l 以上で最小」は next_missing(l - 1)
    int next_missing(int x) const {
        return kth_missing(count_missing(0, x + 1));
    }
    // x より小さい最大の欠番（無ければ -1）
    int prev_missing(int x) const {
        long long c = count_missing(0, x);
        return c == 0 ? -1 : kth_missing(c - 1);
    }
};

// 座標圧縮 + MEX 管理クラス
// ・欠番管理は BIT による 0/1 管理
// ・返却は「ID版」と「実際の値版」の両方を提供
template<class T>
struct MexCompressed {
    // ========= 内部構造 =========
    CoordinateCompress<T> comp; // 座標圧縮
    BIT bit;                    // 欠番管理用 BIT（1 = 欠番, 0 = 存在）
    vector<long long> cnt;      // 各値の出現回数
    long long total_cnt = 0;    // 追加されている要素数（重複込み）
    // ========= コンストラクタ =========
    // candidates : 「欠番候補になりうる値の全集合」（mex を求めるなら 0〜N も入れておく）
    MexCompressed(const vector<T>& candidates) {
        comp.add(candidates);
        comp.build();
        int n = comp.size();
        bit = BIT(n, 1);   // 初期状態ではすべて欠番（O(N) で作る）
        cnt.assign(n, 0);
    }
    // ========= 基本操作 =========
    // 値 x を追加（重複あり。候補にない値は無視）
    void add(const T& x) {
        if (!comp.contains(x)) return;
        int id = comp.get(x);
        if (cnt[id] == 0) bit.add(id, -1); // 欠番でなくなる
        cnt[id]++;
        total_cnt++;
    }
    // 値 x を 1 個削除（無ければ何もしない）
    void erase(const T& x) {
        if (!comp.contains(x)) return;
        int id = comp.get(x);
        if (cnt[id] == 0) return;
        cnt[id]--;
        if (cnt[id] == 0) bit.add(id, 1); // 再び欠番になる
        total_cnt--;
    }
    // 値 x の個数
    long long count(const T& x) const { return comp.contains(x) ? cnt[comp.get(x)] : 0; }
    // ========= 欠番判定 =========
    bool is_missing_value(const T& x) const { return count(x) == 0; }   // 候補にない値も欠番扱い
    bool is_missing_id(int id) const { return cnt[id] == 0; }
    // ========= 欠番（ID 版。ID = 候補を小さい順に並べた番号）=========
    // 最小の欠番の ID（存在しなければ -1）
    int mex_id() const {
        if (missing_count() == 0) return -1;
        return bit.kth(1);
    }
    // 最大の欠番の ID（存在しなければ -1）
    int max_missing_id() const {
        long long tot = missing_count();
        if (tot == 0) return -1;
        return bit.kth(tot);
    }
    // 小さい方から k 番目（0 始まり）の欠番の ID（存在しなければ -1）
    int kth_missing_id(long long k) const {
        if (k < 0 || missing_count() <= k) return -1;
        return bit.kth(k + 1);
    }
    // id より大きい最小の欠番の ID（存在しなければ -1）。「l 以上で最小」は next_missing_id(l - 1)
    int next_missing_id(int id) const {
        if (id + 1 >= comp.size()) return -1;
        long long before = (id < 0 ? 0 : bit.sum(id + 1));
        if (before == missing_count()) return -1;
        return bit.kth(before + 1);
    }
    // ========= 欠番（実際の値の版。存在しなければ T(-1)）=========
    T mex_value() const { int id = mex_id(); return id == -1 ? T(-1) : comp.rev(id); }
    T max_missing_value() const { int id = max_missing_id(); return id == -1 ? T(-1) : comp.rev(id); }
    T kth_missing_value(long long k) const { int id = kth_missing_id(k); return id == -1 ? T(-1) : comp.rev(id); }
    // x より大きい最小の欠番の値
    T next_missing_value(const T& x) const { int id = next_missing_id(comp.upper_idx(x) - 1); return id == -1 ? T(-1) : comp.rev(id); }
    // ========= 欠番数 =========
    long long missing_count() const { return bit.sum(comp.size()); }   // 全体の欠番数
    // 値が [l, r) の候補のうち欠番の数
    long long count_missing(const T& l, const T& r) const {
        int L = comp.lower_idx(l);
        int R = comp.lower_idx(r);
        if (L >= R) return 0;
        return bit.sum(R) - bit.sum(L);
    }
    // ========= 補助 =========
    long long size() const { return total_cnt; }   // 追加されている要素数（重複込み）
};

void solv(){
    // ========= MEX 候補の準備 =========
    // 0～9 と 非連続な巨大値も含める
    vector<ll> candidates;
    for (ll i = 0; i < 10; i++) candidates.push_back(i);
    candidates.push_back(1000000000000000000LL);
    candidates.push_back(1000000000000000001LL);
    candidates.push_back(1000000000000000100LL);

    MexCompressed<ll> mex(candidates);

    cout << "=== 初期状態 ===\n";
    cout << "欠番数: " << mex.missing_count() << "\n";
    cout << "最小の欠番(ID): " << mex.mex_id() << "\n";
    cout << "最小の欠番(値): " << mex.mex_value() << "\n";
    cout << "最大の欠番(値): " << mex.max_missing_value() << "\n\n";

    // ========= 追加 =========
    cout << "=== 2, 5, 5 を追加 ===\n";
    mex.add(2);
    mex.add(5);
    mex.add(5);

    cout << "欠番数: " << mex.missing_count() << "\n";
    cout << "3番目の欠番(ID)（0 始まりで 2）: " << mex.kth_missing_id(2) << "\n";
    cout << "3番目の欠番(値)（0 始まりで 2）: " << mex.kth_missing_value(2) << "\n\n";

    // ========= さらに追加 =========
    cout << "=== 0, 1 を追加 ===\n";
    mex.add(0);
    mex.add(1);

    cout << "最小の欠番(値): " << mex.mex_value() << "\n";
    cout << "3以上の最小欠番(値): "
         << mex.next_missing_value(2) << "\n\n";

    // ========= 削除 =========
    cout << "=== 2 を削除 ===\n";
    mex.erase(2);

    cout << "2 は欠番か？: "
         << (mex.is_missing_value(2) ? "はい" : "いいえ") << "\n";
    cout << "最小の欠番(値): " << mex.mex_value() << "\n\n";

    // ========= 区間欠番数 =========
    cout << "=== [2, 7) の欠番数 ===\n";
    cout << mex.count_missing(2, 7) << "\n\n";

    // ========= 非連続な巨大値テスト =========
    cout << "=== 非連続な巨大値テスト ===\n";
    ll big = 1000000000000000000LL;

    cout << "1e18 は欠番か？: "
         << (mex.is_missing_value(big) ? "はい" : "いいえ") << "\n";
    cout << "1e18+1 は欠番か？: "
         << (mex.is_missing_value(big + 1) ? "はい" : "いいえ") << "\n";

    mex.add(big + 1);
    cout << "1e18+1 を追加後の欠番判定: "
         << (mex.is_missing_value(big + 1) ? "はい" : "いいえ") << "\n";

    cout << "1e18 以上で最小の欠番: "
         << mex.next_missing_value(big - 1) << "\n\n";

    // ========= サイズ確認 =========
    cout << "=== 現在の要素数（重複込み） ===\n";
    cout << mex.size() << "\n";
    return;
}

int main(){
    BUFF;
    int T=1;
    //cin>>T;
    while(T--)solv();
    return 0;
}
