struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// BIT (Fenwick Tree) 
//点加算と区間和の取得を高速で
//転倒数を高速で
//0-index,1-indexに注意する
// ★ 型を変えたいときはここ
// ======================= ここからテンプレート本体（コピー範囲） =======================
// BIT（Fenwick Tree）。0-indexed・半開区間 [l, r)（ACL の fenwick_tree・BasicBIT と同じ約束）
//   BIT bit(n) / BIT<int> bit(n) のように使う（T の既定は long long）
//   add(i, x)        : a[i] += x
//   sum(r)           : a[0] + … + a[r-1]（[0, r)）
//   sum(l, r)        : [l, r) の和     sum_closed(l, r) : [l, r] の和
//   count(x) / kth(k): 個数管理として使うとき（値 x の個数を位置 x に持つ）
template<class T = long long>
struct BIT {
    int n;
    vector<T> bit;   // 内部は 1-indexed

    BIT(int _n=0){ init(_n); }

    void init(int _n){
        n = _n;
        bit.assign(n+1, 0);
    }

    // 初期配列 a から O(N) で構築
    void build(const vector<T>& a){
        init(a.size());
        for(int i=0;i<n;i++) bit[i+1] = a[i];
        for(int i=1;i<=n;i++){
            int j = i + (i & -i);
            if(j <= n) bit[j] += bit[i];
        }
    }

    // a[i] += x（0-indexed）
    void add(int i, T x){
        for(i++; i <= n; i += i & -i){
            bit[i] += x;
        }
    }

    // [0, r) の和
    T sum(int r) const {
        T s = 0;
        for(; r > 0; r -= r & -r){
            s += bit[r];
        }
        return s;
    }

    // [l, r) の和（l >= r なら 0）
    T sum(int l, int r) const {
        if(l >= r) return 0;
        return sum(r) - sum(l);
    }

    // [l, r] の和
    T sum_closed(int l, int r) const { return sum(l, r + 1); }

    // ===== 出現数 =====
    // 出現数と k 番目に小さい値を使う場合は、追加: add(val, 1)、削除: add(val, -1)
    T count(int x) const {
        return sum(x, x + 1);
    }

    // ===== k 番目に小さい値 =====
    // 累積和 sum(i + 1) が k 以上になる最小の位置 i（0-indexed）。k が全体の個数より大きいと n
    int kth(T k) const {
        int x = 0;
        int top = 1;
        while((top << 1) <= n) top <<= 1;   // n 以下の最大の 2 冪（n の大きさに制限なし）
        for(int i = top; i > 0; i >>= 1){
            if(x + i <= n && bit[x+i] < k){
                k -= bit[x+i];
                x += i;
            }
        }
        return x;
    }
};

// 転倒数（i < j かつ a[i] > a[j] の組の数）。O(N log N)
ll inversion_count(vector<ll> a){
    int n = a.size();

    // ===== 座標圧縮（0-indexed） =====
    vector<ll> v = a;
    sort(v.begin(), v.end());
    v.erase(unique(v.begin(), v.end()), v.end());

    for(auto &x : a){
        x = lower_bound(v.begin(), v.end(), x) - v.begin();
    }

    // ===== BIT =====
    int m = v.size();
    BIT<ll> bit(m);

    ll inv = 0;
    for(int i=0;i<n;i++){
        inv += bit.sum(a[i] + 1, m);   // これまでに出た a[i] より大きい値の個数
        bit.add(a[i], 1);
    }
    return inv;
}

// -----------------------------------------------------------------------------
//  実戦での使用例（ALPC B）… 標準入力を読むので main からは呼ばない
// -----------------------------------------------------------------------------

int solve_ALPC_B(){
    BUFF;
    int N,Q;
    cin>>N>>Q;
    vector<ll>A(N);
    rep(i,0,N)cin>>A[i];
    BIT bit(N);
    bit.build(A);
    rep(_,0,Q){
        int t,a,b;
        cin>>t>>a>>b;
        //[a,b)の区間和を取得したい（BIT は 0-indexed・半開区間なのでそのまま渡す）
        if(t)cout<<bit.sum(a,b)<<'\n';
        else bit.add(a,b);
    }
    return 0;
}

//  使い方デモ（入力なしで動く）
// -----------------------------------------------------------------------------
void demo(){
    cout << "===== BIT（0-indexed・半開区間の Fenwick Tree）デモ =====\n";
    BIT bit(5);                                  // 要素数 5（位置は 0〜4）
    bit.build({3, 1, 4, 1, 5});                  // 配列から O(N) で構築
    cout << "bit.sum(3) = " << bit.sum(3) << "   // [0, 3) の和 3+1+4 = 8\n";
    cout << "bit.sum(1, 4) = " << bit.sum(1, 4) << "   // [1, 4) の和 1+4+1 = 6\n";
    cout << "bit.sum_closed(1, 4) = " << bit.sum_closed(1, 4) << "   // [1, 4] の和 1+4+1+5 = 11\n";
    bit.add(1, 10);                              // 位置 1 に +10
    cout << "bit.add(1, 10) 後 sum(0, 2) = " << bit.sum(0, 2) << "   // 3+11 = 14\n";
    // 個数管理として使う（値 v の個数を位置 v に持つ）
    BIT<int> cnt(10);
    for (int v : {3, 7, 3, 9}) cnt.add(v, 1);
    cout << "cnt.count(3) = " << cnt.count(3) << ", cnt.kth(3) = " << cnt.kth(3) << "   // 3 は 2 個、小さい方から 3 番目は 7\n";
    cout << "cnt.kth(5) = " << cnt.kth(5) << "   // 全部で 4 個しかないので n（= 10）\n";
    cout << "inversion_count({3, 1, 2}) = " << inversion_count({3, 1, 2}) << "   // 転倒数 (3,1),(3,2) → 2\n";
}

int main(){
    BUFF;
    demo();
    return 0;
}
