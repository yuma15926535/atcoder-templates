struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

template<class T>
struct SortedArrayHelper {
    vector<T> a;

    /*
     * コンストラクタ
     * v: 元の配列
     * sorted_flag: ソート済みかどうか
     */
    SortedArrayHelper(const vector<T>& v, bool sorted_flag) {
        a = v;
        if (!sorted_flag) sort(a.begin(), a.end());
    }

    SortedArrayHelper(){}

    /*
     * x 以下の要素数 (<= x)
     */
    int count_le(const T& x) const {
        return upper_bound(a.begin(), a.end(), x) - a.begin();
    }

    /*
     * x 未満の要素数 (< x)
     */
    int count_lt(const T& x) const {
        return lower_bound(a.begin(), a.end(), x) - a.begin();
    }

    /*
     * x 以上の要素数 (>= x)
     */
    int count_ge(const T& x) const {
        return a.end() - lower_bound(a.begin(), a.end(), x);
    }

    /*
     * x より大きい要素数 (> x)
     */
    int count_gt(const T& x) const {
        return a.end() - upper_bound(a.begin(), a.end(), x);
    }

    // 半開区間 [l, r) に含まれる要素数（l >= r なら 0）
    int count_range(const T& l, const T& r) const {
        if (!(l < r)) return 0;
        return count_lt(r) - count_lt(l);
    }

    // 閉区間 [l, r] に含まれる要素数（l > r なら 0）
    int count_range_closed(const T& l, const T& r) const {
        if (r < l) return 0;
        return count_le(r) - count_lt(l);
    }

    // k 番目 (0-indexed)
    T kth(int k) const {
        return a[k];
    }

    // 存在判定
    bool contains(const T& x) const {
        auto it = lower_bound(a.begin(), a.end(), x);
        return it != a.end() && *it == x;
    }
};

// X以上Y番目の欠番
// 数列Aの範囲が広い、かつX,Yが大きい
struct ForbiddenList{
    SortedArrayHelper<ll> sah;

    /*
     * コンストラクタ
     * v: 元の配列、重複なし
     * sorted_flag: ソート済みかどうか
     */
    ForbiddenList(const vector<ll>& v, bool sorted_flag){
        sah=SortedArrayHelper<ll>(v,sorted_flag);
        auto& a=sah.a;
        a.erase(unique(a.begin(),a.end()),a.end());   // 重複があっても正しく動くように取り除く
    }

    // num 以上の整数のうち、配列に含まれない（欠番の）もので小さい方から k 番目（k は 1 始まり）
    // O(log N)：num 以上の要素 a[j] について「[num, a[j]) の欠番の数 = a[j] - num - (j - s)」は増えていくので、
    //           それが k 以上になる最初の j を二分探索し、答え = num + k - 1 + (j - s)（s = num 以上が始まる位置）
    ll kth_missing_from(ll num,ll k) const {
        const auto& a=sah.a;
        int s=lower_bound(a.begin(),a.end(),num)-a.begin();
        int lo=s,hi=a.size();   // [lo, hi) で「欠番の数 >= k」となる最初の j を探す
        while(lo<hi){
            int mid=(lo+hi)/2;
            if(a[mid]-num-(mid-s)>=k) hi=mid;
            else lo=mid+1;
        }
        return num+k-1+(lo-s);
    }
};
// ======================= ここまでテンプレート本体 =======================


// -----------------------------------------------------------------------------
//  実戦での使用例（ABC440 D）… 標準入力を読むので main からは呼ばない
// -----------------------------------------------------------------------------

void solve_ABC440D(){
    int N,Q;
    cin>>N>>Q;
    vector<ll>A(N);
    cin>>A;
    ForbiddenList fl(A,false);
    rep(i,0,Q){
        ll X,Y;
        cin>>X>>Y;
        cout<<fl.kth_missing_from(X,Y)<<'\n';
    }
    return;
}

void demo(){
    cout << "===== ForbiddenList（巨大な範囲の欠番）デモ =====\n";
    ForbiddenList fl({3, 5, 6, 10}, false);    // 重複なしの配列（ソート済みなら第 2 引数 true）
    cout << "fl.kth_missing_from(3, 1) = " << fl.kth_missing_from(3, 1) << "   // 3 以上で 3,5,6,10 以外: 4,7,8,9,11,... の 1 番目 → 4\n";
    cout << "fl.kth_missing_from(3, 3) = " << fl.kth_missing_from(3, 3) << "   // → 8\n";
    cout << "fl.kth_missing_from(11, 1000000000) = " << fl.kth_missing_from(11, 1000000000) << "   // 巨大でも O(log)\n";
    SortedArrayHelper<ll> sah({5, 1, 3, 3}, false);
    cout << "SortedArrayHelper: count_le(3) = " << sah.count_le(3) << ", count_range(2, 5) = " << sah.count_range(2, 5) << "（[2,5)）, count_range_closed(2, 5) = " << sah.count_range_closed(2, 5) << "   // [2,5]（閉区間）に 3,3,5\n";
}
