    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;

            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;

                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }

/*
============================================================
 Functional Graph COMPLETE (FINAL)
------------------------------------------------------------
 outdegree = 1 graph 専用

取得可能:
 ★ cycle 判定
 ★ cycle entry
 ★ cycle size
 ★ depth（cycleまで距離）
 ★ k回後頂点
 ★ k回総和
 ★ 無限操作後
 ★ loop総和 O(1)

計算量:
 build : O(N)
 query : O(1) or O(logdepth)
============================================================
*/
struct WeightedDoubling {

    using ll = long long;

    int n,LOG;
    vector<vector<int>> nxt;
    vector<vector<ll>> sum;

    WeightedDoubling(int n,ll max_k=(1LL<<60)):n(n){

        LOG=0;
        while((1LL<<LOG)<=max_k) LOG++;

        nxt.assign(LOG,vector<int>(n,-1));
        sum.assign(LOG,vector<ll>(n,0));
    }

    void set_next(int v,int to,ll cost){
        nxt[0][v]=to;
        sum[0][v]=cost;
    }

    void build(){
        for(int k=1;k<LOG;k++){
            for(int v=0;v<n;v++){

                int mid=nxt[k-1][v];
                if(mid==-1) continue;

                nxt[k][v]=nxt[k-1][mid];
                // 2^k 回ぶんの和は巨大になりうるので符号なしで足す（溢れても未定義動作にしない。
                // 実際に使う和が long long に収まれば、途中が溢れていても答えは正しい）
                sum[k][v]=(ll)((unsigned long long)sum[k-1][v]+(unsigned long long)sum[k-1][mid]);
            }
        }
    }

    pair<int,ll> kth_with_sum(int v,ll K) const{

        ll res=0;

        for(int i=0;i<LOG;i++){
            if(!(K&(1LL<<i))) continue;

            res=(ll)((unsigned long long)res+(unsigned long long)sum[i][v]);
            v=nxt[i][v];
        }
        return {v,res};
    }

    int kth(int v,ll k)const{
        return kth_with_sum(v,k).first;
    }

    ll kth_sum(int v,ll k)const{
        return kth_with_sum(v,k).second;
    }
};

struct FunctionalGraph {

    using ll = long long;

    int N;
    vector<int> to;

    vector<int> indeg;
    vector<int> depth;
    vector<int> comp;
    vector<int> pos;
    vector<int> entry;

    vector<vector<int>> cycles;
    vector<int> cycle_sz;

    // ⭐ 完成体追加
    vector<vector<ll>> cycle_prefix;
    vector<ll> cycle_sum;

    FunctionalGraph(int n)
        :N(n),
         to(n),
         indeg(n,0),
         depth(n,-1),
         comp(n,-1),
         pos(n,-1),
         entry(n,-1){}

    //------------------------------------------------
    void set_edge(int u,int v){
        to[u]=v;
        indeg[v]++;
    }

    //------------------------------------------------
    void build(){

        queue<int> q;

        for(int i=0;i<N;i++)
            if(indeg[i]==0)
                q.push(i);

        while(!q.empty()){
            int v=q.front();q.pop();
            if(--indeg[to[v]]==0)
                q.push(to[v]);
        }

        vector<int> used(N);

        for(int i=0;i<N;i++){
            if(indeg[i]>0&&!used[i]){

                vector<int> cyc;
                int v=i;

                while(!used[v]){
                    used[v]=1;
                    cyc.push_back(v);
                    v=to[v];
                }

                int id=cycles.size();

                for(int j=0;j<(int)cyc.size();j++){
                    int x=cyc[j];
                    comp[x]=id;
                    pos[x]=j;
                    depth[x]=0;
                    entry[x]=x;
                }

                cycles.push_back(cyc);
                cycle_sz.push_back(cyc.size());
            }
        }

        vector<vector<int>> rev(N);
        for(int i=0;i<N;i++)
            rev[to[i]].push_back(i);

        queue<int> qq;
        for(int i=0;i<N;i++)
            if(depth[i]==0)
                qq.push(i);

        while(!qq.empty()){
            int v=qq.front();qq.pop();

            for(int u:rev[v]){
                if(depth[u]!=-1) continue;

                depth[u]=depth[v]+1;
                comp[u]=comp[v];
                entry[u]=entry[v];
                qq.push(u);
            }
        }
    }

    //------------------------------------------------
    void build_weighted(
        WeightedDoubling& db,
        const vector<ll>& cost
    ){

        for(int i=0;i<N;i++)
            db.set_next(i,to[i],cost[i]);

        db.build();

        int C=cycles.size();

        cycle_prefix.assign(C,{});
        cycle_sum.assign(C,0);

        for(int cid=0;cid<C;cid++){

            int sz=cycles[cid].size();
            cycle_prefix[cid].assign(sz+1,0);

            for(int i=0;i<sz;i++){
                int v=cycles[cid][i];
                cycle_prefix[cid][i+1]
                    =cycle_prefix[cid][i]+cost[v];
            }

            cycle_sum[cid]=cycle_prefix[cid][sz];
        }
    }

    //------------------------------------------------
    bool on_cycle(int v)const{
        return depth[v]==0;
    }

    int cycle_entry(int v)const{
        return entry[v];
    }

    int cycle_size(int v)const{
        return cycle_sz[comp[v]];
    }

    //------------------------------------------------
    int after_infinite(
        int v,
        const WeightedDoubling& db
    )const{
        return db.kth(v,depth[v]);
    }

    //------------------------------------------------
    // k 回動いた結果を「行き先・木の部分と端数の和 rest・周回数 loops・1 周の和 loop_sum」に分ける
    //   総和 = rest + loops * loop_sum（rest は高々 2N 回ぶんの和なので long long に収まる前提）
    template<class WDB>
    tuple<int,long long,long long,long long>
    move_split(int v,long long k,const WDB& db) const{
        long long take=min<long long>(k,depth[v]);   // ---- 木部分 ----
        auto [nv,s1]=db.kth_with_sum(v,take);
        v=nv;
        k-=take;
        if(k==0) return {v,s1,0,0};
        int cid=comp[v];                              // ---- cycle ----
        int sz=cycle_sz[cid];
        auto [_,loop_sum]=db.kth_with_sum(v,sz);
        long long loops=k/sz;
        auto [fv,s2]=db.kth_with_sum(v,k%sz);
        return {fv,s1+s2,loops,loop_sum};
    }

    // k 回動いた後の頂点と、通った辺の重みの総和（総和が long long に収まる前提）
    template<class WDB>
    pair<int,long long>
    move_weighted(int v,long long k,const WDB& db) const{
        auto [fv,rest,loops,loop_sum]=move_split(v,k,db);
        return {fv,rest+loops*loop_sum};
    }

    // 総和を __int128 で返す版（k ≤ 10^18・重み ≤ 10^18 程度でも溢れない）
    template<class WDB>
    pair<int,__int128>
    move_weighted_i128(int v,long long k,const WDB& db) const{
        auto [fv,rest,loops,loop_sum]=move_split(v,k,db);
        return {fv,(__int128)rest+(__int128)loops*loop_sum};
    }

    // 総和を mod で返す版（0 ≤ 結果 < mod。重みが負でもよい）
    template<class WDB>
    pair<int,long long>
    move_weighted_mod(int v,long long k,const WDB& db,long long mod) const{
        auto [fv,rest,loops,loop_sum]=move_split(v,k,db);
        auto md=[&](__int128 x){ x%=mod; if(x<0) x+=mod; return (long long)x; };
        return {fv,md((__int128)md(rest)+(__int128)md(loops)*md(loop_sum))};
    }
};

void solv(){
    cout<<"=============================\n";
    cout<<" Functional Graph 最終動作確認\n";
    cout<<"=============================\n";

    /*
        構造:

        0 → 1 → 2 → 3 → 4
                  ↑     ↓
                  ← ← ←

        cycle = {2,3,4}
    */

    int N=5;

    FunctionalGraph fg(N);

    fg.set_edge(0,1);
    fg.set_edge(1,2);
    fg.set_edge(2,3);
    fg.set_edge(3,4);
    fg.set_edge(4,2);

    vector<long long> cost={
        10,20,30,40,50
    };

    //------------------------------------------------
    cout<<"\n▼ build\n";

    fg.build();

    WeightedDoubling db(N,1000);
    fg.build_weighted(db,cost);

    //------------------------------------------------
    cout<<"\n▼ 基本情報\n";

    for(int v=0;v<N;v++){

        cout<<"頂点 "<<v<<"\n";
        cout<<"  次頂点: "<<fg.to[v]<<"\n";
        cout<<"  cycleまで距離: "<<fg.depth[v]<<"\n";
        cout<<"  cycle入口: "<<fg.cycle_entry(v)<<"\n";
        cout<<"  cycleサイズ: "<<fg.cycle_size(v)<<"\n";
        cout<<"  cycle上か: "
            <<(fg.on_cycle(v)?"はい":"いいえ")<<"\n\n";
    }

    //------------------------------------------------
    cout<<"-----------------------------\n";
    cout<<"▼ 1回遷移確認\n";

    for(int v=0;v<N;v++){
        auto [nv,sum]=db.kth_with_sum(v,1);

        cout<<"頂点 "<<v
            <<" → "<<nv
            <<" / 加算="<<sum<<"\n";
    }

    //------------------------------------------------
    cout<<"-----------------------------\n";
    cout<<"▼ k回遷移 + 総和 (start=0)\n";

    for(int k=0;k<=12;k++){

        auto [v,sum]
            =fg.move_weighted(0,k,db);

        cout<<k<<" 回後 → "
            <<"頂点 "<<v
            <<" / 総和="<<sum<<"\n";
    }

    //------------------------------------------------
    cout<<"-----------------------------\n";
    cout<<"▼ cycle1周確認\n";

    int entry=fg.cycle_entry(0);
    int sz=fg.cycle_size(entry);

    auto [_,loop_sum]=db.kth_with_sum(entry,sz);

    cout<<"cycleサイズ = "<<sz<<"\n";
    cout<<"1周コスト総和 = "<<loop_sum<<"\n";

    //------------------------------------------------
    cout<<"-----------------------------\n";
    cout<<"▼ 無限操作後\n";

    for(int v=0;v<N;v++){
        cout<<"頂点 "<<v
            <<" → "
            <<fg.after_infinite(v,db)
            <<"\n";
    }

    //------------------------------------------------
    cout<<"-----------------------------\n";
    cout<<"▼ 超巨大Kテスト (1e17)\n";

    // 1e18 回だと総和（約 4×10^19）が long long を超えるので 1e17 回にしている
    long long BIG=1e17;

    auto [v,sum]
        =fg.move_weighted(0,BIG,db);

    cout<<"1e17 回後\n";
    cout<<"頂点 = "<<v<<"\n";
    cout<<"総和 = "<<sum<<"\n";

    //------------------------------------------------
    cout<<"-----------------------------\n";
    cout<<"▼ 総和が long long を超える場合 (1e18 回)\n";
    {
        auto [v2,s128]=fg.move_weighted_i128(0,(long long)1e18,db);   // __int128 で受け取る
        string str; __int128 x=s128; while(x>0){ str+=char('0'+(int)(x%10)); x/=10; } reverse(str.begin(),str.end());
        cout<<"move_weighted_i128(0, 1e18): 頂点 = "<<v2<<", 総和 = "<<str<<"\n";
        auto [v3,smod]=fg.move_weighted_mod(0,(long long)1e18,db,998244353);
        cout<<"move_weighted_mod(0, 1e18, 998244353): 頂点 = "<<v3<<", 総和 mod = "<<smod<<"\n";
        cout<<"（確認）__int128 の総和 mod 998244353 = "<<(long long)(s128%998244353)<<"\n";
    }

    cout<<"=============================\n";
    return;
}

int main(){
    //BUFF;
    int T=1;
    //cin>>T;
    while(T--)solv();
    return 0;
