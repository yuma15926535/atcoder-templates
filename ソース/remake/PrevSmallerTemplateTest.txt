struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

//  Next / Prev Smaller & Greater (Mono Stack Template)
//  （作り直し版：元ファイル PrevSmallerTemplateTest.txt）
// -----------------------------------------------------------------------------
//  説明
//    - 左右で最も近い引数以上、以下、より大きい、より小さい要素を持つindexを返却
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
// ======================= ここからテンプレート本体（コピー範囲） =======================
/*
==================================================
 Next / Prev Smaller & Greater (Mono Stack Template)
==================================================

与えられた数列 a に対して、単調スタックを用いて
以下の 8 種類の「隣接要素」をすべて O(N) で前処理する。

【smaller 系】
- prev_smaller_strict(i)   : 左で最も近い a[j] <  a[i]
- prev_smaller_or_equal(i) : 左で最も近い a[j] <= a[i]
- next_smaller_strict(i)   : 右で最も近い a[j] <  a[i]
- next_smaller_or_equal(i) : 右で最も近い a[j] <= a[i]

【greater 系】
- prev_greater_strict(i)   : 左で最も近い a[j] >  a[i]
- prev_greater_or_equal(i) : 左で最も近い a[j] >= a[i]
- next_greater_strict(i)   : 右で最も近い a[j] >  a[i]
- next_greater_or_equal(i) : 右で最も近い a[j] >= a[i]

存在しない場合は -1

--------------------------------------------------
 典型用途:
 - ヒストグラム最大長方形
 - 各要素が最小 / 最大となる支配区間
 - 区間DPの遷移制限
 - Cartesian Tree 構築
==================================================
*/
template <class T>
struct MonoNeighbor {
private:
    int n;
    vector<T> a;

    vector<int> pss, pso, nss, nso;
    vector<int> pgs, pgo, ngs, ngo;

public:
    MonoNeighbor(const vector<T>& a_) : a(a_) {
        n = (int)a.size();

        pss.assign(n, -1);
        pso.assign(n, -1);
        nss.assign(n, -1);
        nso.assign(n, -1);
        pgs.assign(n, -1);
        pgo.assign(n, -1);
        ngs.assign(n, -1);
        ngo.assign(n, -1);

        build_smaller();
        build_greater();
    }

    // ===== smaller 系 API =====

    // 左で最も近い a[j] < a[i]
    int prev_smaller_strict(int i) const {
        return pss[i];
    }

    // 左で最も近い a[j] <= a[i]
    int prev_smaller_or_equal(int i) const {
        return pso[i];
    }

    // 右で最も近い a[j] < a[i]
    int next_smaller_strict(int i) const {
        return nss[i];
    }

    // 右で最も近い a[j] <= a[i]
    int next_smaller_or_equal(int i) const {
        return nso[i];
    }

    // ===== greater 系 API =====

    // 左で最も近い a[j] > a[i]
    int prev_greater_strict(int i) const {
        return pgs[i];
    }

    // 左で最も近い a[j] >= a[i]
    int prev_greater_or_equal(int i) const {
        return pgo[i];
    }

    // 右で最も近い a[j] > a[i]
    int next_greater_strict(int i) const {
        return ngs[i];
    }

    // 右で最も近い a[j] >= a[i]
    int next_greater_or_equal(int i) const {
        return ngo[i];
    }

private:
    void build_smaller() {
        // prev smaller strict (<)
        {
            stack<int> st;
            for (int i = 0; i < n; i++) {
                while (!st.empty() && a[st.top()] >= a[i]) st.pop();
                pss[i] = st.empty() ? -1 : st.top();
                st.push(i);
            }
        }

        {
            stack<int> st;
            for (int i = 0; i < n; i++) {
                while (!st.empty() && a[st.top()] > a[i]) st.pop();
                pso[i] = st.empty() ? -1 : st.top();
                st.push(i);
            }
        }

        {
            stack<int> st;
            for (int i = n - 1; i >= 0; i--) {
                while (!st.empty() && a[st.top()] >= a[i]) st.pop();
                nss[i] = st.empty() ? -1 : st.top();
                st.push(i);
            }
        }

        {
            stack<int> st;
            for (int i = n - 1; i >= 0; i--) {
                while (!st.empty() && a[st.top()] > a[i]) st.pop();
                nso[i] = st.empty() ? -1 : st.top();
                st.push(i);
            }
        }
    }

    void build_greater() {
        // prev greater strict (>)
        {
            stack<int> st;
            for (int i = 0; i < n; i++) {
                while (!st.empty() && a[st.top()] <= a[i]) st.pop();
                pgs[i] = st.empty() ? -1 : st.top();
                st.push(i);
            }
        }

        {
            stack<int> st;
            for (int i = 0; i < n; i++) {
                while (!st.empty() && a[st.top()] < a[i]) st.pop();
                pgo[i] = st.empty() ? -1 : st.top();
                st.push(i);
            }
        }

        {
            stack<int> st;
            for (int i = n - 1; i >= 0; i--) {
                while (!st.empty() && a[st.top()] <= a[i]) st.pop();
                ngs[i] = st.empty() ? -1 : st.top();
                st.push(i);
            }
        }

        {
            stack<int> st;
            for (int i = n - 1; i >= 0; i--) {
                while (!st.empty() && a[st.top()] < a[i]) st.pop();
                ngo[i] = st.empty() ? -1 : st.top();
                st.push(i);
            }
        }
    }
};

// =============================================================================
//  愚直解とのランダム比較テスト（作り直しで追加）
// =============================================================================
void random_test_bruteforce(){ mt19937 rng(2); int bad=0;
 for(int it=0;it<3000;it++){ int n=rng()%8+1; vector<int>a(n); for(auto&x:a)x=rng()%4; MonoNeighbor<int> mn(a);
  for(int i=0;i<n;i++){
   auto pv=[&](auto cond){ for(int j=i-1;j>=0;j--) if(cond(a[j],a[i])) return j; return -1; };
   auto nx=[&](auto cond){ for(int j=i+1;j<n;j++) if(cond(a[j],a[i])) return j; return -1; };
   if(mn.prev_smaller_strict(i)!=pv([](int x,int y){return x<y;}))bad++;
   if(mn.prev_smaller_or_equal(i)!=pv([](int x,int y){return x<=y;}))bad++;
   if(mn.next_smaller_strict(i)!=nx([](int x,int y){return x<y;}))bad++;
   if(mn.next_smaller_or_equal(i)!=nx([](int x,int y){return x<=y;}))bad++;
   if(mn.prev_greater_strict(i)!=pv([](int x,int y){return x>y;}))bad++;
   if(mn.prev_greater_or_equal(i)!=pv([](int x,int y){return x>=y;}))bad++;
   if(mn.next_greater_strict(i)!=nx([](int x,int y){return x>y;}))bad++;
   if(mn.next_greater_or_equal(i)!=nx([](int x,int y){return x>=y;}))bad++;
  } } 
    cout << "random_test_bruteforce（愚直解とのランダム比較）: " << (((bad)==0) ? "OK" : "NG") << "\n";
    assert((bad)==0);
}
