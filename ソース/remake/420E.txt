struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// -----------------------------------------------------------------------------
//  説明
//    - 各連結成分(root)に値を持たせ、unite時にmerge関数で統合
//    - 頂点更新がある場合はauto &a=dsu.root_value(u);
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
// Monoid DSU (Union-Find with component value)
// 各連結成分(root)に値を持たせ、unite時にmerge関数で統合する。
// mergeは「モノイド演算」を想定（例: +, max, min, xor など）
//
// 使い方:
//  auto merge = [](ll a, ll b){ return a + b; };
//  vector<ll> init(N);
//  MonoidDSU<ll, decltype(merge)> dsu(N, merge, init);
//
//  dsu.unite(a,b);
//  dsu.get(v);      // vの属する連結成分の値
//
// 頂点更新がある場合は
//  int r = dsu.find(v);
//  dsu.val[r] += delta;   // rootを直接更新
// ======================= ここからテンプレート本体（コピー範囲） =======================
template<class T, class F>
struct MonoidDSU {

    int n;

    // parent[x] = 親
    vector<int> parent;

    // sz[root] = 連結成分サイズ
    vector<int> sz;

    // val[root] = 連結成分の値
    vector<T> val;

    // merge関数
    F merge;

    // コンストラクタ
    MonoidDSU(int n, F merge, const vector<T>& init)
        : n(n), parent(n), sz(n,1), val(init), merge(merge)
    {
        for(int i=0;i<n;i++) parent[i]=i;
    }

    // rootを取得（経路圧縮）
    int find(int x){
        if(parent[x]==x) return x;
        return parent[x] = find(parent[x]);
    }

    // 同じ連結成分か
    bool same(int a,int b){
        return find(a)==find(b);
    }

    // 連結成分サイズ
    int size(int v){
        return sz[find(v)];
    }

    // unite（union by size）
    bool unite(int a,int b){

        a = find(a);
        b = find(b);

        if(a==b) return false;

        if(sz[a] < sz[b]) swap(a,b);

        parent[b] = a;
        sz[a] += sz[b];

        // モノイド演算で値を統合（merge(大きい方の値, 小さい方の値) の順。非可換な演算なら注意）
        val[a] = merge(val[a], val[b]);

        return true;
    }

    // vの属する連結成分の値を取得
    T get(int v){
        return val[find(v)];
    }

    // rootの値を直接参照（更新用途）
    T& root_value(int v){
        return val[find(v)];
    }
};
// ======================= ここまでテンプレート本体 =======================


// -----------------------------------------------------------------------------
//  実戦での使用例（ABC420 E）… 標準入力を読むので main からは呼ばない
// -----------------------------------------------------------------------------

void solve_ABC420E(){
    int N,Q;
    cin>>N>>Q;
    auto merge=[](ll a,ll b){return a+b;};
    vector<ll>init(N,0);
    MonoidDSU<ll,decltype(merge)> dsu(N,merge,init);
    vector<char>color(N,0);
    rep(i,0,Q){
        int q,u;
        cin>>q>>u;
        u--;
        if(q==1){
            int v;
            cin>>v;
            v--;
            dsu.unite(u,v);
        }else if(q==2){
            color[u]^=1;
            auto &a=dsu.root_value(u);
            if(color[u])a++;
            else a--;
        }else{
            if(dsu.get(u))YN;
        }
    }
    return;
}

// -----------------------------------------------------------------------------
//  使い方デモ（入力なしで動く）
// -----------------------------------------------------------------------------
void demo(){
    cout << "===== MonoidDSU デモ（連結成分ごとに値を持つ Union-Find）=====\n";
    auto merge = [](ll a, ll b){ return a + b; };           // 成分の値の合わせ方（ここでは和）
    vector<ll> init = {5, 1, 3, 2};                          // 各頂点の初期値
    MonoidDSU<ll, decltype(merge)> dsu(4, merge, init);
    dsu.unite(0, 1);
    cout << "unite(0,1) 後: dsu.get(1) = " << dsu.get(1) << "   // 成分 {0,1} の値 5+1 = 6\n";
    cout << "dsu.same(0, 2) = " << dsu.same(0, 2) << ", dsu.size(0) = " << dsu.size(0) << "\n";
    dsu.root_value(2) += 10;                                 // 成分の値を直接書き換える（参照が返る）
    dsu.unite(2, 3);
    cout << "root_value(2) += 10 → unite(2,3) 後: dsu.get(3) = " << dsu.get(3) << "   // 13 + 2 = 15\n";
    cout << "dsu.unite(0, 1) = " << dsu.unite(0, 1) << "   // すでに同じ成分なら false\n";
}
