struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// 区間最大を保持するセグメントツリー（値: pair<length, index>）
// ======================= ここからテンプレート本体（コピー範囲） =======================
struct SegTree {
    int n;
    vector<pair<ll,int>> seg; // (length, index)

    void init(int sz){
        n = 1;
        while(n < sz) n <<= 1;
        seg.assign(2*n, {0, -1});
    }

    // [l, r) の最大を返す
    pair<ll,int> prod(int l, int r){
        pair<ll,int> res = {0, -1};
        l += n; r += n;
        while(l < r){
            if(l & 1) res = max(res, seg[l++]);
            if(r & 1) res = max(res, seg[--r]);
            l >>= 1; r >>= 1;
        }
        return res;
    }

    // pos の値を val によって更新（より大きければ置き換え）
    void update(int pos, pair<ll,int> val){
        pos += n;
        if(val.first <= seg[pos].first) return; // 既存の方が良ければ何もしない
        seg[pos] = val;
        pos >>= 1;
        while(pos){
            seg[pos] = max(seg[pos<<1], seg[(pos<<1)|1]);
            pos >>= 1;
        }
    }
};

//最長増加部分列(LIS)(seg木)を返却
vector<ll> lis_values(const vector<ll>& a){
    int m = a.size();
    if(m == 0) return {};

    // 座標圧縮（値の大小関係のみ使う）
    vector<ll> xs = a;
    sort(xs.begin(), xs.end());
    xs.erase(unique(xs.begin(), xs.end()), xs.end());
    int K = xs.size();

    SegTree st;
    st.init(K);

    vector<int> parent(m, -1);
    vector<ll> length(m, 0);

    int best_end = -1;
    ll best_len = 0;

    for(int i = 0; i < m; ++i){
        int rank = lower_bound(xs.begin(), xs.end(), a[i]) - xs.begin();
        // strictly increasing -> 値 < a[i] の範囲を検索 => [0, rank)
        pair<ll,int> q = {0, -1};
        if(rank > 0) q = st.prod(0, rank);
        length[i] = q.first + 1;
        parent[i] = q.second;

        // rank の位置を length で更新（同じ値の位置は長さの大きい方を保持）
        st.update(rank, {length[i], i});

        if(length[i] > best_len){
            best_len = length[i];
            best_end = i;
        }
    }

    // 復元
    vector<ll> lis;
    int cur = best_end;
    while(cur != -1){
        lis.push_back(a[cur]);
        cur = parent[cur];
    }
    reverse(lis.begin(), lis.end());
    return lis;
}

//  使い方デモ（入力なしで動く）

void demo(){
    cout << "===== lis_values（セグメント木版）デモ =====\n";
    vector<ll> a = {3, 1, 4, 1, 5, 9, 2, 6};
    auto v = lis_values(a);
    cout << "lis_values({3,1,4,1,5,9,2,6}) = "; for (ll x : v) cout << x << " "; cout << "  // 長さ " << v.size() << "（狭義増加）\n";
    // SegTree 単体：位置ごとに (長さ, 添字) の最大を持つ。update は「大きければ置き換え」
    SegTree st; st.init(5);
    st.update(1, {3, 10}); st.update(3, {2, 11}); st.update(1, {1, 12});   // 3 回目は小さいので無視
    auto q = st.prod(0, 4);
    cout << "st.prod(0, 4) = (" << q.first << ", " << q.second << ")   // [0,4) で長さ最大は (3, 10)\n";
}
