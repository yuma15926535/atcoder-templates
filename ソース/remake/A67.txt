struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

//  説明
//    - コスト付きグラフの辺をできるだけコストが少なくなり連続になるように辺を削除
//    - 辺と重みを渡し最小全域木の重みを求める
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）

//https://atcoder.jp/contests/tessoku-book/tasks/tessoku_book_bo
// ======================= ここからテンプレート本体（コピー範囲） =======================
struct UnionFind
{
    // P[x] < 0 なら x は根で、-P[x] がその集合の大きさ
    vector<int> P;
    UnionFind() {}
    UnionFind(int n){
        P.assign(n,-1);
    }

    // 根を返す（非再帰・経路圧縮）
    int find(int x){
        int r=x;
        while(P[r]>=0) r=P[r];
        while(P[x]>=0){ int nx=P[x]; P[x]=r; x=nx; }
        return r;
    }

    // 併合（大きい方に小さい方をつなぐ）。別の集合だったら true
    bool unite(int x,int y){
        x=find(x);
        y=find(y);
        if(x==y) return false;
        if(P[x]>P[y]) swap(x,y);
        P[x]+=P[y];
        P[y]=x;
        return true;
    }

    bool same(int x,int y){ return find(x)==find(y); }
    int size(int x){ return -P[find(x)]; }
};
struct Edge {
    long long u;
    long long v;
    long long cost;
};
bool comp_e(const Edge &e1, const Edge &e2) { return e1.cost < e2.cost; } // 辺を直接比較するための関数

/* Kruskal :クラスカル法で minimum spanning tree (最小全域木)を求める構造体
    入力: 辺のvector, 頂点数V
    sum       : 選んだ辺の重みの総和（非連結なら最小全域森の総和）
    mst       : 選んだ辺の一覧（重みの小さい順）
    connected : 全域木ができたか（グラフが連結か）
    components: 連結成分の数
    計算量: O(|E|log|E|)
*/
struct Kruskal {
    UnionFind uft;
    long long sum;  // 最小全域木の重みの総和
    vector<Edge> edges;
    vector<Edge> mst;  // 選んだ辺
    bool connected;
    int components;
    int V;
    Kruskal(const vector<Edge> &edges_, int V_) : edges(edges_), V(V_) { init(); }
    void init() {
        sort(edges.begin(), edges.end(), comp_e); // 辺の重みでソート
        uft = UnionFind(V);
        sum = 0;
        mst.clear();
        for (auto e : edges) {
            if (uft.unite(e.u, e.v)) { // 閉路にならなければ加える
                sum += e.cost;
                mst.push_back(e);
            }
        }
        components = V - (int)mst.size();
        connected = (components <= 1);
    }
};
// ======================= ここまでテンプレート本体 =======================


// -----------------------------------------------------------------------------
//  実戦での使用例（鉄則本 A67）… 標準入力を読むので main からは呼ばない
// -----------------------------------------------------------------------------

int solve_TessokuA67(){
    int N,M;
    cin>>N>>M;
    vector<Edge> edges(M);
    rep(i,0,M){
        ll a,b,c;
        cin>>a>>b>>c;
        a--;b--;
        Edge e = {a, b, c};
        edges[i] = e;
    }
    Kruskal krs(edges, N);
    cout << krs.sum << endl;
    return 0;
}

//  使い方デモ（入力なしで動く）

void demo(){
    cout << "===== Kruskal（最小全域木）デモ =====\n";
    int V = 4;
    vector<Edge> edges = {{0, 1, 3}, {1, 2, 1}, {0, 2, 2}, {2, 3, 5}, {1, 3, 4}};   // {u, v, cost}（0-indexed）
    Kruskal krs(edges, V);
    cout << "krs.sum = " << krs.sum << "   // 選ばれる辺 (1,2)=1, (0,2)=2, (1,3)=4 → 7\n";
    cout << "krs.mst（選んだ辺 {u, v, cost}）: ";
    for (auto& e : krs.mst) cout << "{" << e.u << ", " << e.v << ", " << e.cost << "} ";
    cout << "\n";
    cout << "krs.connected = " << krs.connected << "（全域木ができた）, krs.components = " << krs.components << "\n";
    // 非連結なグラフ：全域木はできない（最小全域森になる）
    Kruskal krs2({{0, 1, 5}, {2, 3, 7}}, 5);
    cout << "非連結: krs2.sum = " << krs2.sum << ", connected = " << krs2.connected << ", components = " << krs2.components << "   // {0,1} {2,3} {4} の 3 成分\n";
    UnionFind uf(3);
    uf.unite(0, 2);
    cout << "UnionFind: uf.find(0) == uf.find(2) → " << (uf.find(0) == uf.find(2)) << ", uf.find(1) == uf.find(2) → " << (uf.find(1) == uf.find(2)) << "\n";
    cout << "uf.same(0, 2) = " << uf.same(0, 2) << ", uf.size(0) = " << uf.size(0) << ", uf.unite(2, 0) = " << uf.unite(2, 0) << "（既に同じ集合なら false）\n";
}
