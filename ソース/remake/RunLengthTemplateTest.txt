struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

    using V = std::decay_t<decltype(*begin(v))>;
    if constexpr(is_iter_v<V>){
        bool first_row=true;
        os<<"\n";
        for(auto&&row:v){
            if(!first_row) os<<"\n";
            first_row=false;
            bool f=0;
            for(auto&&x:row){
                if(f) os<<" ";
                os<<x;
                f=1;
            }
        }
    }
    else{
        os<<"[";
        bool f=0;
        for(auto&&x:v){
            if(f) os<<", ";
            os<<x;
            f=1;
        }
        os<<"]";
    }
    return os;
}

    }

/*
==================================================
 ランレングス圧縮（数列版）
==================================================

【機能】
連続して同じ値が現れる区間を (値, 個数) のペアに圧縮する。

例:
a = [1,1,1,2,2,3,1,1]
→ [(1,3),(2,2),(3,1),(1,2)]

【計算量】
O(N)

【注意】
- 値の大小比較は不要（== のみ）
- 空配列にも対応
*/
template<class T>
vector<pair<T,long long>> run_length(const vector<T>& a) {
    vector<pair<T,long long>> res;
    for (const auto& x : a) {
        if (res.empty() || res.back().first != x) {
            res.emplace_back(x, 1);
        } else {
            res.back().second++;
        }
    }
    return res;
}
/*
==================================================
 ランレングス圧縮（string版）
==================================================

【機能】
文字列を連続する同一文字ごとに (文字, 個数) に圧縮する。

例:
s = "aaabbccccd"
→ [('a',3),('b',2),('c',4),('d',1)]

【計算量】
O(|s|)

【注意】
- 空文字列にも対応
*/
vector<pair<char,int>> run_length(const string& s) {
    vector<pair<char,int>> res;
    for (char c : s) {
        if (res.empty() || res.back().first != c) {
            res.emplace_back(c, 1);
        } else {
            res.back().second++;
        }
    }
    return res;
}
// ======================= ここまでテンプレート本体 =======================

void solv(){
    // ==============================
    // ① 数列版 ランレングス圧縮
    // ==============================
    vector<int> a = {1,1,1,2,2,3,1,1};

    auto rle_a = run_length(a);

    cout << "=== 数列 RLE ===\n";
    cout << "input: ";
    for (int x : a) cout << x << " ";
    cout << "\ncompressed:\n";
    for (auto [v, cnt] : rle_a) {
        cout << "(" << v << ", " << cnt << ") ";
    }
    cout << "\n\n";

    // ==============================
    // ② string版 ランレングス圧縮
    // ==============================
    string s = "aaabbccccd";

    auto rle_s = run_length(s);

    cout << "=== string RLE ===\n";
    cout << "input: " << s << "\n";
    cout << "compressed:\n";
    for (auto [c, cnt] : rle_s) {
        cout << "('" << c << "', " << cnt << ") ";
    }
    cout << "\n";
    return;
}
