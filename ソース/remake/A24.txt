struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

//最長増加部分列(LIS)を返却
// ======================= ここからテンプレート本体（コピー範囲） =======================
// 最長増加部分列（狭義単調増加）を 1 つ返す（値の列）。長さは lis_values(a).size()  O(N log N)
//   広義（同じ値も可）にするなら二分探索の dp[mid] < a[i] を dp[mid] <= a[i] にする
vector<int> lis_values(const vector<int>& a){
    int n = a.size();
    if(n==0) return {};

    vector<int> dp;                 // 値を保持する最小末尾列
    vector<int> idx;                // dp の各要素が指す a のインデックス
    vector<int> parent(n, -1);      // 復元用の親ポインタ

    rep(i,0,n){
    int left = -1,right = (int)dp.size(); // right は「候補の最小インデックス」
    while(right - left > 1){
        int mid = (left + right) / 2;
        if(dp[mid] < a[i]) left = mid; // dp[mid] < target のとき左を動かす
        else right = mid;              // dp[mid] >= target のとき右を動かす
    }
    int pos = right;

    if(pos == (int)dp.size()){
        dp.push_back(a[i]);
        idx.push_back(i);
    } else {
        dp[pos] = a[i];
        idx[pos] = i;
    }
        if(pos > 0) parent[i] = idx[pos-1];
    }

    // 最後の要素のインデックスは idx.back()
    int k = dp.size();
    vector<int> lis(k);
    int cur = idx[k-1];
    for(int i=k-1;i>=0;i--){
        lis[i] = a[cur];
        cur = parent[cur];
    }
    return lis;
}
// ======================= ここまでテンプレート本体 =======================

//  実戦での使用例（鉄則本 A24）… 標準入力を読むので main からは呼ばない

int solve_TessokuA24(){
    int N;
    cin>>N;
    vector<int>A(N);
    rep(i,0,N)cin>>A[i];
    auto v=lis_values(A);
    cout<<v.size()<<'\n';
    return 0;
}
