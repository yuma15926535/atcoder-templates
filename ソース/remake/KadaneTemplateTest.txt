struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// Kadane 最大・最小部分列和（空あり / 空なし）
// 値 + 区間 [l, r) をすべて保持
// ======================= ここからテンプレート本体（コピー範囲） =======================
struct SubarrayAll {
    // ===== 最大（空あり）=====
    long long max_allow_sum;
    int max_allow_l, max_allow_r;

    // ===== 最大（空なし）=====
    long long max_force_sum;
    int max_force_l, max_force_r;

    // ===== 最小（空あり）=====
    long long min_allow_sum;
    int min_allow_l, min_allow_r;

    // ===== 最小（空なし）=====
    long long min_force_sum;
    int min_force_l, min_force_r;

    SubarrayAll(const vector<long long>& a) {
        int n = a.size();

        // ---------- 空の配列 ----------
        // 空ありは和 0・区間 [0, 0)。空なしは部分列が無いので max_force_sum = LLONG_MIN, min_force_sum = LLONG_MAX（区間は [0, 0)）
        if (n == 0) {
            max_allow_sum = min_allow_sum = 0;
            max_force_sum = LLONG_MIN;
            min_force_sum = LLONG_MAX;
            max_allow_l = max_allow_r = max_force_l = max_force_r = 0;
            min_allow_l = min_allow_r = min_force_l = min_force_r = 0;
            return;
        }

        // ---------- 最大（空あり）----------
        {
            long long cur = 0;
            int cur_l = 0;
            max_allow_sum = 0;
            max_allow_l = max_allow_r = 0;

            for (int i = 0; i < n; i++) {
                cur += a[i];
                if (cur > max_allow_sum) {
                    max_allow_sum = cur;
                    max_allow_l = cur_l;
                    max_allow_r = i + 1;
                }
                if (cur < 0) {
                    cur = 0;
                    cur_l = i + 1;
                }
            }
        }

        // ---------- 最大（空なし）----------
        {
            long long cur = a[0];
            int cur_l = 0;
            max_force_sum = a[0];
            max_force_l = 0;
            max_force_r = 1;

            for (int i = 1; i < n; i++) {
                if (cur < 0) {
                    cur = a[i];
                    cur_l = i;
                } else {
                    cur += a[i];
                }
                if (cur > max_force_sum) {
                    max_force_sum = cur;
                    max_force_l = cur_l;
                    max_force_r = i + 1;
                }
            }
        }

        // ---------- 最小（空あり）----------
        {
            long long cur = 0;
            int cur_l = 0;
            min_allow_sum = 0;
            min_allow_l = min_allow_r = 0;

            for (int i = 0; i < n; i++) {
                cur += a[i];
                if (cur < min_allow_sum) {
                    min_allow_sum = cur;
                    min_allow_l = cur_l;
                    min_allow_r = i + 1;
                }
                if (cur > 0) {
                    cur = 0;
                    cur_l = i + 1;
                }
            }
        }

        // ---------- 最小（空なし）----------
        {
            long long cur = a[0];
            int cur_l = 0;
            min_force_sum = a[0];
            min_force_l = 0;
            min_force_r = 1;

            for (int i = 1; i < n; i++) {
                if (cur > 0) {
                    cur = a[i];
                    cur_l = i;
                } else {
                    cur += a[i];
                }
                if (cur < min_force_sum) {
                    min_force_sum = cur;
                    min_force_l = cur_l;
                    min_force_r = i + 1;
                }
            }
        }
    }
};

// ---- ランダムテスト用: 円環も扱える区間和 [l, r)（l > r なら折り返し）----
ll csum_test(const vector<ll>&a,int l,int r){ int n=a.size(); ll s=0; if(l<r){for(int i=l;i<r;i++)s+=a[i%n];} else if(l>r){for(int i=l;i<n;i++)s+=a[i]; for(int i=0;i<r;i++)s+=a[i];} return s; }

void random_test_kadane(){ mt19937 rng(7); int bad[8]={0};
 for(int it=0;it<20000;it++){ int n=rng()%6+1; vector<ll>a(n); for(auto&x:a)x=(ll)(rng()%9)-4;
  ll mxa=0,mxf=LLONG_MIN,mna=0,mnf=LLONG_MAX;
  for(int l=0;l<n;l++){ll s=0; for(int r=l;r<n;r++){s+=a[r]; mxa=max(mxa,s); mxf=max(mxf,s); mna=min(mna,s); mnf=min(mnf,s);} }
  SubarrayAll S(a);
  if(S.max_allow_sum!=mxa||csum_test(a,S.max_allow_l,S.max_allow_r)!=mxa)bad[0]++;
  if(S.max_force_sum!=mxf||csum_test(a,S.max_force_l,S.max_force_r)!=mxf||S.max_force_l>=S.max_force_r)bad[1]++;
  if(S.min_allow_sum!=mna||csum_test(a,S.min_allow_l,S.min_allow_r)!=mna)bad[2]++;
  if(S.min_force_sum!=mnf||csum_test(a,S.min_force_l,S.min_force_r)!=mnf||S.min_force_l>=S.min_force_r)bad[3]++;
#ifdef CIRC
  // circular: subarrays of length 0..n (allow) or 1..n (force)
  ll cmxa=0,cmxf=LLONG_MIN,cmna=0,cmnf=LLONG_MAX;
  for(int s=0;s<n;s++){ll t=0; for(int len=1;len<=n;len++){ t+=a[(s+len-1)%n]; cmxa=max(cmxa,t);cmxf=max(cmxf,t);cmna=min(cmna,t);cmnf=min(cmnf,t);} }
  SubarrayAllCircular C(a);
  auto cs=[&](int l,int r,ll want,bool force){ if(l==r){ return force? false : want==0 ; } return csum_test(a,l,r)==want; };
  if(C.max_allow_sum!=cmxa)bad[4]++;
  if(C.max_force_sum!=cmxf)bad[5]++;
  if(C.min_allow_sum!=cmna)bad[6]++;
  if(C.min_force_sum!=cmnf)bad[7]++;
#endif
 } int tot=0; for(int i=0;i<8;i++) tot+=bad[i];
 cout << "random_test_kadane: " << (tot==0 ? "OK" : "NG") << "（愚直解と 20000 回比較）\n"; assert(tot==0);
}
