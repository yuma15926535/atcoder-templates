/*
 Block-Cut Tree / Bridge / Articulation / BCC 総合テンプレート
 ===============================================================

 無向グラフに対して以下を O(N+M) で構築する：
  - 橋
  - 関節点
  - 二重連結成分（BCC, 辺集合）
  - Block-Cut Tree（BCT）

 さらに：
  - 頂点削除後の連結成分数
  - 頂点削除後の各連結成分サイズ
  - 最大連結成分サイズ

 【ノード番号規約】
  - 0 .. N-1        : 元グラフの頂点
  - N .. N+B-1      : BCC ノード
*/
struct BlockCutTree {
    int N;
    vector<vector<int>> G;    // 隣接頂点
    vector<vector<int>> GE;   // G と同じ並びの辺番号（多重辺を区別するため）
    int M = 0;                // 辺の数

    // lowlink 用
    vector<int> ord, low;
    int timer = 0;

    // 結果
    vector<pair<int,int>> bridges;     // 橋
    vector<int> articulation;           // 関節点
    vector<vector<pair<int,int>>> bcc;  // 二重連結成分（辺集合）

    // 辺スタック（BCC 構築用）
    vector<pair<int,int>> stk;

    // Block-Cut Tree
    vector<vector<int>> tree;

    // BCT 用サイズ管理
    vector<int> parent; // BCT 上の親（各連結成分で番号の小さいノードを根にする）
    vector<int> sub;    // BCT 上の部分木に含まれる「元の頂点」の数
    vector<int> bcc_size; // 各 BCC に含まれる非関節点数

    // 追加：頂点削除用
    vector<int> sz;          // DFS 木での部分木サイズ
    vector<int> comp_size;   // 各頂点が属する連結成分サイズ
    vector<int> root;        // DFS 木の根
    vector<vector<int>> dfs_child;   // DFS 木での子

    BlockCutTree(int n)
        : N(n), G(n), GE(n), ord(n, -1), low(n, -1),
          sz(n, 0), comp_size(n, 0), root(n, -1), dfs_child(n) {}

    // 無向辺を追加する（多重辺・自己ループも可。自己ループは橋・関節点・BCC に影響しないので無視される）
    void add_edge(int u, int v) {
        G[u].push_back(v); GE[u].push_back(M);
        G[v].push_back(u); GE[v].push_back(M);
        M++;
    }

    // lowlink 計算と BCC 構築（非再帰 DFS）。親へは「来た辺」だけを戻らないので多重辺も正しく扱う
    void dfs(int s) {
        struct Frame { int v, pe, i, child; bool art; };
        vector<Frame> st;
        vector<int> visited;
        ord[s] = low[s] = timer++;
        sz[s] = 1;
        visited.push_back(s);
        st.push_back({s, -1, 0, 0, false});
        while (!st.empty()) {
            Frame& f = st.back();
            int v = f.v;
            if (f.i < (int)G[v].size()) {
                int to = G[v][f.i], id = GE[v][f.i];
                f.i++;
                if (id == f.pe) continue;                 // 来た辺は戻らない
                if (ord[to] == -1) {
                    stk.emplace_back(v, to);
                    f.child++;
                    dfs_child[v].push_back(to);
                    root[to] = root[v];
                    ord[to] = low[to] = timer++;
                    sz[to] = 1;
                    visited.push_back(to);
                    st.push_back({to, id, 0, 0, false});  // f はこの後使わない（参照が無効になるため）
                } else if (ord[to] < ord[v]) {            // 後退辺
                    stk.emplace_back(v, to);
                    low[v] = min(low[v], ord[to]);
                }
                continue;
            }
            // v の探索が終わった
            bool art = f.art;
            st.pop_back();
            if (art) articulation.push_back(v);
            if (st.empty()) break;
            Frame& pf = st.back();
            int p = pf.v;
            sz[p] += sz[v];
            low[p] = min(low[p], low[v]);
            if (ord[p] < low[v]) bridges.emplace_back(p, v);  // 橋判定
            if (ord[p] <= low[v]) {                            // BCC の切り出し条件
                if ((int)st.size() > 1 || pf.child > 1) pf.art = true;
                vector<pair<int,int>> comp;
                while (true) {
                    auto e = stk.back(); stk.pop_back();
                    comp.push_back(e);
                    if (e.first == p && e.second == v) break;
                }
                bcc.push_back(comp);
            }
        }
        for (int x : visited) comp_size[x] = sz[s];  // 連結成分サイズ
    }

    // 全体を構築する（必ず最初に呼ぶ）
    void build() {
        for (int i = 0; i < N; i++) {
            if (ord[i] == -1) {
                root[i] = i;
                dfs(i);
            }
        }
        build_block_cut_tree();
        prepare_bct_size();
    }

    // Block-Cut Tree を構築する
    void build_block_cut_tree() {
        int B = bcc.size();
        tree.assign(N + B, {});
        bcc_size.assign(B, 0);
        vector<vector<int>> members(B);
        for (int i = 0; i < B; i++) {
            int id = N + i;
            vector<int>& vs = members[i];
            for (auto [u, v] : bcc[i]) { vs.push_back(u); vs.push_back(v); }
            sort(vs.begin(), vs.end());
            vs.erase(unique(vs.begin(), vs.end()), vs.end());
            for (int v : vs) {
                tree[v].push_back(id);
                tree[id].push_back(v);
            }
        }
        // 非関節点（属する BCC が 1 つだけの頂点）の数を BCC サイズとして持つ（木が完成してから数える）
        for (int i = 0; i < B; i++)
            for (int v : members[i])
                if ((int)tree[v].size() == 1) bcc_size[i]++;
    }

    // BCT の各連結成分を根付き木にして部分木サイズを計算する（非再帰）
    void prepare_bct_size() {
        int SZ = tree.size();
        parent.assign(SZ, -1);
        sub.assign(SZ, 0);
        vector<char> seen(SZ, 0);
        vector<int> order;
        for (int r = 0; r < SZ; r++) {
            if (seen[r]) continue;
            seen[r] = 1;
            order.clear();
            order.push_back(r);
            for (int h = 0; h < (int)order.size(); h++) {
                int v = order[h];
                for (int to : tree[v]) if (!seen[to]) { seen[to] = 1; parent[to] = v; order.push_back(to); }
            }
            for (int h = (int)order.size() - 1; h >= 0; h--) {
                int v = order[h];
                sub[v] += (v < N ? 1 : 0);
                if (parent[v] != -1) sub[parent[v]] += sub[v];
            }
        }
    }

    // 頂点 v を削除したあとにできる各連結成分サイズを返す
    vector<int> components_after_removal(int v) {
        vector<int> res;
        int total = comp_size[v];

        int sum = 0;
        for (int to : dfs_child[v]) {          // DFS 木の子だけを見る（後退辺の先の子孫を数えない）
            if (low[to] >= ord[v]) {
                res.push_back(sz[to]);
                sum += sz[to];
            }
        }

        int rest = total - sum - 1;
        if (rest > 0) res.push_back(rest);
        if (res.empty() && total > 1) res.push_back(total - 1);

        return res;
    }

    // 頂点 v を削除したあとの最大連結成分サイズ
    int max_component_after_removal(int v) {
        int mx = 0;
        for (int sz : components_after_removal(v)) {
            mx = max(mx, sz);
        }
        return mx;
    }

    // 全頂点について「削除後の連結成分数」を返す
    vector<int> component_count_after_vertex_removal() {
        vector<int> res(N, 1);
        for (int v : articulation) {
            res[v] = components_after_removal(v).size();
        }
        return res;
    }
};

void random_test_bruteforce(){ mt19937 rng(14); int bad=0, badb=0, bada=0;
 for(int it=0;it<3000;it++){ int n=rng()%8+1; int m=rng()%12; BlockCutTree bc(n); vector<pair<int,int>> es;
  for(int i=0;i<m;i++){int u=rng()%n,v=rng()%n; if(u>v)swap(u,v); es.push_back({u,v}); bc.add_edge(u,v);}  // 多重辺・自己ループも入れる
  bc.build();
  auto comps=[&](int skipv, int skipe){ vector<int> id(n,-1); int c=0; vector<int> sizes;
    vector<vector<pair<int,int>>> g(n); for(int i=0;i<(int)es.size();i++){ if(i==skipe)continue; auto [u,v]=es[i]; g[u].push_back({v,i}); g[v].push_back({u,i}); }
    for(int s=0;s<n;s++){ if(s==skipv||id[s]!=-1)continue; int cnt=0; vector<int> st={s}; id[s]=c; while(!st.empty()){int x=st.back(); st.pop_back(); cnt++; for(auto [y,ei]:g[x]) if(y!=skipv&&id[y]==-1){id[y]=c; st.push_back(y);} } sizes.push_back(cnt); c++; }
    return make_pair(id,sizes); };
  auto [id0,s0]=comps(-1,-1); int c0=s0.size();
  // bridges
  set<pair<int,int>> br; for(int i=0;i<(int)es.size();i++){ auto [id1,s1]=comps(-1,i); if((int)s1.size()>c0) br.insert(es[i]); }
  set<pair<int,int>> got; for(auto [u,v]:bc.bridges) got.insert({min(u,v),max(u,v)}); if(got!=br)badb++;
  // articulation & components_after_removal
  set<int> art; for(int v=0;v<n;v++){ auto [id1,s1]=comps(v,-1); int cv=id0[v];
    // components inside v's original component
    vector<int> want; map<int,int> cnt; for(int x=0;x<n;x++) if(x!=v && id0[x]==cv) cnt[id1[x]]++; for(auto [k,c]:cnt) want.push_back(c); sort(want.begin(),want.end());
    if(want.size()>1) art.insert(v);
    auto got2=bc.components_after_removal(v); sort(got2.begin(),got2.end()); if(got2!=want)bad++; }
  set<int> ga(bc.articulation.begin(),bc.articulation.end()); if(ga!=art)bada++;
  // BCT：根ごとの sub の合計 = 頂点数、BCC の辺の合計 = 自己ループ以外の辺の数
  { int tot=0; for(int x=0;x<(int)bc.tree.size();x++) if(bc.parent[x]==-1) tot+=bc.sub[x]; if(tot!=n) bada++;
    int ecnt=0; for(auto& c:bc.bcc) ecnt+=c.size(); int want=0; for(auto [u,v]:es) if(u!=v) want++; if(ecnt!=want) bada++; }
 } 
    cout << "random_test_bruteforce（愚直解とのランダム比較）: " << (((bad)==0&&(badb)==0&&(bada)==0) ? "OK" : "NG") << "\n";
    assert((bad)==0&&(badb)==0&&(bada)==0);
}

        cout << "[橋]\n";
        for (auto [u, v] : bc.bridges)
            cout << u << "-" << v << " ";
        cout << "\n";

        cout << "[関節点]\n";
        for (int v : bc.articulation)
            cout << v << " ";
        cout << "\n";

        cout << "[頂点削除後の連結成分数]\n";
        auto cnt = bc.component_count_after_vertex_removal();
        for (int i = 0; i < N; i++)
            cout << "頂点 " << i << " を削除 -> " << cnt[i] << "\n";

        cout << "[頂点削除後の各連結成分サイズ]\n";
        for (int i = 0; i < N; i++) {
            auto comps = bc.components_after_removal(i);
            cout << "頂点 " << i << " を削除: ";
            for (int sz : comps) cout << sz << " ";
            cout << "\n";
        }

        cout << "[頂点削除後の最大連結成分サイズ]\n";
        for (int i = 0; i < N; i++)
            cout << "頂点 " << i << " を削除 -> "
                 << bc.max_component_after_removal(i) << "\n";
        cout << "\n";
    }
    // =====================================================
    // ケース4：多重辺（同じ 2 頂点を結ぶ辺が 2 本）
    // =====================================================
    {
        cout << "===== ケース4：多重辺 =====\n";
        BlockCutTree bc(3);
        bc.add_edge(0, 1);
        bc.add_edge(0, 1);   // 0-1 が 2 本 → どちらも橋ではない
        bc.add_edge(1, 2);   // 1-2 は橋
        bc.build();
        cout << "[橋] ";
        for (auto [u, v] : bc.bridges) cout << u << "-" << v << " ";
        cout << "（1-2 だけ）\n";
        cout << "[関節点] ";
        for (int v : bc.articulation) cout << v << " ";
        cout << "（1 だけ）\n";
        cout << "[BCT の部分木の頂点数] sub[根] = " << bc.sub[0] << "（連結成分の頂点数 3）\n\n";
    }
    random_test_bruteforce();
    return;
