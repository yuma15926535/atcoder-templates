struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
=====================================================
★ モノイド定義（問題ごとに変更）
=====================================================

S : ノードの値
op : 区間結合
id : 単位元

※ prefix探索を使う場合は
累積が単調になる演算であること
=====================================================
*/

using S = ll;

// ★区間結合
S op(S a, S b) {
    return min(a, b); // 区間min
}

// ★単位元
S e() {
    return (1LL << 60);
}

/*
=====================================================
SegTree 本体
=====================================================
*/

struct SegTree {
    int n;
    vector<S> seg;

    SegTree(int sz) {
        n = 1;
        while (n < sz) n <<= 1;
        seg.assign(2*n, e());
    }

    // 配列初期化
    SegTree(const vector<S>& v) {
        int sz = v.size();
        n = 1;
        while (n < sz) n <<= 1;
        seg.assign(2*n, e());

        for (int i = 0; i < sz; i++)
            seg[n+i] = v[i];

        for (int i = n-1; i > 0; i--)
            seg[i] = op(seg[i<<1], seg[i<<1|1]);
    }

    // ========================
    // 点更新
    // ========================
    void set(int i, S x) {
        i += n;
        seg[i] = x;

        while (i >>= 1)
            seg[i] = op(seg[i<<1], seg[i<<1|1]);
    }

    // ========================
    // 区間取得 [l,r)
    // ========================
    S prod(int l, int r) {
        S L = e(), R = e();

        for (l += n, r += n; l < r; l >>= 1, r >>= 1) {
            if (l & 1) L = op(L, seg[l++]);
            if (r & 1) R = op(seg[--r], R);
        }

        return op(L, R);
    }

    S get(int i) {
        return seg[n+i];
    }

    S all_prod() {
        return seg[1];
    }
};

void random_test() {
    random_device rd;
    mt19937 rng(rd());

    rep(tc, 0, 10000) {
        int N = rng() % 20 + 1;

        vector<ll> v(N);
        rep(i, 0, N)
            v[i] = rng() % 100;

        SegTree seg(v);
        vector<ll> naive = v;

        rep(q, 0, 1000) {

            // 0:set
            // 1:range min
            int type = rng() % 2;

            // ========================
            // 点更新
            // ========================
            if (type == 0) {
                int idx = rng() % N;
                ll x = rng() % 100;

                seg.set(idx, x);
                naive[idx] = x;
            }

            // ========================
            // 区間min
            // ========================
            else {
                int l = rng() % N;
                int r = rng() % N;

                if (l > r) swap(l, r);
                r++;

                ll ans1 = seg.prod(l, r);

                ll ans2 = (1LL << 60);
                rep(i, l, r)
                    ans2 = min(ans2, naive[i]);

                if (ans1 != ans2) {
                    cout << "WA\n";

                    cout << "N = " << N << '\n';

                    cout << "array:\n";
                    for (auto x : naive)
                        cout << x << ' ';
                    cout << '\n';

                    cout << "l r = " << l << ' ' << r << '\n';

                    cout << "seg = " << ans1 << '\n';
                    cout << "naive = " << ans2 << '\n';

                    exit(0);
                }
            }
        }
    }

    cout << "AC\n";
}

void demo(){
    cout << "===== 区間 min デモ =====\n";
    vector<ll> a = {5, 3, 8, 6, 1, 7};
    SegTree seg(a);                       // 配列から構築 O(N)  ※ SegTree seg(n) なら全部単位元 e()
    cout << "a = {5, 3, 8, 6, 1, 7}\n";
    cout << "seg.prod(1, 4) = " << seg.prod(1, 4) << "   // [1,4) の最小値 → 3\n";
    cout << "seg.all_prod() = " << seg.all_prod() << "   // 全体の最小値 → 1\n";
    seg.set(2, 0);                        // a[2] = 0 に書き換え O(log N)
    cout << "seg.set(2, 0) 後: seg.prod(1, 4) = " << seg.prod(1, 4) << "   // → 0\n";
    cout << "seg.get(2) = " << seg.get(2) << "   // a[2] の値 → 0\n";
    cout << "seg.prod(3, 3) = " << seg.prod(3, 3) << "   // 空区間は単位元 e() = 2^60\n";
}
