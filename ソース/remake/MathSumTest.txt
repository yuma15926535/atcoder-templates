struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

//  数列の和・連結系テンプレ（非素数 mod 対応）
//  （作り直し版：元ファイル MathSumTest.txt）
// -----------------------------------------------------------------------------
//  説明
//    - 等差数列の和のmod
//    - 等比数列の和のmod
//    - 数値の連結のmod
//    - 数値の連続連結のmod
//    - レピュニットmod
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
// ============================================================
// 数列・連結系テンプレ（非素数 mod 対応）
// ============================================================
//
// 含まれる関数
//
// arith_sum_mod  : 等差数列の和
// geom_sum_core   : r^n と (1+r+...+r^(n-1))
// sum_geometric   : 等比数列の和
// concat_mod          : 2つの数の連結
// repeat_concat_mod   : 同じ数をk回連結
//
// すべて非素数 mod でも使用可能
// ============================================================

// ============================================================
// 等差数列の和
// a + (a+d) + ... + (a+(n-1)d)
// 計算量: O(1)
// ============================================================
// ======================= ここからテンプレート本体（コピー範囲） =======================
// 等差数列の和 a + (a+d) + ... + (a+(n-1)d) mod m（0 以上 m 未満で返す）
ll arith_sum_mod(ll a, ll d, ll n, ll mod){
    if(n == 0) return 0;
    __int128 term = (__int128)n * ((__int128)2*a + (__int128)(n-1)*d);   // 途中も 128 ビットで（溢れ防止）
    term /= 2; // 必ず割り切れる
    ll res = (ll)(term % mod);
    if(res < 0) res += mod;
    return res;
}

// ============================================================
// 等比数列コア
// r^n と (1 + r + r^2 + ... + r^(n-1)) を同時計算
//
// 戻り値
// first  = r^n
// second = 1 + r + ... + r^(n-1)
//
// 計算量: O(log n)
// ============================================================
pair<ll,ll> geom_sum_core(ll r, ll n, ll mod){
    if(n == 0) return {1,0};
    auto [p, s] = geom_sum_core(r, n/2, mod);
    ll p2 = (__int128)p * p % mod;
    ll s2 = (__int128)s * (1 + p) % mod;
    if(n % 2 == 0){
        return {p2, s2};
    }else{
        ll p3 = (__int128)p2 * r % mod;
        ll s3 = (s2 + p2) % mod;
        return {p3, s3};
    }
}

// ============================================================
// 等比数列の和（数学第二世代 00_math_base の geom_sum_mod と同じ）
// 1 + r + r^2 + ... + r^(n-1) mod m（逆元を使わないので m は何でも）
// 計算量: O(log n)
// ============================================================
ll geom_sum_mod(ll r, ll n, ll mod){
    if(mod == 1) return 0;
    ll rr = r % mod; if(rr < 0) rr += mod;
    return geom_sum_core(rr, n, mod).second;
}

// ============================================================
// 等比数列の和（先頭 a つき）
// a + ar + ar^2 + ... + ar^(n-1) ＝ a × geom_sum_mod(r, n, mod)
// 計算量: O(log n)
// ============================================================
ll sum_geometric(ll a, ll r, ll n, ll mod){
    if(mod == 1) return 0;
    ll aa = a % mod; if(aa < 0) aa += mod;
    return (__int128)aa * geom_sum_mod(r, n, mod) % mod;
}

// ============================================================
// 同じ数字 digit を n 回並べた数 mod
//
// 例
// digit=3 n=5 → 33333
// digit=7 n=4 → 7777
//
// 計算量: O(log n)
// 非素数 mod OK
// ============================================================
long long repunit_mod(long long digit, long long n, long long mod){
    digit %= mod;
    return digit * sum_geometric(1, 10, n, mod) % mod;
}

// ============================================================
// b の桁数を取得
// ============================================================
ll num_digits(ll x){
    if(x == 0) return 1;
    ll d = 0;
    while(x){
        d++;
        x/=10;
    }
    return d;
}

// ============================================================
// 10^d mod（繰り返し二乗法 O(log d)）
// ============================================================
ll pow10_mod(ll d, ll mod){
    ll r = 1 % mod, b = 10 % mod;
    for(; d > 0; d >>= 1){
        if(d & 1) r = (__int128)r * b % mod;
        b = (__int128)b * b % mod;
    }
    return r;
}

// ============================================================
// 数の連結
// concat_mod(a,b)
//
// 例
// a=123 b=45
// → 12345
//
// 計算量: O(digit(b) + log digit(b))
// ============================================================
ll concat_mod(ll a, ll b, ll mod){
    ll d = num_digits(b);
    ll p = pow10_mod(d, mod);
    return ((__int128)a * p + b) % mod;
}

// ============================================================
// 同じ数を k 回連結
//
// x=123 k=3
// → 123123123
//
// 計算量: O(log k)
// ============================================================
ll repeat_concat_mod(ll x, ll k, ll mod){
    if(k == 0) return 0;
    ll d = num_digits(x);
    ll r = pow10_mod(d, mod);
    return sum_geometric(x, r, k, mod);
}
// ======================= ここまでテンプレート本体 =======================

ll brute_mod(string s,ll mod){
    ll r=0;
    for(char c:s){
        r=(r*10+(c-'0'))%mod;
    }
    return r;
}

void demo(){
    cout << "===== 数列の和・数の連結（mod）デモ =====\n";
    const ll M = 1000000007;
    cout << "arith_sum_mod(1, 2, 5, M) = " << arith_sum_mod(1, 2, 5, M) << "   // 1+3+5+7+9 = 25\n";
    cout << "arith_sum_mod(-10, 3, 4, 7) = " << arith_sum_mod(-10, 3, 4, 7) << "   // -10-7-4-1 = -22 ≡ 6 (mod 7)\n";
    cout << "sum_geometric(3, 2, 4, M) = " << sum_geometric(3, 2, 4, M) << "   // 3+6+12+24 = 45（先頭 a つき）\n";
    cout << "geom_sum_mod(2, 4, M) = " << geom_sum_mod(2, 4, M) << "   // 1+2+4+8 = 15（数学第二世代と同じ意味）\n";
    cout << "pow10_mod(1000000000000000000, M) = " << pow10_mod(1000000000000000000LL, M) << "   // O(log d) なので d が巨大でもすぐ\n";
    cout << "repunit_mod(7, 4, M) = " << repunit_mod(7, 4, M) << "   // 7777\n";
    cout << "num_digits(12345) = " << num_digits(12345) << ", num_digits(0) = " << num_digits(0) << "\n";
    cout << "concat_mod(123, 45, M) = " << concat_mod(123, 45, M) << ", repeat_concat_mod(123, 3, M) = " << repeat_concat_mod(123, 3, M) << "\n";
    cout << "repeat_concat_mod(7, 1000000000000000000, M) = " << repeat_concat_mod(7, 1000000000000000000LL, M) << "   // 7 を 10^18 回並べた数 mod M\n";
}

void solv(){
    cout<<"===== 数列・連結テンプレ動作確認 =====\n\n";

    ll mod=107; // 非素数mod

    // 等差
    cout<<"[等差数列テスト]\n";
    cout<<"3 + 5 + 7 + 9 + 11\n";
    cout<<"計算結果: "<<arith_sum_mod(3,2,5,mod)<<"\n";
    cout<<"期待値: 35\n\n";

    // 等比
    cout<<"[等比数列テスト]\n";
    cout<<"3 + 6 + 12 + 24 + 48\n";
    cout<<"計算結果: "<<sum_geometric(3,2,5,mod)<<"\n";
    cout<<"期待値: 93\n\n";

    // concat_mod
    cout<<"[concatテスト]\n";
    ll c=concat_mod(123,45,mod);
    cout<<"123 と 45 を連結\n";
    cout<<"計算結果: "<<c<<"\n";
    cout<<"期待値: "<<12345%mod<<"\n\n";

    // repeat concat_mod
    cout<<"[repeat concatテスト]\n";
    ll rc=repeat_concat_mod(123,3,mod);
    cout<<"123 を3回連結\n";
    cout<<"計算結果: "<<rc<<"\n";
    cout<<"期待値: "<<brute_mod("123123123",mod)<<"\n\n";

    // repeated digit
    cout<<"[repeated digitテスト]\n";
    ll rd=repunit_mod(3,5,mod);
    cout<<"3 を5回並べる\n";
    cout<<"計算結果: "<<rd<<"\n";
    cout<<"期待値: "<<brute_mod("33333",mod)<<"\n\n";

    // repunit
    cout<<"[repunitテスト]\n";
    ll ru=repunit_mod(1,6,mod);
    cout<<"111111\n";
    cout<<"計算結果: "<<ru<<"\n";
    cout<<"期待値: "<<brute_mod("111111",mod)<<"\n\n";

    cout<<"===== テスト終了 =====\n";
    demo();
    return;
}
