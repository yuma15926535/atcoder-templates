struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// -----------------------------------------------------------------------------
//  説明
//    - 重複込みの欠番管理
//    - 欠番のmin/max
//    - k番目の欠番
//    - 区間最小欠番
//    - 区間欠番数
//    - 全体の欠番数
//    - 欠番判定
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける

// ======================= ここからテンプレート本体（コピー範囲） =======================
struct BIT {
    int n;
    vector<long long> bit;

    BIT(int n=0): n(n), bit(n+1,0) {}
    // すべての要素を init にして O(N) で作る（各ノードは「長さ lowbit(i) の区間の和」）
    BIT(int n, long long init): n(n), bit(n+1,0) {
        for(int i=1;i<=n;i++) bit[i] = init * (i & -i);
    }

    // index i に v を加算
    void add(int i, long long v){
        for(i++; i<=n; i+=i&-i) bit[i] += v;
    }

    // [0, r) の和（ALPCB.txt の BIT と同じ約束）
    long long sum(int r) const {
        long long s = 0;
        for(; r>0; r-=r&-r) s += bit[r];
        return s;
    }

    // 累積和 sum(i + 1) が k 以上になる最小の index i を返す（0-index。k >= 1 を想定。無ければ n）
    int kth(long long k) const {
        int idx = 0;
        long long cur = 0;
        int top = 1;
        while((top << 1) <= n) top <<= 1;   // n 以下の最大の 2 冪（n の大きさに制限なし）
        for(int d = top; d > 0; d >>= 1){
            if(idx + d <= n && cur + bit[idx + d] < k){
                cur += bit[idx + d];
                idx += d;
            }
        }
        return idx; // 0-index
    }
};

// ==============================================
// MEX / 欠番管理本体
// ==============================================
struct MEX {
    int N;                 // 管理する値は 0 〜 N-1
    vector<int> cnt;       // 各値の個数
    BIT bit;               // 欠番なら 1, 存在するなら 0
    long long sz = 0;      // 現在の要素数（重複込み）
    // 初期状態：すべて欠番
    MEX(int N): N(N), cnt(N,0), bit(N, 1) {}   // 全部 1（欠番）の BIT を O(N) で作る
    // x を 1 個追加（重複可。範囲外は無視）
    void add(int x){
        if(x < 0 || x >= N) return;
        if(cnt[x] == 0) bit.add(x, -1); // 欠番でなくなる
        cnt[x]++;
        sz++;
    }
    // x を 1 個削除（無ければ何もしない）
    void erase(int x){
        if(x < 0 || x >= N || cnt[x] == 0) return;
        cnt[x]--;
        if(cnt[x] == 0) bit.add(x, +1); // 再び欠番
        sz--;
    }
    // 現在の要素数（重複込み）
    long long size() const { return sz; }
    // x の個数 / x があるか / x が欠番か
    int count(int x) const { return (0 <= x && x < N) ? cnt[x] : 0; }
    bool contains(int x) const { return count(x) > 0; }
    bool is_missing(int x) const { return 0 <= x && x < N && cnt[x] == 0; }
    // 全体の欠番数
    long long missing_count() const { return bit.sum(N); }
    // [l, r) に含まれる欠番の個数
    long long count_missing(int l, int r) const {
        l = max(l, 0); r = min(r, N);
        if(l >= r) return 0;
        return bit.sum(r) - bit.sum(l);
    }
    // 最小の欠番（mex）。0 〜 N-1 が全部あるときは N
    int mex() const {
        if(missing_count() == 0) return N;
        return bit.kth(1);
    }
    // 最大の欠番（無ければ -1）
    int max_missing() const {
        long long tot = missing_count();
        if(tot == 0) return -1;
        return bit.kth(tot);
    }
    // 小さい方から k 番目の欠番（k は 0 始まり。無ければ -1）
    int kth_missing(long long k) const {
        if(k < 0 || missing_count() <= k) return -1;
        return bit.kth(k + 1);
    }
    // x より大きい最小の欠番（無ければ -1）。「l 以上で最小」は next_missing(l - 1)
    int next_missing(int x) const {
        return kth_missing(count_missing(0, x + 1));
    }
    // x より小さい最大の欠番（無ければ -1）
    int prev_missing(int x) const {
        long long c = count_missing(0, x);
        return c == 0 ? -1 : kth_missing(c - 1);
    }
};

// -----------------------------------------------------------------------------
//  実戦での使用例（ABC194 E）… 標準入力を読むので main からは呼ばない
// -----------------------------------------------------------------------------

void solve_ABC194E(){
    int N,M;
    cin>>N>>M;
    vector<int>A(N);
    rep(i,0,N)cin>>A[i];
    MEX mex(N+5);
    rep(i,0,M)mex.add(A[i]);
    ll ans=mex.mex();
    rep(i,0,N){
        if(i+M>=N)break;
        mex.erase(A[i]);
        mex.add(A[i+M]);
        chmin(ans,(ll)mex.mex());
    }
    cout<<ans<<'\n';
    return;
}

void demo(){
    cout << "===== MEX（重複ありの欠番管理）デモ =====\n";
    MEX m(8);                                  // 値 0〜7 を管理
    for (int x : {0, 1, 1, 3, 6}) m.add(x);    // 重複して追加できる
    cout << "m.mex() = " << m.mex() << "   // {0,1,1,3,6} の最小の欠番 → 2\n";
    cout << "m.kth_missing(1) = " << m.kth_missing(1) << "   // 欠番 2,4,5,7 の 0 始まり 1 番目 → 4\n";
    cout << "m.next_missing(4) = " << m.next_missing(4) << ", m.prev_missing(4) = " << m.prev_missing(4) << "   // 4 より大きい最小 5、小さい最大 2\n";
    cout << "m.max_missing() = " << m.max_missing() << ", m.missing_count() = " << m.missing_count() << ", m.count_missing(0, 5) = " << m.count_missing(0, 5) << "\n";
    m.erase(1);
    cout << "m.erase(1) 後: m.count(1) = " << m.count(1) << ", m.mex() = " << m.mex() << "   // 1 はまだ 1 個ある\n";
    m.erase(1);
    cout << "もう一度 m.erase(1) 後: m.mex() = " << m.mex() << "   // → 1\n";
}
