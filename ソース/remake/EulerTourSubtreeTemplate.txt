// ======================= ここからテンプレート本体（コピー範囲） =======================
struct EulerTourBase {
    int N;
    std::vector<std::vector<int>> g;

    std::vector<int> par;
    std::vector<int> dep;
    std::vector<int> tin;
    std::vector<int> tout;
    std::vector<int> rev;
    std::vector<int> euler;
    std::vector<int> first;
    std::vector<int> euler_lca;
    std::vector<int> depth_lca;

    EulerTourBase(int n) : N(n), g(n), par(n, -1), dep(n, 0), tin(n, -1), tout(n, -1), first(n, -1) {
        rev.reserve(n);
        euler.reserve(2 * n);
        euler_lca.reserve(2 * n);
        depth_lca.reserve(2 * n);
    }

    void add_edge(int u, int v) {
        g[u].push_back(v);
        g[v].push_back(u);
    }

    void build(int root = 0) {
        std::fill(par.begin(), par.end(), -1);
        std::fill(dep.begin(), dep.end(), 0);
        std::fill(tin.begin(), tin.end(), -1);
        std::fill(tout.begin(), tout.end(), -1);
        std::fill(first.begin(), first.end(), -1);
        rev.clear();
        euler.clear();
        euler_lca.clear();
        depth_lca.clear();

        int timer = 0;
        // 非再帰 DFS（再帰版と同じ順番で tin / euler / euler_lca を作る。深い木でもスタックが溢れない）
        std::vector<std::pair<int, int>> st;  // (頂点, 次に見る隣接の番号)
        auto enter = [&](int u, int p) {
            par[u] = p;
            dep[u] = (p == -1 ? 0 : dep[p] + 1);
            tin[u] = timer++;
            rev.push_back(u);

            euler.push_back(u);

            first[u] = euler_lca.size();
            euler_lca.push_back(u);
            depth_lca.push_back(dep[u]);
            st.push_back({u, 0});
        };
        enter(root, -1);
        while (!st.empty()) {
            auto& [u, i] = st.back();
            if (i < (int)g[u].size()) {
                int v = g[u][i++];
                if (v == par[u]) continue;
                enter(v, u);  // ここで st が伸びるので u, i はこの後使わない
                continue;
            }
            tout[u] = timer;
            euler.push_back(~u);
            st.pop_back();
            if (!st.empty()) {  // 子から戻ってきたら親をもう一度並べる
                int p = st.back().first;
                euler_lca.push_back(p);
                depth_lca.push_back(dep[p]);
            }
        }
    }

    int parent(int v) const {
        return par[v];
    }

    int depth(int v) const {
        return dep[v];
    }

    std::pair<int, int> subtree_range(int v) const {
        return {tin[v], tout[v]};
    }

    int subtree_size(int v) const {
        return tout[v] - tin[v];
    }

    int vertex_at(int pos) const {
        return rev[pos];
    }

    bool is_ancestor(int u, int v) const {
        return tin[u] <= tin[v] && tout[v] <= tout[u];
    }

    const std::vector<int>& get_first() const { return first; }
    const std::vector<int>& get_euler_lca() const { return euler_lca; }
    const std::vector<int>& get_depth_lca() const { return depth_lca; }
};

struct S {
    ll sum;
    int len;
};

using F = ll;

/*
=====================================================
★ 演算定義
=====================================================
*/

// ノード結合
S op(S a, S b) {
    return {a.sum + b.sum, a.len + b.len};
}

// 更新適用
S mapping(F f, S x) {
    x.sum += f * x.len;
    return x;
}

/*
composition(f, g)

意味：
既存更新 g の後に
新しい更新 f を適用

つまり：

new_lazy = f ∘ g

加算なら単純に足す
*/
F composition(F f, F g) {
    return f + g;
}

S e() { return {0,0}; }
F id() { return 0; }

/*
=====================================================
Lazy SegTree 本体
=====================================================
*/

struct LazySegTree {
    int n;
    std::vector<S> seg;
    std::vector<F> lazy;

    LazySegTree(const std::vector<ll>& v) {
        int sz = v.size();
        n = 1;
        while (n < sz) n <<= 1;

        seg.assign(2*n, e());
        lazy.assign(2*n, id());

        for (int i = 0; i < sz; i++)
            seg[n+i] = {v[i],1};
        for (int i = sz; i < n; i++)
            seg[n+i] = {0,1};

        for (int i = n-1; i > 0; i--)
            seg[i] = op(seg[i<<1], seg[i<<1|1]);
    }

    // ========================
    // lazy伝播
    // ========================
    void push(int k) {
        if (lazy[k] == id()) return;

        apply_node(k<<1, lazy[k]);
        apply_node(k<<1|1, lazy[k]);

        lazy[k] = id();
    }

    // ---- 非再帰の区間作用・区間取得（ACL の lazy_segtree と同じやり方。再帰版より 2〜3 倍速い）----
    void apply_range(int l, int r, F f) {
        if (l >= r) return;
        const int lg = __builtin_ctz((unsigned)n);
        l += n; r += n;
        for (int i = lg; i >= 1; i--) {               // 上から遅延を下ろす
            if (((l >> i) << i) != l) push(l >> i);
            if (((r >> i) << i) != r) push((r - 1) >> i);
        }
        for (int a = l, b = r; a < b; a >>= 1, b >>= 1) {
            if (a & 1) apply_node(a++, f);
            if (b & 1) apply_node(--b, f);
        }
        for (int i = 1; i <= lg; i++) {               // 下から集約し直す
            if (((l >> i) << i) != l) { int k = l >> i; seg[k] = op(seg[k << 1], seg[k << 1 | 1]); }
            if (((r >> i) << i) != r) { int k = (r - 1) >> i; seg[k] = op(seg[k << 1], seg[k << 1 | 1]); }
        }
    }
    S prod_range(int l, int r) {
        if (l >= r) return e();
        const int lg = __builtin_ctz((unsigned)n);
        l += n; r += n;
        for (int i = lg; i >= 1; i--) {
            if (((l >> i) << i) != l) push(l >> i);
            if (((r >> i) << i) != r) push((r - 1) >> i);
        }
        S sml = e(), smr = e();
        for (; l < r; l >>= 1, r >>= 1) {
            if (l & 1) sml = op(sml, seg[l++]);
            if (r & 1) smr = op(seg[--r], smr);
        }
        return op(sml, smr);
    }

    void apply_node(int k, F f) {
        seg[k] = mapping(f, seg[k]);
        if (k < n)
            lazy[k] = composition(f, lazy[k]);
    }

    // ========================
    // 区間更新
    // ========================
    void apply(int l, int r, F f) {
        apply_range(l, r, f);
    }

    // ========================
    // 区間取得
    // ========================
    // 区間 [l, r) の集約値 S を返す（ACL の prod と同じ）
    S prod(int l, int r) { return prod_range(l, r); }
    // 区間 [l, r) の和
    ll range_sum(int l, int r) {
        return prod_range(l, r).sum;
    }

    // ========================
    // prefix探索
    // ========================
    int ge(ll x) {
        if (seg[1].sum < x) return n;
        int k = 1;
        ll acc = 0;
        while (k < n) {               // 根から下る（非再帰）
            push(k);
            if (acc + seg[k << 1].sum >= x) k <<= 1;
            else { acc += seg[k << 1].sum; k = k << 1 | 1; }
        }
        return k - n;
    }

    int gt(ll x){ return ge(x+1); }
    int le(ll x){ return ge(x+1)-1; }
    int lt(ll x){ return ge(x)-1; }
};

struct EulerTourSubtree {
    const EulerTourBase* et;
    LazySegTree seg;

    EulerTourSubtree(const EulerTourBase* et, const std::vector<ll>& val)
        : et(et), seg([&]() {
            std::vector<ll> base(et->N, 0);
            for (int i = 0; i < et->N; ++i) {
                base[et->tin[i]] = val[i];
            }
            return LazySegTree(base);
        }()) {}

    ll get_vertex(int v) {
        return seg.range_sum(et->tin[v], et->tin[v] + 1);
    }

    void set_vertex(int v, ll x) {
        ll cur = get_vertex(v);
        apply_vertex(v, x - cur);
    }

    void apply_vertex(int v, ll f) {
        seg.apply(et->tin[v], et->tin[v] + 1, f);
    }

    ll query_subtree(int v) {
        auto [l, r] = et->subtree_range(v);
        return seg.range_sum(l, r);
    }

    void apply_subtree(int v, ll f) {
        auto [l, r] = et->subtree_range(v);
        seg.apply(l, r, f);
    }

    ll query_range_vertex(int l, int r) {
        return seg.range_sum(l, r);
    }

    void apply_range_vertex(int l, int r, ll f) {
        seg.apply(l, r, f);
    }
};

// ランダムな木を生成する
vector<vector<int>> generate_random_tree(int N, mt19937& rng) {
    vector<vector<int>> g(N);
    for (int i = 1; i < N; ++i) {
        int p = uniform_int_distribution<int>(0, i - 1)(rng);
        g[p].push_back(i);
        g[i].push_back(p);
    }
    return g;
}

// 愚直なDFSによる木情報の構築
struct BruteTree {
    int N;
    vector<vector<int>> g;
    int root;

    vector<int> par;
    vector<int> dep;
    vector<int> tin;
    vector<int> tout;
    vector<int> rev;
    vector<set<int>> ancestors;
    vector<vector<int>> subtree_verts;

    BruteTree(int n, const vector<vector<int>>& graph, int r) 
        : N(n), g(graph), root(r), par(n, -1), dep(n, 0), tin(n, -1), tout(n, -1), ancestors(n), subtree_verts(n) {}

    void run() {
        int timer = 0;
        vector<int> path;
        auto dfs = [&](auto self, int u, int p) -> void {
            par[u] = p;
            dep[u] = (p == -1 ? 0 : dep[p] + 1);
            tin[u] = timer++;
            rev.push_back(u);

            path.push_back(u);
            for (int anc : path) {
                ancestors[u].insert(anc);
            }

            subtree_verts[u].push_back(u);

            for (int v : g[u]) {
                if (v == p) continue;
                self(self, v, u);
                for (int x : subtree_verts[v]) {
                    subtree_verts[u].push_back(x);
                }
            }
            tout[u] = timer;
            path.pop_back();
        };
        dfs(dfs, root, -1);
    }
};

// 愚直なDFSで部分木の頂点リストを得るためのヘルパー
struct BruteSubtreeHelper {
    int N;
    vector<vector<int>> g;
    int root;
    vector<vector<int>> subtree_verts;

    BruteSubtreeHelper(int n, const vector<vector<int>>& graph, int r)
        : N(n), g(graph), root(r), subtree_verts(n) {}

    void run() {
        auto dfs = [&](auto self, int u, int p) -> void {
            subtree_verts[u].push_back(u);
            for (int v : g[u]) {
                if (v == p) continue;
                self(self, v, u);
                for (int x : subtree_verts[v]) {
                    subtree_verts[u].push_back(x);
                }
            }
        };
        dfs(dfs, root, -1);
    }
};

void run_random_tests() {
    mt19937 rng(1337);

    const int NUM_TESTS = 1000;
    const int MAX_N = 100;

    for (int t = 0; t < NUM_TESTS; ++t) {
        int N = uniform_int_distribution<int>(1, MAX_N)(rng);
        auto g = generate_random_tree(N, rng);
        int root = uniform_int_distribution<int>(0, N - 1)(rng);

        EulerTourBase et(N);
        for (int u = 0; u < N; ++u) {
            for (int v : g[u]) {
                if (u < v) {
                    et.add_edge(u, v);
                }
            }
        }
        et.build(root);

        BruteTree brute(N, g, root);
        brute.run();

        for (int i = 0; i < N; ++i) {
            // 1. parent の比較
            assert(et.parent(i) == brute.par[i]);

            // 2. depth の比較
            assert(et.depth(i) == brute.dep[i]);

            // 3. subtree_range / subtree_size の比較
            auto prod = et.subtree_range(i);
            int size = et.subtree_size(i);
            assert(size == (int)brute.subtree_verts[i].size());
            assert(prod.second - prod.first == size);

            vector<int> actual_subtree;
            for (int pos = prod.first; pos < prod.second; ++pos) {
                actual_subtree.push_back(et.vertex_at(pos));
            }
            assert(actual_subtree == brute.subtree_verts[i]);

            // 4. tin/rev の整合性
            assert(et.vertex_at(et.tin[i]) == i);
            assert(et.rev[et.tin[i]] == i);
            int v_at_pos = et.vertex_at(i);
            assert(et.tin[v_at_pos] == i);

            // 5. is_ancestor の比較
            for (int j = 0; j < N; ++j) {
                bool expected_ancestor = (brute.ancestors[j].count(i) > 0);
                assert(et.is_ancestor(i, j) == expected_ancestor);
            }
        }

        // LCA用オイラーツアーの整合性チェック
        const auto& elca = et.get_euler_lca();
        const auto& dlca = et.get_depth_lca();
        const auto& first = et.get_first();

        if (N == 1) {
            assert(elca.size() == 1);
            assert(dlca.size() == 1);
            assert(elca[0] == root);
            assert(dlca[0] == 0);
            assert(first[root] == 0);
        } else {
            assert((int)elca.size() == 2 * N - 1);
            assert((int)dlca.size() == 2 * N - 1);
            for (int i = 0; i < N; ++i) {
                assert(first[i] >= 0 && first[i] < (int)elca.size());
                assert(elca[first[i]] == i);
                for (int pos = 0; pos < first[i]; ++pos) {
                    assert(elca[pos] != i);
                }
            }
            for (int pos = 0; pos < (int)elca.size(); ++pos) {
                int u = elca[pos];
                assert(dlca[pos] == et.depth(u));
            }
            for (int pos = 0; pos < (int)elca.size() - 1; ++pos) {
                int u = elca[pos];
                int v = elca[pos + 1];
                bool has_edge = false;
                for (int neighbor : g[u]) {
                    if (neighbor == v) has_edge = true;
                }
                assert(has_edge || u == v);
                assert(abs(dlca[pos] - dlca[pos + 1]) == 1);
            }
        }
    }

    cerr << "Step1 Random Test Passed\n";
}

void run_random_tests2() {
    mt19937 rng(42);

    const int NUM_TESTS = 300;
    const int MAX_N = 100;
    const int NUM_QUERIES = 200;

    for (int t = 0; t < NUM_TESTS; ++t) {
        int N = uniform_int_distribution<int>(1, MAX_N)(rng);
        auto g = generate_random_tree(N, rng);
        int root = uniform_int_distribution<int>(0, N - 1)(rng);

        EulerTourBase et(N);
        for (int u = 0; u < N; ++u) {
            for (int v : g[u]) {
                if (u < v) {
                    et.add_edge(u, v);
                }
            }
        }
        et.build(root);

        // 初期値
        vector<ll> initial_values(N);
        for (int i = 0; i < N; ++i) {
            initial_values[i] = uniform_int_distribution<ll>(-1000, 1000)(rng);
        }

        EulerTourSubtree et_sub(&et, initial_values);

        // 愚直用データ
        vector<ll> brute = initial_values;
        BruteSubtreeHelper helper(N, g, root);
        helper.run();

        for (int q = 0; q < NUM_QUERIES; ++q) {
            int type = uniform_int_distribution<int>(0, 6)(rng);
            if (type == 0) {
                // get_vertex
                int v = uniform_int_distribution<int>(0, N - 1)(rng);
                ll act = et_sub.get_vertex(v);
                ll exp = brute[v];
                assert(act == exp);
            }
            else if (type == 1) {
                // set_vertex
                int v = uniform_int_distribution<int>(0, N - 1)(rng);
                ll val = uniform_int_distribution<ll>(-2000, 2000)(rng);
                et_sub.set_vertex(v, val);
                brute[v] = val;
            }
            else if (type == 2) {
                // apply_vertex
                int v = uniform_int_distribution<int>(0, N - 1)(rng);
                ll f = uniform_int_distribution<ll>(-100, 100)(rng);
                et_sub.apply_vertex(v, f);
                brute[v] += f;
            }
            else if (type == 3) {
                // query_subtree
                int v = uniform_int_distribution<int>(0, N - 1)(rng);
                ll act = et_sub.query_subtree(v);
                ll exp = 0;
                for (int u : helper.subtree_verts[v]) {
                    exp += brute[u];
                }
                assert(act == exp);
            }
            else if (type == 4) {
                // apply_subtree
                int v = uniform_int_distribution<int>(0, N - 1)(rng);
                ll f = uniform_int_distribution<ll>(-50, 50)(rng);
                et_sub.apply_subtree(v, f);
                for (int u : helper.subtree_verts[v]) {
                    brute[u] += f;
                }
            }
            else if (type == 5) {
                // query_range_vertex
                int l = uniform_int_distribution<int>(0, N - 1)(rng);
                int r = uniform_int_distribution<int>(l + 1, N)(rng);
                ll act = et_sub.query_range_vertex(l, r);
                ll exp = 0;
                for (int pos = l; pos < r; ++pos) {
                    int u = et.rev[pos];
                    exp += brute[u];
                }
                assert(act == exp);
            }
            else if (type == 6) {
                // apply_range_vertex
                int l = uniform_int_distribution<int>(0, N - 1)(rng);
                int r = uniform_int_distribution<int>(l + 1, N)(rng);
                ll f = uniform_int_distribution<ll>(-50, 50)(rng);
                et_sub.apply_range_vertex(l, r, f);
                for (int pos = l; pos < r; ++pos) {
                    int u = et.rev[pos];
                    brute[u] += f;
                }
            }
        }
    }

    cerr << "Step2 Random Test Passed\n";
}

void demo(){
    cout << "===== EulerTourSubtree（部分木の和・部分木への加算）デモ =====\n";
    // 木（根 0）: 0 の子が 1, 2, 3 / 1 の子が 4, 5 / 3 の子が 6
    int N = 7;
    EulerTourBase et(N);
    for (auto [u, v] : vector<pair<int,int>>{{0,1},{0,2},{0,3},{1,4},{1,5},{3,6}}) et.add_edge(u, v);
    et.build(0);                                    // 根 0 で前計算
    vector<ll> val = {10, 1, 2, 3, 4, 5, 6};
    EulerTourSubtree es(&et, val);                   // 頂点の値から作る
    cout << "es.query_subtree(1) = " << es.query_subtree(1) << "   // 1+4+5 = 10\n";
    es.apply_subtree(3, 100);                        // 頂点 3 の部分木（3, 6）に +100
    cout << "es.apply_subtree(3, 100) 後 es.query_subtree(0) = " << es.query_subtree(0) << "   // 31 + 200 = 231\n";
    es.set_vertex(4, 0);
    cout << "es.set_vertex(4, 0) 後 es.get_vertex(4) = " << es.get_vertex(4) << ", es.query_subtree(1) = " << es.query_subtree(1) << "\n";
    auto [l, r] = et.subtree_range(1);               // 部分木はオイラーツアーの位置で連続 [l, r)
    cout << "et.subtree_range(1) = [" << l << ", " << r << ")\n";
}
