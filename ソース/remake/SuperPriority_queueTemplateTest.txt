    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){

        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);

            int u, v;

            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }

    ){
        vector<vector<pair<int,long long>>> adj(n);

// OrderedPriorityQueue: multisetを使った順序付き拡張優先度付きキュー
// ======================= ここからテンプレート本体（コピー範囲） =======================
struct OrderedPriorityQueue {
    multiset<long long> s; // 要素を昇順に保持
    // --- 基本操作 ---
    void push(long long x) {
        s.insert(x); // O(log n)
    }

    void pop() {
        if(!s.empty()) s.erase(prev(s.end())); // 最大値を削除 O(log n)
    }

    long long top() {
        return *prev(s.end()); // 最大値を取得 O(1)
    }

    bool empty() {
        return s.empty(); // O(1)
    }

    int size() {
        return s.size(); // O(1)
    }

    // --- 順序操作 ---
    // x の前の値（predecessor）
    long long predecessor(long long x) {
        auto it = s.lower_bound(x);
        if(it == s.begin()) return LLONG_MIN; // predecessorなし
        return *prev(it); // O(log n)
    }

    // x の後の値（successor）
    long long successor(long long x) {
        auto it = s.upper_bound(x);
        if(it == s.end()) return LLONG_MAX; // successorなし
        return *it; // O(log n)
    }

    // x の順位（0-indexedで何番目か）
    int rank(long long x) {
        return distance(s.begin(), s.lower_bound(x)); // O(log n)
    }

    // k番目の値 (0-indexed)
    long long select(int k) {
        if(k < 0 || k >= size()) return LLONG_MIN;
        auto it = s.begin();
        advance(it, k);
        return *it; // O(log n) multisetでは線形だが平衡BSTならO(log n)
    }

    // x の個数
    int count(long long x) {
        return s.count(x); // O(log n)
    }

    // 最大値 / 最小値
    long long global_max() { return *prev(s.end()); } // O(1)
    long long global_min() { return *s.begin(); }     // O(1)

    // x を1つ削除
    void erase_value(long long x) {
        auto it = s.find(x);
        if(it != s.end()) s.erase(it); // O(log n)
    }

    // x を全て削除
    void erase_value_all(long long x) {
        s.erase(x); // O(log n)
    }

    void dump(){
        cout<<"現在の集合: ";
        for(auto v:s) cout<<v<<" ";
        cout<<"\n";
    }
};

// Treapノード
struct Node {
    long long val, sum, min_val, max_val;
    int size;
    int priority;
    Node *l, *r;
    Node(long long v) : val(v), sum(v), min_val(v), max_val(v), size(1), priority(rand()), l(nullptr), r(nullptr) {}
};

int getSize(Node* t) { return t ? t->size : 0; }
long long getSum(Node* t) { return t ? t->sum : 0; }
long long getMin(Node* t) { return t ? t->min_val : LLONG_MAX; }
long long getMax(Node* t) { return t ? t->max_val : LLONG_MIN; }

void update(Node* t) {
    if(!t) return;
    t->size = 1 + getSize(t->l) + getSize(t->r);
    t->sum = t->val + getSum(t->l) + getSum(t->r);
    t->min_val = min({t->val, getMin(t->l), getMin(t->r)});
    t->max_val = max({t->val, getMax(t->l), getMax(t->r)});
}

// Split by position: [0, k-1], [k, ...]
void split(Node* t, int k, Node*& l, Node*& r) {
    if(!t) { l = r = nullptr; return; }
    if(getSize(t->l) >= k) { 
        split(t->l, k, l, t->l); r = t; 
    } else { 
        split(t->r, k - getSize(t->l) - 1, t->r, r); l = t; 
    }
    update(t);
}

Node* merge(Node* l, Node* r) {
    if(!l || !r) return l ? l : r;
    if(l->priority > r->priority) { l->r = merge(l->r, r); update(l); return l; }
    else { r->l = merge(l, r->l); update(r); return r; }
}

// Treap-based priority queue with range query
struct TreapQueue {
    Node* root = nullptr;

    void push(long long x) { // push_back
        Node* n = new Node(x);
        root = merge(root, n);
    }

    void pop() { // pop_back
        Node *l, *r;
        split(root, getSize(root)-1, l, r);
        delete r;
        root = l;
    }

    long long top() { // back
        Node* t = root;
        while(t && t->r) t = t->r;
        return t->val;
    }

    bool empty() { return !root; }
    int size() { return getSize(root); }

    void insert(int pos, long long x) {
        Node *l, *r;
        split(root, pos, l, r);
        root = merge(merge(l, new Node(x)), r);
    }

    void erase_pos(int pos) {
        Node *l, *m, *r;
        split(root, pos, l, r);
        split(r, 1, m, r);
        delete m;
        root = merge(l, r);
    }

    // 区間クエリ [l, r) の sum / min / max
    tuple<long long,long long,long long> range_query(int lpos, int rpos) {
        Node *l, *m, *r;
        split(root, lpos, l, r);
        split(r, rpos-lpos, m, r);
        long long s = getSum(m), mn = getMin(m), mx = getMax(m);
        root = merge(merge(l, m), r);
        return {s, mn, mx};
    }

    // ノードの位置取得（0-indexed）
    int index(Node* node) {
        if(!node) return -1;
        int idx = getSize(node->l);
        Node* t = node;
        while(t != root) {
            // 親情報がない場合、TreapだけではO(n)探索
            // 実際には親ポインタを持つとO(log n)可能
            t = nullptr; // placeholder
        }
        return idx;
    }

    void dump(Node*t){
        if(!t)return;
        dump(t->l);
        cout<<t->val<<" ";
        dump(t->r);
    }

    void dump(){
        cout<<"現在の配列: ";
        dump(root);
        cout<<"\n";
    }
};

    cout<<"successor(10): "<<pq.successor(10)<<"\n";

    pq.erase_value(10);

    pq.dump();

    pq.erase_value_all(5);
    cout<<"5を全削除\n";
    pq.dump();

    pq.pop();
    cout<<"最大値をpop\n";
    pq.dump();

    cout<<"\n=============================\n";
    cout<<" TreapQueue 動作確認\n";
    cout<<"=============================\n";

    TreapQueue tq;

    tq.push(10);
    tq.push(20);
    tq.push(30);
    tq.dump();

    cout<<"末尾(top): "<<tq.top()<<"\n";

    tq.insert(1,99);
    cout<<"位置1に99を挿入\n";
    tq.dump();

    tq.erase_pos(2);
    cout<<"位置2を削除\n";
    tq.dump();

    cout<<"  最小 = "<<mn<<"\n";
    cout<<"  最大 = "<<mx<<"\n";

    tq.pop();
    cout<<"末尾pop\n";
    tq.dump();

    cout<<"\n===== 動作確認終了 =====\n";

    return;
}
