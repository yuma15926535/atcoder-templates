struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

//セグメント木（遅延なし）
//点更新、区間max
// ======================= ここからテンプレート本体（コピー範囲） =======================
// セグメント木（1 点更新・区間取得）。op / e を引数で渡すので、演算の違う木を同時にいくつも持てる
//   auto op = [](ll a, ll b){ return max(a, b); };   // ★ マージ関数
//   auto e  = [](){ return 0LL; };                    // ★ 単位元（S の型は e の返り値の型になる）
//   SegTree sg(N, op, e);
//   set(i, x) / get(i) / prod(l, r)（[l, r)）/ all_prod() / max_right(l, f) / min_left(r, f)
template<class Op, class E>
struct SegTree {
    using S = decltype(std::declval<E>()());   // 要素の型

    int n;
    int sz;
    vector<S> seg;
    Op op;
    E e;

    SegTree(int _n, Op _op, E _e) : op(_op), e(_e) {
        sz=_n;
        n = 1;
        while(n < sz) n <<= 1;
        seg.assign(2*n, e());
    }

    void build(const vector<S>& v){
        rep(i,0,sz) seg[i+n] = v[i];
        rrep(i,n,1) seg[i] = op(seg[i<<1], seg[i<<1|1]);
    }

    // 点更新
    void set(int i, S x){
        i += n;
        // ★ 更新
        seg[i] = x;
        while(i >>= 1){
            seg[i] = op(seg[i<<1], seg[i<<1|1]);
        }
    }

    // 1 点取得
    S get(int i) const { return seg[i + n]; }
    // 全体の集約
    S all_prod() const { return seg[1]; }
    // 区間取得 [l, r)
    S prod(int l, int r) const {
        S L = e(), R = e();
        for(l+=n, r+=n; l<r; l>>=1, r>>=1){
            if(l&1) L = op(L, seg[l++]);
            if(r&1) R = op(seg[--r], R);
        }
        return op(L, R);
    }

    // f(op(a[l..r))) == true となる最大の r を返す
    //fとしてboolを返却する関数を渡す
    //最小のrを返却したいならfalseとなる関数を渡す
    template<class F>
    int max_right(int l, F f) const {
        if(l == sz) return sz;
        S sm = e();
        l += n;
        do {
            while((l & 1) == 0) l >>= 1;
            if(!f(op(sm, seg[l]))){
                // このノード内で壊れる
                while(l < n){
                    l <<= 1;
                    if(f(op(sm, seg[l]))){
                        sm = op(sm, seg[l]);
                        l++;
                    }
                }
                return l - n;
            }
            sm = op(sm, seg[l]);
            l++;
        } while((l & -l) != l);
        return sz;
    }

    // f(op(a[l..r))) == true となる最小の l を返す
    template<class F>
    int min_left(int r, F f) const {
        if(r == 0) return 0;
        S sm = e();
        r += n;
        do {
            r--;
            while(r > 1 && (r & 1)) r >>= 1;
            if(!f(op(seg[r], sm))){
                while(r < n){
                    r = r<<1 | 1;
                    if(f(op(seg[r], sm))){
                        sm = op(seg[r], sm);
                        r--;
                    }
                }
                return r + 1 - n;
            }
            sm = op(seg[r], sm);
        } while((r & -r) != r);
        return 0;
    }
};

//  実戦での使用例（ALPC J）… 標準入力を読むので main からは呼ばない

int solve_ALPC_J(){
    BUFF;
    int N,Q;
    cin>>N>>Q;
    vector<ll>A(N);
    rep(i,0,N)cin>>A[i];
    auto op=[](ll a,ll b){ return max(a,b); };   // ★ max
    auto e=[](){ return 0LL; };                    // ★ 単位元
    SegTree sg(N,op,e);
    sg.build(A);
    rep(i,0,Q){
        int t,a,b;
        cin>>t>>a>>b;
        if(t==1)sg.set(a-1,b);
        else if(t==2)cout<<sg.prod(a-1,b)<<'\n';
        else cout<<sg.max_right(a-1,[&](ll e){return e<b;})+1<<'\n';
    }
    return 0;
}

// -----------------------------------------------------------------------------
//  使い方デモ（入力なしで動く）
// -----------------------------------------------------------------------------
void demo(){
    cout << "===== SegTree（op / e を引数で渡す版）デモ =====\n";
    auto op_max = [](ll a, ll b){ return max(a, b); };   // ★ マージ関数
    auto e_max  = [](){ return 0LL; };                    // ★ 単位元（値は 0 以上を想定）
    SegTree sg(6, op_max, e_max);
    sg.build({1, 5, 2, 4, 3, 6});
    cout << "sg.prod(0, 3) = " << sg.prod(0, 3) << "   // [0,3) の最大 5\n";
    sg.set(2, 9);
    cout << "sg.set(2, 9) 後 sg.prod(1, 4) = " << sg.prod(1, 4) << ", sg.get(2) = " << sg.get(2) << ", sg.all_prod() = " << sg.all_prod() << "\n";
    // max_right(l, f): f(op(a[l..r))) が true のままの最大の r（ここでは「最大値 < 6」が続く範囲）
    int r = sg.max_right(0, [](ll x){ return x < 6; });
    cout << "sg.max_right(0, x < 6) = " << r << "   // a[2]=9 で壊れるので r = 2\n";
    // ALPC J の「x 以上の値が最初に現れる位置」は max_right(l, v < x) の返り値
    int l = sg.min_left(6, [](ll x){ return x < 9; });
    cout << "sg.min_left(6, x < 9) = " << l << "   // 右から見て a[2]=9 の手前まで → 3\n";
    // 演算の違う木を同時に持てる（区間和の木）
    SegTree sum_tree(6, [](ll a, ll b){ return a + b; }, [](){ return 0LL; });
    sum_tree.build({1, 5, 2, 4, 3, 6});
    cout << "sum_tree.prod(0, 3) = " << sum_tree.prod(0, 3) << "   // 区間和 1+5+2 = 8（max の木 sg と同時に使える）\n";
}

int main(){
    BUFF;
    demo();
    return 0;
}
