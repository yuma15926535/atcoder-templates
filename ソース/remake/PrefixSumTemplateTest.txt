struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

//累積和(1D)
// ======================= ここからテンプレート本体（コピー範囲） =======================
struct PrefixSum1D {
    int N;
    vector<long long> ps; // ps[i]: [0, i) の和

    // 0-index の配列を渡す
    PrefixSum1D(const vector<long long>& a) {
        N = a.size();
        ps.assign(N + 1, 0);
        for (int i = 0; i < N; i++) {
            ps[i + 1] = ps[i] + a[i];
        }
    }

    // ===== 0-index =====

    // 半開区間 [l, r)
    long long sum(int l, int r) {
        return ps[r] - ps[l];
    }

    // 閉区間 [l, r]
    long long sum_closed(int l, int r) {
        return ps[r + 1] - ps[l];
    }

    // ===== 1-index =====

    // 半開区間 [l, r)（1-index）
    long long sum1(int l, int r) {
        // [l, r) -> [l-1, r-1)
        return ps[r - 1] - ps[l - 1];
    }

    // 閉区間 [l, r]（1-index）
    long long sum1_closed(int l, int r) {
        // [l, r] -> [l-1, r)
        return ps[r] - ps[l - 1];
    }
};

//累積和(2D)
struct PrefixSum2D {
    int H, W;
    vector<vector<long long>> ps;

    // a は 0-index の H×W 行列
    PrefixSum2D(const vector<vector<long long>>& a) {
        H = a.size();
        W = a[0].size();
        ps.assign(H + 1, vector<long long>(W + 1, 0));

        for (int i = 0; i < H; i++) {
            for (int j = 0; j < W; j++) {
                ps[i + 1][j + 1] =
                    ps[i][j + 1] + ps[i + 1][j] - ps[i][j] + a[i][j];
            }
        }
    }

    // ===== 0-index =====

    // 半開区間 [x1, y1) × [x2, y2)
    long long sum(int x1, int y1, int x2, int y2) {
        return ps[y1][y2]
             - ps[x1][y2]
             - ps[y1][x2]
             + ps[x1][x2];
    }

    // 閉区間 [x1, y1] × [x2, y2]
    long long sum_closed(int x1, int y1, int x2, int y2) {
        return sum(x1, y1 + 1, x2, y2 + 1);
    }

    // ===== 1-index =====

    // 半開区間 [x1, y1) × [x2, y2)
    long long sum1(int x1, int y1, int x2, int y2) {
        return sum(x1 - 1, y1 - 1, x2 - 1, y2 - 1);
    }

    // 閉区間 [x1, y1] × [x2, y2]
    long long sum1_closed(int x1, int y1, int x2, int y2) {
        return sum(x1 - 1, y1, x2 - 1, y2);
    }
};
// ======================= ここまでテンプレート本体 =======================

void solv(){
    // ===== PrefixSum1D 動作確認 =====
    vector<long long> a = {1, 2, 3, 4, 5};
    PrefixSum1D ps1(a);

    cout << "=== PrefixSum1D ===\n";
    cout << "array: ";
    for (auto x : a) cout << x << " ";
    cout << "\n";

    cout << "[0-index]\n";
    cout << "sum(1, 4)  = " << ps1.sum(1, 4)
         << " (2+3+4=9)\n";
    cout << "sum_closed(1, 3) = " << ps1.sum_closed(1, 3)
         << " (2+3+4=9)\n";

    cout << "[1-index]\n";
    cout << "sum1(2, 5)  = " << ps1.sum1(2, 5)
         << " (2+3+4=9)\n";
    cout << "sum1_closed(2, 4) = " << ps1.sum1_closed(2, 4)
         << " (2+3+4=9)\n";

    cout << "\n";

    // ===== PrefixSum2D 動作確認 =====
    vector<vector<long long>> mat = {
        {1, 2, 3},
        {4, 5, 6},
        {7, 8, 9}
    };
    PrefixSum2D ps2(mat);

    cout << "=== PrefixSum2D ===\n";
    cout << "matrix:\n";
    for (auto& row : mat) {
        for (auto x : row) cout << x << " ";
        cout << "\n";
    }

    cout << "[0-index]\n";
    cout << "sum(0,2,0,2)  = " << ps2.sum(0, 2, 0, 2)
         << " (1+2+4+5=12)\n";
    cout << "sum_closed(1,2,1,2) = " << ps2.sum_closed(1, 2, 1, 2)
         << " (5+6+8+9=28)\n";

    cout << "[1-index]\n";
    cout << "sum1(1,3,1,3)  = " << ps2.sum1(1, 3, 1, 3)
         << " (1+2+4+5=12)\n";
    cout << "sum1_closed(2,3,2,3) = " << ps2.sum1_closed(2, 3, 2, 3)
         << " (5+6+8+9=28)\n";
    return;
}
