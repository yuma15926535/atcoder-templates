struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
=====================================================
★ ノード情報（区間代入 + 区間max + 最大値個数）
=====================================================
*/

struct S {
    ll max_val;
    int cnt;
    int len;
};

/*
=====================================================
★ 更新情報
=====================================================
*/

struct F{
    ll val;
    bool flag;
};

/*
=====================================================
★ 演算定義
=====================================================
*/

// ノード結合
S op(S a, S b) {

    if(a.max_val > b.max_val){

        return {
            a.max_val,
            a.cnt,
            a.len + b.len
        };
    }

    if(a.max_val < b.max_val){

        return {
            b.max_val,
            b.cnt,
            a.len + b.len
        };
    }

    return {
        a.max_val,
        a.cnt + b.cnt,
        a.len + b.len
    };
}

// 更新適用
S mapping(F f, S x) {

    if(f.flag && x.len != 0){

        x.max_val = f.val;

        x.cnt = x.len;
    }

    return x;
}

// 更新の合成
// 先に f、その後に g
F composition(F f, F g) {

    if(g.flag)
        return g;

    return f;
}

S e() {
    return {-LINF, 0, 0};
}

F id() {
    return {0, false};
}

/*
=====================================================
Lazy SegTree 本体
=====================================================
*/

struct LazySegTree {

    int n;

    vector<S> seg;

    vector<F> lazy;

    LazySegTree(const vector<ll>& v) {

        int sz = v.size();

        n = 1;

        while (n < sz)
            n <<= 1;

        seg.assign(2 * n, e());

        lazy.assign(2 * n, id());

        for (int i = 0; i < sz; i++)
            seg[n + i] = {v[i], 1, 1};

        for (int i = n - 1; i > 0; i--)
            seg[i] = op(seg[i << 1], seg[i << 1 | 1]);
    }

    void apply_node(int k, F f) {

        seg[k] = mapping(f, seg[k]);

        if (k < n)
            lazy[k] = composition(lazy[k], f);
    }

    void push(int k) {

        if (!lazy[k].flag)
            return;

        apply_node(k << 1, lazy[k]);

        apply_node(k << 1 | 1, lazy[k]);

        lazy[k] = id();
    }

    // ---- 非再帰の区間作用・区間取得（ACL の lazy_segtree と同じやり方。再帰版より 2〜3 倍速い）----
    void apply_range(int l, int r, F f) {
        if (l >= r) return;
        const int lg = __builtin_ctz((unsigned)n);
        l += n; r += n;
        for (int i = lg; i >= 1; i--) {               // 上から遅延を下ろす
            if (((l >> i) << i) != l) push(l >> i);
            if (((r >> i) << i) != r) push((r - 1) >> i);
        }
        for (int a = l, b = r; a < b; a >>= 1, b >>= 1) {
            if (a & 1) apply_node(a++, f);
            if (b & 1) apply_node(--b, f);
        }
        for (int i = 1; i <= lg; i++) {               // 下から集約し直す
            if (((l >> i) << i) != l) { int k = l >> i; seg[k] = op(seg[k << 1], seg[k << 1 | 1]); }
            if (((r >> i) << i) != r) { int k = (r - 1) >> i; seg[k] = op(seg[k << 1], seg[k << 1 | 1]); }
        }
    }
    S prod_range(int l, int r) {
        if (l >= r) return e();
        const int lg = __builtin_ctz((unsigned)n);
        l += n; r += n;
        for (int i = lg; i >= 1; i--) {
            if (((l >> i) << i) != l) push(l >> i);
            if (((r >> i) << i) != r) push((r - 1) >> i);
        }
        S sml = e(), smr = e();
        for (; l < r; l >>= 1, r >>= 1) {
            if (l & 1) sml = op(sml, seg[l++]);
            if (r & 1) smr = op(seg[--r], smr);
        }
        return op(sml, smr);
    }

    void apply(int l, int r, ll x) {
        apply_range(l, r, F{x, true});
    }

    S prod(int l, int r) {
        return prod_range(l, r);
    }

    ll range_max(int l, int r) {
        return prod(l, r).max_val;
    }

    int range_count(int l, int r) {
        return prod(l, r).cnt;
    }
};

void random_test() {

    mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());

    rep(_,0,10000){

        // ========================
        // サイズ
        // ========================

        int N = rng()%20 + 1;

        vector<ll> A(N);

        rep(i,0,N)
            A[i] = rng()%21 - 10;

        // ========================
        // segtree 構築
        // ========================

        LazySegTree seg(A);

        // ========================
        // クエリ
        // ========================

        rep(q,0,1000){

            int type = rng()%2;

            // ====================================
            // 区間代入
            // ====================================

            if(type == 0){

                int l = rng()%N;
                int r = rng()%N;

                if(l > r)
                    swap(l,r);

                r++;

                ll x = rng()%21 - 10;

                // 愚直
                rep(i,l,r)
                    A[i] = x;

                // segtree
                seg.apply(l,r,x);
            }

            // ====================================
            // 区間取得
            // ====================================

            else{

                int l = rng()%N;
                int r = rng()%N;

                if(l > r)
                    swap(l,r);

                r++;

                // ------------------------
                // 愚直
                // ------------------------

                ll mx = -LINF;
                int cnt = 0;

                rep(i,l,r){

                    if(A[i] > mx){

                        mx = A[i];
                        cnt = 1;
                    }

                    else if(A[i] == mx){

                        cnt++;
                    }
                }

                // ------------------------
                // segtree
                // ------------------------

                auto res = seg.prod(l,r);

                // ------------------------
                // check
                // ------------------------

                if(res.max_val != mx || res.cnt != cnt){

                    cout << "ERROR\n";

                    cout << "N = " << N << endl;

                    cout << "A : ";

                    for(auto x : A)
                        cout << x << " ";

                    cout << endl;

                    cout << "l r = " << l << " " << r << endl;

                    cout << "expected : ";
                    cout << mx << " " << cnt << endl;

                    cout << "actual   : ";
                    cout << res.max_val << " " << res.cnt << endl;

                    exit(0);
                }
            }
        }
    }

    cout << "OK\n";
}

void demo(){
    cout << "===== 区間代入・区間 max とその個数 デモ =====\n";
    vector<ll> a = {1, 5, 2, 5, 3};
    LazySegTree seg(a);                       // 配列から構築 O(N)
    cout << "a = {1, 5, 2, 5, 3}\n";
    cout << "seg.range_max(0, 5) = " << seg.range_max(0, 5) << ", seg.range_count(0, 5) = " << seg.range_count(0, 5) << "   // 5 が 2 個\n";
    seg.apply(0, 3, 7);                   // a[0..2] = 7 に代入→ {7,7,7,5,3}
    cout << "seg.apply(0, 3, 7) 後: range_max = " << seg.range_max(0, 5) << ", range_count = " << seg.range_count(0, 5) << "   // 7 が 3 個\n";
}
