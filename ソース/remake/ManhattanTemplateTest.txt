struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

            }

            if(f) os<<", ";
            os<<x;
            f=1;
        }

    }
    return os;

//  マンハッタン距離 座標変換テンプレ（復元付き）
//  （作り直し版：元ファイル ManhattanTemplateTest.txt）
// -----------------------------------------------------------------------------
//  説明
//    - 座標を45度回転して持つことで長方形に帰着
//    - 最大マンハッタン距離
//    - 最大距離ペア復元
//    - 任意点からの最遠距離
//    - 任意点からの最遠距離復元
//    - マンハッタン距離制限
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
// ======================= ここからテンプレート本体（コピー範囲） =======================
/*
==================================================
 マンハッタン距離 座標変換テンプレ（復元付き）
==================================================
45度回転して座標を内部で持つことにより
・点集合の最大マンハッタン距離
・ある点からの最遠距離
・ひし形範囲判定
をO(1)で処理できる。
==================================================
*/
struct Manhattan {
    // 元の点を保存（復元用）
    vector<pair<ll,ll>> pts;

    // 変換後座標
    vector<ll> u, v;

    // 極値
    ll umin, umax;
    ll vmin, vmax;

    // 極値のインデックス
    int idx_umin, idx_umax;
    int idx_vmin, idx_vmax;

    //------------------------------------------------
    // コンストラクタ（空）
    //------------------------------------------------
    Manhattan() {
        umin = vmin = LLONG_MAX;
        umax = vmax = LLONG_MIN;
    }

    //------------------------------------------------
    // コンストラクタ（初期点集合）
    //------------------------------------------------
    Manhattan(const vector<pair<ll,ll>>& init) : Manhattan() {
        for(auto [x,y] : init) add_point(x,y);
    }

    //------------------------------------------------
    // 点追加
    //------------------------------------------------
    void add_point(ll x, ll y) {
        int id = pts.size();

        pts.push_back({x,y});

        ll uu = x + y;
        ll vv = x - y;

        u.push_back(uu);
        v.push_back(vv);

        if(uu < umin){ umin = uu; idx_umin = id; }
        if(uu > umax){ umax = uu; idx_umax = id; }

        if(vv < vmin){ vmin = vv; idx_vmin = id; }
        if(vv > vmax){ vmax = vv; idx_vmax = id; }
    }

    //------------------------------------------------
    // 最大マンハッタン距離
    //------------------------------------------------
    ll max_dist() const {
        if(pts.empty()) return 0;
        return max(umax - umin, vmax - vmin);
    }

    //------------------------------------------------
    // 最大距離ペアを復元
    //------------------------------------------------
    pair<pair<ll,ll>,pair<ll,ll>> max_pair() const {
        if(umax-umin >= vmax-vmin)
            return {pts[idx_umin], pts[idx_umax]};
        else
            return {pts[idx_vmin], pts[idx_vmax]};
    }

    //------------------------------------------------
    // 任意点からの最遠距離
    //------------------------------------------------
    ll farthest_dist(ll x, ll y) const {
        if(pts.empty()) return 0;

        ll uu = x+y;
        ll vv = x-y;

        return max({
            llabs(uu-umin),
            llabs(uu-umax),
            llabs(vv-vmin),
            llabs(vv-vmax)
        });
    }

    //------------------------------------------------
    // 任意点からの最遠点を復元
    //------------------------------------------------
    pair<ll,ll> farthest_point(ll x, ll y) const {
        ll uu = x+y;
        ll vv = x-y;

        vector<pair<ll,int>> cand = {
            {llabs(uu-umin), idx_umin},
            {llabs(uu-umax), idx_umax},
            {llabs(vv-vmin), idx_vmin},
            {llabs(vv-vmax), idx_vmax}
        };

        auto best = max_element(cand.begin(), cand.end());
        return pts[best->second];
    }

    //------------------------------------------------
    // ひし形範囲判定
    //------------------------------------------------
    static bool in_diamond(
        ll cx, ll cy, // 中心
        ll D,         // マンハッタン距離上限
        ll x, ll y    // 判定点
    ){
        ll u = x+y;
        ll v = x-y;
        ll u0 = cx+cy;
        ll v0 = cx-cy;

        return max(llabs(u-u0), llabs(v-v0)) <= D;
    }
};
// ======================= ここまでテンプレート本体 =======================

void solv(){

    cout << "=============================\n";
    cout << " マンハッタン距離テンプレ動作確認\n";
    cout << "=============================\n\n";

    vector<pair<long long,long long>> points = {
        {1,5},
        {4,2},
        {3,7},
        {0,0},
        {6,1}
    };

    Manhattan man(points);

    cout << "【登録された点】\n";
    for(int i=0;i<points.size();i++){
        cout << "点" << i << " : ("
             << points[i].first << ", "
             << points[i].second << ")\n";
    }

    cout << "\n-----------------------------\n";

    cout << "■ 点集合の最大マンハッタン距離\n";
    cout << "最大距離 = "
         << man.max_dist() << "\n";

    auto p = man.max_pair();
    cout << "そのペア = ("
         << p.first.first << "," << p.first.second
         << ") と ("
         << p.second.first << "," << p.second.second
         << ")\n";

    cout << "\n-----------------------------\n";

    long long qx = 2, qy = 3;

    cout << "■ 任意点 ("<<qx<<","<<qy<<") からの最遠距離\n";

    cout << "最遠距離 = "
         << man.farthest_dist(qx,qy) << "\n";

    auto fp = man.farthest_point(qx,qy);

    cout << "最遠点 = ("
         << fp.first << ","
         << fp.second << ")\n";

    cout << "\n-----------------------------\n";

    cout << "■ ひし形範囲判定\n";

    long long cx=3, cy=3, D=3;

    cout << "中心 = ("<<cx<<","<<cy<<")\n";
    cout << "距離制限 = "<<D<<"\n\n";

    vector<pair<ll,ll>> test = {
        {3,6},
        {5,5},
        {0,0}
    };

    for(auto [x,y] : test){
        bool ok = Manhattan::in_diamond(cx,cy,D,x,y);

        cout << "("<<x<<","<<y<<") → ";

        if(ok) cout << "範囲内\n";
        else cout << "範囲外\n";
    }

    cout << "\n=============================\n";
    cout << " 動作確認終了\n";
    cout << "=============================\n";
    return;
}

int main(){
    BUFF;
    int T=1;
    //cin>>T;
    while(T--)solv();
    return 0;
}
