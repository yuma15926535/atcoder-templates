/* ==================================================

// ---------- 0/1 ナップサック（YES / NO） ----------
// res[w] = 重さ w が作れるか
vector<bool> knapsack_bool(int N, int W, const vector<int>& wt) {
    vector<bool> dp(W+1, false);
    dp[0] = true;

    for (int i = 0; i < N; i++) {
        for (int w = W; w >= wt[i]; w--) {
            if (dp[w - wt[i]]) dp[w] = true;
        }
    }
    return dp;
}

// ---------- 完全ナップサック（YES / NO） ----------
vector<bool> knapsack_bool_unbounded(int N, int W, const vector<int>& wt) {
    vector<bool> dp(W+1, false);
    dp[0] = true;

    for (int i = 0; i < N; i++) {
        for (int w = wt[i]; w <= W; w++) {
            if (dp[w - wt[i]]) dp[w] = true;
        }
    }
    return dp;
}

// ---------- 個数制限付き（YES / NO） ----------
// O(N W)：used[w] = 重さ w を作るのにこの品物を使った最小の個数 として、個数が cnt[i] 以下のときだけ広げる
vector<bool> knapsack_bool_bounded(int N, int W,
                                   const vector<int>& wt,
                                   const vector<int>& cnt) {
    vector<bool> dp(W+1, false);
    dp[0] = true;
    vector<int> used(W+1);

    for (int i = 0; i < N; i++) {
        fill(used.begin(), used.end(), 0);
        for (int w = wt[i]; w <= W; w++) {
            if (!dp[w] && dp[w - wt[i]] && used[w - wt[i]] < cnt[i]) {
                dp[w] = true;
                used[w] = used[w - wt[i]] + 1;
            }
        }
    }
    return dp;
}

/* ==================================================
   2. YES / NO + 最小個数
   ================================================== */

// ---------- 0/1 ナップサック（最小個数） ----------
// res[w] = 重さ w を作るための最小使用個数（作れない場合 -1）
vector<int> knapsack_min_count(int N, int W, const vector<int>& wt) {
    const int INF = 1e9;
    vector<int> dp(W+1, INF);
    dp[0] = 0;

    for (int i = 0; i < N; i++) {
        for (int w = W; w >= wt[i]; w--) {
            dp[w] = min(dp[w], dp[w - wt[i]] + 1);
        }
    }

    vector<int> res(W+1, -1);
    for (int w = 0; w <= W; w++) {
        if (dp[w] != INF) res[w] = dp[w];
    }
    return res;
}

// ---------- 完全ナップサック（最小個数） ----------
vector<int> knapsack_min_count_unbounded(int N, int W, const vector<int>& wt) {
    const int INF = 1e9;
    vector<int> dp(W+1, INF);
    dp[0] = 0;

    for (int i = 0; i < N; i++) {
        for (int w = wt[i]; w <= W; w++) {
            dp[w] = min(dp[w], dp[w - wt[i]] + 1);
        }
    }

    vector<int> res(W+1, -1);
    for (int w = 0; w <= W; w++) {
        if (dp[w] != INF) res[w] = dp[w];
    }
    return res;
}

// ---------- 個数制限付き（最小個数） ----------
// 二進分割：個数 cnt を 1, 2, 4, …, 残り のまとまりに分けて 0/1 ナップサックにする。O(W × Σ log cnt)
vector<int> knapsack_min_count_bounded(int N, int W,
                                       const vector<int>& wt,
                                       const vector<int>& cnt) {
    const int INF = 1e9;
    vector<int> dp(W+1, INF);
    dp[0] = 0;

    for (int i = 0; i < N; i++) {
        int rest = cnt[i];
        for (int s = 1; rest > 0; s <<= 1) {
            int k = min(s, rest);   // k 個のまとまり（重さ k * wt、個数 k）
            rest -= k;
            long long kw = (long long)k * wt[i];
            if (kw > W) continue;
            for (int w = W; w >= kw; w--) {
                if (dp[w - kw] != INF) dp[w] = min(dp[w], dp[w - kw] + k);
            }
        }
    }

    vector<int> res(W+1, -1);
    for (int w = 0; w <= W; w++) {
        if (dp[w] != INF) res[w] = dp[w];
    }
    return res;
}

/* ==================================================
   3. 最大個数ナップサック（0/1）
   ================================================== */

// dp[w] = 重さ w を作るための最大使用個数
vector<int> knapsack_max_count(int N, int W, const vector<int>& wt) {
    const int NEG = -1e9;
    vector<int> dp(W+1, NEG);
    dp[0] = 0;

    for (int i = 0; i < N; i++) {
        for (int w = W; w >= wt[i]; w--) {
            if (dp[w - wt[i]] != NEG) {
                dp[w] = max(dp[w], dp[w - wt[i]] + 1);
            }
        }
    }
    return dp;
}

/* ==================================================
   4. 0/1 ナップサック（価値最大化 + 復元）
   ================================================== */

void knapsack01_restore(int N, int W,
                        const vector<int>& wt,
                        const vector<int>& val,
                        vector<int>& chosen) {
    vector<vector<long long>> dp(N+1, vector<long long>(W+1, -1));
    vector<vector<bool>> use(N+1, vector<bool>(W+1, false));

    dp[0][0] = 0;

    for (int i = 0; i < N; i++) {
        for (int w = 0; w <= W; w++) {
            if (dp[i][w] < 0) continue;

            // 使わない
            if (dp[i+1][w] < dp[i][w]) {
                dp[i+1][w] = dp[i][w];
                use[i+1][w] = false;
            }

            // 使う
            if (w + wt[i] <= W) {
                if (dp[i+1][w + wt[i]] < dp[i][w] + val[i]) {
                    dp[i+1][w + wt[i]] = dp[i][w] + val[i];
                    use[i+1][w + wt[i]] = true;
                }
            }
        }
    }

    int best_w = max_element(dp[N].begin(), dp[N].end()) - dp[N].begin();

    chosen.clear();
    for (int i = N; i >= 1; i--) {
        if (use[i][best_w]) {
            chosen.push_back(i-1);
            best_w -= wt[i-1];
        }
    }
    reverse(chosen.begin(), chosen.end());
}

// ---------- 価値最大化 + 復元 ----------
void knapsack_value_restore(int N, int W,
                            const vector<int>& wt,
                            const vector<int>& val,
                            vector<int>& chosen) {
    const long long INF = 4e18;
    int Vsum = accumulate(val.begin(), val.end(), 0);

    vector<vector<long long>> dp(N+1, vector<long long>(Vsum+1, INF));
    vector<vector<bool>> use(N+1, vector<bool>(Vsum+1, false));
    dp[0][0] = 0;

    for (int i = 0; i < N; i++) {
        for (int v = 0; v <= Vsum; v++) {
            if (dp[i][v] == INF) continue;

            if (dp[i+1][v] > dp[i][v]) {
                dp[i+1][v] = dp[i][v];
                use[i+1][v] = false;
            }
            if (v + val[i] <= Vsum &&
                dp[i+1][v + val[i]] > dp[i][v] + wt[i]) {
                dp[i+1][v + val[i]] = dp[i][v] + wt[i];
                use[i+1][v + val[i]] = true;
            }
        }
    }

    int best_v = 0;
    for (int v = 0; v <= Vsum; v++) {
        if (dp[N][v] <= W) best_v = v;
    }

    chosen.clear();
    for (int i = N; i >= 1; i--) {
        if (use[i][best_v]) {
            chosen.push_back(i-1);
            best_v -= val[i-1];
        }
    }
    reverse(chosen.begin(), chosen.end());
}

// ---------- 価値DP × 最小個数 ----------
// res[v] = 価値の合計をちょうど v にする最小の個数（作れない場合 -1。他の関数と同じ）
vector<int> knapsack_value_min_count(int N, const vector<int>& val) {
    const int INF = 1e9;
    int Vsum = accumulate(val.begin(), val.end(), 0);

    vector<int> dp(Vsum+1, INF);
    dp[0] = 0;

    for (int i = 0; i < N; i++) {
        for (int v = Vsum; v >= val[i]; v--) {
            dp[v] = min(dp[v], dp[v - val[i]] + 1);
        }
    }
    for (int& x : dp) if (x == INF) x = -1;
    return dp;
}

// ---------- 価値DP × 最小個数 + 復元 ----------
// 価値の合計をちょうど V にする最小個数を返し（作れなければ -1）、選んだ品物の番号（昇順）を chosen に入れる
int knapsack_value_min_count_restore(int N,
                                      const vector<int>& val,
                                      int V,
                                      vector<int>& chosen) {
    const int INF = 1e9;
    int Vsum = accumulate(val.begin(), val.end(), 0);
    chosen.clear();
    if (V < 0 || V > Vsum) return -1;

    vector<vector<int>> dp(N+1, vector<int>(Vsum+1, INF));
    vector<vector<bool>> use(N+1, vector<bool>(Vsum+1, false));
    dp[0][0] = 0;

    for (int i = 0; i < N; i++) {
        for (int v = 0; v <= Vsum; v++) {
            if (dp[i][v] == INF) continue;

            if (dp[i+1][v] > dp[i][v]) {
                dp[i+1][v] = dp[i][v];
                use[i+1][v] = false;
            }
            if (v + val[i] <= Vsum &&
                dp[i+1][v + val[i]] > dp[i][v] + 1) {
                dp[i+1][v + val[i]] = dp[i][v] + 1;
                use[i+1][v + val[i]] = true;
            }
        }
    }

    if (dp[N][V] == INF) return -1;
    int best_v = V;
    for (int i = N; i >= 1; i--) {
        if (use[i][best_v]) {
            chosen.push_back(i-1);
            best_v -= val[i-1];
        }
    }
    reverse(chosen.begin(), chosen.end());
    return dp[N][V];
}

   ================================================== */

// ---------- 0/1 通り数 ----------
long long knapsack_count(int N, int W, const vector<int>& wt, long long MOD) {
    vector<long long> dp(W+1, 0);
    dp[0] = 1;

    for (int i = 0; i < N; i++) {
        for (int w = W; w >= wt[i]; w--) {
            dp[w] = (dp[w] + dp[w - wt[i]]) % MOD;
        }
    }
    return dp[W];
}

// ---------- 完全ナップサック 通り数 ----------
long long knapsack_count_unbounded(int N, int W,
                                   const vector<int>& wt,
                                   long long MOD) {
    vector<long long> dp(W+1, 0);
    dp[0] = 1;

    for (int i = 0; i < N; i++) {
        for (int w = wt[i]; w <= W; w++) {
            dp[w] = (dp[w] + dp[w - wt[i]]) % MOD;
        }
    }
    return dp[W];
}

// =============================================================================
//  愚直解とのランダム比較テスト（作り直しで追加）
// =============================================================================
void random_test_bruteforce(){ mt19937 rng(6); int bad=0;
 for(int it=0;it<3000;it++){ int N=rng()%5+1, W=rng()%15; vector<int> wt(N),val(N),cnt(N); for(int i=0;i<N;i++){wt[i]=rng()%6+1; val[i]=rng()%10+1; cnt[i]=rng()%3+1;}
  // brute over 0/1 subsets
  vector<int> mn01(W+1,INT_MAX), mx01(W+1,-1); vector<long long> ways(W+1,0); long long bestv=0;
  for(int m=0;m<(1<<N);m++){ int w=0,v=0,c=0; for(int i=0;i<N;i++) if(m>>i&1){w+=wt[i];v+=val[i];c++;} if(w<=W){ mn01[w]=min(mn01[w],c); mx01[w]=max(mx01[w],c); ways[w]++; bestv=max(bestv,(long long)v);} }
  auto b=knapsack_bool(N,W,wt); auto mc=knapsack_min_count(N,W,wt); auto xc=knapsack_max_count(N,W,wt);
  for(int w=0;w<=W;w++){ if(b[w]!=(mn01[w]!=INT_MAX))bad++; if(mc[w]!=(mn01[w]==INT_MAX?-1:mn01[w]))bad++; if(mx01[w]>=0 && xc[w]!=mx01[w])bad++; }
  if(knapsack_count(N,W,wt,1000000007)!=ways[W]%1000000007)bad++;
  vector<int> ch; knapsack01_restore(N,W,wt,val,ch); long long sv=0,sw=0; for(int i:ch){sv+=val[i];sw+=wt[i];} if(sv!=bestv||sw>W)bad++;
  ch.clear(); knapsack_value_restore(N,W,wt,val,ch); sv=0;sw=0; for(int i:ch){sv+=val[i];sw+=wt[i];} if(sv!=bestv||sw>W)bad++;
  // unbounded / bounded via brute recursion on counts
  vector<int> ub(W+1,INT_MAX), bb(W+1,INT_MAX); vector<long long> uw(W+1,0);
  function<void(int,int,int,bool)> rec=[&](int i,int w,int c,bool bounded){ if(w>W)return; if(i==N){ if(bounded) bb[w]=min(bb[w],c); else {ub[w]=min(ub[w],c); uw[w]++;} return;} for(int k=0;;k++){ if(bounded&&k>cnt[i])break; if(w+k*wt[i]>W)break; rec(i+1,w+k*wt[i],c+k,bounded);} };
  rec(0,0,0,false); rec(0,0,0,true);
  auto bu=knapsack_bool_unbounded(N,W,wt), bbd=knapsack_bool_bounded(N,W,wt,cnt); auto mu=knapsack_min_count_unbounded(N,W,wt), mbd=knapsack_min_count_bounded(N,W,wt,cnt);
  for(int w=0;w<=W;w++){ if(bu[w]!=(ub[w]!=INT_MAX))bad++; if(bbd[w]!=(bb[w]!=INT_MAX))bad++; if(mu[w]!=(ub[w]==INT_MAX?-1:ub[w]))bad++; if(mbd[w]!=(bb[w]==INT_MAX?-1:bb[w]))bad++; }
  if(knapsack_count_unbounded(N,W,wt,1000000007)!=uw[W]%1000000007)bad++;
  // 価値DP × 最小個数（+ 復元）
  { int Vs=accumulate(val.begin(),val.end(),0); vector<int> want(Vs+1,INT_MAX);
    for(int m=0;m<(1<<N);m++){ int v=0,c=0; for(int i=0;i<N;i++) if(m>>i&1){v+=val[i];c++;} want[v]=min(want[v],c); }
    auto vm=knapsack_value_min_count(N,val); for(int v=0;v<=Vs;v++) if(vm[v]!=(want[v]==INT_MAX?-1:want[v]))bad++;
    int V=rng()%(Vs+2); vector<int> ch; int r=knapsack_value_min_count_restore(N,val,V,ch);
    int wv=(V<=Vs&&want[V]!=INT_MAX)?want[V]:-1; if(r!=wv)bad++;
    if(r>=0){ int sv2=0; for(int i:ch) sv2+=val[i]; if(sv2!=V||(int)ch.size()!=r)bad++; } else if(!ch.empty())bad++; }
 } 
    cout << "random_test_bruteforce（愚直解とのランダム比較）: " << (((bad)==0) ? "OK" : "NG") << "\n";
    assert((bad)==0);
}
