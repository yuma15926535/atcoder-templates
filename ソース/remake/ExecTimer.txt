struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

template<class T>
std::enable_if_t<is_iter_v<T>,ostream&>
operator<<(ostream&os,const T&v){
    using V = std::decay_t<decltype(*begin(v))>;
    if constexpr(is_iter_v<V>){
        bool first_row=true;
        os<<"\n";
        for(auto&&row:v){
            if(!first_row) os<<"\n";
            first_row=false;
            bool f=0;
            for(auto&&x:row){
                if(f) os<<" ";
                os<<x;
                f=1;
            }
        }
    }
    else{
        os<<"[";
        bool f=0;
        for(auto&&x:v){
            if(f) os<<", ";
            os<<x;
            f=1;
        }
        os<<"]";
    }
    return os;
}

template<class T,class...U>
void _dbg(const char*names,T&&x,U&&...y){
    const char*comma=strchr(names,',');
    if(!comma){
        cerr<<DBG_COLOR<<names<<" = "<<x<<"\n"<<DBG_RESET;
    }else{
        cerr<<DBG_COLOR;
        cerr.write(names,comma-names)<<" = "<<x<<" | ";
        _dbg(comma+1,y...);
    }
}

// ======================= ここからテンプレート本体（コピー範囲） =======================
struct ExecTimer {
    using Clock = std::chrono::steady_clock;
    Clock::time_point start;

    // コンストラクタで自動開始
    ExecTimer() {
        start = Clock::now();
    }

    // 計り直す（今から 0 ms）
    void reset() { start = Clock::now(); }

    // 開始からの経過時間（ミリ秒）。焼きなましなどで while (timer.elapsed_ms() < 1900) { … } のように使う
    long long elapsed_ms() const {
        return std::chrono::duration_cast<std::chrono::milliseconds>(Clock::now() - start).count();
    }
    // 経過時間（秒・小数）
    double elapsed_sec() const {
        return std::chrono::duration<double>(Clock::now() - start).count();
    }

    // 経過時間を標準エラーに表示する
    void end() const {
        std::cerr << "[Execution Time] " << elapsed_ms() << " ms\n";
    }
};
// ======================= ここまでテンプレート本体 =======================

void solv(){
    // 使い方：main の先頭で ExecTimer e; を作り、最後に e.end(); を呼ぶと実行時間が標準エラーに出る
    long long cnt = 0;
    rep(i, 0, 30000000) cnt += i % 7;
    cout << "cnt = " << cnt << "   // 実行時間は標準エラーに [Execution Time] xx ms と出る\n";

    // 時間いっぱい探索を回す（焼きなまし・乱択など）：経過時間を値で受け取る
    ExecTimer t;
    long long loops = 0;
    while (t.elapsed_ms() < 50) loops++;   // 50 ms だけ回す（実戦では 1900 ms など）
    cout << "50 ms 回した: elapsed_ms() = " << t.elapsed_ms() << " 以上, elapsed_sec() = " << (t.elapsed_sec() >= 0.05) << "（0.05 秒以上なら 1）, ループ回数 > 0 → " << (loops > 0) << "\n";
    t.reset();
    cout << "reset() 直後の elapsed_ms() = " << t.elapsed_ms() << "\n";
    return;
}

int main(){
    BUFF;
    ExecTimer e;
    int T=1;
    //cin>>T;
    while(T--)solv();
    e.end();
    return 0;
}
