struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// =============================================================================
//  DistantSelector
//  （作り直し版：元ファイル ARC182B.txt）
// -----------------------------------------------------------------------------
//  説明
//    - 区間の最大値の管理
//    - 最初に区間を指定
//    - getで呼び出し以下を満たす区間内の整数を返却
//    - getで返却された値を両端に持つ区間のうち最大の区間の中央値を返却
//    - データとして持つのが区間であるため範囲が広くてもgetの2倍程度にしかならない
//  メモ
//    - 区間長の最大を管理する場合priority_queueを使うのが最適
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
// getされた数を両端とする区間の最大の者の中央値を返却
// priority_queueを使う
// ======================= ここからテンプレート本体（コピー範囲） =======================
class DistantSelector {
    long long L, R, total_n;
    long long count = 0;

    // 区間 [l, r] を管理する構造体
    struct Segment {
        long long l, r;
        long long mid_dist; // 中央値から端までの距離

        Segment(long long _l, long long _r) : l(_l), r(_r) {
            // (r - l) / 2 が、中央に置いた時の両端からの距離になる
            mid_dist = (r - l) / 2;
        }

        // 優先度付きキューのための比較演算子
        // 1. 距離が遠いものを優先
        // 2. 距離が同じなら、左側の区間を優先（これは好みでOK）
        bool operator<(const Segment& other) const {
            if (mid_dist != other.mid_dist) {
                return mid_dist < other.mid_dist;
            }
            return l > other.l;
        }
    };

    std::priority_queue<Segment> pq;

public:
    DistantSelector(long long l, long long r) : L(l), R(r) {
        if (L <= R) {
            total_n = R - L + 1;
        } else {
            total_n = 0;
        }
    }

    long long get() {
        count++;

        // 1. すべての整数を出し切った後の処理
        if (count > total_n) {
            return L;
        }

        // 2. 1回目は最小値 L
        if (count == 1) {
            return L;
        }

        // 3. 2回目は最大値 R
        if (count == 2) {
            // 3回目以降のために、[L, R] という大きな隙間を登録
            if (R - L > 0) {
                pq.push(Segment(L, R));
            }
            return R;
        }

        // 4. 3回目以降：最も広い隙間の中央を選択
        while (!pq.empty()) {
            Segment top = pq.top();
            pq.pop();

            // 隙間に整数が残っていない場合は次を探す
            if (top.r - top.l <= 1) continue;

            long long mid = top.l + (top.r - top.l) / 2;

            // 新しく分割された2つの隙間をキューに追加
            if (mid - top.l > 1) pq.push(Segment(top.l, mid));
            if (top.r - mid > 1) pq.push(Segment(mid, top.r));

            return mid;
        }

        // 万が一のフォールバック（理論上ここには来ない）
        return L;
    }
};
// ======================= ここまでテンプレート本体 =======================

void solve_ARC182B(){
    int N,K;

}
