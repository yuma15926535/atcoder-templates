struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// =============================================================================
//  Moソート
//  （作り直し版：元ファイル MoSortTemplate.txt）
// -----------------------------------------------------------------------------
//  説明
//    - 1次元区間の形でクエリが与えられて区間の移動で答えを更新できる場合に使える
//    - 区間の移動距離の和を小さくなるように並び替える(最小ではない)
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
// ======================= ここからテンプレート本体（コピー範囲） =======================
struct Query {
    int l, r, idx;
};

/*
    Mo's Algorithm

    Usage:
        1. add_query(l, r) で区間 [l, r) のクエリを追加する。
        2. build() を呼ぶと、Mo順にソートされたクエリ列を取得できる。
        3. 取得したクエリを先頭から順に処理し、
           現在区間 [L, R) を add/remove により更新しながら答えを求める。

        4. または run(add, erase, out) / run(add_left, add_right, erase_left, erase_right, out) に
           「位置 i を入れる・外す」「今の答えをクエリ idx に記録する」関数を渡すと最後まで回してくれる。

    Complexity:
        build() : O(Q log Q)

        クエリ走査(add/removeが O(1) の場合):
            O(N * √Q + Q * N / √Q)  ＝ O(N √Q)（ブロックの大きさ N / √Q のとき）

        ※ add/remove の計算量を T とすると、さらに T 倍
*/
class Mo {
private:
    int n;
    int block_size;
    int fixed_block;   // 0 なら build() で N / √Q に決める
    vector<Query> qs;

public:
    // block を指定しなければ build() のときに max(1, N / √Q) にする（N と Q の差が大きくても移動の合計が小さい）
    explicit Mo(int n, int block = 0)
        : n(n), block_size(1), fixed_block(block) {}

    void add_query(int l, int r) {
        qs.push_back({l, r, (int)qs.size()});
    }

    int block() const { return block_size; }   // build() で決まったブロックの大きさ

    const vector<Query>& build() {
        int q = max<int>(1, qs.size());
        block_size = fixed_block > 0 ? fixed_block : max(1, (int)(n / sqrt((double)q)));
        sort(qs.begin(), qs.end(), [&](const Query& a, const Query& b) {
            int ab = a.l / block_size;
            int bb = b.l / block_size;
            if (ab != bb) return ab < bb;
            if (ab & 1) return a.r > b.r;
            return a.r < b.r;
        });
        return qs;
    }

    // Mo の順に区間を動かしながら全クエリを処理する
    //   add_left(i) / add_right(i)     : 位置 i を区間に入れる（左端から広げる・右端から広げる）
    //   erase_left(i) / erase_right(i) : 位置 i を区間から外す
    //   out(idx)                       : 今の区間がクエリ idx の区間になったときに呼ばれる（答えを記録する）
    //   区間は「広げてから縮める」ので、途中で区間が空より小さくなることはない
    template<class AL, class AR, class EL, class ER, class O>
    void run(AL add_left, AR add_right, EL erase_left, ER erase_right, O out) {
        build();
        int L = 0, R = 0;   // 今の区間 [L, R)
        for (const Query& q : qs) {
            while (L > q.l) add_left(--L);
            while (R < q.r) add_right(R++);
            while (L < q.l) erase_left(L++);
            while (R > q.r) erase_right(--R);
            out(q.idx);
        }
    }
    // 左右で同じ処理なら 3 つでよい
    template<class A, class E, class O>
    void run(A add, E erase, O out) { run(add, add, erase, erase, out); }
};
// ======================= ここまでテンプレート本体 =======================

void random_test() {
    mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());

    constexpr int TEST = 200000;

    for (int tc = 1; tc <= TEST; tc++) {
        int N = uniform_int_distribution<int>(1, 300)(rng);
        int Q = uniform_int_distribution<int>(1, 300)(rng);

        Mo mo(N);

        for (int i = 0; i < Q; i++) {
            int l = uniform_int_distribution<int>(0, N - 1)(rng);
            int r = uniform_int_distribution<int>(l + 1, N)(rng); // [l,r)
            mo.add_query(l, r);
        }

        auto &qs = mo.build();

        int B = mo.block();

        for (int i = 1; i < (int)qs.size(); i++) {
            int b1 = qs[i - 1].l / B;
            int b2 = qs[i].l / B;

            // ブロック番号は非減少
            assert(b1 <= b2);

            // 同じブロック内ではrの並びを確認
            if (b1 == b2) {
                if (b1 & 1)
                    assert(qs[i - 1].r >= qs[i].r);
                else
                    assert(qs[i - 1].r <= qs[i].r);
            }
        }
    }

    cout << "Random Test Passed!\n";
}

void demo(){
    cout << "===== Mo's Algorithm デモ（区間 [l, r) の値の種類数）=====\n";
    vector<int> a = {1, 2, 1, 3, 2, 2, 4};
    vector<pair<int,int>> qs = {{0, 3}, {2, 7}, {1, 5}, {4, 6}};
    vector<int> cnt(5, 0), ans(qs.size());
    int kinds = 0;
    auto add = [&](int i){ if (cnt[a[i]]++ == 0) kinds++; };
    auto remove = [&](int i){ if (--cnt[a[i]] == 0) kinds--; };

    // (1) run に任せる（ふつうはこちら）
    Mo mo(a.size());
    for (auto [l, r] : qs) mo.add_query(l, r);       // クエリを追加（[l, r)）
    mo.run(add, remove, [&](int idx){ ans[idx] = kinds; });
    for (int i = 0; i < (int)qs.size(); i++) cout << "[" << qs[i].first << ", " << qs[i].second << ") の種類数 = " << ans[i] << "\n";
    cout << "mo.block() = " << mo.block() << "   // N / √Q = 7 / 2 → 3\n";

    // (2) 自分でループを書く場合（build() で並べ替えた順に処理）
    Mo mo2(a.size());
    for (auto [l, r] : qs) mo2.add_query(l, r);
    fill(cnt.begin(), cnt.end(), 0); kinds = 0;
    int L = 0, R = 0;                                // 今の区間 [L, R)
    for (const Query& q : mo2.build()) {             // Mo の順に処理
        while (R < q.r) add(R++);                    // 先に広げてから
        while (L > q.l) add(--L);
        while (R > q.r) remove(--R);                 // 縮める
        while (L < q.l) remove(L++);
        ans[q.idx] = kinds;                          // 元のクエリ番号に答えを入れる
    }
    cout << "自分でループ: "; for (int x : ans) cout << x << " "; cout << "\n";
}

void random_test_run() {
    mt19937 rng(31);
    for (int tc = 0; tc < 2000; tc++) {
        int N = rng() % 30 + 1, Q = rng() % 30 + 1;
        vector<int> a(N);
        for (auto& x : a) x = rng() % 6;
        Mo mo(N);
        vector<pair<int,int>> qs;
        for (int i = 0; i < Q; i++) { int l = rng() % N, r = l + 1 + rng() % (N - l); qs.push_back({l, r}); mo.add_query(l, r); }
        vector<int> cnt(6, 0), ans(Q);
        int kinds = 0;
        mo.run([&](int i){ if (cnt[a[i]]++ == 0) kinds++; }, [&](int i){ if (--cnt[a[i]] == 0) kinds--; }, [&](int idx){ ans[idx] = kinds; });
        for (int i = 0; i < Q; i++) { set<int> st(a.begin() + qs[i].first, a.begin() + qs[i].second); assert(ans[i] == (int)st.size()); }
    }
    cout << "random_test_run（run の答えを愚直解と比較）: OK\n";
}

void solv(){
    random_test();
    random_test_run();
    demo();
    return;
}
