struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
==================================================
 Difference Array（差分配列 / 1次元 Imos）
==================================================

【用途】
- 区間加算・区間操作を高速に処理したいとき
- すべての更新後に最終配列が欲しいとき

【計算量】
- add : O(1)
- restore : O(N)

【区間】
- [l, r)（l以上 r未満）
*/
template <class T>
struct DifferenceArray {
    int n;
    vector<T> diff;


    DifferenceArray(int n_) : n(n_), diff(n + 1, T(0)) {}

    // [l,r)にxを加算
    void add(int l, int r, T x) {
        diff[l] += x;
        diff[r] -= x;
    }

    // 差分から元の配列を復元
    vector<T> restore() const {
        vector<T> a(n);
        T cur = 0;
        for (int i = 0; i < n; i++) {
            cur += diff[i];
            a[i] = cur;
        }
        return a;
    }
};

/*
==================================================
 Two Pointers / Sliding Window（判定関数差し替え型）
==================================================
【使い方】window の出し入れと判定をラムダで渡す
  two_pointers(n,
      [&](int i){ ... a[i] を window に入れる ... },        // add(i)
      [&](int i){ ... a[i] を window から出す ... },        // remove(i)
      [&]{ return ...今の window が条件を満たすか...; },    // ok()
      [&](int l, int r){ ... [l, r) で答えを更新 ... });    // on_window(l, r)
【動き】右端を 1 つずつ伸ばし、条件を満たさなくなったら満たすまで左端を縮める。
  on_window(l, r) は右端 r-1 ごとに「条件を満たす最長の window [l, r)」で呼ばれる
  （空の window で呼ばれることもある）。
【前提】条件を満たす window の部分区間も条件を満たす（単調性）
  例: 和 ≤ K（要素が 0 以上）/ 種類数 ≤ K / 最大値 − 最小値 ≤ K
【答えの例】最長: ans = max(ans, r - l) / 条件を満たす区間の個数: ans += r - l
【計算量】全体で O(N)（add / remove / ok が O(1) のとき）
*/
template <class Add, class Remove, class Ok, class OnWindow>
void two_pointers(int n, Add add, Remove remove, Ok ok, OnWindow on_window) {
    int l = 0;
    for (int r = 0; r < n; r++) {
        add(r);                                  // window [l, r] に r を入れる
        while (l <= r && !ok()) remove(l++);     // 条件を満たすまで左端を縮める
        on_window(l, r + 1);                     // [l, r + 1) は条件を満たす
    }
}

void demo(){
    cout << "===== DifferenceArray（いもす法）デモ =====\n";
    DifferenceArray<ll> da(6);
    da.add(1, 4, 5);                      // [1,4) に +5
    da.add(3, 6, 2);                      // [3,6) に +2
    auto a = da.restore();                // 最後にまとめて復元 O(N)
    cout << "da.restore() = "; for (ll x : a) cout << x << " "; cout << "  // 0 5 5 7 2 2\n";

    cout << "===== two_pointers デモ =====\n";
    vector<ll> v = {2, 1, 3, 4, 1, 1, 5};
    ll K = 6;
    // 例1: 和が K 以下の区間の最長と個数
    ll sum = 0, best = 0, cnt = 0;
    two_pointers((int)v.size(),
        [&](int i){ sum += v[i]; },
        [&](int i){ sum -= v[i]; },
        [&]{ return sum <= K; },
        [&](int l, int r){ best = max(best, (ll)(r - l)); cnt += r - l; });
    cout << "v = {2,1,3,4,1,1,5}, 和 <= 6 の最長 = " << best << "（{2,1,3} など）, 区間の個数 = " << cnt << "\n";
    // 例2: 最大値 − 最小値 <= 2 の最長（deque で index を持つ）
    deque<int> mx, mn; ll best2 = 0;
    two_pointers((int)v.size(),
        [&](int i){
            while (!mx.empty() && v[mx.back()] <= v[i]) mx.pop_back(); mx.push_back(i);
            while (!mn.empty() && v[mn.back()] >= v[i]) mn.pop_back(); mn.push_back(i);
        },
        [&](int i){
            if (!mx.empty() && mx.front() == i) mx.pop_front();
            if (!mn.empty() && mn.front() == i) mn.pop_front();
        },
        [&]{ return mx.empty() || v[mx.front()] - v[mn.front()] <= 2; },
        [&](int l, int r){ best2 = max(best2, (ll)(r - l)); });
    cout << "最大 − 最小 <= 2 の最長 = " << best2 << "   // {2,1,3}\n";
}

    return;
}
