struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

//整数の累乗
// ======================= ここからテンプレート本体（コピー範囲） =======================
//整数の累乗（溢れは確認しない。溢れが心配なら math_v2/00_math_base の pow_int / pow_cap）
ll pow_int(ll base,ll exp){
    ll result=1;
    while(exp>0){
        if(exp&1)result*=base;
        exp>>=1;
        if(exp) base*=base; // 最後は 2 乗しない（答えが収まるのに途中で溢れるのを防ぐ）
    }
    return result;
}
//整数の累乗(mod)  base^exp mod m（m は 2^62 程度まで。__int128 で掛ける）
ll mod_pow(ll base,ll exp,ll mod) {
    ll result=1%mod;
    base%=mod;
    if(base<0)base+=mod;
    while(exp>0){
        if(exp&1)result = (ll)((__int128)result * base % mod);
        base=(ll)((__int128)base*base%mod);
        exp>>=1;
    }
    return result;
}
//拡張ユークリッド: a x + b y = g = gcd(a, b) となる x, y を求めて g を返す
ll ext_gcd(ll a,ll b,ll &x,ll &y) {
    if(b==0){
        x=1;
        y=0;
        return a;
    }
    ll g=ext_gcd(b,a%b,y,x);
    y-=(a/b)*x;
    return g;
}
//整数の逆元(mod)  a x ≡ 1 (mod m) となる x（0 ≤ x < m）。m は素数でなくてよい。逆元が無ければ -1
ll mod_inv(ll a,ll mod){
    ll x,y;
    a%=mod; if(a<0)a+=mod;
    ll g=ext_gcd(a,mod,x,y);
    if(g!=1)return -1;// 逆元なし
    return (x%mod+mod)%mod;
}
//nCr を計算する mod（init_comb で設定。素数であること）
ll COMB_MOD;
//表の大きさ（n の最大値。init_comb で決まる）
int COMB_MAXN = -1;
//階乗・階乗の逆数
vector<ll> fact, invfact;
//二項係数計算の初期処理 O(maxn)。最初に 1 回 init_comb(998244353) / init_comb(998244353, 200000) のように呼ぶ
//  maxn：使う n の最大値（既定 2×10^6）。必要な分だけにすると速く、メモリも少ない
//  mod が maxn 以下の素数（例 mod = 3, 10007）なら表は mod - 1 までにして、n ≥ mod は Lucas の定理で計算する
void init_comb(ll mod, int maxn = 2000000){
    COMB_MOD=mod;
    COMB_MAXN=(int)min<ll>(maxn, mod-1);
    fact.assign(COMB_MAXN+1,0);
    invfact.assign(COMB_MAXN+1,0);
    fact[0]=1%mod;
    erep(i,1,COMB_MAXN)fact[i]=fact[i-1]*i%COMB_MOD;
    invfact[COMB_MAXN]=mod_inv(fact[COMB_MAXN],COMB_MOD);
    rrep(i,COMB_MAXN+1,1)invfact[i-1]=invfact[i]*i%COMB_MOD;
}
//nCrを返却(modが素数の場合)
//  n ≤ COMB_MAXN なら O(1)。mod が小さい素数（mod ≤ maxn）なら n が巨大でも Lucas の定理で O(log_mod n)
ll comb(ll n,ll r){
    if(r<0||r>n)return 0;
    if(n>COMB_MAXN){
        assert(COMB_MAXN==COMB_MOD-1 && "n が init_comb の maxn を超えている（mod が大きい素数のとき）");
        ll res=1%COMB_MOD;   // Lucas：mod 進数の各桁の nCr の積
        while(n>0||r>0){
            ll ni=n%COMB_MOD, ri=r%COMB_MOD;
            if(ri>ni)return 0;
            res=res*(fact[ni]*invfact[ri]%COMB_MOD*invfact[ni-ri]%COMB_MOD)%COMB_MOD;
            n/=COMB_MOD; r/=COMB_MOD;
        }
        return res;
    }
    return fact[n]*invfact[r]%COMB_MOD*invfact[n-r]%COMB_MOD;
}
// 重複組合せ(nHr) = (n+r-1)Cr
ll homo(ll n,ll r){
    if(r<0)return 0;
    if(n<=0)return (r == 0)?1%COMB_MOD:0;
    return comb(n+r-1,r);
}
// 順列(nPr) = n (n-1) … (n-r+1)
ll perm(ll n,ll r){
    if(r<0||r>n)return 0;
    if(n>COMB_MAXN){
        assert(COMB_MAXN==COMB_MOD-1 && "n が init_comb の maxn を超えている（mod が大きい素数のとき）");
        // mod が小さい素数：n, n-1, …, n-r+1 に mod の倍数が入れば 0、入らなければ余りだけで計算できる
        ll m=n%COMB_MOD;
        if(r>m)return 0;
        return fact[m]*invfact[m-r]%COMB_MOD;
    }
    return fact[n]*invfact[n - r]%COMB_MOD;
}
//nが小さい場合のnCr（r が小さければ n が巨大でもよい。O(r log mod)。1〜r が mod と互いに素であること）
ll comb_small(ll n,ll r,ll mod) {
    if(r<0||r>n)return 0;
    chmin(r,n-r);
    ll res=1;
    erep(i,1,r){
        res=(ll)((__int128)res*((n-i+1)%mod)%mod);
        res=(ll)((__int128)res*mod_inv(i,mod)%mod);
    }
    return res;
}

//  実戦での使用例（ABC368 D）… 標準入力を読むので main からは呼ばない

int solve_ABC368D(){
    int N,K;
    cin>>N>>K;
    vector<vector<int>>G(N);
    rep(i,0,N-1){
        int a,b;
        cin>>a>>b;
        a--;b--;
        G[a].push_back(b);
        G[b].push_back(a);
    }
    vector<int>degree(N,0);
    rep(i,0,N)degree[i]=G[i].size();
    queue<int>q;
    rep(i,0,N)if(degree[i]==1)q.push(i);
    unordered_set<int>V;
    rep(i,0,K){
        int v;
        cin>>v;
        v--;
        V.insert(v);
    }
    vector<bool>used(N,true);
    //特定の頂点を含む最小の部分木
    //特定の頂点でない葉を消しつづっける
    while(!q.empty()){
        int pos=q.front();
        q.pop();
        if(V.count(pos))continue;
        used[pos]=false;
        for(int nxt:G[pos])if(used[nxt]){
            degree[nxt]--;
            if(degree[nxt]==1)q.push(nxt);
        }
    }
    cout<<count(nall(used),true)<<'\n';
    return 0;
}

    cout << "ext_gcd(12, 18) = " << g << " (x=" << x << ", y=" << y << ")   // 12x + 18y = 6\n";
    init_comb(998244353, 200000);               // 最初に 1 回。使う n の最大値（ここでは 2×10^5）までの表を作る O(maxn)
    cout << "comb(5, 2) = " << comb(5, 2) << ", perm(5, 2) = " << perm(5, 2) << ", homo(3, 2) = " << homo(3, 2) << "   // 10, 20, 6\n";
    cout << "COMB_MAXN = " << COMB_MAXN << "   // 表の大きさ（これを超える n は assert で止まる）\n";

    cout << "comb_small(1000000000000, 3, 998244353) = " << comb_small(1000000000000LL, 3, 998244353) << "   // n が巨大でも r が小さければ OK\n";
}
