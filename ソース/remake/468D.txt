struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// =============================================================================
//  ExpandedString(文字列の中心から探索する場合に中心に文字の有無を疑似でなくす)
//  （作り直し版：元ファイル 468D.txt）
// -----------------------------------------------------------------------------
//  説明
//    - 仮想的に文字の間へ区切り文字('#')を挿入した文字列を扱う。
//    - 元の文字列 "abc" は "#a#b#c#" としてアクセスできる。
//    - 実際には文字列を生成せず、添字から対応する文字を返す。
//    - 回文判定・Manacher法など、奇数長/偶数長を統一して扱いたいときに使う。
//  ファイル構成
//    [テンプレート本体]  ← 「ここから」〜「ここまで」をコピーして Main_base.cpp に貼り付ける
//    それ以降            … テスト・使い方デモ（貼り付け不要）
// =============================================================================
// 仮想的に文字の間へ区切り文字('#')を挿入した文字列を扱う。
// 元の文字列 "abc" は "#a#b#c#" としてアクセスできる。
// 実際には文字列を生成せず、添字から対応する文字を返す。
// 回文判定・Manacher法など、奇数長/偶数長を統一して扱いたいときに使う。
// ======================= ここからテンプレート本体（コピー範囲） =======================
struct ExpandedString{
    string str;
    int sz;
    static constexpr char sep='#';

    ExpandedString(const string& S):str(S),sz(sizei(S)*2+1){}

    int size()const{
        return sz;
    }

    bool in(int idx)const{
        return 0<=idx&&idx<sz;
    }

    char get(int idx)const{
        if(!in(idx))return 0;
        return idx%2?str[idx/2]:sep;
    }

    char operator[](int idx)const{
        return get(idx);
    }
};
// ======================= ここまでテンプレート本体 =======================


// -----------------------------------------------------------------------------
//  実戦での使用例（ABC468 D）… 標準入力を読むので main からは呼ばない
// -----------------------------------------------------------------------------

void solve_ABC468D(){
    string S;
    cin>>S;
    ExpandedString es(S);
    ll ans=0;
    rep(i,sizei(es)){
        int cnt=0,tmp=0,add=0;
        while(1){
            if(!es.in(i-tmp))break;
            if(!es.in(i+tmp))break;
            if(es[i-tmp]!=es[i+tmp])cnt++;
            if(cnt==2)break;
            if(es[i-tmp]!='#')add++;
            tmp++;
        }
        ans+=add;
    }
    cout<<ans<<'\n';
    return;
}

void demo(){
    cout << "===== ExpandedString デモ =====\n";
    ExpandedString es("abc");             // 実際には "#a#b#c#" を作らずにアクセスできる
    cout << "es.size() = " << es.size() << "   // 2*3+1 = 7\n";
    cout << "es[0..6] = ";
    for(int i = 0; i < es.size(); i++) cout << es[i];
    cout << "   // 偶数番目は区切り '#'、奇数番目 i は元の文字 S[i/2]\n";
    cout << "es.in(7) = " << es.in(7) << ", es.get(-1) = " << (int)es.get(-1) << "   // 範囲外は in が false、get は 0\n";
    // 奇数長・偶数長の回文をまとめて中心から広げる（Manacher の素朴版）
    ExpandedString t("abba");
    int best = 0;
    for(int c = 0; c < t.size(); c++){
        int r = 0;
        while(t.in(c - r - 1) && t.in(c + r + 1) && t[c - r - 1] == t[c + r + 1]) r++;
        best = max(best, r);               // 拡張後の半径 r = 元の回文の長さ
    }
    cout << "\"abba\" の最長回文の長さ = " << best << "   // 4\n";
}
