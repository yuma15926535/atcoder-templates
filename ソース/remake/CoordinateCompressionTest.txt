struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

// 座標圧縮（テンプレート）
// ======================= ここからテンプレート本体（コピー範囲） =======================
template<class T>
struct CoordinateCompress {
    vector<T> xs;

    //座標圧縮対象
    void add(const T& x) { xs.push_back(x); }
    void add(const vector<T>& v) { for (auto &x : v) xs.push_back(x); }

    //座標圧縮
    void build() {
        sort(xs.begin(), xs.end());
        xs.erase(unique(xs.begin(), xs.end()), xs.end());
    }

    //引数の座標圧縮後の値(0-index)
    int get(const T& x) const {
        return lower_bound(xs.begin(), xs.end(), x) - xs.begin();
    }

    //引数の座標圧縮後の値(0-index) 存在しなければ-1
    int get_or_minus1(const T& x) const {
        auto it = lower_bound(xs.begin(), xs.end(), x);
        if (it == xs.end() || *it != x) return -1;
        return it - xs.begin();
    }

    //実際の値から座標圧縮後の値
    bool contains(const T& x) const {
        return binary_search(xs.begin(), xs.end(), x);
    }

    //以上の最小の座標圧縮後の値
    int lower_idx(const T& x) const {
        return lower_bound(xs.begin(), xs.end(), x) - xs.begin();
    }

    //より大きい最小の座標圧縮後の値
    int upper_idx(const T& x) const {
        return upper_bound(xs.begin(), xs.end(), x) - xs.begin();
    }

    //座標圧縮後の値から実際の値
    T rev(int i) const { return xs[i]; }

    //座標圧縮後の値の範囲から実際の値の範囲長
    T range_len(int l, int r) const {
        return xs[r] - xs[l];
    }

    int size() const { return xs.size(); }
};

void random_test_bruteforce(){ mt19937 rng(5); int bad=0;
 for(int it=0;it<3000;it++){ int n=rng()%8+1; vector<ll>a(n); for(auto&x:a)x=(ll)(rng()%10)-5; CoordinateCompress<ll> cc; cc.add(a); cc.build();
  vector<ll> u=a; sort(u.begin(),u.end()); u.erase(unique(u.begin(),u.end()),u.end());
  if(cc.size()!=(int)u.size())bad++;
  for(ll x=-6;x<=6;x++){ int lo=lower_bound(u.begin(),u.end(),x)-u.begin(), up=upper_bound(u.begin(),u.end(),x)-u.begin();
   if(cc.lower_idx(x)!=lo||cc.upper_idx(x)!=up)bad++; bool c=binary_search(u.begin(),u.end(),x); if(cc.contains(x)!=c)bad++;
   if(c && (cc.get(x)!=lo || cc.rev(lo)!=x))bad++; if(cc.get_or_minus1(x)!=(c?lo:-1))bad++; } }
 
    cout << "random_test_bruteforce（愚直解とのランダム比較）: " << (((bad)==0) ? "OK" : "NG") << "\n";
    assert((bad)==0);
}

void solv(){
    vector<long long> a = {100, 500, 100, -20};

    CoordinateCompress<long long> cc;

    // add（圧縮対象）
    cc.add(a);

    // build
    cc.build();

    cout << "=== original ===\n";
    for (auto x : a) cout << x << " ";
    cout << "\n\n";

    cout << "=== compressed ===\n";
    for (auto x : a) {
        cout << cc.get(x) << " ";
    }
    cout << "\n\n";

    cout << "=== rev (index -> value) ===\n";
    for (int i = 0; i < cc.size(); i++) {
        cout << i << " -> " << cc.rev(i) << "\n";
    }
    cout << "\n";

    cout << "=== contains ===\n";
    cout << "contains(100): " << cc.contains(100) << "\n";
    cout << "contains(200): " << cc.contains(200) << "\n\n";

    cout << "=== get_or_minus1 ===\n";
    cout << "get(500): " << cc.get_or_minus1(500) << "\n";
    cout << "get(200): " << cc.get_or_minus1(200) << "\n\n";

    cout << "=== lower_idx / upper_idx ===\n";
    cout << "lower_idx(100): " << cc.lower_idx(100) << "\n";
    cout << "upper_idx(100): " << cc.upper_idx(100) << "\n\n";

    cout << "=== range_len ===\n";
    int l = cc.get(-20);
    int r = cc.get(500);
    cout << "range_len(-20, 500): " << cc.range_len(l, r) << "\n";
    random_test_bruteforce();
    return;
}
