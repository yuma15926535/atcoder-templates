#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int N,M;
struct UnionFind
{
    vector<int> P;
    UnionFind(int n){
        P.resize(n,-1);
    }

    int find(int x){
        if(P[x]==-1)return x;
        else return P[x]=find(P[x]);
    }

    void unite(int x,int y){
        x=find(x);
        y=find(y);
        if(x!=y)P[x]=y;
    }
};

int main()
{
    cin>>N>>M;
    //単純でなくても無向グラフの連結成分の管理はUnion-Findで可能
    UnionFind uf=UnionFind(N);
    vector<int> mc(N,0);
    rep(i,0,M){
        int u,v;
        cin>>u>>v;
        u--;
        v--;
        uf.unite(u,v);
        mc[u]++;
        mc[v]++;
    }
    //親の連結成分の頂点の数と辺の数(2回カウントされることに注意)
    map<int,pair<int,int>> m;
    rep(i,0,N){
        if(m.count(uf.find(i))){
            auto p=m[uf.find(i)];
            m[uf.find(i)]={p.first+1,p.second+mc[i]};
        }else m[uf.find(i)]={1,mc[i]};
    }
    for(auto p:m){
        if(p.second.first!=p.second.second/2){
            NO;
            return 0;
        }
    }
    YES;
    return 0;
}
