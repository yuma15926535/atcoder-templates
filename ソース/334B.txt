ll M;
//植木算
ll ueki(ll x){
    ll ret=x/M;
    if(x>=0)ret++;
    else if(x%M==0)ret++;
    return ret;
}

int main()
{
    ll A,L,R;
    cin>>A>>M>>L>>R;
    L-=A;
    R-=A;
    cout<<ueki(R)-ueki(L-1)<<"\n";
    return 0;
}
