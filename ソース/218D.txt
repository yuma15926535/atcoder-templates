#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

bool comp(int a,int b){
    return a>b;
}

int main()
{
    int N;
    cin>>N;
    //デフォルトではunordered_setにpairを使えない
    //https://qiita.com/tqk/items/7c4df1eeb5ebaedcf714
    //以上のリンク先に解決法がある
    set<pair<int,int>> s;
    unordered_map<int,vector<int>> mx,my;
    rep(i,0,N){
        int x,y;
        cin>>x>>y;
        s.insert({x,y});
        if(mx.count(x))mx[x].push_back(y);
        else mx[x]={y};
        if(my.count(y))my[y].push_back(x);
        else my[y]={x};
    }
    ll ans=0;
    unordered_set<int> sx,sy;
    //左下の座標を固定し全探索
    for(auto p:s){
        if(mx.count(p.first)&&my.count(p.second)){
            if(!sx.count(p.first)){
                sort(nall(mx[p.first]),comp);
                sx.insert(p.first);
            }
            if(!sy.count(p.second)){
                sort(nall(my[p.second]),comp);
                sy.insert(p.second);
            }
            for(int y:mx[p.first]){
                if(y<=p.second)break;
                for(int x:my[p.second]){
                    if(x<=p.first)break;
                    if(s.count({x,y}))ans++;
                }
            }
        }
    }
    cout<<ans<<"\n";
    return 0;
}
