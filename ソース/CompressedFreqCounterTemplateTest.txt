struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
============================================================
 CompressedFreqCounter
------------------------------------------------------------
【概要】

座圧された Frequency Counter。

値の追加・削除を行いながら

・count(x)     : 出現回数
・distinct()   : 出現している値の種類数
・max_freq()   : 最大出現回数
・min_freq()   : 最小出現回数

を取得できる。

------------------------------------------------------------
【計算量】

N = 座圧後の種類数

add / remove / count : O(log N)
distinct             : O(1)
max_freq / min_freq  : O(1)

------------------------------------------------------------
【用途】

・sliding window distinct
・最頻値
・multiset管理
・頻度制約問題
============================================================
*/
struct CompressedFreqCounter {

    vector<ll> xs;                 // 座圧後の値
    vector<ll> cnt;                // 各値の出現回数

    map<ll,ll> freq_cnt;           // freq -> その頻度の値の種類数

    ll distinct_count = 0;         // 出現している値の種類数
    ll maxfreq = 0;                // 最大頻度
    ll minfreq = 0;                // 最小頻度（0は除く）

    //--------------------------------------------------------
    // コンストラクタ
    //--------------------------------------------------------
    CompressedFreqCounter(const vector<ll>& a){

        xs = a;

        sort(xs.begin(), xs.end());
        xs.erase(unique(xs.begin(), xs.end()), xs.end());

        cnt.assign(xs.size(),0);
    }

    //--------------------------------------------------------
    // 内部用：値 → index
    //--------------------------------------------------------
    int id(ll x) const{

        auto it = lower_bound(xs.begin(), xs.end(), x);

        if(it == xs.end() || *it != x) return -1;

        return it - xs.begin();
    }

    //--------------------------------------------------------
    // 値を1個追加
    //--------------------------------------------------------
    void add(ll x){

        int i = id(x);
        if(i==-1) return;

        ll old = cnt[i];
        ll now = old + 1;

        cnt[i] = now;

        if(old>0){
            freq_cnt[old]--;
            if(freq_cnt[old]==0) freq_cnt.erase(old);
        }else{
            distinct_count++;
        }

        freq_cnt[now]++;

        maxfreq = freq_cnt.rbegin()->first;
        minfreq = freq_cnt.begin()->first;
    }

    //--------------------------------------------------------
    // 値を1個削除（安全版）
    //--------------------------------------------------------
    void remove(ll x){

        int i = id(x);

        if(i==-1) return; 

        ll old = cnt[i];

        if(old==0) return;

        ll now = old - 1;

        cnt[i] = now;

        freq_cnt[old]--;
        if(freq_cnt[old]==0) freq_cnt.erase(old);

        if(now>0){
            freq_cnt[now]++;
        }else{
            distinct_count--;
        }

        if(freq_cnt.empty()){
            maxfreq = 0;
            minfreq = 0;
            return;
        }

        maxfreq = freq_cnt.rbegin()->first;
        minfreq = freq_cnt.begin()->first;
    }

    //--------------------------------------------------------
    // 出現回数
    //--------------------------------------------------------
    ll count(ll x) const{

        int i = id(x);

        if(i==-1) return 0; 

        return cnt[i];
    }

    //--------------------------------------------------------
    // 出現している値の種類数
    //--------------------------------------------------------
    ll distinct() const{

        return distinct_count;
    }

    //--------------------------------------------------------
    // 最大頻度
    //--------------------------------------------------------
    ll max_freq() const{

        return maxfreq;
    }

    //--------------------------------------------------------
    // 最小頻度（0を除く）
    //--------------------------------------------------------
    ll min_freq() const{

        return minfreq;
    }
};

void print_state(CompressedFreqCounter &fc){

    cout<<"-----------------------------\n";

    cout<<"distinct種類数 = "<<fc.distinct()<<"\n";
    cout<<"最大頻度 = "<<fc.max_freq()<<"\n";
    cout<<"最小頻度 = "<<fc.min_freq()<<"\n";

    cout<<"各値のカウント\n";

    for(ll x : fc.xs){
        cout<<"  "<<x<<" : "<<fc.count(x)<<"\n";
    }

    cout<<"-----------------------------\n";
}
