int main(){
    int N;
    cin>>N;
    vector<vector<ll>>cnt(2002,vector<ll>(2002,0)),rec(2002,vector<ll>(2002,0));
    //imos
    erep(i,1,N){
        int u,d,l,r;
        cin>>u>>d>>l>>r;
        d++;r++;
        cnt[u][l]++;
        cnt[d][r]++;
        cnt[u][r]--;
        cnt[d][l]--;
        rec[u][l]+=i;
        rec[d][r]+=i;
        rec[u][r]-=i;
        rec[d][l]-=i;
    }
    rep(i,0,2002)rep(j,0,2001)cnt[i][j+1]+=cnt[i][j],rec[i][j+1]+=rec[i][j];
    rep(i,0,2002)rep(j,0,2001)cnt[j+1][i]+=cnt[j][i],rec[j+1][i]+=rec[j][i];
    vector<ll>ans(N+1,0);
    erep(i,1,2000)erep(j,1,2000){
        if(cnt[i][j]==0)ans[0]++;
        else if(cnt[i][j]==1)ans[rec[i][j]]++;
    }
    erep(i,1,N)cout<<ans[0]+ans[i]<<'\n';
    return 0;
}
