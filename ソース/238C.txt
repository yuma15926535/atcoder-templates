pair<ll,ll> ppow(int d){
    ll n=1,rec;
    while(d>1){
        n*=10;
        d--;
    }
    rec=n;
    n*=10;
    n--;
    return {rec,n};
}

int main()
{
    ll N,M;
    cin>>N;
    M=N;
    ll ans=0;
    //rはmodmでの2の逆元
    ll m=998244353,r=499122177;
    vector<int> v;
    while(N){
        v.push_back(N%10);
        N/=10;
    }
    reverse(nall(v));
    erep(i,1,v.size()){
        auto p=ppow(i);
        if(i!=(int)v.size()){
            ll s=p.second-p.first+1;
            ans+=((((s%m)*((s+1)%m))%m)*r)%m;
            ans%=m;
        }else{
            ll s=M-p.first+1;
            ans+=((((s%m)*((s+1)%m))%m)*r)%m;
            ans%=m;
        }
    }
    cout<<ans<<"\n";
    return 0;
}
