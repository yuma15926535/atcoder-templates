int main(){
    int N;
    cin>>N;
    ll m=998244353;
    vector<int>f(N);
    rep(i,0,N){
        cin>>f[i];
        f[i]--;
    }
    int yet=-1;
    //visitedをboolでなくintにすることでサイクルを一度しかカウントしない
    vector<int>visited(N,yet);
    int cnt=0;
    rep(i,0,N)if(visited[i]==yet){
        int pos=i;
        while(1){
            visited[pos]=i;
            pos=f[pos];
            if(visited[pos]==i){
                cnt++;
                break;
            }else if(visited[pos]!=yet)break;
        }
    }
    ll ans=1;
    while(cnt){
        ans*=2;
        ans%=m;
        cnt--;
    }
    ans+=m-1;
    ans%=m;
    cout<<ans<<'\n';
    return 0;
}
