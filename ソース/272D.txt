#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main(){
    int N;
    ll M;
    cin>>N>>M;
    //進み方
    vector<pair<int,int>>dij;
    //原点からの距離がちょうど√Mであるときにdijに追加
    rep(i,-N,N)rep(j,-N,N)if(i*i+j*j==M)dij.push_back({i,j});
    int INF=1e10;
    //距離
    vector dist(N,vector<int>(N,INF));
    queue<pair<int,int>>q;
    //queueへの追加とdistへの記録を行う関数クラス内でを定義
    auto push = [&](int i, int j, int d) {
        //範囲チェック
        if (i<0 || j<0 || i>=N || j>=N) return;
        //BFSで探索するのでINFであるかのみ確かめればよい(距離が小さい順に探索する)
        if (dist[i][j] != INF) return;
        dist[i][j] = d;
        q.emplace(i,j);
    };
    //初期値
    push(0,0,0);
    //BFS
    while(!q.empty()){
        //変数の宣言、代入をpairで同時に行う
        auto [i,j]=q.front();
        q.pop();
        for(auto [di,dj]:dij)push(i+di,j+dj,dist[i][j]+1);
    }
    rep(i,0,N){
        rep(j,0,N){
            int d=dist[i][j];
            if(d==INF)d=-1;
            cout<<d<<" ";
        }
        cout<<"\n";
    }
    return 0;
}
