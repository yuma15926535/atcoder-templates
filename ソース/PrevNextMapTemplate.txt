struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/**
 * @brief 大小関係のあるキーと値を管理するMap
 *
 * キーに対応する値の管理に加え、
 * prev / prev_eq / next / next_eq による前後検索を提供する。
 *
 * キー型は long long 固定。
 *
 * 計算量:
 *   add         : O(logN)
 *   set         : O(logN)
 *   erase       : O(logN)
 *   contains    : O(logN)
 *   get         : O(logN)
 *   prev        : O(logN)
 *   prev_eq     : O(logN)
 *   next        : O(logN)
 *   next_eq     : O(logN)
 *   min         : O(1)
 *   max         : O(1)
 *   size        : O(1)
 *   empty       : O(1)
 *   clear       : O(N)
 */
template <class T>
struct PrevNextMap {
    /**
     * @brief キーと値を取得するための構造体
     *
     * exist == false の場合は検索失敗を表す。
     */
    struct Node {
        bool exist = false;
        ll key = 0;
        T value{};
    };

private:
    map<ll, T> mp;

public:
    //==================================================
    // Constructor
    //==================================================

    PrevNextMap() = default;

    //==================================================
    // Update
    //==================================================

    /**
     * @brief キーを追加する（存在する場合は上書き）
     *
     * @param key キー
     * @param value 値
     *
     * 計算量: O(logN)
     */
    void add(ll key, const T& value) {
        mp[key] = value;
    }

    /**
     * @brief 値を更新する
     *
     * キーが存在する場合のみ更新する。
     *
     * @param key キー
     * @param value 新しい値
     * @return 更新に成功したか
     *
     * 計算量: O(logN)
     */
    bool set(ll key, const T& value) {
        auto it = mp.find(key);
        if (it == mp.end()) return false;
        it->second = value;
        return true;
    }

    /**
     * @brief キーを削除する
     *
     * @param key キー
     * @return 削除できたか
     *
     * 計算量: O(logN)
     */
    bool erase(ll key) {
        return mp.erase(key) > 0;
    }

    //==================================================
    // Search
    //==================================================

    /**
     * @brief キーが存在するか判定する
     *
     * @param key キー
     * @return 存在するか
     *
     * 計算量: O(logN)
     */
    bool contains(ll key) const {
        return mp.find(key) != mp.end();
    }

    /**
     * @brief 指定キーの値を取得する
     *
     * @param key キー
     * @return 対応するNode
     *
     * exist == false の場合は存在しない。
     *
     * 計算量: O(logN)
     */
    Node get(ll key) const {
        auto it = mp.find(key);
        if (it == mp.end()) return {};

        return {true, it->first, it->second};
    }

    /**
     * @brief 指定キー未満の最大のキーを取得する
     *
     * @param key キー
     * @return 条件を満たすNode
     *
     * exist == false の場合は存在しない。
     *
     * 計算量: O(logN)
     */
    Node prev(ll key) const {
        auto it = mp.lower_bound(key);
        if (it == mp.begin()) return {};

        --it;
        return {true, it->first, it->second};
    }

     /**
      * @brief 指定キー以下の最大のキーを取得する
      *
      * @param key キー
      * @return 条件を満たすNode
      *
      * exist == false の場合は存在しない。
      *
      * 計算量: O(logN)
      */
    Node prev_eq(ll key) const {
        auto it = mp.upper_bound(key);
        if (it == mp.begin()) return {};

        --it;
        return {true, it->first, it->second};
    }

    /**
     * @brief 指定キーより大きい最小のキーを取得する
     *
     * @param key キー
     * @return 条件を満たすNode
     *
     * exist == false の場合は存在しない。
     *
     * 計算量: O(logN)
     */
    Node next(ll key) const {
        auto it = mp.upper_bound(key);
        if (it == mp.end()) return {};

        return {true, it->first, it->second};
    }

    /**
     * @brief 指定キー以上の最小のキーを取得する
     *
     * @param key キー
     * @return 条件を満たすNode
     *
     * exist == false の場合は存在しない。
     *
     * 計算量: O(logN)
     */
    Node next_eq(ll key) const {
        auto it = mp.lower_bound(key);
        if (it == mp.end()) return {};

        return {true, it->first, it->second};
    }

    /**
     * @brief 最小のキーを取得する
     *
     * @return 最小キーのNode
     *
     * exist == false の場合は要素が存在しない。
     *
     * 計算量: O(1)
     */
    Node min() const {
        if (mp.empty()) return {};

        auto it = mp.begin();
        return {true, it->first, it->second};
    }

    /**
     * @brief 最大のキーを取得する
     *
     * @return 最大キーのNode
     *
     * exist == false の場合は要素が存在しない。
     *
     * 計算量: O(1)
     */
    Node max() const {
        if (mp.empty()) return {};

        auto it = std::prev(mp.end());
        return {true, it->first, it->second};
    }

    //==================================================
    // Utility
    //==================================================

    /**
     * @brief 要素数を取得する
     *
     * @return 要素数
     *
     * 計算量: O(1)
     */
    int size() const {
        return (int)mp.size();
    }

    /**
     * @brief 空か判定する
     *
     * @return 空ならtrue
     *
     * 計算量: O(1)
     */
    bool empty() const {
        return mp.empty();
    }

    /**
     * @brief 全要素を削除する
     *
     * 計算量: O(N)
     */
    void clear() {
        mp.clear();
    }
};

void random_test_PrevNextMap() {
    cout << "===== PrevNextMap Random Test =====" << endl;

    mt19937 rng(0);

    PrevNextMap<int> a;
    map<ll, int> b;

    rep(_, 1000000) {
        int op = rng() % 10;
        ll key = rng() % 30 - 15;
        int value = rng() % 1000;

        if (op == 0) {
            // add
            a.add(key, value);
            b[key] = value;
        }
        else if (op == 1) {
            // set
            bool r1 = a.set(key, value);

            bool r2 = false;
            auto it = b.find(key);
            if (it != b.end()) {
                it->second = value;
                r2 = true;
            }

            assert(r1 == r2);
        }
        else if (op == 2) {
            // erase
            bool r1 = a.erase(key);
            bool r2 = b.erase(key);

            assert(r1 == (bool)r2);
        }
        else if (op == 3) {
            // contains
            assert(a.contains(key) == (b.count(key) != 0));
        }
        else if (op == 4) {
            // get
            auto x = a.get(key);
            auto it = b.find(key);

            if (it == b.end()) {
                assert(!x.exist);
            }
            else {
                assert(x.exist);
                assert(x.key == it->first);
                assert(x.value == it->second);
            }
        }
        else if (op == 5) {
            // prev
            auto x = a.prev(key);

            auto it = b.lower_bound(key);
            if (it == b.begin()) {
                assert(!x.exist);
            }
            else {
                --it;
                assert(x.exist);
                assert(x.key == it->first);
                assert(x.value == it->second);
            }
        }
        else if (op == 6) {
            // prev_eq
            auto x = a.prev_eq(key);

            auto it = b.upper_bound(key);
            if (it == b.begin()) {
                assert(!x.exist);
            }
            else {
                --it;
                assert(x.exist);
                assert(x.key == it->first);
                assert(x.value == it->second);
            }
        }
        else if (op == 7) {
            // next
            auto x = a.next(key);

            auto it = b.upper_bound(key);
            if (it == b.end()) {
                assert(!x.exist);
            }
            else {
                assert(x.exist);
                assert(x.key == it->first);
                assert(x.value == it->second);
            }
        }
        else if (op == 8) {
            // next_eq
            auto x = a.next_eq(key);

            auto it = b.lower_bound(key);
            if (it == b.end()) {
                assert(!x.exist);
            }
            else {
                assert(x.exist);
                assert(x.key == it->first);
                assert(x.value == it->second);
            }
        }
        else {
            // utility
            assert(a.size() == (int)b.size());
            assert(a.empty() == b.empty());

            auto mn = a.min();
            auto mx = a.max();

            if (b.empty()) {
                assert(!mn.exist);
                assert(!mx.exist);
            }
            else {
                assert(mn.exist);
                assert(mx.exist);

                assert(mn.key == b.begin()->first);
                assert(mn.value == b.begin()->second);

                auto it = prev(b.end());
                assert(mx.key == it->first);
                assert(mx.value == it->second);
            }

            // ときどきclear
            if (rng() % 20 == 0) {
                a.clear();
                b.clear();
            }
        }
    }

    cout << "ALL TEST PASSED" << endl;
}
