int main(){
    int N;
    cin>>N;
    modint::set_mod(998244353);
    vector<ll>A(N);
    rep(i,0,N)cin>>A[i];
    //minとmaxを固定する代わりにminのみを固定して考える
    sort(nall(A),[&](ll a,ll b){return a<b;});
    A.push_back(0);
    modint ans=0,sum=0;
    rrep(i,N,0){
        ans+=A[i]*A[i];
        sum*=2;
        sum+=A[i+1];
        ans+=A[i]*sum;
    }
    cout<<ans.val()<<'\n';
    return 0;
}
