//エラストテネスの篩(llver)
vector<ll> makepl(int n){
    vector<bool> rec(n+1,true);
    vector<ll> ret;
    erep(i,2,n)if(rec[i]){
        ret.push_back(i);
        for(ll j=2;i*j<=n;j++)rec[i*j]=false;
    }
    return ret;
}

int main()
{
    ll N;
    cin>>N;
    vector<ll> pl=makepl(1e6);
    ll ans=0;
    for(ll q:pl){
        if(q*q*q>N)break;
        for(ll p:pl){
            if(p*q*q*q>N||p>=q)break;
            ans++;
        }
    }
    cout<<ans<<"\n";
    return 0;
}
