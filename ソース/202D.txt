int main(){
    int A,B;
    ll K;
    cin>>A>>B>>K;
    //パスカルの三角形
    vector<vector<ll>>C(65,vector<ll>(65,0));
    C[0][0]=1;
    erep(i,0,60)rep(j,0,i+1){
        C[i+1][j] += C[i][j];
        C[i+1][j+1] += C[i][j];
    }
    K--;
    vector<char>ans(A+B);
    int idx=0;
    while(K!=0&&A!=0&&B!=0){
        if(C[A+B-1][A-1]<=K){
            ans[idx]='b';
            K-=C[A+B-1][A-1];
            B--;
        }else{
            ans[idx]='a';
            A--;
        }
        idx++;
    }
    while(A)ans[idx]='a',A--,idx++;
    while(B)ans[idx]='b',B--,idx++;
    for(char c:ans)cout<<c;
    cout<<'\n';
    return 0;
}
