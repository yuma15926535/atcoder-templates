#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    int N;
    cin>>N;
    vector<ll> T(N),K(N);
    vector<vector<int>> A(N);
    rep(i,0,N){
        cin>>T[i]>>K[i];
        A[i].resize(K[i]);
        rep(j,0,K[i])cin>>A[i][j];
    }
    ll ans=0;
    set<int> s;
    queue<int> q;
    q.push(N);
    s.insert(N);
    //グリッドでないがBFS
    while(!q.empty()){
        int a=q.front();
        q.pop();
        a--;
        ans+=T[a];
        for(int b:A[a])if(!s.count(b)){
            s.insert(b);
            q.push(b);
        }
    }
    cout<<ans<<"\n";
    return 0;
}
