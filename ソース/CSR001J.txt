// BIT(Fenwick Tree)
//点加算と区間和の取得を高速で
//転倒数を高速で
//0-index,1-indexに注意する
// ★ 型を変えたいときはここ
using T = long long;

struct BIT {
    int n;
    vector<T> bit;

    BIT(int _n=0){ init(_n); }

    void init(int _n){
        n = _n;
        bit.assign(n+1, 0);
    }

    // ★ 初期配列 a (0-index) から構築
    void build(const vector<T>& a){
        init(a.size());
        for(int i=0;i<n;i++){
            add(i+1, a[i]);
        }
    }

    // a[i] += x (1-indexed)
    // 点加算
    void add(int i, T x){
        for(; i <= n; i += i & -i){
            bit[i] += x;
        }
    }

    // sum[1..i] (1-indexed)
    // iまでの和
    T sum(int i){
        T s = 0;
        for(; i > 0; i -= i & -i){
            s += bit[i];
        }
        return s;
    }

    // sum[l..r] (1-indexed)
    // 区間和
    T range_sum(int l, int r){
        if(l > r) return 0;
        return sum(r) - sum(l-1);
    }

    // ===== 出現数 =====
    //出現数とk番目に小さい値を使う場合は、追加:add(val,1),削除:add(val,-1)
    T count(int x){
        return range_sum(x, x);
    }

    // ===== k番目に小さい値 =====
    // ★ n <= 2^20 想定（必要ならビット幅を変更）
    int kth(T k){
        int x = 0;
        for(int i = 1 << 20; i > 0; i >>= 1){
            if(x + i <= n && bit[x+i] < k){
                k -= bit[x+i];
                x += i;
            }
        }
        return x + 1;
    }
};

ll inversion_count(vector<ll> a){
    int n = a.size();

    // ===== 座標圧縮 =====
    vector<ll> v = a;
    sort(v.begin(), v.end());
    v.erase(unique(v.begin(), v.end()), v.end());

    for(auto &x : a){
        x = lower_bound(v.begin(), v.end(), x) - v.begin() + 1;
    }

    // ===== BIT =====
    BIT bit(v.size());

    ll inv = 0;
    for(int i=0;i<n;i++){
        inv += bit.range_sum(a[i]+1, v.size());
        bit.add(a[i], 1);
    }
    return inv;
}

int main(){
    int N;
    cin>>N;
    vector<ll>A(N);
    rep(i,0,N)cin>>A[i];
    //転倒数を高速で取得
    cout<<inversion_count(A)<<'\n';
    return 0;
}
