struct SubarrayAll {
    // ===== 最大（空あり）=====
    long long max_allow_sum;
    int max_allow_l, max_allow_r;

    // ===== 最大（空なし）=====
    long long max_force_sum;
    int max_force_l, max_force_r;

    // ===== 最小（空あり）=====
    long long min_allow_sum;
    int min_allow_l, min_allow_r;

    // ===== 最小（空なし）=====
    long long min_force_sum;
    int min_force_l, min_force_r;

    SubarrayAll(const vector<long long>& a) {
        int n = a.size();

        // ---------- 最大（空あり）----------
        {
            long long cur = 0;
            int cur_l = 0;
            max_allow_sum = 0;
            max_allow_l = max_allow_r = 0;

            for (int i = 0; i < n; i++) {
                cur += a[i];
                if (cur > max_allow_sum) {
                    max_allow_sum = cur;
                    max_allow_l = cur_l;
                    max_allow_r = i + 1;
                }
                if (cur < 0) {
                    cur = 0;
                    cur_l = i + 1;
                }
            }
        }

        // ---------- 最大（空なし）----------
        {
            long long cur = a[0];
            int cur_l = 0;
            max_force_sum = a[0];
            max_force_l = 0;
            max_force_r = 1;

            for (int i = 1; i < n; i++) {
                if (cur < 0) {
                    cur = a[i];
                    cur_l = i;
                } else {
                    cur += a[i];
                }
                if (cur > max_force_sum) {
                    max_force_sum = cur;
                    max_force_l = cur_l;
                    max_force_r = i + 1;
                }
            }
        }

        // ---------- 最小（空あり）----------
        {
            long long cur = 0;
            int cur_l = 0;
            min_allow_sum = 0;
            min_allow_l = min_allow_r = 0;

            for (int i = 0; i < n; i++) {
                cur += a[i];
                if (cur < min_allow_sum) {
                    min_allow_sum = cur;
                    min_allow_l = cur_l;
                    min_allow_r = i + 1;
                }
                if (cur > 0) {
                    cur = 0;
                    cur_l = i + 1;
                }
            }
        }

        // ---------- 最小（空なし）----------
        {
            long long cur = a[0];
            int cur_l = 0;
            min_force_sum = a[0];
            min_force_l = 0;
            min_force_r = 1;

            for (int i = 1; i < n; i++) {
                if (cur > 0) {
                    cur = a[i];
                    cur_l = i;
                } else {
                    cur += a[i];
                }
                if (cur < min_force_sum) {
                    min_force_sum = cur;
                    min_force_l = cur_l;
                    min_force_r = i + 1;
                }
            }
        }
    }
};

// 円環 Kadane 最大・最小部分列和（空あり / 空なし）
// 値 + 区間 [l, r) をすべて保持
// ※ l > r の場合は「末尾→先頭をまたぐ円環区間」
struct SubarrayAllCircular {
    int n;

    // ===== 円環 最大（空あり）=====
    long long max_allow_sum;
    int max_allow_l, max_allow_r;

    // ===== 円環 最大（空なし）=====
    long long max_force_sum;
    int max_force_l, max_force_r;

    // ===== 円環 最小（空あり）=====
    long long min_allow_sum;
    int min_allow_l, min_allow_r;

    // ===== 円環 最小（空なし）=====
    long long min_force_sum;
    int min_force_l, min_force_r;

    SubarrayAllCircular(const vector<long long>& a) {
        n = a.size();

        // 通常（非円環）Kadane をまず計算
        SubarrayAll base(a);

        // 全体和
        long long total = 0;
        for (long long x : a) total += x;

        // ==================================================
        // 円環 最大（空あり）
        // ・通常最大（空あり）
        // ・円環を使う場合 = total - 最小（空なし）
        // ==================================================
        {
            long long cand1 = base.max_allow_sum;                 // 円を使わない
            long long cand2 = total - base.min_force_sum;         // 円をまたぐ

            if (cand1 >= cand2) {
                max_allow_sum = cand1;
                max_allow_l = base.max_allow_l;
                max_allow_r = base.max_allow_r;
            } else {
                max_allow_sum = cand2;
                // 「最小区間を除いた残り」が答え
                max_allow_l = base.min_force_r;
                max_allow_r = base.min_force_l;
            }
        }

        // ==================================================
        // 円環 最大（空なし）
        // ※ 全部取るケースは禁止されない
        // ==================================================
        {
            long long cand1 = base.max_force_sum;
            long long cand2 = LLONG_MIN;

            // 最小区間が「全体」でない場合のみ円環を許可
            if (!(base.min_force_l == 0 && base.min_force_r == n)) {
                cand2 = total - base.min_force_sum;
            }

            if (cand1 >= cand2) {
                max_force_sum = cand1;
                max_force_l = base.max_force_l;
                max_force_r = base.max_force_r;
            } else {
                max_force_sum = cand2;
                max_force_l = base.min_force_r;
                max_force_r = base.min_force_l;
            }
        }

        // ==================================================
        // 円環 最小（空あり）
        // ・通常最小（空あり）
        // ・円環を使う場合 = total - 最大（空なし）
        // ==================================================
        {
            long long cand1 = base.min_allow_sum;                 // 非円環
            long long cand2 = total - base.max_force_sum;         // 円環

            if (cand1 <= cand2) {
                min_allow_sum = cand1;
                min_allow_l = base.min_allow_l;
                min_allow_r = base.min_allow_r;
            } else {
                min_allow_sum = cand2;
                min_allow_l = base.max_force_r;
                min_allow_r = base.max_force_l;
            }
        }

        // ==================================================
        // 円環 最小（空なし）
        // ==================================================
        {
            long long cand1 = base.min_force_sum;                 // 非円環
            long long cand2 = total - base.max_force_sum;         // 円環

            if (cand1 <= cand2) {
                min_force_sum = cand1;
                min_force_l = base.min_force_l;
                min_force_r = base.min_force_r;
            } else {
                min_force_sum = cand2;
                min_force_l = base.max_force_r;
                min_force_r = base.max_force_l;
            }
        }
    }
};

void solv(){
    // ===== テストケース =====
    // 円環で効いてくる典型例
    vector<vector<ll>> tests = {
        {1, -2, 3, -2},
        {5, -3, 5},
        {-3, -2, -3},
        {2, -1, 2, -1, 2},
        {1, -1, 1, -1, 1}
    };

    for (int tc = 0; tc < (int)tests.size(); tc++) {
        auto &a = tests[tc];
        int n = a.size();

        cout << "=============================\n";
        cout << "Test " << tc + 1 << "\n";
        cout << "Array: ";
        for (ll x : a) cout << x << " ";
        cout << "\n";

        SubarrayAll base(a);
        SubarrayAllCircular cir(a);

        // ---------- 非円環 ----------
        cout << "\n[Non-Circular]\n";
        cout << "max_allow : " << base.max_allow_sum
             << " [" << base.max_allow_l << "," << base.max_allow_r << ")\n";
        cout << "max_force : " << base.max_force_sum
             << " [" << base.max_force_l << "," << base.max_force_r << ")\n";
        cout << "min_allow : " << base.min_allow_sum
             << " [" << base.min_allow_l << "," << base.min_allow_r << ")\n";
        cout << "min_force : " << base.min_force_sum
             << " [" << base.min_force_l << "," << base.min_force_r << ")\n";

        // ---------- 円環 ----------
        cout << "\n[Circular]\n";
        cout << "max_allow : " << cir.max_allow_sum
             << " [" << cir.max_allow_l << "," << cir.max_allow_r << ")";
        if (cir.max_allow_l > cir.max_allow_r) cout << " (wrap)";
        cout << "\n";

        cout << "max_force : " << cir.max_force_sum
             << " [" << cir.max_force_l << "," << cir.max_force_r << ")";
        if (cir.max_force_l > cir.max_force_r) cout << " (wrap)";
        cout << "\n";

        cout << "min_allow : " << cir.min_allow_sum
             << " [" << cir.min_allow_l << "," << cir.min_allow_r << ")";
        if (cir.min_allow_l > cir.min_allow_r) cout << " (wrap)";
        cout << "\n";

        cout << "min_force : " << cir.min_force_sum
             << " [" << cir.min_force_l << "," << cir.min_force_r << ")";
        if (cir.min_force_l > cir.min_force_r) cout << " (wrap)";
        cout << "\n";
    }
    return;
}
