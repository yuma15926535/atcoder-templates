struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

template<class T>
struct CoordinateCompress {
    vector<T> xs;

    //座標圧縮対象
    void add(const T& x) { xs.push_back(x); }
    void add(const vector<T>& v) { for (auto &x : v) xs.push_back(x); }

    //座標圧縮
    void build() {
        sort(xs.begin(), xs.end());
        xs.erase(unique(xs.begin(), xs.end()), xs.end());
    }

    //引数の座標圧縮後の値(0-index)
    int get(const T& x) const {
        return lower_bound(xs.begin(), xs.end(), x) - xs.begin();
    }

    //引数の座標圧縮後の値(0-index) 存在しなければ-1
    int get_or_minus1(const T& x) const {
        auto it = lower_bound(xs.begin(), xs.end(), x);
        if (it == xs.end() || *it != x) return -1;
        return it - xs.begin();
    }

    //実際の値から座標圧縮後の値
    bool contains(const T& x) const {
        return binary_search(xs.begin(), xs.end(), x);
    }

    //以上の最小の座標圧縮後の値
    int lower_idx(const T& x) const {
        return lower_bound(xs.begin(), xs.end(), x) - xs.begin();
    }

    //より大きい最小の座標圧縮後の値
    int upper_idx(const T& x) const {
        return upper_bound(xs.begin(), xs.end(), x) - xs.begin();
    }

    //座標圧縮後の値から実際の値
    T rev(int i) const { return xs[i]; }

    //座標圧縮後の値の範囲から実際の値の範囲長
    T range_len(int l, int r) const {
        return xs[r] - xs[l];
    }

    int size() const { return xs.size(); }
};

struct LinkedList {
    int n;
    vector<int> nxt, prv;

    LinkedList(int n) : n(n), nxt(n+1, -1), prv(n+1, -1) {}

    // x の後ろに y をつなぐ（x -> y）
    void link(int x, int y) {
        nxt[x] = y;
        if (y != -1) prv[y] = x;
    }

    // x をリストから外す
    void cut(int x) {
        int p = prv[x];
        int q = nxt[x];
        if (p != -1) nxt[p] = q;
        if (q != -1) prv[q] = p;
        prv[x] = nxt[x] = -1;
    }

    // x の後ろに y を挿入（y は単体 or 別リストでもOK）
    void insert_after(int x, int y) {
        int z = nxt[x];
        link(x, y);
        link(y, z);
    }

    // 区間 [l, r] を切り出す（l と r は同一リスト内）
    // 戻り値：(前のノード, 次のノード)
    pair<int,int> cut_segment(int l, int r) {
        int p = prv[l];
        int q = nxt[r];
        if (p != -1) nxt[p] = q;
        if (q != -1) prv[q] = p;
        prv[l] = nxt[r] = -1;
        return {p, q};
    }

    // 区間 [l, r] を x の後ろに挿入
    void insert_segment(int x, int l, int r) {
        int z = nxt[x];
        link(x, l);
        link(r, z);
    }
};

void solv(){
    int N;
    cin>>N;
    vector<int>A(N);
    vector<int>num(N);
    cin>>A;
    rep(i,0,N)num[i]=A[i];
    CoordinateCompress<int> cc;
    cc.add(num);
    int Q;
    cin>>Q;
    vector<qxy>query(Q);
    rep(i,0,Q){
        int q,x,y=0;
        cin>>q>>x;
        if(q==1){
            cin>>y;
            num.push_back(y);
            cc.add(y);
        }
        query[i]={q,x,y};
    }
    cc.build();
    LinkedList llist(cc.size());
    llist.prv[cc.get(A[0])]=-2;
    if(N>1)llist.nxt[cc.get(A[0])]=cc.get(A[1]);
    llist.nxt[cc.get(A[N-1])]=-2;
    if(N>1)llist.prv[cc.get(A[N-1])]=cc.get(A[N-2]);
    rep(i,1,N-1){
        llist.prv[cc.get(A[i])]=cc.get(A[i-1]);
        llist.nxt[cc.get(A[i])]=cc.get(A[i+1]);
    }
    int s=cc.get(A[0]);
    rep(i,0,Q){
        auto [q,x,y]=query[i];
        if(q==1){
            if(llist.nxt[cc.get(x)]==-2){
                llist.nxt[cc.get(x)]=cc.get(y);
                llist.prv[cc.get(y)]=cc.get(x);
                llist.nxt[cc.get(y)]=-2;
            }else{
                int n=llist.nxt[cc.get(x)];
                llist.nxt[cc.get(x)]=cc.get(y);
                llist.prv[cc.get(y)]=cc.get(x);
                llist.nxt[cc.get(y)]=n;
                llist.prv[n]=cc.get(y);
            }
        }else{
            int p=llist.prv[cc.get(x)];
            int n=llist.nxt[cc.get(x)];
            if(p==-2){
                llist.prv[n]=-2;
                s=n;
            }else if(n==-2){
                llist.nxt[p]=-2;
            }else{
                llist.prv[n]=p;
                llist.nxt[p]=n;
            }
            llist.prv[cc.get(x)]=-1;
            llist.nxt[cc.get(x)]=-1;
        }
    }
    while(llist.nxt[s]!=-2){
        cout<<cc.rev(s)<<' ';
        s=llist.nxt[s];
    }
    cout<<cc.rev(s)<<'\n';
    return;
}

int main(){
    BUFF;
    int T=1;
    //cin>>T;
    while(T--)solv();
    return 0;
}
