template<class T>

operator<<(ostream&os,const T&v){
    using V = std::decay_t<decltype(*begin(v))>;
    if constexpr(is_iter_v<V>){
        bool first_row=true;
        os<<"\n";
        for(auto&&row:v){
            if(!first_row) os<<"\n";
            first_row=false;
            bool f=0;
            for(auto&&x:row){
                if(f) os<<" ";
                os<<x;
                f=1;
            }
        }
    }
    else{
        os<<"[";
        bool f=0;
        for(auto&&x:v){
            if(f) os<<", ";
            os<<x;
            f=1;
        }
        os<<"]";
    }
    return os;
}

template<class T>
struct FastBoundMultiset {
private:
    multiset<T> ms;

public:

    // =========================
    // 基本操作
    // =========================

    void insert(const T& x) {
        ms.insert(x);
    }

    void erase_one(const T& x) {
        auto it = ms.find(x);
        if (it != ms.end()) ms.erase(it);
    }

    int size() const { return ms.size(); }
    bool empty() const { return ms.empty(); }

    // =========================
    // x以上 (>=)
    // =========================
    bool ge(const T& x, T& out) const {
        auto it = ms.lower_bound(x);
        if (it == ms.end()) return false;
        out = *it;
        return true;
    }

    // =========================
    // xより大きい (>)
    // =========================
    bool gt(const T& x, T& out) const {
        auto it = ms.upper_bound(x);
        if (it == ms.end()) return false;
        out = *it;
        return true;
    }

    // =========================
    // x以下 (<=)
    // =========================
    bool le(const T& x, T& out) const {
        auto it = ms.upper_bound(x);
        if (it == ms.begin()) return false;
        --it;
        out = *it;
        return true;
    }

    // =========================
    // xより小さい (<)
    // =========================
    bool lt(const T& x, T& out) const {
        auto it = ms.lower_bound(x);
        if (it == ms.begin()) return false;
        --it;
        out = *it;
        return true;
    }

    // =========================
    // 最小値 O(1)
    // =========================
    bool get_min(T& out) const {
        if (ms.empty()) return false;
        out = *ms.begin();
        return true;
    }

    // =========================
    // 最大値 O(1)
    // =========================
    bool get_max(T& out) const {
        if (ms.empty()) return false;
        out = *ms.rbegin();
        return true;
    }

    // =========================
    // 高速 count
    // O(log N)
    // =========================
    int count_fast(const T& x) const {
        auto l = ms.lower_bound(x);
        if (l == ms.end() || *l != x) return 0;
        auto r = ms.upper_bound(x);
        return distance(l, r);
    }

    // =========================
    // 高速 存在判定
    // O(log N)
    // =========================
    bool contains(const T& x) const {
        return ms.find(x) != ms.end();
    }

    // =========================
    // 高速 範囲要素存在判定
    // 判定範囲:[l,r]
    // O(log N)
    // =========================
    bool exists_range(const T& l, const T& r) const {
        auto it = ms.lower_bound(l);
        return it != ms.end() && *it <= r;
    }
};

void solv(){
    int N;
    ll D;
    cin>>N>>D;
    vector<ll>A(N);
    rep(i,0,N)cin>>A[i];
    FastBoundMultiset<ll>fbm;
    int left=0,right=0;
    ll ans=0;
    while(left<N){
        if(fbm.size()==0)fbm.insert(A[left]);
        if(right<left)right=left;
        while(right+1<N){
            if(fbm.count_fast(A[right+1])!=0)break;
            ll g=-1,l=-1;
            fbm.gt(A[right+1],g);
            fbm.lt(A[right+1],l);
            if(g!=-1&&g-A[right+1]<D)break;
            if(l!=-1&&A[right+1]-l<D)break;
            right++;
            fbm.insert(A[right]);
        }
        ans+=right-left+1;
        debug(left,right);
        fbm.erase_one(A[left]);
        left++;
    }
    cout<<ans<<'\n';
    return;
}

    while(T--)solv();
