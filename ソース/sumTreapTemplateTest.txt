/**
 * Treap（乱択平衡二分探索木）+ sum拡張版
 * 
 * 【概要】
 * - 順序付き多重集合（multiset）をTreapで実装
 * - 各ノードに部分木の「サイズ」と「総和」を持たせる
 * - 期待計算量：各操作 O(log N)
 * 
 * 【できること】
 * - insert / erase（多重集合）
 * - kth（k番目の値）
 * - order_of_key（未満個数）
 * - lower_bound / upper_bound / prev / next
 * 
 * 【sum系】
 * - sum_k_smallest（小さい方からk個の和）
 * - sum_k_largest（大きい方からk個の和）
 * - median / median_cost（中央値・絶対値和）
 * 
 * 【区間系（indexベース）】
 * - split_kth による分割
 * - range_sum
 * - range_kth
 * - range_median
 * - range_cost
 * 
 * 【設計】
 * - keyでBST、priorityでヒープ
 * - split / merge ベースで安全に操作
 * - 集約情報（sz, sum）はupdateで維持
 * 
 * 【注意】
 * - kthは1-index
 * - rangeは [l, r)（0-index）
 * - split後は必ずmergeで戻す
 */
struct Treap {
    struct Node {
        int key;
        uint32_t pri;
        int sz;
        long long sum;
        Node *l, *r;

        Node(int key, uint32_t pri)
            : key(key), pri(pri), sz(1), sum(key), l(nullptr), r(nullptr) {}
    };

    using NP = Node*;

    // RNG（高速・衝突少）
    mt19937 rng;

    Treap() : rng(chrono::steady_clock::now().time_since_epoch().count()) {}

    // ノード生成（priorityは乱数）
    NP new_node(int key) {
        return new Node(key, rng());
    }

    // 部分木サイズ取得（nullは0）
    int size(NP t) const {
        return t ? t->sz : 0;
    }

    // 部分木の総和取得
    long long sum(NP t) const {
        return t ? t->sum : 0;
    }

    // 部分木サイズを更新（子から再計算）
    void update(NP t) {
        if (!t) return;
        t->sz = 1 + size(t->l) + size(t->r);
        t->sum = t->key + sum(t->l) + sum(t->r);
    }

    // 2つのTreapをマージ（lの全要素 <= rの全要素が前提）
    NP merge(NP l, NP r) {
        if (!l || !r) return l ? l : r;

        if (l->pri > r->pri) {
            l->r = merge(l->r, r);
            update(l);
            return l;
        } else {
            r->l = merge(l, r->l);
            update(r);
            return r;
        }
    }

    // key未満 / key以上 に分割
    // 戻り値: {< key, >= key}
    pair<NP, NP> split(NP t, int key) {
        if (!t) return {nullptr, nullptr};

        if (t->key < key) {
            auto [l, r] = split(t->r, key);
            t->r = l;
            update(t);
            return {t, r};
        } else {
            auto [l, r] = split(t->l, key);
            t->l = r;
            update(t);
            return {l, t};
        }
    }

    // 要素keyを1つ追加（multiset）
    // split→mergeで安全に挿入
    NP insert(NP root, int key) {
        auto [l, r] = split(root, key);
        return merge(merge(l, new_node(key)), r);
    }

    // 要素keyを1つ削除（multiset）
    // 存在すれば1つだけ削除
    NP erase(NP root, int key) {
        auto [l, r] = split(root, key);       // < key | >= key
        auto [m, r2] = split(r, key + 1);     // == key | > key

        // mの中から1つだけ削除
        if (m) {
            m = merge(m->l, m->r);
        }

        return merge(l, merge(m, r2));
    }

    // k番目の値を取得（1-index）
    // k=1が最小
    int kth(NP t, int k) {
        if (!t) return -1; // 異常（テストで弾く前提でもOK）

        int left_size = size(t->l);

        if (k <= left_size) {
            return kth(t->l, k);
        } else if (k == left_size + 1) {
            return t->key;
        } else {
            return kth(t->r, k - left_size - 1);
        }
    }

    // key未満の要素数
    int order_of_key(NP t, int key) {
        if (!t) return 0;

        if (key <= t->key) {
            // 全部左にある
            return order_of_key(t->l, key);
        } else {
            // 左 + 自分 + 右の一部
            return size(t->l) + 1 + order_of_key(t->r, key);
        }
    }

    // keyの出現回数
    int count(NP t, int key) {
        return order_of_key(t, key + 1) - order_of_key(t, key);
    }

    // key以上で最小の値（lower_bound）
    // 存在しなければ -1
    int lower_bound(NP t, int key) {
        int k = order_of_key(t, key) + 1;
        if (k > size(t)) return -1;
        return kth(t, k);
    }

    // keyより大きい最小の値（upper_bound）
    // 存在しなければ -1
    int upper_bound(NP t, int key) {
        int k = order_of_key(t, key + 1) + 1;
        if (k > size(t)) return -1;
        return kth(t, k);
    }

    // key未満で最大の値（前駆）
    // 存在しなければ -1
    int prev(NP t, int key) {
        int k = order_of_key(t, key);
        if (k == 0) return -1;
        return kth(t, k);
    }

    // keyより大きい最小の値（後継）
    // 存在しなければ -1
    int next(NP t, int key) {
        int k = order_of_key(t, key + 1) + 1;
        if (k > size(t)) return -1;
        return kth(t, k);
    }

    // 小さい方からk個の和（k個未満なら全部）
    long long sum_k_smallest(NP t, int k) {
        if (!t || k <= 0) return 0;

        int left_size = size(t->l);

        if (k <= left_size) {
            return sum_k_smallest(t->l, k);
        } else if (k == left_size + 1) {
            return sum(t->l) + t->key;
        } else {
            return sum(t->l) + t->key + sum_k_smallest(t->r, k - left_size - 1);
        }
    }

    // 全体和
    long long sum_all(NP t) {
        return sum(t);
    }

    // 大きい方からk個の和
    long long sum_k_largest(NP t, int k) {
        int n = size(t);
        return sum(t) - sum_k_smallest(t, n - k);
    }

    // 中央値（下側）
    int median(NP t) {
        int n = size(t);
        if (n == 0) return -1;
        return kth(t, (n + 1) / 2);
    }

    // 絶対値和（中央値基準）
    long long median_cost(NP t) {
        int n = size(t);
        if (n == 0) return 0;

        int m = median(t);

        int k = order_of_key(t, m);   // m未満の個数
        int cnt_m = count(t, m);      // mの個数

        long long left_sum = sum_k_smallest(t, k);
        long long right_sum = sum(t) - sum_k_smallest(t, k + cnt_m);

        // 左側: m*k - left_sum
        // 右側: right_sum - m*(n - k - cnt_m)

        long long cost = 0;
        cost += 1LL * m * k - left_sum;
        cost += right_sum - 1LL * m * (n - k - cnt_m);

        return cost;
    }

    // [l, r) の和（0-index）
    long long range_sum(NP &t, int l, int r) {
        if (l >= r) return 0;

        // [0,l), [l,r), [r,n)
        auto [a, bc] = split_kth(t, l);
        auto [b, c] = split_kth(bc, r - l);

        long long res = sum(b);

        // 元に戻す（超重要）
        t = merge(a, merge(b, c));

        return res;
    }

    // 左にk個、右に残り
    pair<NP, NP> split_kth(NP t, int k) {
        if (!t) return {nullptr, nullptr};

        if (k <= size(t->l)) {
            auto [l, r] = split_kth(t->l, k);
            t->l = r;
            update(t);
            return {l, t};
        } else {
            auto [l, r] = split_kth(t->r, k - size(t->l) - 1);
            t->r = l;
            update(t);
            return {t, r};
        }
    }

    // [l, r) の要素数
    int range_count(NP &t, int l, int r) {
        return r - l;
    }

    // [l, r) のk番目（1-index）
    int range_kth(NP &t, int l, int r, int k) {
        auto [a, bc] = split_kth(t, l);
        auto [b, c] = split_kth(bc, r - l);

        int res = kth(b, k);

        t = merge(a, merge(b, c));
        return res;
    }

    // [l, r) の中央値
    int range_median(NP &t, int l, int r) {
        return range_kth(t, l, r, (r - l + 1) / 2);
    }

    // [l, r) の中央値コスト
    long long range_cost(NP &t, int l, int r) {
        auto [a, bc] = split_kth(t, l);
        auto [b, c] = split_kth(bc, r - l);

        long long res = median_cost(b);

        t = merge(a, merge(b, c));
        return res;
    }

    // 中順走査（昇順配列に変換）
    void inorder(NP t, vector<int>& v) const {
        if (!t) return;
        inorder(t->l, v);
        v.push_back(t->key);
        inorder(t->r, v);
    }
};

void random_test_sum_step1() {
    using NP = Treap::Node*;
    Treap tr;

    mt19937 rng(0);

    for (int iter = 0; iter < 2000; iter++) {
        vector<int> A;

        int N = rng() % 20;
        for (int i = 0; i < N; i++) {
            A.push_back(rng() % 50);
        }

        vector<int> sorted = A;
        sort(sorted.begin(), sorted.end());

        // mergeだけで構築
        NP root = nullptr;
        for (int x : sorted) {
            root = tr.merge(root, tr.new_node(x));
        }

        // inorder確認
        vector<int> res;
        tr.inorder(root, res);
        if (res != sorted) {
            cout << "Step1: inorder NG\n";
            exit(0);
        }

        // sum確認
        long long expected = 0;
        for (int x : sorted) expected += x;

        if (tr.sum(root) != expected) {
            cout << "Step1: sum NG\n";
            exit(0);
        }

        // split → merge をランダムに繰り返す
        for (int t = 0; t < 50; t++) {
            int k = rng() % 50;

            auto [l, r] = tr.split(root, k);
            root = tr.merge(l, r);

            // inorder再確認
            vector<int> res2;
            tr.inorder(root, res2);
            if (res2 != sorted) {
                cout << "Step1: split-merge inorder NG\n";
                exit(0);
            }

            // sum再確認
            if (tr.sum(root) != expected) {
                cout << "Step1: split-merge sum NG\n";
                exit(0);
            }
        }
    }

    cout << "random_test_sum_step1 OK\n";
}

void random_test_sum_step2() {
    using NP = Treap::Node*;
    Treap tr;

    mt19937 rng(0);

    for (int iter = 0; iter < 2000; iter++) {
        NP root = nullptr;
        vector<int> vec;

        int Q = 100;

        for (int q = 0; q < Q; q++) {
            int type = rng() % 2;

            if (type == 0 || vec.empty()) {
                // insert
                int x = rng() % 50;
                root = tr.insert(root, x);
                vec.push_back(x);
            } else {
                // erase
                int idx = rng() % vec.size();
                int x = vec[idx];

                root = tr.erase(root, x);

                auto it = find(vec.begin(), vec.end(), x);
                vec.erase(it);
            }

            // 正解
            vector<int> sorted = vec;
            sort(sorted.begin(), sorted.end());

            long long expected_sum = 0;
            for (int x : sorted) expected_sum += x;

            // inorderチェック
            vector<int> res;
            tr.inorder(root, res);
            if (res != sorted) {
                cout << "Step2: inorder NG\n";
                exit(0);
            }

            // sumチェック（ここが本体）
            if (tr.sum(root) != expected_sum) {
                cout << "Step2: sum NG\n";
                exit(0);
            }

            // split→mergeチェック
            int k = rng() % 50;
            auto [l, r] = tr.split(root, k);
            root = tr.merge(l, r);

            vector<int> res2;
            tr.inorder(root, res2);
            if (res2 != sorted) {
                cout << "Step2: split-merge inorder NG\n";
                exit(0);
            }

            if (tr.sum(root) != expected_sum) {
                cout << "Step2: split-merge sum NG\n";
                exit(0);
            }
        }
    }

    cout << "random_test_sum_step2 OK\n";
}

void random_test_sum_step4() {

    Treap tr;

    mt19937 rng(0);

    for (int iter = 0; iter < 2000; iter++) {
        NP root = nullptr;
        vector<int> vec;

        int Q = 100;

        for (int q = 0; q < Q; q++) {
            int type = rng() % 2;

            if (type == 0 || vec.empty()) {
                // insert
                int x = rng() % 50;
                root = tr.insert(root, x);
                vec.push_back(x);
            } else {
                // erase
                int idx = rng() % vec.size();
                int x = vec[idx];

                root = tr.erase(root, x);

                auto it = find(vec.begin(), vec.end(), x);
                vec.erase(it);
            }

            // 正解
            vector<int> sorted = vec;
            sort(sorted.begin(), sorted.end());

            int n = sorted.size();

            // medianチェック
            if (n > 0) {
                int expected_median = sorted[(n - 1) / 2];
                int got_median = tr.median(root);

                if (expected_median != got_median) {
                    cout << "Step4: median NG\n";
                    exit(0);
                }
            }

            // median_costチェック
            long long expected_cost = 0;
            if (n > 0) {
                int m = sorted[(n - 1) / 2];

                for (int v : sorted) {
                    expected_cost += abs(v - m);
                }
            }

            long long got_cost = tr.median_cost(root);

            if (expected_cost != got_cost) {
                cout << "Step4: median_cost NG\n";
                exit(0);
            }

            // split→mergeチェック
            int k = rng() % 50;
            auto [l, r] = tr.split(root, k);
            root = tr.merge(l, r);
        }
    }

    cout << "random_test_sum_step4 OK\n";
}

void random_test_sum_step5() {
    using NP = Treap::Node*;
    Treap tr;

    mt19937 rng(0);

    for (int iter = 0; iter < 2000; iter++) {
        NP root = nullptr;
        vector<int> vec;

        int Q = 100;

        for (int q = 0; q < Q; q++) {
            int type = rng() % 2;

            if (type == 0 || vec.empty()) {
                int x = rng() % 50;
                root = tr.insert(root, x);
                vec.push_back(x);
            } else {
                int idx = rng() % vec.size();
                int x = vec[idx];

                root = tr.erase(root, x);

                auto it = find(vec.begin(), vec.end(), x);
                vec.erase(it);
            }

            vector<int> sorted = vec;
            sort(sorted.begin(), sorted.end());

            int n = sorted.size();

            // 全区間チェック
            for (int l = 0; l <= n; l++) {
                for (int r = l; r <= n; r++) {
                    long long expected = 0;
                    for (int i = l; i < r; i++) {
                        expected += sorted[i];
                    }

                    long long got = tr.range_sum(root, l, r);

                    if (expected != got) {
                        cout << "Step5: range_sum NG\n";
                        cout << "l=" << l << " r=" << r << "\n";
                        exit(0);
                    }
                }
            }
        }
    }

    cout << "random_test_sum_step5 OK\n";
}

void random_test_sum_step6() {
    using NP = Treap::Node*;
    Treap tr;

    mt19937 rng(0);

    for (int iter = 0; iter < 2000; iter++) {
        NP root = nullptr;
        vector<int> vec;

        int Q = 100;

        for (int q = 0; q < Q; q++) {
            int type = rng() % 2;

            if (type == 0 || vec.empty()) {
                // insert
                int x = rng() % 50;
                root = tr.insert(root, x);
                vec.push_back(x);
            } else {
                // erase
                int idx = rng() % vec.size();
                int x = vec[idx];

                root = tr.erase(root, x);

                auto it = find(vec.begin(), vec.end(), x);
                vec.erase(it);
            }

            vector<int> sorted = vec;
            sort(sorted.begin(), sorted.end());
            int n = sorted.size();

            // --- sum_k_largest ---
            for (int k = 0; k <= n; k++) {
                long long expected = 0;
                for (int i = n - k; i < n; i++) {
                    if (i >= 0) expected += sorted[i];
                }

                long long got = tr.sum_k_largest(root, k);

                if (expected != got) {
                    cout << "Step6: sum_k_largest NG\n";
                    cout << "k=" << k << "\n";
                    exit(0);
                }
            }

            // --- range系 ---
            for (int l = 0; l <= n; l++) {
                for (int r = l; r <= n; r++) {
                    int len = r - l;

                    if (len == 0) continue;

                    // range_kth
                    for (int k = 1; k <= len; k++) {
                        int expected = sorted[l + k - 1];
                        int got = tr.range_kth(root, l, r, k);

                        if (expected != got) {
                            cout << "Step6: range_kth NG\n";
                            cout << "l=" << l << " r=" << r << " k=" << k << "\n";
                            exit(0);
                        }
                    }

                    // range_median
                    int expected_median = sorted[l + (len - 1) / 2];
                    int got_median = tr.range_median(root, l, r);

                    if (expected_median != got_median) {
                        cout << "Step6: range_median NG\n";
                        cout << "l=" << l << " r=" << r << "\n";
                        exit(0);
                    }

                    // range_cost
                    int m = expected_median;
                    long long expected_cost = 0;
                    for (int i = l; i < r; i++) {
                        expected_cost += abs(sorted[i] - m);
                    }

                    long long got_cost = tr.range_cost(root, l, r);

                    if (expected_cost != got_cost) {
                        cout << "Step6: range_cost NG\n";
                        cout << "l=" << l << " r=" << r << "\n";
                        exit(0);
                    }
                }
            }
        }
    }

    cout << "random_test_sum_step6 OK\n";
}

void demo_sum_treap() {

    cout << "\n--- 中央値 ---\n";
    cout << "median = " << tr.median(root) << "\n";
    cout << "median_cost = " << tr.median_cost(root) << "\n";

    cout << "\n--- 区間 ---\n";
    cout << "[1,4) の和 = " << tr.range_sum(root, 1, 4) << "\n";
    cout << "[1,4) の2番目 = " << tr.range_kth(root, 1, 4, 2) << "\n";

    tr.inorder(root, v);

    cout << "結果: ";
    for (int x : v) cout << x << " ";
    cout << "\n";

    cout << "==============================\n";
}
