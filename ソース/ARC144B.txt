#define chmax(x,y) x = max(x,y)

#define popcount __builtin_popcountll

int main(){
    int N,a,b;
    cin>>N>>a>>b;
    vector<int>A(N);
    rep(i,0,N)cin>>A[i];
    ll left=0,right=1e9+1;
    while(right-left>1){
        ll mid=(right+left)/2;
        ll lcnt=0,rcnt=0;
        rep(i,0,N){
            if(A[i]<mid){
                lcnt+=(mid-A[i])/a;
                if((mid-A[i])%a)lcnt++;
            }else{
                rcnt+=(A[i]-mid)/b;
            }
        }
        if(lcnt<=rcnt)left=mid;
        else right=mid;
    }
    cout<<left<<'\n';
    return 0;
}
