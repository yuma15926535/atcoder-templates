struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

vector<int> indices(ull mask){
    vector<int> res;
    for(int i=0;mask;i++){
        if(mask&1) res.push_back(i);
        mask>>=1;
    }
    return res;
}

//popcountの固定(nCk列挙)
struct CombinationBits {
    using ull = unsigned long long;

    int N, K;
    ull cur;
    ull limit;

    //n桁,popcount=k
    CombinationBits(int n, int k) : N(n), K(k) {
        cur = (1ULL << K) - 1;   // 最初: 下位Kbitが1
        limit = (1ULL << N);
    }

    bool has_next() const {
        return cur < limit;
    }

    ull next() {
        ull ret = cur;

        // --- Gosper's Hack ---
        ull c = cur & -cur;
        ull r = cur + c;
        cur = (((r ^ cur) >> 2) / c) | r;

        return ret;
    }
};

//1固定(superset列挙)
struct ForcedOneMaskGenerator {
    using ull = unsigned long long;

    int N;
    ull base;
    vector<int> free_pos;

    ull state = 0;
    ull limit;

    //n桁,bの1は固定
    ForcedOneMaskGenerator(int n, ull b)
        : N(n)
    {
        ull valid = (N==64 ? ~0ULL : ((1ULL<<N)-1));
        base = b & valid;

        for(int i=0;i<N;i++)
            if(!(base>>i & 1))
                free_pos.push_back(i);

        limit = 1ULL << free_pos.size();
    }

    bool has_next() const {
        return state < limit;
    }

    ull next() {
        ull add = 0;

        for(int i=0;i<(int)free_pos.size();i++)
            if(state>>i & 1)
                add |= 1ULL << free_pos[i];

        state++;
        return base | add;
    }
};

//0固定(subset制限)
struct ForcedZeroMaskGenerator {
    using ull = unsigned long long;

    int N;
    vector<int> free_pos;

    ull state = 0;
    ull limit;

    //n桁,zero_maskの1を0に固定
    ForcedZeroMaskGenerator(int n, ull zero_mask)
        : N(n)
    {
        ull valid = (N==64 ? ~0ULL : ((1ULL<<N)-1));
        zero_mask &= valid;

        for(int i=0;i<N;i++)
            if(!(zero_mask>>i & 1))
                free_pos.push_back(i);

        limit = 1ULL << free_pos.size();
    }

    bool has_next() const {
        return state < limit;
    }

    ull next() {
        ull res = 0;

        for(int i=0;i<(int)free_pos.size();i++)
            if(state>>i & 1)
                res |= 1ULL << free_pos[i];

        state++;
        return res;
    }
};

//1固定+popcount固定(条件付き選択)
struct ForcedOnePopcountK {
    using ull = unsigned long long;

    int N;
    ull base;

    vector<int> free_pos;

    ull comb = 0;
    bool valid = true;

    //n桁,bの1は固定,popcount=k
    ForcedOnePopcountK(int n, ull b, int K)
        : N(n)
    {
        ull valid_mask =
            (N==64 ? ~0ULL : ((1ULL<<N)-1));

        base = b & valid_mask;

        int fixed = __builtin_popcountll(base);
        int R = K - fixed;

        if(R < 0){
            valid = false;
            return;
        }

        for(int i=0;i<N;i++)
            if(!(base>>i & 1))
                free_pos.push_back(i);

        if(R > (int)free_pos.size()){
            valid = false;
            return;
        }

        comb = (1ULL<<R)-1;
    }

    bool has_next() const {
        return valid;
    }

    ull next() {
        ull add = 0;

        for(int i=0;i<(int)free_pos.size();i++)
            if(comb>>i & 1)
                add |= 1ULL<<free_pos[i];

        ull res = base | add;

        // --- Gosper ---
        ull c = comb & -comb;
        ull r = comb + c;
        if(r >= (1ULL<<free_pos.size()))
            valid = false;
        else
            comb = (((r^comb)>>2)/c)|r;

        return res;
    }
};

//0固定+popcount固定(禁止付き選択)
struct ForcedZeroPopcountK {

    //n桁,zero_maskの1を0に固定,popcount=k
    ForcedZeroPopcountK(int n, ull zero_mask, int K)
        : N(n)
    {
        ull valid_mask =
            (N==64 ? ~0ULL : ((1ULL<<N)-1));

        zero_mask &= valid_mask;

        for(int i=0;i<N;i++)

                free_pos.push_back(i);

        if(K > (int)free_pos.size()){
            valid=false;
            return;
        }

    }

    bool has_next() const {
        return valid;
    }

    ull next() {
        ull res = 0;

        for(int i=0;i<(int)free_pos.size();i++)

        ull c = comb & -comb;
        ull r = comb + c;
        if(r >= (1ULL<<free_pos.size()))
            valid=false;
        else
            comb = (((r^comb)>>2)/c)|r;

        return res;
    }
};
