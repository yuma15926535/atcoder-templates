    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }

/**
 * @brief ハッシュマップ（unordered_map ラッパー）
 *
 * ll をキーとする競技プログラミング向け HashMap。
 * custom_hash を使用してハッシュ衝突への耐性を向上させている。
 *
 * 特徴
 * - 平均 O(1) の検索・追加・削除
 * - イテレータを隠蔽したシンプルな API
 * - contains・get・try_get・keys・values・vec など便利関数を提供
 * - random() は内部の先頭要素を返却（ランダムではない）
 *
 * @tparam T 値の型
 */
template<class T>
class HashMap {
private:
    struct custom_hash {
        static uint64_t splitmix64(uint64_t x) {
            x += 0x9e3779b97f4a7c15;
            x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9;
            x = (x ^ (x >> 27)) * 0x94d049bb133111eb;
            return x ^ (x >> 31);
        }

        size_t operator()(uint64_t x) const {
            static const uint64_t FIXED_RANDOM =
                chrono::steady_clock::now().time_since_epoch().count();
            return splitmix64(x + FIXED_RANDOM);
        }

        size_t operator()(ll x) const {
            return (*this)(static_cast<uint64_t>(x));
        }
    };

    unordered_map<ll, T, custom_hash> mp;

public:

    /**
     * @brief 空の HashMap を構築する
     *
     * reserve と max_load_factor を設定し、
     * リハッシュ回数を抑える。
     */
    HashMap() {
        mp.reserve(1 << 16);
        mp.max_load_factor(0.7f);
    }

    /**
     * @brief 初期予約サイズを指定して構築する
     *
     * @param reserve_size 予約する要素数
     */
    explicit HashMap(size_t reserve_size) {
        mp.reserve(reserve_size);
        mp.max_load_factor(0.7f);
    }

    //==================================================
    // 基本
    //==================================================

    /**
     * @brief 要素数を返す
     *
     * @return 要素数
     */
    ll size() const {
        return (ll)mp.size();
    }

    /**
     * @brief 空か判定する
     *
     * @return 空なら true
     */
    bool empty() const {
        return mp.empty();
    }

    /**
     * @brief 全要素を削除する
     */
    void clear() {
        mp.clear();
    }

    //==================================================
    // 更新
    //==================================================

    /**
     * @brief 要素を追加する
     *
     * 既にキーが存在する場合は何もしない。
     *
     * @param key キー
     * @param value 値
     * @return 追加に成功した場合 true
     */
    bool insert(ll key, const T& value) {
        return mp.emplace(key, value).second;
    }

    /**
     * @brief 要素を追加または更新する
     *
     * キーが存在しない場合は追加し、
     * 存在する場合は値を更新する。
     *
     * @param key キー
     * @param value 値
     */
    void set(ll key, const T& value) {
        mp[key] = value;
    }

    /**
     * @brief 指定キーを削除する
     *
     * @param key キー
     * @return 削除した場合 true
     */
    bool erase(ll key) {
        return mp.erase(key);
    }

    //==================================================
    // 検索
    //==================================================

    /**
     * @brief キーが存在するか判定する
     *
     * @param key キー
     * @return 存在する場合 true
     */
    bool contains(ll key) const {
        return mp.find(key) != mp.end();
    }

    /**
     * @brief 値を取得する
     *
     * キーが存在しない場合はデフォルト値で追加する。
     *
     * @param key キー
     * @return 値への参照
     */
    T& get(ll key) {
        return mp[key];
    }

    /**
     * @brief 値を取得する
     *
     * キーが存在しない場合は例外を送出する。
     *
     * @param key キー
     * @return 値への参照
     */
    const T& at(ll key) const {
        return mp.at(key);
    }

    /**
     * @brief 値を取得する
     *
     * キーが存在しない場合はデフォルト値で追加する。
     *
     * @param key キー
     * @return 値への参照
     */
    T& operator[](ll key) {
        return mp[key];
    }

    /**
     * @brief 全キーを取得する
     *
     * @return 全キーの配列（順序は保証しない）
     */
    const T& operator[](ll key) const {
        return mp.at(key);
    }

    //==================================================
    // 一覧取得
    //==================================================

    /**
     * @brief 全キーを取得する
     *
     * @return 全キーの配列（順序は保証しない）
     */
    vector<ll> keys() const {
        vector<ll> res;
        res.reserve(mp.size());
        for (const auto& [key, value] : mp) {
            res.push_back(key);
        }
        return res;
    }

    /**
     * @brief 全値を取得する
     *
     * @return 全値の配列（順序は保証しない）
     */
    vector<T> values() const {
        vector<T> res;
        res.reserve(mp.size());
        for (const auto& [key, value] : mp) {
            res.push_back(value);
        }
        return res;
    }

    /**
     * @brief 全要素を取得する
     *
     * @return (キー, 値) の配列（順序は保証しない）
     */
    vector<pair<ll, T>> vec() const {
        vector<pair<ll, T>> res;
        res.reserve(mp.size());
        for (const auto& [key, value] : mp) {
            res.emplace_back(key, value);
        }
        return res;
    }

    //==================================================
    // 便利関数
    //==================================================

    /**
     * @brief 値を加算する
     *
     * キーが存在しない場合はデフォルト値で追加してから加算する。
     *
     * @param key キー
     * @param delta 加算値
     */
    void add(ll key, const T& delta) {
        mp[key] += delta;
    }

    /**
     * @brief 値を取得する
     *
     * キーが存在しない場合は {false, T{}} を返す。
     *
     * @param key キー
     * @return {存在したか, 値}
     */
    pair<bool, T> try_get(ll key) const {
        auto it = mp.find(key);
        if (it == mp.end()) {
            return {false, T{}};
        }
        return {true, it->second};
    }

    /**
     * @brief いずれかの要素を取得する
     *
     * 内部の begin() が指す (キー, 値) を返す。
     * 要素が存在することのみ保証し、返却順は保証しない。
     *
     * @return (キー, 値)
     */
    pair<ll, T> random() const {
        assert(!mp.empty());
        return *mp.begin();
    }
};

void random_test_HashMap() {
    cout << "===== HashMap Random Test =====" << endl;

    mt19937 rng((uint32_t)chrono::steady_clock::now().time_since_epoch().count());

    rep(_, 500) {
        HashMap<ll> hm;
        unordered_map<ll, ll> mp;

        rep(__, 5000) {
            int op = rng() % 12;
            ll key = rng() % 30;
            ll val = rng() % 1000;

            if (op == 0) {
                // insert
                bool a = hm.insert(key, val);
                bool b = mp.emplace(key, val).second;
                assert(a == b);
            }

            if (op == 1) {
                // set
                hm.set(key, val);
                mp[key] = val;
            }

            if (op == 2) {
                // erase
                bool a = hm.erase(key);
                bool b = (bool)mp.erase(key);
                assert(a == b);
            }

            if (op == 3) {
                // contains
                assert(hm.contains(key) == (mp.find(key) != mp.end()));
            }

            if (op == 4) {
                // get
                assert(hm.get(key) == mp[key]);
            }

            if (op == 5) {
                // operator[]
                hm[key] += val;
                mp[key] += val;
            }

            if (op == 6) {
                // add
                hm.add(key, val);
                mp[key] += val;
            }

            if (op == 7) {
                // try_get
                auto a = hm.try_get(key);

                auto it = mp.find(key);
                if (it == mp.end()) {
                    assert(!a.first);
                } else {
                    assert(a.first);
                    assert(a.second == it->second);
                }
            }

            if (op == 8) {
                // clear
                hm.clear();
                mp.clear();
            }

            if (op == 9) {
                // size / empty
                assert(hm.size() == (ll)mp.size());
                assert(hm.empty() == mp.empty());
            }

            if (op == 10) {
                // keys
                auto a = hm.keys();
                vector<ll> b;
                for (auto [k, v] : mp) b.push_back(k);
                sort(a.begin(), a.end());
                sort(b.begin(), b.end());
                assert(a == b);
            }

            if (op == 11) {
                // vec, values, random
                auto va = hm.vec();
                vector<pair<ll, ll>> vb(mp.begin(), mp.end());
                sort(va.begin(), va.end());
                sort(vb.begin(), vb.end());
                assert(va == vb);

                auto vals = hm.values();
                vector<ll> vals2;
                for (auto [k, v] : mp) vals2.push_back(v);
                sort(vals.begin(), vals.end());
                sort(vals2.begin(), vals2.end());
                assert(vals == vals2);

                if (!mp.empty()) {
                    auto p = hm.random();
                    auto it = mp.find(p.first);
                    assert(it != mp.end());
                    assert(it->second == p.second);
                }
            }

            // 常に状態一致確認
            assert(hm.size() == (ll)mp.size());

            auto va = hm.vec();
            vector<pair<ll, ll>> vb(mp.begin(), mp.end());
            sort(va.begin(), va.end());
            sort(vb.begin(), vb.end());
            assert(va == vb);
        }
    }

    cout << "ALL TEST PASSED!" << endl;
}
