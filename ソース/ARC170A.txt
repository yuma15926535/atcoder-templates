int main(){
    int N;
    string S, T;
    cin >> N >> S >> T;
    rep(i,0,N){
        if(T[i]=='A')break;
        if(S[i]=='A'&&T[i]=='B'){
            cout<<-1<<endl;
            return 0;
        }
    }
    rrep(i,N,0){
        if(T[i]=='B')break;
        if(S[i]=='B'&&T[i]=='A'){
            cout<<-1<<endl;
            return 0;
        }
    }
    int ba=0,ans=0;
    rep(i,0,N){
        if(S[i]!=T[i]){
            if(T[i]=='A'){
                ba++;
            }else{
                if(ba>0){
                    ba--;
                    ans++;
                }else{
                    ans++;
                }
            }
        }
    }
    cout<<ans+ba<<'\n';
    return 0;
}
