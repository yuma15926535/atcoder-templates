using S = ll;

// ★区間結合
S combine(S a, S b) {
    return a + b; // 例：区間和
}

// ★単位元
S id() {
    return 0;
}

/*
=====================================================
SegTree 本体
=====================================================
*/

struct SegTree {
    int n;
    vector<S> seg;

    SegTree(int sz) {
        n = 1;
        while (n < sz) n <<= 1;
        seg.assign(2*n, id());
    }

    // 配列初期化
    SegTree(const vector<S>& v) {
        int sz = v.size();
        n = 1;
        while (n < sz) n <<= 1;
        seg.assign(2*n, id());

        for (int i = 0; i < sz; i++)
            seg[n+i] = v[i];

        for (int i = n-1; i > 0; i--)
            seg[i] = combine(seg[i<<1], seg[i<<1|1]);
    }

    // ========================
    // 点更新
    // ========================
    void set(int i, S x) {
        i += n;
        seg[i] = x;

        while (i >>= 1)
            seg[i] = combine(seg[i<<1], seg[i<<1|1]);
    }

    // 和専用の便利関数
    void add(int i, S x) {
        set(i, get(i) + x);
    }

    // ========================
    // 区間取得 [l,r)
    // ========================
    S range(int l, int r) {
        S L = id(), R = id();
        for (l += n, r += n; l < r; l >>= 1, r >>= 1) {
            if (l & 1) L = combine(L, seg[l++]);
            if (r & 1) R = combine(seg[--r], R);
        }
        return combine(L, R);
    }

    S get(int i) {
        return seg[n+i];
    }

    S all() {
        return seg[1];
    }

    // ========================
    // prefix探索
    // ========================
    // 累積 >= x の最小 index
    int ge(S x) {
        if (all() < x) return n;

        int node = 1;
        S acc = id();

        while (node < n) {
            S nxt = combine(acc, seg[node<<1]);
            if (nxt >= x) {
                node <<= 1;
            } else {
                acc = nxt;
                node = node<<1 | 1;
            }
        }
        return node - n;
    }

    // 累積 > x
    int gt(S x) { return ge(x+1); }

    // 累積 ≤ x
    int le(S x) {
        int pos = ge(x+1);
        return pos - 1;
    }

    // 累積 < x
    int lt(S x) {
        int pos = ge(x);
        return pos - 1;
    }

    // ========================
    // デバッグ
    // ========================
    void dump(int sz) {
        for (int i = 0; i < sz; i++)
            cout << get(i) << " ";
        cout << "\n";
    }
};

void solv(){
    cout<<"=============================\n";
    cout<<" SegTree 動作確認\n";
    cout<<"=============================\n\n";

    vector<ll> v={5,3,7,1,4};
    SegTree seg(v);

    cout<<"▼ 初期配列:\n";
    seg.dump(v.size());

    cout<<"\n▼ get(2): index2 の値 = "
        <<seg.get(2)<<"\n";

    cout<<"\n▼ range(1,4): 区間[1,4) の和 = "
        <<seg.range(1,4)<<"\n";

    cout<<"\n▼ all(): 全体の和 = "
        <<seg.all()<<"\n";

    cout<<"\n▼ set(1,10): index1 を 10 に更新\n";
    seg.set(1,10);
    seg.dump(v.size());

    cout<<"\n▼ add(2,5): index2 に +5\n";
    seg.add(2,5);
    seg.dump(v.size());

    cout<<"\n▼ range(0,5): 全区間和 = "
        <<seg.range(0,5)<<"\n";

    cout<<"\n=============================\n";
    cout<<" prefix探索テスト\n";
    cout<<"=============================\n";

    for(int x=1;x<=25;x+=5){
        cout<<"\nx="<<x<<"\n";

        cout<<" 累積 >= x : index "
            <<seg.ge(x)<<"\n";

        cout<<" 累積 > x : index "
            <<seg.gt(x)<<"\n";

        cout<<" 累積 <= x : index "
            <<seg.le(x)<<"\n";

        cout<<" 累積 < x : index "
            <<seg.lt(x)<<"\n";
    }

    cout<<"\n=============================\n";
    cout<<" 最終状態\n";
    cout<<"=============================\n";

    seg.dump(v.size());

    cout<<"\n✅ 動作確認完了\n";
    return;
}
