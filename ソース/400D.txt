#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    //ダイクストラ法の特殊ver
    //01BFS
    //重みが0,1のみの場合pqをdequeで管理できる
    //0は先頭、1は末尾に追加
    int H,W,a,b,c,d;
    cin>>H>>W;
    vector<string> S(H);
    rep(i,0,H)cin>>S[i];
    cin>>a>>b>>c>>d;
    a--;b--;c--;d--;
    vector<vector<int>>ans(H,vector<int>(W,H*W));
    vector<int> dx={-1,1,0,0},dy={0,0,-1,1};
    ans[a][b]=0;
    deque<int> dq;
    dq.push_back(a*W+b);
    while(!dq.empty()){
        int z=dq.front();
        dq.pop_front();
        if(z==c*W+d){
            cout<<ans[c][d]<<"\n";
            return 0;
        }
        int x=z/W,y=z%W;
        rep(i,0,4){
            bool wall=false;
            erep(j,1,2){
                int nx=x+dx[i]*j,ny=y+dy[i]*j;
                if(nx<0||nx>=H||ny<0||ny>=W)break;
                if(S[nx][ny]=='#')wall=true;
                //壁を通過しているか否か(前蹴りの有無)
                if(!wall){
                    //壁無しで移動する場合はコスト0(2マス貫通は無視)
                    if(j==1){
                        if(ans[nx][ny]>ans[x][y]){
                            ans[nx][ny]=ans[x][y];
                            dq.push_front(nx*W+ny);
                        }
                    }
                }else{
                    //壁有りの場合前蹴りするのでコスト1
                    if(ans[nx][ny]>ans[x][y]+1){
                        ans[nx][ny]=ans[x][y]+1;
                        dq.push_back(nx*W+ny);
                    }
                }
            }
        }
    }
    return 0;
}
