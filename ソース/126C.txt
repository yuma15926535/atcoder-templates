#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    int N,K;
    cin>>N>>K;
    map<int,ll> m;
    int num=1,c=0;
    rrep(i,N+1,1){
        while(num*i<K){
            num*=2;
            c++;
        }
        if(m.count(c))m[c]++;
        else m[c]=1;
    }
    double ans=0;
    auto q=*m.begin();
    ll num2=num;
    rep(i,0,q.first)num/=2;
    for(auto p:m){
        ans+=p.second*num;
        num/=2;
    }
    //出力の桁数を指定
    cout<<setprecision(11)<<ans/(N*num2)<<"\n";
    //or printf("%.11f\n",ans/(N*num2));
    return 0;
}
