#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる

template<class T> using PQ = priority_queue<T, vector<T>>;//大きい順
template<class T> using PQ_g = priority_queue<T, vector<T>, greater<T>>;//小さい順
#define chmax(x,y) x = max(x,y)
#define chmin(x,y) x = min(x,y)

int main(){
    //LCS 最長共通部分文字列
    string S,T;
    cin>>S>>T;
    vector<vector<ll>>dp((int)S.size()+1,vector<ll>(T.size()+1,0));
    erep(i,1,S.size())erep(j,1,T.size()){
        dp[i][j]=max(dp[i-1][j],dp[i][j-1]);
        if(S[i-1]==T[j-1])chmax(dp[i][j],dp[i-1][j-1]+1);
    }
    //復元
    string ans="";
    int si=(int)S.size(),ti=(int)T.size();
    ll num=dp[si][ti];
    while(num){
        if(num==dp[si-1][ti])si--;
        else if(num==dp[si][ti-1])ti--;
        else{
            si--;
            ti--;
            ans=S[si]+ans;
            num--;
        }
    }
    cout<<ans<<'\n';
    return 0;
}
