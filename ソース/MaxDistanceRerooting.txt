struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

vector<vector<int>> make_random_tree(
    int N,
    mt19937 &rng,
    bool shuffle_vertex = true
){
    vector<int> perm(N);
    iota(perm.begin(), perm.end(), 0);

    // 頂点番号をシャッフル
    if(shuffle_vertex){
        shuffle(perm.begin(), perm.end(), rng);
    }

    vector<vector<int>> G(N);

    for(int i = 1; i < N; i++){

        // [0, i-1] のどれかに接続
        int p = uniform_int_distribution<int>(0, i - 1)(rng);

        int u = perm[i];
        int v = perm[p];

        G[u].push_back(v);
        G[v].push_back(u);
    }

    return G;
}

struct Rerooting {

    int N;
    vector<vector<int>> g;

    // ★ 部分木DP
    // dp[v] =
    // v から部分木内への最大距離
    vector<int> dp;

    // ★ 全方位DP
    // ans[v] =
    // v から全頂点への最大距離
    vector<int> ans;

    // コンストラクタ
    Rerooting(const vector<vector<int>>& graph)
        : g(graph)
    {
        N = g.size();

        // ★ 初期化
        dp.assign(N, 0);
        ans.assign(N, 0);
    }

    // ===== 1回目DFS（部分木DP） =====
    void dfs1(int v, int p){

        // ★ 葉なら 0
        dp[v] = 0;

        for(int to : g[v]){

            if(to == p) continue;

            dfs1(to, v);

            // ★ 子方向へ 1 辺進む
            dp[v] = max(
                dp[v],
                dp[to] + 1
            );
        }
    }

    // ===== 2回目DFS（全方位木DP） =====
    void dfs2(int v, int p, int from_parent){

        int m = g[v].size();

        // ★ 各方向から来る値
        vector<int> val(m);

        for(int i = 0; i < m; i++){

            int to = g[v][i];

            // ★ 親側
            if(to == p){

                val[i] = from_parent;
            }

            // ★ 子側
            else{

                val[i] = dp[to] + 1;
            }
        }

        // ===== prefix/suffix max =====

        vector<int> pref(m + 1, 0);
        vector<int> suff(m + 1, 0);

        for(int i = 0; i < m; i++){

            pref[i + 1] =
                max(pref[i], val[i]);
        }

        for(int i = m - 1; i >= 0; i--){

            suff[i] =
                max(suff[i + 1], val[i]);
        }

        // ★ v を根とした答え
        ans[v] = pref[m];

        // ===== 子へ配る =====

        for(int i = 0; i < m; i++){

            int to = g[v][i];

            if(to == p) continue;

            // ★ 自分以外の最大
            int best =
                max(pref[i], suff[i + 1]);

            // ★ 子から見ると +1
            int nxt = best + 1;

            dfs2(to, v, nxt);
        }
    }

    // 実行
    void build(int root=0){

        // ★ 部分木DP
        dfs1(root, -1);

        // ★ 全方位木DP
        dfs2(root, -1, 0);
    }
};

void random_test(){

    mt19937 rng(
        chrono::steady_clock::now().time_since_epoch().count()
    );

    for(int tc = 0; tc < 10000; tc++){

        // ===== ランダム木生成 =====
        int N = uniform_int_distribution<int>(1, 100)(rng);

        auto G = make_random_tree(N, rng);

        // ===== root =====
        int root =
            uniform_int_distribution<int>(0, N - 1)(rng);

        // ===== 全方位木DP実行 =====
        Rerooting rerooting(G);

        rerooting.build(root);

        // ===== BFSでナイーブ計算 =====
        vector<int> naive(N, 0);

        for(int s = 0; s < N; s++){

            vector<int> dist(N, -1);

            queue<int> q;

            dist[s] = 0;

            q.push(s);

            while(!q.empty()){

                int v = q.front();
                q.pop();

                for(int to : G[v]){

                    if(dist[to] != -1) continue;

                    dist[to] = dist[v] + 1;

                    q.push(to);
                }
            }

            naive[s] =
                *max_element(dist.begin(), dist.end());
        }

        // ===== 比較 =====
        for(int v = 0; v < N; v++){

            if(rerooting.ans[v] != naive[v]){

                cout << "NG\n";

                cout << "N = " << N << "\n";

                cout << "root = " << root << "\n";

                cout << "vertex = " << v << "\n";

                cout << "Rerooting = "
                     << rerooting.ans[v]
                     << "\n";

                cout << "Naive = "
                     << naive[v]
                     << "\n";

                cout << "\nEdges\n";

                for(int i = 0; i < N; i++){
                    for(int to : G[i]){
                        if(i < to){
                            cout << i
                                 << " "
                                 << to
                                 << "\n";
                        }
                    }
                }

                return;
            }
        }
    }

    cout << "Random Test OK\n";
}

void test(){

    /*
              0
            /   \
           1     2
                / \
               3   4
                    \
                     5

        各頂点からの最大距離:

        0 -> 3
        1 -> 4
        2 -> 2
        3 -> 3
        4 -> 3
        5 -> 4
    */

    int N = 6;

    vector<vector<int>> G(N);

    auto add_edge = [&](int u, int v){

        G[u].push_back(v);
        G[v].push_back(u);
    };

    add_edge(0, 1);
    add_edge(0, 2);
    add_edge(2, 3);
    add_edge(2, 4);
    add_edge(4, 5);

    // ===== 全方位木DP実行 =====
    Rerooting rerooting(G);

    rerooting.build(0);

    // ===== 出力 =====
    cout << "各頂点からの最大距離\n";

    for(int v = 0; v < N; v++){

        cout << "頂点 "
             << v
             << " : "
             << rerooting.ans[v]
             << "\n";
    }
}
