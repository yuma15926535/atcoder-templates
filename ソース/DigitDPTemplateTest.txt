struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
============================================================
 Digit DP Template (Competitive Programming Version)
------------------------------------------------------------

巨大整数 N 以下の整数の中で
条件を満たす数を数える。

例
・digit sum
・mod
・特定数字の出現回数
・禁止数字
・bitmask
など

------------------------------------------------------------
DP状態

dp[pos][state][tight]

pos
    何桁目か

state
    問題固有状態

tight
    上限一致
    1 → まだ一致
    0 → 既に小さい

※ leading zero は state 内で管理する

------------------------------------------------------------
計算量

O(桁数 × STATE × 10 × 2)

例
18 × 200 × 10 ≈ 36000
============================================================
*/

/*
------------------------------------------------------------
stateの最大サイズ

問題に応じて変更
今回はlong longの20桁が各桁10種で200あれば十分
------------------------------------------------------------
*/
const int STATE = 200;


/*
------------------------------------------------------------
leading zero 管理

digitがまだ始まってないとき
stateを変更しないケースが多い
------------------------------------------------------------
*/
inline bool is_leading_zero(int state){
    return state == 0;
}

/*
------------------------------------------------------------
state更新関数

digit追加時の状態更新
------------------------------------------------------------
*/
int next_state(int state, int digit, bool started){

    if(!started) return state;

    /*
    ===== ここを書き換える =====

    例1: digit sum
    */

    return state + digit;

    /*
    例2: mod

    return (state*10 + digit) % K;

    例3: 出現回数

    return state + (digit==7);

    例4: bitmask

    return state | (1<<digit);
    */
}

/*
------------------------------------------------------------
digit禁止チェック

例
4と9禁止
------------------------------------------------------------
*/
bool valid_digit(int digit){

    /*
    ===== 必要なら書く =====

    if(digit==4 || digit==9) return false;
    */

    return true;
}

/*
------------------------------------------------------------
最終条件

stateが条件を満たすか
------------------------------------------------------------
*/
bool check(int state){

    /*
    ===== 条件を書く =====

    例
    digit sum ≤ 100
    */

    return state <= 100;
}

/*
============================================================
桁DP本体
============================================================
*/
long long digit_dp(const vector<int>& digits){

    int L = digits.size();

    static long long dp[200][STATE][2];
    memset(dp,0,sizeof(dp));

    dp[0][0][1] = 1;

    for(int pos=0; pos<L; pos++){

        int nd = digits[pos];

        for(int state=0; state<STATE; state++){

            for(int tight=0; tight<=1; tight++){

                long long cur = dp[pos][state][tight];
                if(!cur) continue;

                int limit = tight ? nd : 9;

                for(int digit=0; digit<=limit; digit++){

                    if(!valid_digit(digit)) continue;

                    bool started =
                        !(is_leading_zero(state) && digit==0);

                    int ns = next_state(state,digit,started);

                    if(ns>=STATE) continue;

                    int nt = (tight && digit==limit);

                    dp[pos+1][ns][nt] += cur;
                }
            }
        }
    }

    long long ans = 0;

    for(int state=0; state<STATE; state++){
        for(int tight=0; tight<=1; tight++){

            if(check(state))
                ans += dp[L][state][tight];
        }
    }

    return ans;
}

/*
============================================================
整数 → digit配列
============================================================
*/
vector<int> to_digits(long long x){

    string s = to_string(x);

    vector<int> d;

    for(char c : s)
        d.push_back(c-'0');

    return d;
}

/*
============================================================
0〜N
============================================================
*/
long long solve(long long N){

    if(N<0) return 0;

    return digit_dp(to_digits(N));
}

/*
============================================================
L〜R
============================================================
*/
long long solve_range(long long L,long long R){

    return solve(R) - solve(L-1);
}

void solv(){
    /*
    例

    digit sum ≤ 100

    */

    long long L,R;
    cin >> L >> R;

    cout << solve_range(L,R) << endl;
    cout << digit_dp({9,9,9,9,9,9,9,9,9,9}) << endl;
    return;
}
