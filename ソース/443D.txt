void solv(){
    int N;
    cin>>N;
    vector<int>R(N);
    rep(i,0,N)cin>>R[i];
    ll ans=0;
    //操作で2種類の最適化->左右に分割してもよい　パターン
    //左右の最適化を左からの最適化+右からの最適化に分けて行う
    rep(i,1,N){
        if(R[i-1]+1<R[i])ans+=R[i]-(R[i-1]+1),R[i]=R[i-1]+1;
    }
    rrep(i,N-1,0){
        if(R[i+1]+1<R[i])ans+=R[i]-(R[i+1]+1),R[i]=R[i+1]+1;
    }
    cout<<ans<<'\n';
    return;
}

int main(){
    BUFF;
    int T=1;
    cin>>T;
    while(T--)solv();
    return 0;
}
