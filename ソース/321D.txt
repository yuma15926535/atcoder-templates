#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる

#define rall(a) a.rbegin(),a.rend()

template<class T> using PQ = priority_queue<T, vector<T>>;//大きい順
template<class T> using PQ_g = priority_queue<T, vector<T>, greater<T>>;//小さい順
#define chmax(x,y) x = max(x,y)
#define chmin(x,y) x = min(x,y)
#define popcount __builtin_popcountll

int main(){
    int N,M,P;
    cin>>N>>M>>P;
    vector<ll>A(N),B(M);
    rep(i,0,N)cin>>A[i];
    rep(i,0,M)cin>>B[i];
    sort(nall(A),[](ll a,ll b){return a<b;});
    sort(nall(B),[](ll a,ll b){return a<b;});
    vector<ll>cntA(N),cntB(M,0);
    ll cntP=0;
    {
        int idx=M-1;
        rep(i,0,N){
            while(0<=idx-1&&A[i]+B[idx-1]>=P)idx--;
            if(A[i]+B[idx]>=P){
                cntA[i]=idx;
                cntB[idx]++;
                cntP+=M-idx;
            }else cntA[i]=M;
        }
        rep(i,0,M-1)cntB[i+1]+=cntB[i];
        rep(i,0,M)cntB[i]=N-cntB[i];
    }
    ll ans=cntP*P;
    rep(i,0,N)ans+=cntA[i]*A[i];
    rep(i,0,M)ans+=cntB[i]*B[i];
    cout<<ans<<'\n';
    return 0;
}
