struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

struct GridStateBFS {
    int H, W, S; // S = 状態数
    vector<string> grid;

    vector<vector<vector<int>>> dist;
    vector<vector<vector<tuple<int,int,int>>>> prev;

    int dh[4] = {1,-1,0,0};
    int dw[4] = {0,0,1,-1};
    char dc[4] = {'D','U','R','L'};

    GridStateBFS(int h, int w, int s, const vector<string>& g)
        : H(h), W(w), S(s), grid(g)
    {
        dist.assign(H, vector(W, vector<int>(S, -1)));
        prev.assign(H, vector(W, vector<tuple<int,int,int>>(S)));
    }

    bool in(int i,int j){
        return (0<=i && i<H && 0<=j && j<W);
    }

    // ========================
    // BFS
    // ========================
    template<class F>
    void bfs(int si,int sj, F transition){
        queue<tuple<int,int,int>> q;

        for(int s=0;s<S;s++){
            dist[si][sj][s] = 0;
            prev[si][sj][s] = {-1,-1,-1};
            q.push({si,sj,s});
        }

        while(!q.empty()){
            auto [i,j,s] = q.front(); q.pop();

            // 遷移を外から渡す
            for(auto [ni,nj,ns] : transition(i,j,s)){
                if(!in(ni,nj)) continue;
                if(grid[ni][nj]=='#') continue;
                if(dist[ni][nj][ns]!=-1) continue;

                dist[ni][nj][ns] = dist[i][j][s] + 1;
                prev[ni][nj][ns] = {i,j,s};
                q.push({ni,nj,ns});
            }
        }
    }

    // ========================
    // 経路復元
    // ========================
    vector<char> restore(int gi,int gj){
        int best = -1;
        for(int s=0;s<S;s++){
            if(dist[gi][gj][s]!=-1){
                best = s;
                break;
            }
        }
        if(best==-1) return {};

        vector<char> res;
        int i=gi,j=gj,s=best;

        while(true){
            auto [pi,pj,ps] = prev[i][j][s];
            if(pi==-1) break;

            // 方向推定
            for(int d=0;d<4;d++){
                if(pi + dh[d] == i && pj + dw[d] == j){
                    res.push_back(dc[d]);
                }
            }

            i=pi; j=pj; s=ps;
        }

        reverse(res.begin(),res.end());
        return res;
    }
};

void solv(){
    int H,W;
    cin>>H>>W;
    vector<string>S(H);
    cin>>S;
    int sh,sw,gh,gw;
    rep(i,0,H)rep(j,0,W){
        if(S[i][j]=='S'){
            sh=i;
            sw=j;
        }else if(S[i][j]=='G'){
            gh=i;
            gw=j;
        }
    }

    GridStateBFS bfs(H,W,4,S);

    auto transition = [&](int i,int j,int dir){
        vector<tuple<int,int,int>> res;

        for(int ndir=0; ndir<4; ndir++){
            int ni = i + bfs.dh[ndir];
            int nj = j + bfs.dw[ndir];

            char c = S[i][j];

            if(c=='o' && ndir!=dir) continue;
            if(c=='x' && ndir==dir) continue;

            res.push_back({ni,nj,ndir});
        }

        return res;
    };

    bfs.bfs(sh,sw,transition);

    auto path = bfs.restore(gh,gw);

    if(path.empty()){
        cout<<"No\n";
    }else{
        cout<<"Yes\n";
        for(char c:path) cout<<c;
        cout<<"\n";
    }
    return;
}

int main(){
    BUFF;
    int T=1;
    //cin>>T;
    while(T--)solv();
    return 0;
