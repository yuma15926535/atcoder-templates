struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
==================================================
        Doubling (FINAL COMPLETE VERSION)
--------------------------------------------------
・f(v) を k 回適用
・functional graph
・permutation
・teleport
・親をk個上
全部対応

★ first_true（状態二分探索）
★ step制限版（安全修正版）

計算量:
 build       : O(N logK)
 jump        : O(logK)
 first_true  : O(logK)

IMPORTANT:
 check(v) は O(1) 必須
==================================================
*/
struct Doubling {

    using ll = long long;

    int n;
    int LOG;
    vector<vector<int>> nxt;

    // --------------------------------------------
    // コンストラクタ
    // --------------------------------------------
    Doubling(int n, ll max_k=(1LL<<60)) : n(n){
        LOG=0;
        while((1LL<<LOG)<=max_k) LOG++;
        nxt.assign(LOG, vector<int>(n,-1));
    }

    // --------------------------------------------
    // f(v)
    // --------------------------------------------
    void set_next(int v,int to){
        nxt[0][v]=to;
    }

    // --------------------------------------------
    // build
    // --------------------------------------------
    void build(){
        for(int k=1;k<LOG;k++){
            for(int v=0;v<n;v++){
                int mid=nxt[k-1][v];
                if(mid==-1) continue;
                nxt[k][v]=nxt[k-1][mid];
            }
        }
    }

    // --------------------------------------------
    // k 回遷移
    // --------------------------------------------
    int jump(int v,ll k) const{
        for(int i=0;i<LOG;i++){
            if(k&(1LL<<i)){
                if(v==-1) return -1;
                v=nxt[i][v];
            }
        }
        return v;
    }

    // ==================================================
    // first_true (UNLIMITED SAFE)
    // --------------------------------------------------
    // check(f^k(v)) が単調であること必須
    //
    // true が続く最大地点を返す
    // ==================================================
    template<class F>
    int first_true_unlimited(int start, F ok) {

        if(start == -1) return -1;

        // すでに true
        if(ok(start)) return start;

        int v = start;

        // 「まだ false の最大地点」へ進む
        for(int k = LOG-1; k >= 0; k--) {

            int nv = nxt[k][v];
            if(nv == -1) continue;

            if(!ok(nv)) {
                v = nv;
            }
        }

        // 次が答え候補
        int ans = nxt[0][v];

        if(ans == -1) return -1;
        if(ok(ans)) return ans;

        return -1;
    }

    // ==================================================
    // first_true (STEP LIMITED SAFE)
    // --------------------------------------------------
    // 最大 step 回以内で
    // check を満たす最大地点
    //
    // ★ binary lifting 正式版
    // ==================================================
    template<class F>
    int first_true_limited(int start, long long K, F ok) {

        if(start == -1) return -1;

        if(ok(start)) return start;

        int v = start;

        for(int k = LOG-1; k >= 0; k--) {

            if(!(K & (1LL<<k))) continue;

            int nv = nxt[k][v];
            if(nv == -1) continue;

            if(!ok(nv)) {
                v = nv;
            } else {
                // true側に入ったので細かく降りる
                for(int j = k-1; j >= 0; j--) {
                    int tv = nxt[j][v];
                    if(tv != -1 && !ok(tv))
                        v = tv;
                }
                return nxt[0][v];
            }
        }

        return -1;
    }
};

void solv(){
    cout<<"=============================\n";
    cout<<" Doubling 動作確認テスト開始\n";
    cout<<"=============================\n";

    /*
        グラフ:
        0 → 1 → 2 → 3 → 4 → -1
    */

    int N = 5;
    Doubling db(N,100);

    vector<int> nxt = {1,2,3,4,-1};

    for(int i=0;i<N;i++)
        db.set_next(i,nxt[i]);

    db.build();

    // ------------------------------------------------
    // jump テスト
    // ------------------------------------------------
    cout<<"\n▼ jump テスト\n";

    for(int k=0;k<=6;k++){
        cout<<"0 から "<<k<<" 回遷移 → ";
        cout<<db.jump(0,k)<<"\n";
    }

    // ------------------------------------------------
    // first_true_unlimited
    // ------------------------------------------------
    cout<<"\n▼ first_true_unlimited テスト\n";
    cout<<"条件: 頂点 >= 3\n";

    auto ok1 = [&](int v){
        return v>=3;
    };

    int ans1 = db.first_true_unlimited(0,ok1);

    cout<<"0 から条件を初めて満たす頂点 = ";
    cout<<ans1<<"\n";


    // ------------------------------------------------
    // first_true_limited
    // ------------------------------------------------
    cout<<"\n▼ first_true_limited テスト\n";
    cout<<"条件: 頂点 >= 3\n";

    for(int step=0;step<=5;step++){
        int res = db.first_true_limited(0,step,ok1);

        cout<<"step="<<step
            <<" → "<<res<<"\n";
    }

    cout<<"\n=============================\n";
    cout<<" Doubling 動作確認終了\n";
    cout<<"=============================\n";
    return;
}
