/*
========================================================
 ExtendedDeque（dequeのmax/min/count方向への拡張）
========================================================
【機能】
- push_front / push_back
- pop_front / pop_back
- front / back 取得
- get_min / get_max
- pop_min / pop_max
- count(x) : 値 x の個数
========================================================
*/
template<class T>
struct ExtendedDeque {

    // --- deque本体 ---
    list<T> dq;

    // multiset用ノード
    struct Node {
        T val;
        typename list<T>::iterator it;

        bool operator<(const Node& other) const {
            if (val != other.val) return val < other.val;
            return &(*it) < &(*other.it);
        }
    };

    multiset<Node> ms;

    // 値の個数管理
    map<T, int> freq;

    //============================
    // 基本操作
    //============================

    void push_front(const T& x) {
        dq.push_front(x);
        ms.insert({x, dq.begin()});
        freq[x]++;
    }

    void push_back(const T& x) {
        dq.push_back(x);
        auto it = prev(dq.end());
        ms.insert({x, it});
        freq[x]++;
    }

    void pop_front() {
        if (dq.empty()) return;
        auto it = dq.begin();

        ms.erase(ms.find({*it, it}));
        if (--freq[*it] == 0) freq.erase(*it);

        dq.pop_front();
    }

    void pop_back() {
        if (dq.empty()) return;
        auto it = prev(dq.end());

        ms.erase(ms.find({*it, it}));
        if (--freq[*it] == 0) freq.erase(*it);

        dq.pop_back();
    }

    //============================
    // front / back
    //============================

    T& front() { return dq.front(); }
    T& back()  { return dq.back(); }

    //============================
    // min / max
    //============================

    const T& get_min() const {
        return ms.begin()->val;
    }

    const T& get_max() const {
        return ms.rbegin()->val;
    }

    T pop_min() {
        auto it = ms.begin();
        T val = it->val;

        dq.erase(it->it);
        ms.erase(it);

        if (--freq[val] == 0) freq.erase(val);
        return val;
    }

    T pop_max() {
        auto it = prev(ms.end());
        T val = it->val;

        dq.erase(it->it);
        ms.erase(it);

        if (--freq[val] == 0) freq.erase(val);
        return val;
    }

    //============================

    int count(const T& x) const {
        auto it = freq.find(x);
        return (it == freq.end() ? 0 : it->second);
    }

    int size() const { return dq.size(); }
    bool empty() const { return dq.empty(); }
};

template<class T>
struct OrderStatisticDeque {

    struct Node {
        T val, sum, mn, mx;
        int prio, sz;
        Node *l, *r;

        Node(T v)
            : val(v), sum(v), mn(v), mx(v),
              prio(rand()), sz(1), l(nullptr), r(nullptr) {}
    };

    using np = Node*;
    np root = nullptr;

    //============================
    // 補助関数
    //============================

    int sz(np t){ return t ? t->sz : 0; }
    T sum(np t){ return t ? t->sum : 0; }
    T mn(np t){ return t ? t->mn : numeric_limits<T>::max(); }
    T mx(np t){ return t ? t->mx : numeric_limits<T>::lowest(); }

    void upd(np t){
        if(!t) return;
        t->sz = 1 + sz(t->l) + sz(t->r);
        t->sum = t->val + sum(t->l) + sum(t->r);
        t->mn = min({t->val, mn(t->l), mn(t->r)});
        t->mx = max({t->val, mx(t->l), mx(t->r)});
    }

    // split [0,k) | [k,n)
    void split(np t, np &a, np &b, int k){
        if(!t){ a=b=nullptr; return; }

        if(sz(t->l) >= k){
            split(t->l, a, t->l, k);
            b = t;
        }else{
            split(t->r, t->r, b, k - sz(t->l) - 1);
            a = t;
        }
        upd(t);
    }

    np merge(np a, np b){
        if(!a||!b) return a?a:b;
        if(a->prio > b->prio){
            a->r = merge(a->r, b);
            upd(a);
            return a;
        }else{
            b->l = merge(a, b->l);
            upd(b);
            return b;
        }
    }

    //============================
    // 基本deque操作
    //============================

    void push_front(T x){
        root = merge(new Node(x), root);
    }

    void push_back(T x){
        root = merge(root, new Node(x));
    }

    void pop_front(){
        np a,b;
        split(root,a,b,1);
        root=b;
    }

    void pop_back(){
        np a,b;
        split(root,a,b,sz(root)-1);
        root=a;
    }

    //============================
    // insert / erase
    //============================

    void insert(int k, T x){
        np a,b;
        split(root,a,b,k);
        root = merge(merge(a,new Node(x)),b);
    }

    void erase(int k){
        np a,b,c;
        split(root,a,b,k);
        split(b,b,c,1);
        root = merge(a,c);
    }

    //============================
    // k番目アクセス
    //============================

    T& kth(np t,int k){
        int left=sz(t->l);
        if(k<left) return kth(t->l,k);
        if(k==left) return t->val;
        return kth(t->r,k-left-1);
    }

    T& operator[](int k){
        return kth(root,k);
    }

    //============================
    // range query
    //============================

    tuple<T,T,T> range_query(int l,int r){
        np a,b,c;
        split(root,a,b,l);
        split(b,b,c,r-l+1);

        T S=sum(b);
        T MN=mn(b);
        T MX=mx(b);

        root=merge(a,merge(b,c));
        return {S,MN,MX};
    }

    //============================

    int size(){ return sz(root); }
};
