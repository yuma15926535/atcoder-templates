int main(){
    int N;
    cin>>N;
    vector<ll>A(N),B(N),C(N);
    rep(i,0,N)cin>>A[i];
    rep(i,0,N)cin>>B[i];
    rep(i,0,N)cin>>C[i];
    vector<vector<ll>>dp(3,vector<ll>(N+1,-1e18));
    dp[0][0]=0;
    rep(i,0,N)dp[0][i+1]=dp[0][i]+A[i];
    rep(i,1,N)dp[1][i+1]=max(dp[1][i],dp[0][i])+B[i];
    rep(i,2,N)dp[2][i+1]=max(dp[2][i],dp[1][i])+C[i];
    cout<<dp[2][N]<<'\n';
    return 0;
}
