struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

template<class T>
std::enable_if_t<is_iter_v<T>,ostream&>
operator<<(ostream&os,const T&v){
    using V = std::decay_t<decltype(*begin(v))>;
    if constexpr(is_iter_v<V>){
        bool first_row=true;
        os<<"\n";
        for(auto&&row:v){
            if(!first_row) os<<"\n";
            first_row=false;
            bool f=0;
            for(auto&&x:row){
                if(f) os<<" ";
                os<<x;
                f=1;
            }
        }
    }
    else{
        os<<"[";
        bool f=0;
        for(auto&&x:v){
            if(f) os<<", ";
            os<<x;
            f=1;
        }
        os<<"]";
    }
    return os;
}

    const char*comma=strchr(names,',');
    if(!comma){
        cerr<<DBG_COLOR<<names<<" = "<<x<<"\n"<<DBG_RESET;
    }else{
        cerr<<DBG_COLOR;
        cerr.write(names,comma-names)<<" = "<<x<<" | ";
        _dbg(comma+1,y...);
    }

/*
============================================================
ACL floor_sum 使用テンプレ
============================================================

floor_sum(n, m, a, b)

次の値を計算する

    sum_{i=0}^{n-1} floor((a*i + b) / m)

計算量
    O(log m)

引数
    n : 項数
    m : 割る数
    a : i の係数
    b : 定数

戻り値
    Σ floor((a*i + b)/m)

注意
    i = 0 から n-1 まで
============================================================
*/

void example_basic(){
    cout << "=== 基本例 ===\n";

    ll n = 10;
    ll m = 7;
    ll a = 3;
    ll b = 4;

    /*
    計算するもの

    Σ_{i=0..9} floor((3*i + 4)/7)
    */

    ll ans = floor_sum(n, m, a, b);

    cout << "Σ floor((3*i+4)/7) i=0..9 = " << ans << "\n\n";
}

void example_floor_ai_div_m(){
    cout << "=== よくある形1 ===\n";

    /*
    Σ floor(a*i / m)

    → b=0 にすればいい
    */

    ll n = 10;
    ll a = 5;
    ll m = 7;

    ll ans = floor_sum(n, m, a, 0);

    cout << "Σ floor((5*i)/7) i=0..9 = " << ans << "\n\n";
}

void example_floor_ai_plus_b(){
    cout << "=== よくある形2 ===\n";

    /*
    Σ floor((a*i + b) / m)

    そのまま使える
    */

    ll n = 10;
    ll a = 4;
    ll b = 2;
    ll m = 5;

    ll ans = floor_sum(n, m, a, b);

    cout << "Σ floor((4*i+2)/5) i=0..9 = " << ans << "\n\n";
}

void example_floor_k_i_div_n(){
    cout << "=== よくある形3 ===\n";

    /*
    Σ floor((k*i)/n)

    これも典型
    */

    ll n = 10;
    ll k = 6;

    ll ans = floor_sum(n, n, k, 0);

    cout << "Σ floor((6*i)/10) i=0..9 = " << ans << "\n\n";
}

void example_shift(){
    cout << "=== 区間シフト ===\n";

    /*
    Σ floor((a*(i+L) + b) / m)
    i=0..n-1

    → b をずらす
    */

    ll n = 5;
    ll L = 10;

    ll a = 3;
    ll b = 4;
    ll m = 7;

    ll ans = floor_sum(n, m, a, a*L + b);

    cout << "Σ floor((3*(i+10)+4)/7) i=0..4 = " << ans << "\n\n";
}

//Σ floor(N / i)をO(√N)で計算
long long floor_grouping(long long N){
    long long ans = 0;
    long long i = 1;
    while(i <= N){
        long long v = N / i;
        long long r = N / v;
        ans += v * (r - i + 1);
        i = r + 1;
    }
    return ans;
}

void solv(){
    cout << "ACL floor_sum 動作確認\n\n";
    example_basic();
    example_floor_ai_div_m();

    cout<<"floor_grouping(5)=floor(5/1)+floor(5/2)+floor(5/3)+floor(5/4)+floor(5/5)=5+2+1+1+1\n";
    cout<<floor_grouping(5)<<'\n';;
    return;
}
