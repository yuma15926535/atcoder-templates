/**
 * 2^61 - 1 を法とするローリングハッシュ用のユーティリティ
 */
struct RollingHash {
    static const uint64_t MOD = (1ULL << 61) - 1;
    uint64_t base;
    vector<uint64_t> power;

    RollingHash(uint64_t n, uint64_t b = 0) {
        if (b == 0) {
            mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());
            uniform_int_distribution<uint64_t> dist(128, MOD - 1);
            base = dist(rng);
        } else {
            base = b;
        }
        power.assign(n + 1, 1);
        for (int i = 0; i < n; i++) {
            power[i + 1] = mul(power[i], base);
        }
    }

    static uint64_t add(uint64_t a, uint64_t b) {
        if ((a += b) >= MOD) a -= MOD;
        return a;
    }

    static uint64_t mul(uint64_t a, uint64_t b) {
        __uint128_t ans = (__uint128_t)a * b;
        uint64_t res = (uint64_t)(ans >> 61) + (uint64_t)(ans & MOD);
        if (res >= MOD) res -= MOD;
        return res;
    }
};

// グローバルなRollingHash設定
static RollingHash* rh_ptr_pali = nullptr;

struct S {
    uint64_t hash;
    int len;
};

S op(S a, S b) {
    if (a.len == 0) return b;
    if (b.len == 0) return a;
    return {RollingHash::add(RollingHash::mul(a.hash, rh_ptr_pali->power[b.len]), b.hash), a.len + b.len};
}

S e() {
    return {0, 0};
}

#include <vector>
#include <algorithm>
#include <cassert>

template <class S, S (*op)(S, S), S (*e)()>
struct SegTree {
  public:
    SegTree() : SegTree(0) {}
    explicit SegTree(int n) : SegTree(std::vector<S>(n, e())) {}
    explicit SegTree(const std::vector<S>& v) : _n(int(v.size())) {
        size = 1;
        log = 0;
        while (size < _n) {
            size <<= 1;
            log++;
        }
        d = std::vector<S>(2 * size, e());
        for (int i = 0; i < _n; i++) d[size + i] = v[i];
        for (int i = size - 1; i >= 1; i--) {
            update(i);
        }
    }

    void set(int p, S x) {
        assert(0 <= p && p < _n);
        p += size;
        d[p] = x;
        for (int i = 1; i <= log; i++) update(p >> i);
    }

    S get(int p) const {
        assert(0 <= p && p < _n);
        return d[p + size];
    }

    S prod(int l, int r) const {
        assert(0 <= l && l <= r && r <= _n);
        S sml = e(), smr = e();
        l += size;
        r += size;

        while (l < r) {
            if (l & 1) sml = op(sml, d[l++]);
            if (r & 1) smr = op(d[--r], smr);
            l >>= 1;
            r >>= 1;
        }
        return op(sml, smr);
    }

    S all_prod() const { return d[1]; }

    template <class F> int max_right(int l, F f) const {
        assert(0 <= l && l <= _n);
        assert(f(e()));
        if (l == _n) return _n;
        l += size;
        S sm = e();
        do {
            while (l % 2 == 0) l >>= 1;
            if (!f(op(sm, d[l]))) {
                while (l < size) {
                    l = (2 * l);
                    if (f(op(sm, d[l]))) {
                        sm = op(sm, d[l]);
                        l++;
                    }
                }
                return l - size;
            }
            sm = op(sm, d[l]);
            l++;
        } while ((l & -l) != l);
        return _n;
    }

    template <class F> int min_left(int r, F f) const {
        assert(0 <= r && r <= _n);
        assert(f(e()));
        if (r == 0) return 0;
        r += size;
        S sm = e();
        do {
            r--;
            while (r > 1 && (r % 2)) r >>= 1;
            if (!f(op(d[r], sm))) {
                while (r < size) {
                    r = (2 * r + 1);
                    if (f(op(d[r], sm))) {
                        sm = op(d[r], sm);
                        r--;
                    }
                }
                return r + 1 - size;
            }
            sm = op(d[r], sm);
        } while ((r & -r) != r);
        return 0;
    }

  private:
    int _n, size, log;
    std::vector<S> d;

    void update(int k) { d[k] = op(d[2 * k], d[2 * k + 1]); }
};

/**
 * 前方向と後ろ方向のRollingHashをSegTreeで管理し、回文判定を行うクラス
 */
class PalindromeHash {
    int n;
    SegTree<S, op, e> seg_fw, seg_bw;

public:
    PalindromeHash(const string& s) : n(s.size()) {
        if (rh_ptr_pali == nullptr || (int)rh_ptr_pali->power.size() < n + 1) {
            if (rh_ptr_pali) delete rh_ptr_pali;
            rh_ptr_pali = new RollingHash(n);
        }
        vector<S> fw(n), bw(n);
        for (int i = 0; i < n; i++) {
            fw[i] = { (uint64_t)s[i], 1 };
            bw[i] = { (uint64_t)s[n - 1 - i], 1 };
        }
        seg_fw = SegTree<S, op, e>(fw);
        seg_bw = SegTree<S, op, e>(bw);
    }

    void set(int i, char c) {
        seg_fw.set(i, { (uint64_t)c, 1 });
        seg_bw.set(n - 1 - i, { (uint64_t)c, 1 });
    }

    bool is_palindrome(int l, int r) {
        if (l >= r) return true;
        uint64_t h_fw = seg_fw.prod(l, r).hash;
        uint64_t h_bw = seg_bw.prod(n - r, n - l).hash;
        return h_fw == h_bw;
    }
};

/**
 * PalindromeHashのランダムテスト
 */
void test_palindrome_hash() {
    int n = 50;
    string s = "";
    mt19937 rng(42);
    for (int i = 0; i < n; i++) s += (char)('a' + rng() % 2);

    PalindromeHash ph(s);

    for (int q = 0; q < 500; q++) {
        if (rng() % 2 == 0) { // set
            int i = rng() % n;
            char c = (char)('a' + rng() % 2);
            s[i] = c;
            ph.set(i, c);
        } else { // is_palindrome
            int l = rng() % n, r = l + rng() % (n - l + 1);
            bool res = ph.is_palindrome(l, r);
            string sub = s.substr(l, r - l);
            string rev = sub;
            reverse(rev.begin(), rev.end());
            bool expected = (sub == rev);
            assert(res == expected);
        }
    }
    cout << "PalindromeHash test passed!" << endl;
}
