#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    cin.tie(nullptr);
    ios::sync_with_stdio(false);
    ll N;
    ll S;
    cin>>N>>S;
    vector<ll> A(N);
    rep(i,0,N)cin>>A[i];
    //累積和
    //回転を考えるので2倍用意
    vector<ll> sa(2*N+1,0);
    rep(i,0,2*N)sa[i+1]=sa[i]+A[i%N];
    //modにすることで2周で十分
    //初期化、参照でindexに注意する
    S%=sa[N];
    ll i=0,j=1;
    while(j<=2*N){
        if(sa[j]-sa[i]==S){
            YES;
            return 0;
        }
        if(sa[j]-sa[i]>S)i++;
        else j++;
    }
    NO;
    return 0;
}
