struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
========================================================
 ExtendedDeque（dequeのmax/min/count方向への拡張）
========================================================
【機能】
- push_front / push_back
- pop_front / pop_back
- front / back 取得
- get_min / get_max
- pop_min / pop_max
- count(x) : 値 x の個数
========================================================
*/
template<class T>
struct ExtendedDeque {

    // --- deque本体 ---
    list<T> dq;

    // multiset用ノード
    struct Node {
        T val;
        typename list<T>::iterator it;

        bool operator<(const Node& other) const {
            if (val != other.val) return val < other.val;
            return &(*it) < &(*other.it);
        }
    };

    multiset<Node> ms;

    // 値の個数管理
    map<T, int> freq;

    //============================
    // 基本操作
    //============================

    void push_front(const T& x) {
        dq.push_front(x);
        ms.insert({x, dq.begin()});
        freq[x]++;
    }

    void push_back(const T& x) {
        dq.push_back(x);
        auto it = prev(dq.end());
        ms.insert({x, it});
        freq[x]++;
    }

    void pop_front() {
        if (dq.empty()) return;
        auto it = dq.begin();

        ms.erase(ms.find({*it, it}));
        if (--freq[*it] == 0) freq.erase(*it);

        dq.pop_front();
    }

    void pop_back() {
        if (dq.empty()) return;
        auto it = prev(dq.end());

        ms.erase(ms.find({*it, it}));
        if (--freq[*it] == 0) freq.erase(*it);

        dq.pop_back();
    }

    //============================
    // front / back
    //============================

    T& front() { return dq.front(); }
    T& back()  { return dq.back(); }

    //============================
    // min / max
    //============================

    const T& get_min() const {
        return ms.begin()->val;
    }

    const T& get_max() const {
        return ms.rbegin()->val;
    }

    T pop_min() {
        auto it = ms.begin();
        T val = it->val;

        dq.erase(it->it);
        ms.erase(it);

        if (--freq[val] == 0) freq.erase(val);
        return val;
    }

    T pop_max() {
        auto it = prev(ms.end());
        T val = it->val;

        dq.erase(it->it);
        ms.erase(it);

        if (--freq[val] == 0) freq.erase(val);
        return val;
    }

    //============================

    int count(const T& x) const {
        auto it = freq.find(x);
        return (it == freq.end() ? 0 : it->second);
    }

    int size() const { return dq.size(); }
    bool empty() const { return dq.empty(); }
};

void solv(){
    int N,M;
    cin>>N>>M;
    ExtendedDeque<ll> ed;
    rep(i,0,N){
        ll a;
        cin>>a;
        ed.push_back(a);
    }
    rep(i,0,M){
        ll a=ed.get_max();
        ed.pop_max();
        ed.push_back(a/2);
    }
    ll ans=0;
    while(!ed.empty()){
        ans+=ed.front();
        ed.pop_front();
    }
    cout<<ans<<'\n';
    return;
}
