                os<<x;
                f=1;
            }

        }

    }

// FastModInt (定数mod)
//////////////////////////////////////////////////////////////
template<uint32_t MOD>
struct FastModInt {

    using u32 = uint32_t;
    using u64 = uint64_t;

    struct Barrett {
        u32 m;
        u64 im;
        Barrett(u32 m):m(m),im((u64)(-1)/m+1){}
        u32 reduce(u64 z) const{
            u64 x=(__uint128_t(z)*im)>>64;
            u32 v=z-x*m;
            if(v>=m) v+=m;
            return v;
        }
    };

    static Barrett& bt(){
        static Barrett b(MOD);
        return b;
    }

    u32 v;

    FastModInt():v(0){}
    template<class T>
    FastModInt(T x){
        long long y=x%(long long)MOD;
        if(y<0)y+=MOD;
        v=y;
    }

    u32 val() const{return v;}

    FastModInt& operator+=(const FastModInt& o){
        v+=o.v; if(v>=MOD)v-=MOD; return *this;
    }
    FastModInt& operator-=(const FastModInt& o){
        if(v<o.v)v+=MOD; v-=o.v; return *this;
    }
    FastModInt& operator*=(const FastModInt& o){
        v=bt().reduce((u64)v*o.v); return *this;
    }
    FastModInt& operator/=(const FastModInt& o){
        return *this*=o.inv();
    }

    template<class T> FastModInt& operator+=(T x){return *this+=FastModInt(x);}
    template<class T> FastModInt& operator-=(T x){return *this-=FastModInt(x);}
    template<class T> FastModInt& operator*=(T x){return *this*=FastModInt(x);}
    template<class T> FastModInt& operator/=(T x){return *this/=FastModInt(x);}

    FastModInt pow(long long e) const{
        FastModInt r=1,x=*this;
        while(e){
            if(e&1) r*=x;
            x*=x;
            e>>=1;
        }
        return r;
    }

    FastModInt inv() const{
        return pow(MOD-2);
    }

    FastModInt& operator++(){*this+=1;return *this;}
    FastModInt operator++(int){auto t=*this;++*this;return t;}
    FastModInt& operator--(){*this-=1;return *this;}
    FastModInt operator--(int){auto t=*this;--*this;return t;}

    friend FastModInt operator+(FastModInt a,const FastModInt& b){return a+=b;}
    friend FastModInt operator-(FastModInt a,const FastModInt& b){return a-=b;}
    friend FastModInt operator*(FastModInt a,const FastModInt& b){return a*=b;}
    friend FastModInt operator/(FastModInt a,const FastModInt& b){return a/=b;}

    friend ostream& operator<<(ostream& os,const FastModInt& x){
        return os<<x.v;
    }
};

//////////////////////////////////////////////////////////////
// DynBarrettModInt (動的mod)
//////////////////////////////////////////////////////////////
struct DynBarrettModInt {

    using u32=uint32_t;
    using u64=uint64_t;

    struct Barrett{
        u32 m; u64 im;
        void set(u32 mod){m=mod;im=(u64)(-1)/mod+1;}
        u32 reduce(u64 z) const{
            u64 x=(__uint128_t(z)*im)>>64;
            u32 v=z-x*m;
            if(v>=m)v+=m;
            return v;
        }
    };

    static Barrett& bt(){static Barrett b;return b;}
    static void set_mod(u32 m){bt().set(m);}
    static u32 mod(){return bt().m;}

    u32 v;

    DynBarrettModInt():v(0){}
    template<class T>
    DynBarrettModInt(T x){
        long long y=x%(long long)mod();
        if(y<0)y+=mod();
        v=y;
    }

    u32 val() const{return v;}

    DynBarrettModInt& operator+=(const DynBarrettModInt& o){
        v+=o.v;if(v>=mod())v-=mod();return *this;
    }
    DynBarrettModInt& operator-=(const DynBarrettModInt& o){
        if(v<o.v)v+=mod();v-=o.v;return *this;
    }
    DynBarrettModInt& operator*=(const DynBarrettModInt& o){
        v=bt().reduce((u64)v*o.v);return *this;
    }
    DynBarrettModInt& operator/=(const DynBarrettModInt& o){
        return *this*=o.inv();
    }

    template<class T> DynBarrettModInt& operator+=(T x){return *this+=DynBarrettModInt(x);}
    template<class T> DynBarrettModInt& operator-=(T x){return *this-=DynBarrettModInt(x);}
    template<class T> DynBarrettModInt& operator*=(T x){return *this*=DynBarrettModInt(x);}
    template<class T> DynBarrettModInt& operator/=(T x){return *this/=DynBarrettModInt(x);}

    DynBarrettModInt pow(long long e) const{
        DynBarrettModInt r=1,x=*this;
        while(e){
            if(e&1) r*=x;
            x*=x;
            e>>=1;
        }
        return r;
    }

    DynBarrettModInt inv() const{
        return pow(mod()-2);
    }

    DynBarrettModInt& operator++(){*this+=1;return *this;}
    DynBarrettModInt operator++(int){auto t=*this;++*this;return t;}
    DynBarrettModInt& operator--(){*this-=1;return *this;}
    DynBarrettModInt operator--(int){auto t=*this;--*this;return t;}

    friend DynBarrettModInt operator+(DynBarrettModInt a,const DynBarrettModInt& b){return a+=b;}
    friend DynBarrettModInt operator-(DynBarrettModInt a,const DynBarrettModInt& b){return a-=b;}
    friend DynBarrettModInt operator*(DynBarrettModInt a,const DynBarrettModInt& b){return a*=b;}
    friend DynBarrettModInt operator/(DynBarrettModInt a,const DynBarrettModInt& b){return a/=b;}

    friend ostream& operator<<(ostream& os,const DynBarrettModInt& x){
        return os<<x.v;
    }
};

    cout<<"--- FastModInt (静的mod) ---\n";

    cout<<"b = "<<b<<"\n";

    cout<<"inv(a) = "<<a.inv()<<"\n";

    cout<<"\n[++ / --]\n";
    SM x = a;
    cout<<"初期 x = "<<x<<"\n";
    cout<<"前置++ = "<<++x<<"\n";
    cout<<"後置++ = "<<x++<<" → 現在 "<<x<<"\n";
