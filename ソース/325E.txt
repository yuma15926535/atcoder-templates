template<class T> using PQ = priority_queue<T, vector<T>>;//大きい順
template<class T> using PQ_g = priority_queue<T, vector<T>, greater<T>>;//小さい順
#define chmax(x,y) x = max(x,y)

#define popcount __builtin_popcountll

struct dpb{
    ll dis;
    ll pos;
};

int main(){
    int N;
    ll A,B,C;
    cin>>N>>A>>B>>C;
    vector<vector<ll>>D(N,vector<ll>(N));
    rep(i,0,N)rep(j,0,N)cin>>D[i][j];
    auto comp=[&](dpb a,dpb b){return a.dis>b.dis;};
    priority_queue<dpb,vector<dpb>,decltype(comp)>pq1(comp),pq2(comp);
    vector<bool>rec1(N,true),rec2(N,true);
    pq1.push(dpb{0, 0});
    pq1.push(dpb{1LL<<60, 0});
    pq2.push(dpb{1LL<<60, 0});
    vector<ll>dist1(N,1e18),dist2(N,1e18);
    dist1[0]=0;
    //頂点を倍化させたダイクストラでも2つのの距離が決まるまで処理を続ける必要がある
    while(rec1[N-1]||rec2[N-1]){
        bool b=pq1.top().dis<=pq2.top().dis;
        ll dis;
        int pos;
        if(b){
            dis=pq1.top().dis;
            pos=pq1.top().pos;
            pq1.pop();
        }else{
            dis=pq2.top().dis;
            pos=pq2.top().pos;
            pq2.pop();
        }
        if((b&&(!rec1[pos]))||(!b&&(!rec2[pos])))continue;
        else{
            if(b)rec1[pos]=false;
            else rec2[pos]=false;
            if(b)dist1[pos]=dis;
            else dist2[pos]=dis;
            rep(i,0,N)if(rec1[i]||rec2[i]){
                if(b&&rec1[i])pq1.push(dpb{dis+A*D[pos][i],i});
                if(rec2[i])pq2.push(dpb{dis+B*D[pos][i]+C,i});
            }
        }
    }
    cout<<min(dist1[N-1],dist2[N-1])<<'\n';
    return 0;
}
