    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す

        bool directed = false,

    ){

            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }

    ){

                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }

/**
 * @brief Max BIT (Prefix Max Fenwick Tree)
 * 
 * @details 単調増加更新および Prefix 最大値 (累積最大値) の取得を高速に行うデータ構造。
 * 
 * 通常の BIT との違い:
 * - 通常の BIT は加算 (逆元を持つ演算) を管理するため、任意の区間和 [l, r) を sum(r) - sum(l) で算出できます。
 *   一方、最大値演算 (max) は逆元を持たないため、差分による区間最大値の計算ができません。
 * 
 * なぜ Prefix しか取得できないのか:
 * - 各 tree[idx] にはそのノードがカバーする区間の最大値が記録されており、
 *   インデックスを 1 ずつ減らす方向 (Prefix 方向) へ走査することで、
 *   複数のカバー区間を過不足なくマージして [0, r) の最大値を正しく求められます。
 *   任意の区間 [l, r) では、余分な区間 [0, l) の影響を引いて取り除くことができないため、Prefix クエリに限定されます。
 * 
 * なぜ単調更新しかできないのか:
 * - a[i] の値を元の値より大きくする (単調増加) 更新は、i を含む tree の各位置の最大値を
 *   単に max(tree[idx], x) で更新するだけで正しく伝播できます (他の要素の値に依存しない)。
 *   しかし、値を小さくする更新を行おうとすると、そのノードがカバーする他の要素の最大値を
 *   すべて再計算する必要が生じ、O(log N) での更新が不可能になります。
 * 
 * DPでの使い方:
 * - LIS (最長増加部分列) や、dp[i] = max_{j < i, A[j] < A[i]} { dp[j] } + 1 のような遷移において、
 *   j < i の範囲の最大値を query(i) によって O(log N) で高速に取得するために用います。
 * 
 * 計算量:
 * - 構築: O(N log N)
 * - 単調増加更新 (update): O(log N)
 * - Prefix 最大値取得 (query, all_query): O(log N)
 * - 内部配列取得 (values): O(N)
 * - クリア (clear): O(N)
 * 
 * API一覧:
 * - MaxBIT(n), MaxBIT(a)
 * - build(a), assign(a), clear()
 * - update(i, x)
 * - query(r), all_query()
 * - values()
 * - size(), empty()
 */
template <class T>
class MaxBIT {
private:
    int n;
    T INF_NEG;
    std::vector<T> a;
    std::vector<T> tree;

public:
    /**
     * @brief コンストラクタ (サイズ指定)
     * @param n 要素数
     * @note 計算量: O(N)
     */
    explicit MaxBIT(int n = 0) : n(n), INF_NEG(std::numeric_limits<T>::lowest()), a(n, INF_NEG), tree(n + 1, INF_NEG) {}

    /**
     * @brief コンストラクタ (初期配列指定)
     * @param a_init 初期配列
     * @note 計算量: O(N log N)
     */
    explicit MaxBIT(const std::vector<T>& a_init) : n(a_init.size()), INF_NEG(std::numeric_limits<T>::lowest()), a(a_init.size(), INF_NEG), tree(a_init.size() + 1, INF_NEG) {
        build(a_init);
    }

    /**
     * @brief 初期配列から O(N log N) で構築
     * @param a_init 初期配列
     * @note 計算量: O(N log N)。最大値演算は逆元を持たないため、逐次 update による O(N log N) 構築を行う。
     */
    void build(const std::vector<T>& a_init) {
        n = a_init.size();
        a.assign(n, INF_NEG);
        tree.assign(n + 1, INF_NEG);
        for (int i = 0; i < n; ++i) {
            update(i, a_init[i]);
        }
    }

    /**
     * @brief 新しい配列で再初期化
     * @param a_init 新しい配列
     * @note 計算量: O(N log N)
     */
    void assign(const std::vector<T>& a_init) {
        build(a_init);
    }

    /**
     * @brief 要素数を取得
     * @return 要素数
     * @note 計算量: O(1)
     */
    int size() const {
        return n;
    }

    /**
     * @brief 空であるか判定
     * @return 空ならtrue、そうでなければfalse
     * @note 計算量: O(1)
     */
    bool empty() const {
        return n == 0;
    }

    /**
     * @brief すべての要素を極小値 (-INF) に戻す
     * @note 計算量: O(N)
     */
    void clear() {
        std::fill(a.begin(), a.end(), INF_NEG);
        std::fill(tree.begin(), tree.end(), INF_NEG);
    }

    /**
     * @brief 位置 i の値を x で更新する (単調増加更新: a[i] = max(a[i], x))
     * @param i インデックス (0-index)
     * @param x 更新値
     * @note 計算量: O(log N)。x が現在の a[i] 以下の場合は何も行わない。
     */
    void update(int i, T x) {
        assert(0 <= i && i < n);
        if (a[i] >= x) return;
        a[i] = x;
        for (int idx = i + 1; idx <= n; idx += idx & -idx) {
            tree[idx] = std::max(tree[idx], x);
        }
    }

    /**
     * @brief 半開区間 [0, r) の最大値を取得する
     * @param r 右端のインデックス (0-index)
     * @return 区間内の最大値。区間が空の場合は極小値 (-INF)
     * @note 計算量: O(log N)
     */
    T query(int r) const {
        assert(0 <= r && r <= n);
        T res = INF_NEG;
        for (int idx = r; idx > 0; idx -= idx & -idx) {
            res = std::max(res, tree[idx]);
        }
        return res;
    }

    /**
     * @brief 全体の最大値を取得する
     * @return 全体の最大値
     * @note 計算量: O(log N)
     */
    T all_query() const {
        return query(n);
    }

    /**
     * @brief 現在保持している配列をそのまま取得する
     * @return 内部配列
     * @note 計算量: O(N)
     */
    std::vector<T> values() const {
        return a;
    }
};


/**
 * @brief Min BIT (Prefix Min Fenwick Tree)
 * 
 * @details 単調減少更新および Prefix 最小値 (累積最小値) の取得を高速に行うデータ構造。
 * 
 * 通常の BIT との違い、Prefix 限定の理由、更新制約、および DP での用途は MaxBIT と同様である
 * (最大化と最小化が反転している点のみ異なる)。
 * 
 * 計算量:
 * - 構築: O(N log N)
 * - 単調減少更新 (update): O(log N)
 * - Prefix 最小値取得 (query, all_query): O(log N)
 * - 内部配列取得 (values): O(N)
 * - クリア (clear): O(N)
 * 
 * API一覧:
 * - MinBIT(n), MinBIT(a)
 * - build(a), assign(a), clear()
 * - update(i, x)
 * - query(r), all_query()
 * - values()
 * - size(), empty()
 */
template <class T>
class MinBIT {
private:
    int n;
    T INF_POS;
    std::vector<T> a;
    std::vector<T> tree;

public:
    /**
     * @brief コンストラクタ (サイズ指定)
     * @param n 要素数
     * @note 計算量: O(N)
     */
    explicit MinBIT(int n = 0) : n(n), INF_POS(std::numeric_limits<T>::max()), a(n, INF_POS), tree(n + 1, INF_POS) {}

    /**
     * @brief コンストラクタ (初期配列指定)
     * @param a_init 初期配列
     * @note 計算量: O(N log N)
     */
    explicit MinBIT(const std::vector<T>& a_init) : n(a_init.size()), INF_POS(std::numeric_limits<T>::max()), a(a_init.size(), INF_POS), tree(a_init.size() + 1, INF_POS) {
        build(a_init);
    }

    /**
     * @brief 初期配列から O(N log N) で構築
     * @param a_init 初期配列
     * @note 計算量: O(N log N)
     */
    void build(const std::vector<T>& a_init) {
        n = a_init.size();
        a.assign(n, INF_POS);
        tree.assign(n + 1, INF_POS);
        for (int i = 0; i < n; ++i) {
            update(i, a_init[i]);
        }
    }

    /**
     * @brief 新しい配列で再初期化
     * @param a_init 新しい配列
     * @note 計算量: O(N log N)
     */
    void assign(const std::vector<T>& a_init) {
        build(a_init);
    }

    /**
     * @brief 要素数を取得
     * @return 要素数
     * @note 計算量: O(1)
     */
    int size() const {
        return n;
    }

    /**
     * @brief 空であるか判定
     * @return 空ならtrue、そうでなければfalse
     * @note 計算量: O(1)
     */
    bool empty() const {
        return n == 0;
    }

    /**
     * @brief すべての要素を極大値 (INF) に戻す
     * @note 計算量: O(N)
     */
    void clear() {
        std::fill(a.begin(), a.end(), INF_POS);
        std::fill(tree.begin(), tree.end(), INF_POS);
    }

    /**
     * @brief 位置 i の値を x で更新する (単調減少更新: a[i] = min(a[i], x))
     * @param i インデックス (0-index)
     * @param x 更新値
     * @note 計算量: O(log N)。x が現在の a[i] 以上の場合は何も行わない。
     */
    void update(int i, T x) {
        assert(0 <= i && i < n);
        if (a[i] <= x) return;
        a[i] = x;
        for (int idx = i + 1; idx <= n; idx += idx & -idx) {
            tree[idx] = std::min(tree[idx], x);
        }
    }

    /**
     * @brief 半開区間 [0, r) の最小値を取得する
     * @param r 右端のインデックス (0-index)
     * @return 区間内の最小値。区間が空の場合は極大値 (INF)
     * @note 計算量: O(log N)
     */
    T query(int r) const {
        assert(0 <= r && r <= n);
        T res = INF_POS;
        for (int idx = r; idx > 0; idx -= idx & -idx) {
            res = std::min(res, tree[idx]);
        }
        return res;
    }

    /**
     * @brief 全体の最小値を取得する
     * @return 全体の最小値
     * @note 計算量: O(log N)
     */
    T all_query() const {
        return query(n);
    }

    /**
     * @brief 現在保持している配列をそのまま取得する
     * @return 内部配列
     * @note 計算量: O(N)
     */
    std::vector<T> values() const {
        return a;
    }
};

// ランダムテスト
void random_test_MaxBIT() {
    std::cout << "--- random_test_MaxBIT 開始 ---" << std::endl;

    std::mt19937 mt(42);
    int n = 500;
    int queries = 5000;

    std::vector<long long> naive(n, std::numeric_limits<long long>::lowest());
    MaxBIT<long long> bit(n);

    std::uniform_int_distribution<long long> val_dist(-10000, 10000);
    std::uniform_int_distribution<int> idx_dist(0, n - 1);

    for (int q = 0; q < queries; ++q) {
        int op = mt() % 3;
        if (op == 0) {
            // update
            int i = idx_dist(mt);
            long long x = val_dist(mt);
            naive[i] = std::max(naive[i], x);
            bit.update(i, x);
        } else if (op == 1) {
            // query
            int r = mt() % (n + 1);
            long long naive_max = std::numeric_limits<long long>::lowest();
            for (int i = 0; i < r; ++i) {
                naive_max = std::max(naive_max, naive[i]);
            }
            assert(bit.query(r) == naive_max);
        } else {
            // values, all_query
            std::vector<long long> b_vals = bit.values();
            assert(b_vals == naive);
            
            long long naive_all = std::numeric_limits<long long>::lowest();
            for (int i = 0; i < n; ++i) naive_all = std::max(naive_all, naive[i]);
            assert(bit.all_query() == naive_all);
        }
    }

    std::cout << "OK" << std::endl;
    std::cout << "--- random_test_MaxBIT 終了 ---" << std::endl << std::endl;
}

void random_test_MinBIT() {
    std::cout << "--- random_test_MinBIT 開始 ---" << std::endl;

    std::mt19937 mt(42);
    int n = 500;
    int queries = 5000;

    std::vector<long long> naive(n, std::numeric_limits<long long>::max());
    MinBIT<long long> bit(n);

    std::uniform_int_distribution<long long> val_dist(-10000, 10000);
    std::uniform_int_distribution<int> idx_dist(0, n - 1);

    for (int q = 0; q < queries; ++q) {
        int op = mt() % 3;
        if (op == 0) {
            // update
            int i = idx_dist(mt);
            long long x = val_dist(mt);
            naive[i] = std::min(naive[i], x);
            bit.update(i, x);
        } else if (op == 1) {
            // query
            int r = mt() % (n + 1);
            long long naive_min = std::numeric_limits<long long>::max();
            for (int i = 0; i < r; ++i) {
                naive_min = std::min(naive_min, naive[i]);
            }
            assert(bit.query(r) == naive_min);
        } else {
            // values, all_query
            std::vector<long long> b_vals = bit.values();
            assert(b_vals == naive);
            
            long long naive_all = std::numeric_limits<long long>::max();
            for (int i = 0; i < n; ++i) naive_all = std::min(naive_all, naive[i]);
            assert(bit.all_query() == naive_all);
        }
    }

    std::cout << "OK" << std::endl;
    std::cout << "--- random_test_MinBIT 終了 ---" << std::endl << std::endl;
}

    random_test_MinBIT();
    return;
}
