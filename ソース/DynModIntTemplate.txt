    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す

// ============================================================================
//  (2) DynModInt : mod を実行時に決める modint（素数 mod 用・最速）
// ============================================================================
//  ・int / long long と同じ感覚で使える（暗黙変換・全演算子・cout 直接出力）
//  ・内部は「モンゴメリ乗算 + 遅延簡約」。__int128 不使用なので MSVC でも動く
//  ・atcoder::dynamic_modint の機能はすべて持っている
//    （set_mod / raw / mod / val / pow / inv、id で複数 mod を同時に持てる）
//
//  ★ set_mod に渡せる mod の条件 ★
//        奇数 かつ 1 <= mod <= 2^30 (= 1073741824)
//    ・偶数 mod（＝素数 2）や 2^30 より大きい mod は AnyModInt（③）を使ってください
//    ・条件は DynModInt<>::can_set_mod(m) で事前に確認できます
//    ・mod は素数でなくても正しく動きます（inv が自動で拡張ユークリッドに切り替わる）
//      素数のときは Fermat（pow(mod-2)）を使うので速いです
//
//  速度（g++ -O2 / Xeon(Skylake) / mod=998244353 で実測。
//        現行の DynBarrettModInt ＝ atcoder::dynamic_modint ＝ Barrett との比）
//        x = x*y+c の直列   : 1.22 倍速
//        内積               : 1.35 倍速
//        O(n^2) 畳み込み    : 1.30 倍速
//
//  使い方:
//      using mint = DynModInt<>;
//      mint::set_mod(m);            // m は奇数、m <= 2^30
//      mint a = 5, b = -3;  cout << a * b + 1 << "\n";   // .val() 不要
// ============================================================================
template <int id = -1> struct DynModInt {
  private:
    using mint = DynModInt;
    using u32 = unsigned int;
    using u64 = unsigned long long;
    using i64 = long long;

    // ---- 実行時パラメータ（set_mod で設定）---------------------------------
    inline static u32 M = 1;          // mod
    inline static u32 M2 = 2;         // 2*mod （内部表現の上限。_v < M2）
    inline static u32 NINV = ~0u;     // -M^{-1} mod 2^32
    inline static u32 R2 = 0;         // 2^64 mod M
    inline static u32 ONE_I = 0;      // 1 の内部表現（= 2^32 mod M）
    inline static bool PRIME = false; // M が素数か（inv の方式選択用）

    // モンゴメリ簡約。t < M * 2^32 が前提、戻り値は [0, 2M)
    static u32 mr(u64 t) {
        u32 q = (u32)t * NINV;
        return (u32)((t + (u64)q * M) >> 32);
    }
    static u32 to_internal(u32 x) { return mr((u64)x * R2); }   // 0<=x<M → 内部表現
    u32 norm() const { return _v >= M ? _v - M : _v; }          // 内部表現の正規形

    static u64 cpow(u64 a, u64 n, u64 m) { u64 r = 1 % m; a %= m;
        while (n) { if (n & 1) r = r * a % m; a = a * a % m; n >>= 1; } return r; }
    static bool miller_rabin(u32 n) {            // n < 2^32 で決定的
        if (n <= 1) return false;
        if (n == 2 || n == 7 || n == 61) return true;
        if (n % 2 == 0) return false;
        u32 d = n - 1;
        while (d % 2 == 0) d /= 2;
        const u32 bases[3] = {2u, 7u, 61u};
        for (int i = 0; i < 3; i++) {
            u32 a = bases[i], t = d;
            u64 y = cpow(a, t, n);
            while (t != n - 1 && y != 1 && y != n - 1) { y = y * y % n; t <<= 1; }
            if (y != n - 1 && t % 2 == 0) return false;
        }
        return true;
    }
    // {gcd(a,b), x}  ただし a*x ≡ gcd (mod b), 0 <= x < b/gcd
    static std::pair<i64, i64> inv_gcd(i64 a, i64 b) {
        a %= b; if (a < 0) a += b;
        if (a == 0) return {b, 0};
        i64 s = b, t = a, m0 = 0, m1 = 1;
        while (t) { i64 u = s / t; s -= t * u; m0 -= m1 * u;
                    i64 tmp = s; s = t; t = tmp; tmp = m0; m0 = m1; m1 = tmp; }
        if (m0 < 0) m0 += b / s;
        return {s, m0};
    }

    u32 _v;

  public:
    // ---- mod の設定 --------------------------------------------------------
    static constexpr bool can_set_mod(long long m) {   // 事前チェック用
        return 1 <= m && m <= (1LL << 30) && (m & 1);
    }
    static void set_mod(int m) {
        assert(1 <= m && "DynModInt::set_mod : mod は 1 以上にしてください");
        assert((m & 1) &&
               "DynModInt::set_mod : mod が偶数です。偶数 mod は AnyModInt を使ってください");
        assert(m <= (1 << 30) &&
               "DynModInt::set_mod : mod が 2^30 を超えています。AnyModInt を使ってください");
        M = (u32)m;
        M2 = M * 2;
        u32 x = M;                                   // -M^{-1} mod 2^32 をニュートン法で
        for (int i = 0; i < 5; i++) x *= 2u - M * x;
        NINV = (u32)(0u - x);
        R2 = (u32)((((u64)(-1)) % M) + 1) % M;       // 2^64 mod M
        ONE_I = (u32)((((u64)1) << 32) % M);         // 2^32 mod M
        PRIME = miller_rabin(M);
    }
    static int mod() { return (int)M; }
    static bool mod_is_prime() { return PRIME; }

    // ---- 構築 --------------------------------------------------------------
    DynModInt() : _v(0) {}

    template <class T, std::enable_if_t<std::is_integral_v<T> && std::is_signed_v<T>>* = nullptr>
    DynModInt(T x) {
        if constexpr (sizeof(T) <= 4) {
            int y = (int)x % (int)M; if (y < 0) y += (int)M; _v = to_internal((u32)y);
        } else {
            i64 y = (i64)x % (i64)M; if (y < 0) y += (i64)M; _v = to_internal((u32)y);
        }
    }
    template <class T, std::enable_if_t<std::is_integral_v<T> && !std::is_signed_v<T>>* = nullptr>
    DynModInt(T x) {
        if constexpr (sizeof(T) <= 4) _v = to_internal((u32)x % M);
        else _v = to_internal((u32)((u64)x % M));
    }

    // 0 <= x < mod がわかっているときの高速構築（剰余算を省く）
    static mint raw(int x) {
        assert(0 <= x && (u32)x < M);
        mint r; r._v = to_internal((u32)x); return r;
    }

    // ---- 取得 --------------------------------------------------------------
    int val() const { u32 r = mr(_v); return (int)(r >= M ? r - M : r); }
    explicit operator bool() const { return norm() != 0; }

    // ---- 加減乗除 ----------------------------------------------------------
    mint& operator+=(const mint& o) { _v += o._v; if (_v >= M2) _v -= M2; return *this; }
    mint& operator-=(const mint& o) { if (_v < o._v) _v += M2; _v -= o._v; return *this; }
    mint& operator*=(const mint& o) { _v = mr((u64)_v * o._v); return *this; }
    mint& operator/=(const mint& o) { return *this *= o.inv(); }

    mint operator+() const { return *this; }
    mint operator-() const { mint r; r._v = _v ? M2 - _v : 0; return r; }

    mint& operator++() { _v += ONE_I; if (_v >= M2) _v -= M2; return *this; }
    mint& operator--() { if (_v < ONE_I) _v += M2; _v -= ONE_I; return *this; }
    mint operator++(int) { mint t = *this; ++*this; return t; }
    mint operator--(int) { mint t = *this; --*this; return t; }

    friend mint operator+(mint a, const mint& b) { return a += b; }
    friend mint operator-(mint a, const mint& b) { return a -= b; }
    friend mint operator*(mint a, const mint& b) { return a *= b; }
    friend mint operator/(mint a, const mint& b) { return a /= b; }

    // ---- 比較（値 val() での比較。map / sort でそのまま使える）--------------
    friend bool operator==(const mint& a, const mint& b) { return a.norm() == b.norm(); }
    friend bool operator!=(const mint& a, const mint& b) { return a.norm() != b.norm(); }
    friend bool operator<(const mint& a, const mint& b) { return a.val() < b.val(); }
    friend bool operator>(const mint& a, const mint& b) { return a.val() > b.val(); }
    friend bool operator<=(const mint& a, const mint& b) { return a.val() <= b.val(); }
    friend bool operator>=(const mint& a, const mint& b) { return a.val() >= b.val(); }

    // ---- pow / inv ---------------------------------------------------------
    mint pow(long long n) const {          // 負の指数もOK（dynamic_modint の上位互換）
        mint x = *this, r = 1;
        unsigned long long e;
        if (n < 0) { x = x.inv(); e = (unsigned long long)(-(n + 1)) + 1ULL; }
        else e = (unsigned long long)n;
        while (e) { if (e & 1) r *= x; x *= x; e >>= 1; }
        return r;
    }
    mint inv() const {
        if (PRIME) {   // mod が素数なら Fermat の小定理で（分岐は inv 呼び出し時のみ）
            assert(norm() != 0 && "DynModInt::inv() : 0 の逆元は存在しません");
            return pow((long long)M - 2);
        }
        auto eg = inv_gcd((i64)val(), (i64)M);
        assert(eg.first == 1 && "DynModInt::inv() : 逆元が存在しません（gcd != 1）");
        return mint(eg.second);
    }

    // ---- 入出力 ------------------------------------------------------------
    friend std::ostream& operator<<(std::ostream& os, const mint& a) { return os << a.val(); }
    friend std::istream& operator>>(std::istream& is, mint& a) { long long x; is >> x; a = x; return is; }
};

struct Ref {
    long long m;
    explicit Ref(long long m) : m(m) {}
    long long nrm(long long x) const { x %= m; if (x < 0) x += m; return x; }
    long long add(long long a, long long b) const { return (a + b) % m; }
    long long sub(long long a, long long b) const { return ((a - b) % m + m) % m; }
    long long mul(long long a, long long b) const { return a * b % m; }
    long long pw(long long a, long long e) const {
        long long r = 1 % m, x = a % m;
        while (e) { if (e & 1) r = mul(r, x); x = mul(x, x); e >>= 1; }
        return r;
    }
    long long gcd(long long a, long long b) const { while (b) { long long t = a % b; a = b; b = t; } return a; }
};

template <int id> void test_one_mod(int MOD, int iters, std::mt19937_64& rng) {
    using M = DynModInt<id>;
    M::set_mod(MOD);
    Ref ref(MOD);
    CHECK(M::mod() == MOD, "mod()");
    CHECK(M().val() == 0, "default ctor");

    std::vector<long long> cand = {0, 1, 2, MOD - 1LL, MOD - 2LL, (long long)MOD, MOD + 1LL,
                                   -1, -2, -(long long)MOD, -(long long)MOD - 1};
    for (int it = 0; it < iters; it++) {
        long long ra, rb;
        if (it < (int)(cand.size() * cand.size())) { ra = cand[it / cand.size()]; rb = cand[it % cand.size()]; }
        else {
            ra = (long long)(rng() % (unsigned)MOD);
            rb = (long long)(rng() % (unsigned)MOD);
            if (it % 7 == 0) ra = -(long long)(rng() % 1000000000000000000ULL);
            if (it % 11 == 0) rb = (long long)(rng() % 1000000000000000000ULL);
        }
        M a = ra, b = rb;
        long long x = ref.nrm(ra), y = ref.nrm(rb);

        CHECK(a.val() == (int)x && b.val() == (int)y, "ctor / val");
        CHECK(0 <= a.val() && a.val() < MOD, "val range");
        CHECK((a + b).val() == (int)ref.add(x, y), "operator+");
        CHECK((a - b).val() == (int)ref.sub(x, y), "operator-");
        CHECK((a * b).val() == (int)ref.mul(x, y), "operator*");
        CHECK((-a).val() == (int)ref.sub(0, x), "unary -");
        CHECK((+a).val() == (int)x, "unary +");
        { M t = a; t += b; CHECK(t.val() == (int)ref.add(x, y), "+="); }
        { M t = a; t -= b; CHECK(t.val() == (int)ref.sub(x, y), "-="); }
        { M t = a; t *= b; CHECK(t.val() == (int)ref.mul(x, y), "*="); }

        CHECK((a + 5).val() == (int)ref.add(x, ref.nrm(5)), "mint + int");
        CHECK((5 + a).val() == (int)ref.add(ref.nrm(5), x), "int + mint");
        CHECK((a - 7LL).val() == (int)ref.sub(x, ref.nrm(7)), "mint - ll");
        CHECK((7LL - a).val() == (int)ref.sub(ref.nrm(7), x), "ll - mint");
        CHECK((a * 3u).val() == (int)ref.mul(x, ref.nrm(3)), "mint * uint");
        CHECK((3u * a).val() == (int)ref.mul(ref.nrm(3), x), "uint * mint");

        CHECK((a == b) == (x == y), "operator==");
        CHECK((a != b) == (x != y), "operator!=");
        CHECK((a < b) == (x < y), "operator<");
        CHECK((a >= b) == (x >= y), "operator>=");
        CHECK((bool)a == (x != 0), "operator bool");

        { M t = a; CHECK((++t).val() == (int)ref.add(x, 1 % MOD), "pre ++"); }
        { M t = a; CHECK((t++).val() == (int)x, "post ++");
                   CHECK(t.val() == (int)ref.add(x, 1 % MOD), "post ++ result"); }
        { M t = a; CHECK((--t).val() == (int)ref.sub(x, 1 % MOD), "pre --"); }
        { M t = a; CHECK((t--).val() == (int)x, "post --");
                   CHECK(t.val() == (int)ref.sub(x, 1 % MOD), "post -- result"); }

        long long e = (long long)(rng() % 1000000);
        CHECK(a.pow(e).val() == (int)ref.pw(x, e), "pow");
        CHECK(a.pow(0).val() == (int)(1 % MOD), "pow 0");

        if (ref.gcd(y, MOD) == 1) {
            M ib = b.inv();
            CHECK((b * ib).val() == (int)(1 % MOD), "inv");
            CHECK((a / b).val() == (int)ref.mul(x, ib.val()), "operator/");
            CHECK(((a / b) * b).val() == (int)x, "div then mul");
            { M t = a; t /= b; CHECK(t.val() == (int)ref.mul(x, ib.val()), "/="); }
            CHECK(b.pow(-1) == ib, "pow(-1)");
            CHECK(b.pow(-3) == ib * ib * ib, "pow(-3)");
        }
        CHECK(M::raw((int)x).val() == (int)x, "raw");
        { std::ostringstream os; os << a; CHECK(os.str() == std::to_string(x), "operator<<"); }
        { std::istringstream is(std::to_string(ra)); M t; is >> t; CHECK(t.val() == (int)x, "operator>>"); }
    }
    for (int it = 0; it < iters; it++) {
        M a = (long long)(rng() % (unsigned)MOD), b = (long long)(rng() % (unsigned)MOD),
          c = (long long)(rng() % (unsigned)MOD);
        CHECK((a + b) + c == a + (b + c), "add assoc");
        CHECK((a * b) * c == a * (b * c), "mul assoc");
        CHECK(a * (b + c) == a * b + a * c, "distrib");
        CHECK(a - b == a + (-b), "sub == add neg");
    }
}

void random_test() {
    std::cout << "[1] ランダムテスト（DynModInt vs 素朴実装）\n";
    std::mt19937_64 rng(20260909);
    const int N = 4000;

    std::vector<std::pair<int, const char*>> mods = {
        {1, "境界"}, {3, "素数"}, {5, "素数"}, {97, "素数"}, {65537, "素数"},
        {167772161, "NTT素数"}, {469762049, "NTT素数"}, {754974721, "NTT素数"},
        {998244353, "NTT素数"}, {1000000007, "素数"}, {1000000009, "素数"},
        {1073741789, "2^30 未満で最大の素数"},
        {9, "非素数(奇)"}, {15, "非素数(奇)"}, {999999999, "非素数(奇)"}, {1073741823, "2^30-1(非素数)"},
    };
    for (auto& [m, note] : mods) {
        std::cout << "    mod = " << m << " ... " << std::flush;
        test_one_mod<-1>(m, N, rng);
        std::cout << "OK  (" << note << ")\n";
    }

    // ランダムな奇数 mod でも総当り的に確認
    std::cout << "    ランダムな奇数 mod 300 種 ... " << std::flush;
    for (int i = 0; i < 300; i++) {
        int m = (int)(rng() % (unsigned)(1 << 30)) | 1;
        if (m < 1) m = 1;
        test_one_mod<-1>(m, 60, rng);
    }
    std::cout << "OK\n";

    // id を変えると別 mod を同時に持てること（dynamic_modint と同じ）
    DynModInt<0>::set_mod(998244353);
    DynModInt<1>::set_mod(1000000007);
    CHECK(DynModInt<0>::mod() == 998244353 && DynModInt<1>::mod() == 1000000007, "multi id");
    CHECK((DynModInt<0>(-1)).val() == 998244352, "multi id value");
    CHECK((DynModInt<1>(-1)).val() == 1000000006, "multi id value");
    std::cout << "    id 違いで mod を同時保持 ... OK\n";

    // can_set_mod
    CHECK(DynModInt<>::can_set_mod(998244353), "can_set_mod ok");
    CHECK(!DynModInt<>::can_set_mod(2), "can_set_mod even");
    CHECK(!DynModInt<>::can_set_mod((1LL << 30) + 1), "can_set_mod big");
    std::cout << "    can_set_mod の判定 ... OK\n";

    std::cout << "    → 全 " << g_checks << " 項目 一致\n\n";
}

// ============================================================================
void demo() {
    using namespace std;
    using DM = DynModInt<>;

    cout << "[2] 動作確認 : DynModInt\n";
    cout << "=============================\n";

    DM::set_mod(1000000007);
    cout << "set_mod(1000000007)   mod() = " << DM::mod()
         << " , 素数か = " << DM::mod_is_prime() << "\n\n";

    DM p = 555555555;
    DM q = 333333333;
    cout << "p = " << p << "\n";
    cout << "q = " << q << "\n";

    cout << "\n[基本演算]\n";
    cout << "p + q = " << p + q << "\n";
    cout << "p - q = " << p - q << "\n";
    cout << "p * q = " << p * q << "\n";
    cout << "p / q = " << p / q << "\n";
    cout << "-p    = " << -p << "\n";

    cout << "\n[pow / inv]\n";
    cout << "pow(p,10) = " << p.pow(10) << "\n";
    cout << "pow(p,-1) = " << p.pow(-1) << "   (負の指数も可)\n";
    cout << "inv(p)    = " << p.inv() << "\n";
    cout << "p*inv(p)  = " << p * p.inv() << "\n";

    cout << "\n[++ / --]\n";
    DM y = p;
    cout << "初期 y = " << y << "\n";
    cout << "前置++ = " << ++y << "\n";
    cout << "後置++ = " << y++ << " → 現在 " << y << "\n";
    cout << "前置-- = " << --y << "\n";
    cout << "後置-- = " << y-- << " → 現在 " << y << "\n";

    cout << "\n[整数混在]\n";
    cout << "p + 7  = " << p + 7 << "\n";
    cout << "7 + p  = " << 7 + p << "\n";
    cout << "p - 50 = " << p - 50 << "\n";
    cout << "p * 9  = " << p * 9 << "\n";
    cout << "p / 11 = " << p / 11 << "\n";

    cout << "\n[チェーン演算]\n";
    cout << "(p+8)*(q-3)/6 = " << (p + 8) * (q - 3) / 6 << "\n";

    cout << "\n[正規化確認]\n";
    cout << "DM(-999) = " << DM(-999) << "\n";
    cout << "DM(1e18) = " << DM(1000000000000000000LL) << "\n";

    cout << "\n[mod を途中で変えられる]\n";
    DM::set_mod(998244353);
    cout << "set_mod(998244353) → DM(-1) = " << DM(-1) << "\n";
    DM::set_mod(13);
    cout << "set_mod(13)        → DM(-1) = " << DM(-1) << " , 100/7 = " << DM(100) / 7 << "\n";

    cout << "\n[非素数 mod でも動く（逆元は拡張ユークリッド）]\n";
    DM::set_mod(15);
    cout << "set_mod(15) 素数か = " << DM::mod_is_prime() << "\n";
    cout << "7 の逆元 = " << DM(7).inv() << " , 7*inv = " << DM(7) * DM(7).inv() << "\n";

    cout << "\n[id で複数 mod を同時に使う]\n";
    DynModInt<0>::set_mod(998244353);
    DynModInt<1>::set_mod(1000000007);
    cout << "DynModInt<0>(-1) = " << DynModInt<0>(-1) << "\n";
    cout << "DynModInt<1>(-1) = " << DynModInt<1>(-1) << "\n";

    cout << "\n[実用例] mod が入力で与えられる階乗・二項係数\n";
    DM::set_mod(1000000007);
    {
        const int N = 200000;
        vector<DM> fac(N + 1), ifac(N + 1);
        fac[0] = 1;
        for (int i = 1; i <= N; i++) fac[i] = fac[i - 1] * i;
        ifac[N] = fac[N].inv();
        for (int i = N; i >= 1; i--) ifac[i - 1] = ifac[i] * i;
        auto C = [&](int n, int r) { return (r < 0 || r > n) ? DM(0) : fac[n] * ifac[r] * ifac[n - r]; };
        cout << "C(200000,100000) = " << C(200000, 100000) << "\n";
        cout << "C(10,3)          = " << C(10, 3) << "   (= 120)\n";
    }

    cout << "\n=============================\n";
    cout << "動作確認 完了\n\n";
}

// ============================================================================
//  簡易ベンチ（あくまで目安。タイトなループはコード配置の影響で
//  同じ処理でも 1〜2 割ぶれることがあります）
// ============================================================================
void benchmark() {
    using namespace std;
    using namespace std::chrono;
    cout << "[3] 簡易ベンチ (mod = 998244353)\n";
    using M = DynModInt<>;
    M::set_mod(998244353);
    {
        M x = 12345, y = 123456789, c = 7;
        auto t0 = steady_clock::now();
        for (int i = 0; i < 30000000; i++) x = x * y + c;
        auto t1 = steady_clock::now();
        cout << "    x = x*y+c を 3*10^7 回 : " << duration<double, milli>(t1 - t0).count()
             << " ms   (結果 " << x << ")\n";
    }
    {
        const int n = 2000;
        vector<M> a(n), b(n), c(2 * n);
        mt19937 rng(1);
        for (int i = 0; i < n; i++) { a[i] = (long long)rng(); b[i] = (long long)rng(); }
        auto t0 = steady_clock::now();
        for (int i = 0; i < n; i++) for (int j = 0; j < n; j++) c[i + j] += a[i] * b[j];
        auto t1 = steady_clock::now();
        cout << "    2000x2000 の畳み込み   : " << duration<double, milli>(t1 - t0).count()
             << " ms   (結果 " << c[n] << ")\n";
    }
    cout << "\n";
}
