struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
============================================================
 Layer0 : BaseCounter
------------------------------------------------------------
【役割】
値 → 個数 を管理する唯一のコア構造

すべての Counter の土台。
insert / erase はここにのみ存在する。

------------------------------------------------------------
【保持】
cnt[x]        : 出現回数
total_size    : 総要素数
distinct_size : 種類数

------------------------------------------------------------
【計算量】
insert / erase : O(1) average
query系        : O(1)
============================================================
*/
template<class T>
struct BaseCounter {

    unordered_map<T,long long> cnt;

    long long total_size = 0;
    long long distinct_size = 0;

protected:
    /*
        Hook関数
        派生Layerが更新検知に使う
    */
    virtual void on_update(
        const T&, long long, long long){}

public:

    // =============================
    // insert
    // =============================
    void insert(const T& x, long long k=1){
        if(k<=0) return;

        long long old = cnt[x];
        long long now = old + k;

        cnt[x] = now;

        if(old==0) distinct_size++;
        total_size += k;

        on_update(x, old, now);
    }

    // =============================
    // erase
    // =============================
    void erase(const T& x, long long k=1){

        auto it = cnt.find(x);
        if(it==cnt.end()) return;

        long long old = it->second;
        long long dec = min(k, old);
        long long now = old - dec;

        total_size -= dec;

        if(now==0){
            cnt.erase(it);
            distinct_size--;
        }else{
            it->second = now;
        }

        on_update(x, old, now);
    }

    long long count(const T& x) const{
        auto it=cnt.find(x);
        return it==cnt.end()?0:it->second;
    }

    bool exist(const T& x) const{
        return count(x)>0;
    }

    long long size() const{
        return total_size;
    }

    long long distinct() const{
        return distinct_size;
    }

    bool empty() const{
        return total_size==0;
    }
};

/*
============================================================
 Layer1 : OrderedCounter
------------------------------------------------------------
【追加能力】
順序付き取得

min / max / lower_bound

------------------------------------------------------------
【計算量】
insert / erase : O(logN)
query          : O(1) or O(logN)
============================================================
*/
template<class T>
struct OrderedCounter : BaseCounter<T>{

    using Base = BaseCounter<T>;
    set<T> S;

protected:
    void on_update(
        const T& x,
        long long old,
        long long now
    ) override {

        if(old==0 && now>0)
            S.insert(x);

        if(old>0 && now==0)
            S.erase(x);
    }

public:

    const T& min() const{
        return *S.begin();
    }

    const T& max() const{
        return *S.rbegin();
    }

    const T& get_any() const{
        return *S.begin();
    }

    auto lower_bound(const T& x) const{
        return S.lower_bound(x);
    }
};

/*
============================================================
 Layer2 : FrequencyCounter
------------------------------------------------------------
【追加能力】
頻度情報管理

freq_cnt[f]
 = 出現回数fの値の種類数

------------------------------------------------------------
【計算量】
すべて O(1)
============================================================
*/
template<class T>
struct FrequencyCounter : BaseCounter<T>{

    using Base = BaseCounter<T>;

    unordered_map<long long,long long> freq_cnt;
    long long max_freq = 0;

protected:
    void on_update(
        const T&,
        long long old,
        long long now
    ) override {

        if(old>0) freq_cnt[old]--;

        if(now>0){
            freq_cnt[now]++;
            max_freq = max(max_freq, now);
        }

        while(max_freq>0 &&
              freq_cnt[max_freq]==0)
            max_freq--;
    }

public:

    long long freq_kind(long long f) const{
        auto it=freq_cnt.find(f);
        return it==freq_cnt.end()?0:it->second;
    }

    long long get_max_freq() const{
        return max_freq;
    }
};

/*
============================================================
 Layer3 : BucketCounter
------------------------------------------------------------
【追加能力】
mode（最頻値）取得

bucket[f] = 頻度fの値集合

------------------------------------------------------------
【計算量】
insert / erase : O(logN)
mode           : O(1)
============================================================
*/
template<class T>
struct BucketCounter : BaseCounter<T>{

    using Base = BaseCounter<T>;

    unordered_map<long long,set<T>> bucket;
    long long max_freq = 0;

protected:
    void on_update(
        const T& x,
        long long old,
        long long now
    ) override {

        if(old>0)
            bucket[old].erase(x);

        if(now>0){
            bucket[now].insert(x);
            max_freq=max(max_freq,now);
        }

        while(max_freq>0 &&
              bucket[max_freq].empty())
            max_freq--;
    }

public:

    const T& mode() const{
        return *bucket.at(max_freq).begin();
    }

    long long get_max_freq() const{
        return max_freq;
    }
};

/*
============================================================
 Layer4 : MexCounter
------------------------------------------------------------
【役割】
mex（最小非負未出現整数）を管理

BaseCounter を拡張し、
存在しない値集合を維持する。

------------------------------------------------------------
【前提】
0 <= 値 <= LIMIT

LIMIT をコンストラクタで指定する

------------------------------------------------------------
【保持】
missing :
    現在存在しない整数集合

------------------------------------------------------------
【計算量】
insert / erase : O(logN)
mex()          : O(1)

============================================================
*/
template<class T=int>
struct MexCounter : BaseCounter<T>{

    using Base = BaseCounter<T>;

    set<T> missing;
    T LIMIT;

    // =============================
    // constructor
    // =============================
    MexCounter(T limit = 200000)
        : LIMIT(limit)
    {
        for(T i=0;i<=LIMIT;i++)
            missing.insert(i);
    }

protected:

    /*
        BaseCounter から呼ばれる Hook
    */
    void on_update(
        const T& x,
        long long old,
        long long now
    ) override {

        if(x < 0 || x > LIMIT) return;

        // 出現した
        if(old==0 && now>0)
            missing.erase(x);

        // 消えた
        if(old>0 && now==0)
            missing.insert(x);
    }

public:

    // =============================
    // mex取得
    // =============================
    T mex() const{
        return *missing.begin();
    }

    // =============================
    // clear
    // =============================
    void clear(){

        Base::cnt.clear();
        Base::total_size = 0;
        Base::distinct_size = 0;

        missing.clear();
        for(T i=0;i<=LIMIT;i++)
            missing.insert(i);
    }
};
