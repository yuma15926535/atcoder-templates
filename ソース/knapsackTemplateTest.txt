// ---------- 0/1 ナップサック（YES / NO） ----------
// res[w] = 重さ w が作れるか
vector<bool> knapsack_bool(int N, int W, const vector<int>& wt) {
    vector<bool> dp(W+1, false);
    dp[0] = true;

    for (int i = 0; i < N; i++) {
        for (int w = W; w >= wt[i]; w--) {
            if (dp[w - wt[i]]) dp[w] = true;
        }
    }
    return dp;
}

// ---------- 完全ナップサック（YES / NO） ----------
vector<bool> knapsack_bool_unbounded(int N, int W, const vector<int>& wt) {
    vector<bool> dp(W+1, false);
    dp[0] = true;

    for (int i = 0; i < N; i++) {
        for (int w = wt[i]; w <= W; w++) {
            if (dp[w - wt[i]]) dp[w] = true;
        }
    }
    return dp;
}

// ---------- 個数制限付き（YES / NO） ----------
vector<bool> knapsack_bool_bounded(int N, int W,
                                   const vector<int>& wt,
                                   const vector<int>& cnt) {
    vector<bool> dp(W+1, false);
    dp[0] = true;

    for (int i = 0; i < N; i++) {
        vector<bool> next = dp;
        for (int w = 0; w <= W; w++) {
            if (!dp[w]) continue;
            for (int k = 1; k <= cnt[i]; k++) {
                if (w + k * wt[i] > W) break;
                next[w + k * wt[i]] = true;
            }
        }
        dp.swap(next);
    }
    return dp;
}

// ---------- 0/1 ナップサック（最小個数） ----------
// res[w] = 重さ w を作るための最小使用個数（作れない場合 -1）
vector<int> knapsack_min_count(int N, int W, const vector<int>& wt) {
    const int INF = 1e9;
    vector<int> dp(W+1, INF);
    dp[0] = 0;

    for (int i = 0; i < N; i++) {
        for (int w = W; w >= wt[i]; w--) {
            dp[w] = min(dp[w], dp[w - wt[i]] + 1);
        }
    }

    vector<int> res(W+1, -1);
    for (int w = 0; w <= W; w++) {
        if (dp[w] != INF) res[w] = dp[w];
    }
    return res;
}

// ---------- 完全ナップサック（最小個数） ----------
vector<int> knapsack_min_count_unbounded(int N, int W, const vector<int>& wt) {
    const int INF = 1e9;
    vector<int> dp(W+1, INF);
    dp[0] = 0;

    for (int i = 0; i < N; i++) {
        for (int w = wt[i]; w <= W; w++) {
            dp[w] = min(dp[w], dp[w - wt[i]] + 1);
        }
    }

    vector<int> res(W+1, -1);
    for (int w = 0; w <= W; w++) {
        if (dp[w] != INF) res[w] = dp[w];
    }
    return res;
}

// ---------- 個数制限付き（最小個数） ----------
vector<int> knapsack_min_count_bounded(int N, int W,
                                       const vector<int>& wt,
                                       const vector<int>& cnt) {
    const int INF = 1e9;
    vector<int> dp(W+1, INF);
    dp[0] = 0;

    for (int i = 0; i < N; i++) {
        vector<int> next = dp;
        for (int w = 0; w <= W; w++) {
            if (dp[w] == INF) continue;
            for (int k = 1; k <= cnt[i]; k++) {
                if (w + k * wt[i] > W) break;
                next[w + k * wt[i]] = min(next[w + k * wt[i]], dp[w] + k);
            }
        }
        dp.swap(next);
    }

    vector<int> res(W+1, -1);
    for (int w = 0; w <= W; w++) {
        if (dp[w] != INF) res[w] = dp[w];
    }
    return res;
}

/* ==================================================
   3. 最大個数ナップサック（0/1）
   ================================================== */

// dp[w] = 重さ w を作るための最大使用個数
vector<int> knapsack_max_count(int N, int W, const vector<int>& wt) {
    const int NEG = -1e9;
    vector<int> dp(W+1, NEG);
    dp[0] = 0;

    for (int i = 0; i < N; i++) {
        for (int w = W; w >= wt[i]; w--) {
            if (dp[w - wt[i]] != NEG) {
                dp[w] = max(dp[w], dp[w - wt[i]] + 1);
            }
        }
    }
    return dp;
}

/* ==================================================
   4. 0/1 ナップサック（価値最大化 + 復元）
   ================================================== */

void knapsack01_restore(int N, int W,
                        const vector<int>& wt,
                        const vector<int>& val,
                        vector<int>& chosen) {
    vector<vector<long long>> dp(N+1, vector<long long>(W+1, -1));
    vector<vector<bool>> use(N+1, vector<bool>(W+1, false));

    dp[0][0] = 0;

    for (int i = 0; i < N; i++) {
        for (int w = 0; w <= W; w++) {
            if (dp[i][w] < 0) continue;

            // 使わない
            if (dp[i+1][w] < dp[i][w]) {
                dp[i+1][w] = dp[i][w];
                use[i+1][w] = false;
            }

            // 使う
            if (w + wt[i] <= W) {
                if (dp[i+1][w + wt[i]] < dp[i][w] + val[i]) {
                    dp[i+1][w + wt[i]] = dp[i][w] + val[i];
                    use[i+1][w + wt[i]] = true;
                }
            }
        }
    }

    int best_w = max_element(dp[N].begin(), dp[N].end()) - dp[N].begin();

    chosen.clear();
    for (int i = N; i >= 1; i--) {
        if (use[i][best_w]) {
            chosen.push_back(i-1);
            best_w -= wt[i-1];
        }
    }
    reverse(chosen.begin(), chosen.end());
}

// ---------- 価値最大化 + 復元 ----------
void knapsack_value_restore(int N, int W,
                            const vector<int>& wt,
                            const vector<int>& val,
                            vector<int>& chosen) {
    const long long INF = 4e18;
    int Vsum = accumulate(val.begin(), val.end(), 0);

    vector<vector<long long>> dp(N+1, vector<long long>(Vsum+1, INF));
    vector<vector<bool>> use(N+1, vector<bool>(Vsum+1, false));
    dp[0][0] = 0;

    for (int i = 0; i < N; i++) {
        for (int v = 0; v <= Vsum; v++) {
            if (dp[i][v] == INF) continue;

            if (dp[i+1][v] > dp[i][v]) {
                dp[i+1][v] = dp[i][v];
                use[i+1][v] = false;
            }
            if (v + val[i] <= Vsum &&
                dp[i+1][v + val[i]] > dp[i][v] + wt[i]) {
                dp[i+1][v + val[i]] = dp[i][v] + wt[i];
                use[i+1][v + val[i]] = true;
            }
        }
    }

    int best_v = 0;
    for (int v = 0; v <= Vsum; v++) {
        if (dp[N][v] <= W) best_v = v;
    }

    chosen.clear();
    for (int i = N; i >= 1; i--) {
        if (use[i][best_v]) {
            chosen.push_back(i-1);
            best_v -= val[i-1];
        }
    }
    reverse(chosen.begin(), chosen.end());
}

// ---------- 価値DP × 最小個数 ----------
vector<int> knapsack_value_min_count(int N, const vector<int>& val) {
    const int INF = 1e9;
    int Vsum = accumulate(val.begin(), val.end(), 0);

    vector<int> dp(Vsum+1, INF);
    dp[0] = 0;

    for (int i = 0; i < N; i++) {
        for (int v = Vsum; v >= val[i]; v--) {
            dp[v] = min(dp[v], dp[v - val[i]] + 1);
        }
    }
    return dp;
}

// ---------- 価値DP × 最小個数 + 復元 ----------
void knapsack_value_min_count_restore(int N,
                                      const vector<int>& val,
                                      vector<int>& chosen) {
    const int INF = 1e9;
    int Vsum = accumulate(val.begin(), val.end(), 0);

    vector<vector<int>> dp(N+1, vector<int>(Vsum+1, INF));
    vector<vector<bool>> use(N+1, vector<bool>(Vsum+1, false));
    dp[0][0] = 0;

    for (int i = 0; i < N; i++) {
        for (int v = 0; v <= Vsum; v++) {
            if (dp[i][v] == INF) continue;

            if (dp[i+1][v] > dp[i][v]) {
                dp[i+1][v] = dp[i][v];
                use[i+1][v] = false;
            }
            if (v + val[i] <= Vsum &&
                dp[i+1][v + val[i]] > dp[i][v] + 1) {
                dp[i+1][v + val[i]] = dp[i][v] + 1;
                use[i+1][v + val[i]] = true;
            }
        }
    }

    int best_v = 0;
    for (int v = 0; v <= Vsum; v++) {
        if (dp[N][v] < INF) best_v = v;
    }

    chosen.clear();
    for (int i = N; i >= 1; i--) {
        if (use[i][best_v]) {
            chosen.push_back(i-1);
            best_v -= val[i-1];
        }
    }
    reverse(chosen.begin(), chosen.end());
}

// ---------- 0/1 通り数 ----------
long long knapsack_count(int N, int W, const vector<int>& wt, long long MOD) {
    vector<long long> dp(W+1, 0);
    dp[0] = 1;

    for (int i = 0; i < N; i++) {
        for (int w = W; w >= wt[i]; w--) {
            dp[w] = (dp[w] + dp[w - wt[i]]) % MOD;
        }
    }
    return dp[W];
}

// ---------- 完全ナップサック 通り数 ----------
long long knapsack_count_unbounded(int N, int W,
                                   const vector<int>& wt,
                                   long long MOD) {
    vector<long long> dp(W+1, 0);
    dp[0] = 1;

    for (int i = 0; i < N; i++) {
        for (int w = wt[i]; w <= W; w++) {
            dp[w] = (dp[w] + dp[w - wt[i]]) % MOD;
        }
    }
    return dp[W];
}
