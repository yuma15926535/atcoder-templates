int main(){
    int N,K;
    cin>>N>>K;
    vector<vector<int>>G(N);
    rep(i,0,N-1){
        int a,b;
        cin>>a>>b;
        a--;b--;
        G[a].push_back(b);
        G[b].push_back(a);
    }
    vector<bool>V(N,false);
    int v;
    rep(i,0,K){
        cin>>v;
        v--;
        V[v]=true;
    }
    vector<int>degree(N);
    //葉から開始するBFSで必要のない葉をすべて落とす
    queue<int>q;
    rep(i,0,N){
        degree[i]=G[i].size();
        if(degree[i]==1)q.push(i);
    }
    int ans=N;
    while(!q.empty()){
        int pos=q.front();
        q.pop();
        if(V[pos])continue;
        if(degree[pos]==0)continue;
        degree[pos]=0;
        ans--;
        for(int nxt:G[pos]){
            if(degree[nxt]==0)continue;
            degree[nxt]--;
            if(degree[nxt]==1)q.push(nxt);
        }
    }
    cout<<ans<<'\n';
    return 0;
}
