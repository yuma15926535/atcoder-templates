struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

vector<ll> sorted(vector<ll> A){
    sort(A.begin(), A.end());
    return A;
}

// [L, R] に入る index 範囲と個数を返す
// 戻り値: (l, r, cnt)  ※存在しないとき cnt=0, l=r=-1
tuple<int,int,long long> range_count(const vector<long long>& A, long long L, long long R){
    int n = A.size();
    int l = lower_bound(A.begin(), A.end(), L) - A.begin();
    int r = upper_bound(A.begin(), A.end(), R) - A.begin() - 1;

    if(l > r) return {-1, -1, 0};
    return {l, r, r - l + 1};
}

// [L, R] に入る 個数を返す
long long count_range(const vector<long long>& A, long long L, long long R){
    return upper_bound(A.begin(), A.end(), R)
         - lower_bound(A.begin(), A.end(), L);
}

// A[i] < x の個数
int count_less(const vector<ll>& A, ll x){
    return lower_bound(A.begin(), A.end(), x) - A.begin();
}

// A[i] <= x の個数
int count_leq(const vector<ll>& A, ll x){
    return upper_bound(A.begin(), A.end(), x) - A.begin();
}

// A[i] >= x の個数
int count_geq(const vector<ll>& A, ll x){
    return A.end() - lower_bound(A.begin(), A.end(), x);
}

// A[i] > x の個数
int count_greater(const vector<ll>& A, ll x){
    return A.end() - upper_bound(A.begin(), A.end(), x);
}

// 値 x が存在するか
bool exists(const vector<ll>& A, ll x){
    auto it = lower_bound(A.begin(), A.end(), x);
    return it != A.end() && *it == x;
}

// x に最も近い値との差（最小絶対差）
// ※Aが空でないことが前提
ll closest(const vector<ll>& A, ll x){
    int n = A.size();
    int i = lower_bound(A.begin(), A.end(), x) - A.begin();

    ll res = LLONG_MAX;
    if(i < n) res = min(res, abs(A[i] - x));
    if(i > 0) res = min(res, abs(A[i-1] - x));
    return res;
}

// 固定 i に対して A[j] <= x を満たす最大 j
// 存在しない場合は -1 を返す
int max_right_leq(const vector<ll>& A, int i, ll x){
    int j = upper_bound(A.begin()+i+1, A.end(), x) - A.begin() - 1;
    if(j <= i) return -1;
    return j;
}

// A[i] から見て A[j] - A[i] <= K の最大長
int max_len_diff_leq(const vector<ll>& A, ll K){
    int n = A.size();
    int ans = 0;
    for(int i=0;i<n;i++){
        int j = upper_bound(A.begin(), A.end(), A[i]+K) - A.begin();
        ans = max(ans, j - i);
    }
    return ans;
}

// A[i] + A[j] <= K のペア数（i < j）
// 計算量: O(N log N)
ll count_pair_leq(const vector<ll>& A, ll K){
    int n = A.size();
    ll res = 0;
    for(int i=0;i<n;i++){
        int j = upper_bound(A.begin()+i+1, A.end(), K - A[i]) - A.begin();
        res += j - (i+1);
    }
    return res;
}

// k番目の値（1-indexed）
// ※1 <= k <= N が前提
ll kth_value(const vector<ll>& A, ll k){
    return A[k-1];
}
