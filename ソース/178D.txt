ll mod = 1000000007;
vector<ll> fact, fact_inv, inv;
//二項係数の前処理
//階乗を前計算するので初期化のSIZEはnCkのn以上であれば十分
void init_nCk(int SIZE) {
    fact.resize(SIZE + 5);
    fact_inv.resize(SIZE + 5);
    inv.resize(SIZE + 5);
    fact[0] = fact[1] = 1;
    fact_inv[0] = fact_inv[1] = 1;
    inv[1] = 1;
    rep(i,2,SIZE+5){
        fact[i] = fact[i - 1] * i % mod;
        inv[i] = mod - inv[mod % i] * (mod / i) % mod;
        fact_inv[i] = fact_inv[i - 1] * inv[i] % mod;
    }
}
//二項係数の計算
//k<=n<=1e7かつPが素数の時に使える
ll nCk(ll n, ll k) {
    if (n < k) return 0;
    if (n < 0 || k < 0) return 0;
    return fact[n] * (fact_inv[k] * fact_inv[n - k] % mod) % mod;
}

int main(){
    int S;
    cin>>S;
    init_nCk(S);
    ll ans=0;
    //l:数列長
    erep(l,1,S/3){
        //(S-2*l-1)_C_(l-1)
        ans+=nCk(S-2*l-1,l-1);

    }

    return 0;
}
