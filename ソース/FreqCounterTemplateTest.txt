struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
============================================================
 FreqCounter（Frequency Counter）
------------------------------------------------------------
【Layer0 + Layer2】

値の出現回数と「頻度情報」を同時管理する構造体。

 value → count
 count → 種類数

multiset / Counter の上位互換。

------------------------------------------------------------
【できること】
・個数付き insert / erase
・種類数管理
・総要素数
・頻度=f の種類数
・最大頻度 O(1)

------------------------------------------------------------
【計算量】
insert / erase : O(1) average
count          : O(1)
distinct       : O(1)
freq_kind      : O(1)
max_freq       : O(1) amortized
============================================================
*/
template<class T>
struct FreqCounter {

    //==============================
    // 内部状態
    //==============================

    // x -> 出現回数
    unordered_map<T,long long> cnt;

    // f -> 出現回数fの値が何種類あるか
    unordered_map<long long,long long> freq_cnt;

    long long total_size = 0;     // 全要素数
    long long distinct_size = 0;  // 種類数
    long long max_freq_val = 0;   // 最大頻度

    //==============================
    // 基本情報取得
    //==============================

    // 総要素数
    long long size() const {
        return total_size;
    }

    // 種類数
    long long distinct() const {
        return distinct_size;
    }

    // x の出現回数
    long long count(const T& x) const {
        auto it = cnt.find(x);
        if(it == cnt.end()) return 0;
        return it->second;
    }

    // 存在判定
    bool exist(const T& x) const {
        return count(x) > 0;
    }

    //==============================
    // insert
    //==============================
    /*
        x を k 個追加
    */
    void insert(const T& x, long long k = 1){
        if(k <= 0) return;

        long long old = count(x);
        long long now = old + k;

        // --- freq 更新 ---
        if(old > 0){
            freq_cnt[old]--;
        }else{
            distinct_size++;
        }

        cnt[x] = now;
        freq_cnt[now]++;

        total_size += k;

        // 最大頻度更新
        max_freq_val = max(max_freq_val, now);
    }

    //==============================
    // erase
    //==============================
    /*
        x を最大 k 個削除
        (存在数より多く消そうとしても安全)
    */
    void erase(const T& x, long long k = 1){
        auto it = cnt.find(x);
        if(it == cnt.end() || k <= 0) return;

        long long old = it->second;
        long long dec = min(old, k);
        long long now = old - dec;

        // --- freq 更新 ---
        freq_cnt[old]--;

        if(now == 0){
            cnt.erase(it);
            distinct_size--;
        }else{
            cnt[x] = now;
            freq_cnt[now]++;
        }

        total_size -= dec;

        // 最大頻度補正（amortized O(1)）
        while(max_freq_val > 0 &&
              freq_cnt[max_freq_val] == 0)
            max_freq_val--;
    }

    //==============================
    // 頻度情報
    //==============================

    // 出現回数 = f の種類数
    long long freq_kind(long long f) const {
        auto it = freq_cnt.find(f);
        if(it == freq_cnt.end()) return 0;
        return it->second;
    }

    // 最大頻度
    long long max_freq() const {
        return max_freq_val;
    }

    //==============================
    // utility
    //==============================

    bool empty() const {
        return total_size == 0;
    }

    void clear(){
        cnt.clear();
        freq_cnt.clear();
        total_size = 0;
        distinct_size = 0;
        max_freq_val = 0;
    }
};

void solv(){
    cout<<"===== FreqCounter TEST START =====\n";

    FreqCounter<int> fc;

    //--------------------------------
    cout<<"\n[初期状態]\n";
    cout<<"size = "<<fc.size()<<"\n";
    cout<<"distinct = "<<fc.distinct()<<"\n";

    //--------------------------------
    cout<<"\n[1 を追加]\n";
    fc.insert(1);
    cout<<"count(1) = "<<fc.count(1)<<"\n";
    cout<<"size = "<<fc.size()<<"\n";
    cout<<"distinct = "<<fc.distinct()<<"\n";
    cout<<"max_freq = "<<fc.max_freq()<<"\n";

    //--------------------------------
    cout<<"\n[2 を3個追加]\n";
    fc.insert(2,3);
    cout<<"count(2) = "<<fc.count(2)<<"\n";
    cout<<"size = "<<fc.size()<<"\n";
    cout<<"distinct = "<<fc.distinct()<<"\n";
    cout<<"max_freq = "<<fc.max_freq()<<"\n";

    //--------------------------------
    cout<<"\n[1 を2個追加]\n";
    fc.insert(1,2);
    cout<<"count(1) = "<<fc.count(1)<<"\n";
    cout<<"freq=3 の種類数 = "
        <<fc.freq_kind(3)<<"\n";
    cout<<"max_freq = "<<fc.max_freq()<<"\n";

    //--------------------------------
    cout<<"\n[2 を1個削除]\n";
    fc.erase(2);
    cout<<"count(2) = "<<fc.count(2)<<"\n";
    cout<<"freq=2 の種類数 = "
        <<fc.freq_kind(2)<<"\n";
    cout<<"max_freq = "<<fc.max_freq()<<"\n";

    //--------------------------------
    cout<<"\n[1 を全部削除]\n";
    fc.erase(1,100);
    cout<<"exist(1) = "<<fc.exist(1)<<"\n";
    cout<<"distinct = "<<fc.distinct()<<"\n";
    cout<<"size = "<<fc.size()<<"\n";
    cout<<"max_freq = "<<fc.max_freq()<<"\n";

    //--------------------------------
    cout<<"\n[存在しない値削除]\n";
    fc.erase(99);
    cout<<"(安全に無視される)\n";

    //--------------------------------
    cout<<"\n[全削除テスト]\n";
    fc.erase(2,100);
    cout<<"size = "<<fc.size()<<"\n";
    cout<<"distinct = "<<fc.distinct()<<"\n";
    cout<<"max_freq = "<<fc.max_freq()<<"\n";

    cout<<"\n===== TEST END =====\n";
    return;
}
