struct GraphIO {
    // 非重みグラフ入力
    static vector<vector<int>>
    read_unweighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,   // 必要なら渡す
        vector<int>* outdeg = nullptr,  // 必要なら渡す
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<int>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            cin >> u >> v;
            u -= input_index;
            v -= input_index;
            adj[u].push_back(v);
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back(u);
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj; // RVOでコピーほぼなし
    }
    // 重み付きグラフ入力
    static vector<vector<pair<int,long long>>>
    read_weighted(
        int n,
        int m,
        vector<int>* indeg = nullptr,
        vector<int>* outdeg = nullptr,
        bool directed = false,
        int input_index = 1
    ){
        vector<vector<pair<int,long long>>> adj(n);
        if(indeg)  indeg->assign(n, 0);
        if(outdeg) outdeg->assign(n, 0);
        for(int i = 0; i < m; i++){
            int u, v;
            long long w;
            cin >> u >> v >> w;
            u -= input_index;
            v -= input_index;
            adj[u].push_back({v, w});
            if(outdeg) (*outdeg)[u]++;
            if(indeg)  (*indeg)[v]++;
            if(!directed){
                adj[v].push_back({u, w});
                if(outdeg) (*outdeg)[v]++;
                if(indeg)  (*indeg)[u]++;
            }
        }
        return adj;
    }
};

/*
============================================================
 MonoidPrefix
------------------------------------------------------------
【概要】
  静的配列に対する prefix / suffix モノイド累積構造。

  以下を O(1) で取得できる：
    ・prefix
    ・suffix
    ・区間除外（except i）
    ・左右結合結果

【必要条件（重要）】
  演算 op は「モノイド」である必要がある。

  ▼ モノイド条件
    1. 結合法則が成立
         op(op(a,b),c) == op(a,op(b,c))

    2. 単位元 ID が存在
         op(ID,a) = a
         op(a,ID) = a

【代表例】
    和        op(a,b)=a+b        ID=0
    gcd       op(a,b)=gcd(a,b)   ID=0
    min       op(a,b)=min(a,b)   ID=+INF
    max       op(a,b)=max(a,b)   ID=-INF
    xor       op(a,b)=a^b        ID=0

【用途】
  ・1点除外
  ・左右独立計算
  ・prefix/suffix DP
  ・削除して最適化
  ・SegTree不要な静的問題

============================================================
*/
template<class T, class OP>
struct MonoidPrefix {
    int N;
    std::vector<T> pre, suf;
    OP op;
    T ID;

    // A : 元配列
    // op: モノイド演算
    // id: 単位元
    MonoidPrefix(const std::vector<T>& A, OP op, T id)
        : N((int)A.size()), op(op), ID(id)
    {
        pre.assign(N+1, ID);
        suf.assign(N+1, ID);

        // prefix
        for(int i=0;i<N;i++)
            pre[i+1] = op(pre[i], A[i]);

        // suffix
        for(int i=N-1;i>=0;i--)
            suf[i] = op(A[i], suf[i+1]);
    }

    //--------------------------------------------------------
    // [0,r)
    //--------------------------------------------------------
    T prefix(int r) const {
        return pre[r];
    }

    //--------------------------------------------------------
    // [l,N)
    //--------------------------------------------------------
    T suffix(int l) const {
        return suf[l];
    }

    //--------------------------------------------------------
    // A[i] を除いた全体
    // op(A[0..i-1], A[i+1..N-1])
    //--------------------------------------------------------
    T except(int i) const {
        return op(pre[i], suf[i+1]);
    }

    //--------------------------------------------------------
    // 全体
    //--------------------------------------------------------
    T all() const {
        return pre[N];
    }

    //--------------------------------------------------------
    // 任意結合（左右結果を合成）
    // ※ 非可換モノイドでも安全
    //--------------------------------------------------------
    T combine_lr(int l, int r) const {
        return op(pre[l], suf[r]);
    }
};

template<class T, class OP> MonoidPrefix(const vector<T>&, OP, T) -> MonoidPrefix<T, OP>;

void solv(){
    vector<long long> A = {6, 9, 15, 10};

    cout << "=== SUM TEST ===\n";

    MonoidPrefix sum_mp(
        A,
        [](ll a,ll b){ return a+b; },
        0LL
    );

    cout << "全体和: " << sum_mp.all() << "\n";
    cout << "prefix(2): " << sum_mp.prefix(2) << "\n";
    cout << "suffix(2): " << sum_mp.suffix(2) << "\n";

    for(int i=0;i<A.size();i++)
        cout << "except("<<i<<"): "
             << sum_mp.except(i) << "\n";


    cout << "\n=== GCD TEST ===\n";

    MonoidPrefix gcd_mp(
        A,
        [](ll a,ll b){ return gcd(a,b); },
        0LL
    );

    cout << "全体gcd: " << gcd_mp.all() << "\n";

    for(int i=0;i<A.size();i++)
        cout << "except("<<i<<"): "
             << gcd_mp.except(i) << "\n";


    cout << "\n=== MAX TEST ===\n";

    MonoidPrefix max_mp(
        A,
        [](ll a,ll b){ return max(a,b); },
        LLONG_MIN
    );

    cout << "全体max: " << max_mp.all() << "\n";

    for(int i=0;i<A.size();i++)
        cout << "except("<<i<<"): "
             << max_mp.except(i) << "\n";
    return;
}
