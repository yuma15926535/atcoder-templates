#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int N,M;
struct pc{
    int p;
    ll c;
};
vector<vector<pc>> G;
//ansを負で初期化すればより安全
ll ans=9e18;
void dfs(int pos,set<int> s,ll num){
    if(pos==N-1){
        if(ans==9e19)ans=num;
        else ans=min(ans,num);
        return;
    }else{
        for(auto cp:G[pos])if(!s.count(cp.p)){
            s.insert(cp.p);
            dfs(cp.p,s,num^cp.c);
            s.erase(cp.p);
        }
    }
}

int main()
{
    cin>>N>>M;
    G.resize(N);
    rep(i,0,M){
        int u,v;
        ll w;
        cin>>u>>v>>w;
        u--;
        v--;
        G[u].push_back({v,w});
        G[v].push_back({u,w});
    }
    //Nが十分に小さいので通過済みの点の集合を引数に持たせる
    set<int> s;
    s.insert(0);
    dfs(0,s,0);
    cout<<ans<<"\n";
    return 0;
}
