vector<int> A;
//二分探索
void get(int target){
    int len=A.size();
    if(target<=A[0]||A[len-1]<=target){
        cout<<min(abs(A[0]-target),abs(A[len-1]-target))<<"\n";
        return;
    }
    int left=0,right=len-1;
    while(right-left>1){
        int mid=(right+left)/2;
        if(A[mid]<=target)left=mid;
        else right=mid;
    }
    cout<<min(abs(A[left]-target),abs(A[right]-target))<<"\n";
    return;
}

int main()
{
    int N;
    cin>>N;
    A=vector<int>(N);
    rep(i,0,N)cin>>A[i];
    sort(nall(A));
    int Q;
    cin>>Q;
    vector<int> B(Q);
    rep(i,0,Q)cin>>B[i];
    rep(i,0,Q)get(B[i]);
    return 0;
}
