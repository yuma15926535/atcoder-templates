#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()

int main()
{
    int N;
    cin>>N;
    vector<int> C1(N+1,0),C2(N+1,0);
    //累積和
    erep(i,1,N){
        int c,p;
        cin>>c>>p;
        C1[i]=C1[i-1];C2[i]=C2[i-1];
        if(c==1)C1[i]+=p;
        else C2[i]+=p;
    }
    int Q;
    cin>>Q;
    vector<pair<int,int>> q(Q);
    for(auto &p:q)cin>>p.first>>p.second;
    for(auto p:q)cout<<C1[p.second]-C1[p.first-1]<<" "<<C2[p.second]-C2[p.first-1]<<"\n";
    return 0;
}
