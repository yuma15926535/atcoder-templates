int main()
{
    ll L,R;
    cin>>L>>R;
    int sl=to_string(L).size(),sr=to_string(R).size();
    const int m=1000000007;
    ll ans=0;
    //群数列+数列の和 つまり数列の問題
    erep(l,sl,sr){
        ll s,e;
        if(l==sl)s=L;
        else s=gets(l);
        if(l==sr)e=R;

        ans=(ans+(e-s+1)%m*(s%m+e%m)%m*500000004%m*l)%m;
    }
