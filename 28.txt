#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    int N;
    cin>>N;
    //二次元の領域加算は二次元いもす法(領域加算なら使える)
    vector<vector<int>> v(1002,vector<int>(1002,0));
    rep(i,0,N){
        int a,b,c,d;
        cin>>a>>b>>c>>d;
        v[a][b]++;
        v[a][d]--;
        v[c][b]--;
        v[c][d]++;
    }
    rep(i,0,1001)rep(j,1,1001)v[i][j]+=v[i][j-1];
    rep(i,1,1001)rep(j,0,1001)v[i][j]+=v[i-1][j];
    vector<int> ans(N+1,0);
    rep(i,0,1001)rep(j,0,1001)if(v[i][j]>=1)ans[v[i][j]]++;
    erep(i,1,N)cout<<ans[i]<<"\n";
    return 0;
}
