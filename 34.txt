#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    int N,K;
    cin>>N>>K;
    vector<int> a(N);
    rep(i,0,N)cin>>a[i];
    //iに対してjハ単調増加
    //常にi<=j
    //尺取り法
    map<int,int> m;
    m[a[0]]=1;
    int ans=0,j=0;
    rep(i,0,N){
        if(i!=0){
            m[a[i]]--;
            //eraseに渡すイテレータをfindで取得
            //存在しない場合はm.end()が戻り値
            if(m[a[i]]==0)m.erase(m.find(a[i]));
        }
        while(j<N-1&&(m.count(a[j+1])||m.size()<K)){
            j++;
            if(!m.count(a[j]))m[a[j]]=1;
            else m[a[j]]++;
        }
        ans=max(ans,j-(int)i+1);
    }
    cout<<ans<<"\n";
    return 0;
}
