int N;
vector<vector<int>> G;
vector<int> result;
const int yet=-1;

//BFS
void bfs(int start){
    result=vector<int>(N+1,yet);
    queue<int> q;
    q.push(start);
    result[start]=0;
    while(!q.empty()){
        int pos=q.front();
        q.pop();
        for(int to:G[pos])if(result[to]==yet){
            result[to]=result[pos]+1;
            q.push(to);
        }
    }
}
