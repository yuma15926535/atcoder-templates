using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    int N,L;
    cin>>N>>L;
    vector<ll> dp(N+1);
    //dpによる実装
    dp[0]=1;
    erep(i,1,N){
        if(i<L)dp[i]=dp[i-1]%1000000007;
        else dp[i]=(dp[i-1]+dp[i-L])%1000000007;
    }
    cout<<dp[N]<<"\n";
    return 0;
}

map<int,ll> m;
int L;
const ll l=1000000007;
//再起関数による実装
ll solv(int num){
    if(num<L)return 1;
    if(m.count(num))return m[num];
    return (m[num]=(solv(num-1)+solv(num-L))%l);
}

{
    int N;
    cin>>N>>L;
    cout<<solv(N)<<"\n";
