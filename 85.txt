#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    ll K;
    cin>>K;
    vector<ll> v;
    //O(ルートK)で約数全列挙
    for(ll i=1;i*i<=K;i++){
        if(K%i!=0)continue;
        v.push_back(i);
        if(i*i!=K)v.push_back(K/i);
    }
    sort(nall(v));
    ll ans=0;
    rep(i,0,v.size())rep(j,i,v.size()){
        //オーバーフロー対策
        if(K/v[i]<v[j])continue;
        //積が約数にならない場合対策
        if(K%(v[i]*v[j])!=0)continue;
        //b以上であればカウント
        if(v[j]<=K/(v[i]*v[j]))ans++;
    }
    cout<<ans<<"\n";
    return 0;
}
