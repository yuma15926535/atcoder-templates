#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    int N;
    cin>>N;
    vector<pair<int,int>> A(N);
    rep(i,0,N)cin>>A[i].first>>A[i].second;
    //マンハッタン距離の最小はx,y座標で分けられてかつ中央値でとる
    vector<int> H(N),W(N);
    rep(i,0,N){
        H[i]=A[i].first;
        W[i]=A[i].second;
    }
    sort(nall(H));
    sort(nall(W));
    ll x=H[N/2],y=W[N/2];
    ll ans=0;
    for(auto p:A)ans+=abs(p.first-x)+abs(p.second-y);
    cout<<ans<<"\n";
    return 0;
}
