#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using ull = unsigned long long;
#define rep(i,s,n) for(ll i=s;i<n;i++)//i=s,s+1...n-1 ノーマルfor
#define erep(i,s,n) for(ll i=s;i<=n;i++)//i=s,s+1...n イコールfor
#define rrep(i,s,n) for(ll i = (s)-1; i>=n;i--)//s-1スタートでnまで落ちる
#define nall(a) a.begin(),a.end()
#define rall(a) a.rbegin(),a.rend()
#define YES cout<<"Yes\n"
#define NO cout<<"No\n"
#define YN {cout<<"Yes\n";}else{cout<<"No\n";}
#define next_p(v) next_permutation(v.begin(),v.end())

int main()
{
    int N;
    string S;
    cin>>N>>S;
    const int m=1000000007;
    vector<vector<ll>> dp(8,vector<ll>(N+1,0));
    string target="atcoder";
    //Sを1文字ずつ見ていく+targetの文字を1文字ずつ見ていく二段階のDP
    //DPの更新は1文字目か否かの判定をS,targetの両方で行う
    //初期化には注意する
    //atcoderに重なる文字がないからできること?
    erep(i,1,7)erep(j,1,N){
        if(i==1){
            if(j==1&&S[j-1]==target[i-1])dp[i][j]=1;
            else if(S[j-1]==target[i-1])dp[i][j]=(dp[i][j-1]+1)%m;
            else dp[i][j]=dp[i][j-1];
        }else{
            if(j!=1&&S[j-1]==target[i-1])dp[i][j]=(dp[i-1][j-1]+dp[i][j-1])%m;
            else if(j!=1)dp[i][j]=dp[i][j-1];
        }
    }
    cout<<dp[7][N]<<"\n";
    return 0;
}
